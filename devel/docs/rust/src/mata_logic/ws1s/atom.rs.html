1//! The atoms of WS1S as automata over the bits of their arguments.
2//!
3//! A first-order argument is a position, encoded as a singleton set: its tape reads [`ONE`] exactly once. Every
4//! atom accepts only words where its first-order arguments are singletons, and keeps accepting when zero columns
5//! follow, as [`Padding::Zero`](crate::Padding::Zero) needs.
6
7use mata_lazy::{Automaton, Symbol};
8
9use crate::{ONE, ZERO};
10
11/// How a value compares to a bound.
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub(crate) enum Cmp {
14	Eq,
15	Le,
16	Ge,
17}
18
19impl Cmp {
20	fn holds(self, value: i64, bound: i64) -> bool {
21		match self {
22			Cmp::Eq => value == bound,
23			Cmp::Le => value <= bound,
24			Cmp::Ge => value >= bound,
25		}
26	}
27}
28
29/// An atom; `x`, `y` are first-order arguments and `X`, `Y` second-order ones, in this order.
30#[derive(Clone, Copy, Debug, PartialEq, Eq)]
31pub(crate) enum Atom {
32	/// `x` is a position.
33	Singleton,
34	/// `x cmp k`.
35	Position(Cmp, i64),
36	/// `y − x cmp d`.
37	Distance(Cmp, i64),
38	/// `x + k ∈ X`.
39	Member(u32),
40	/// `k ∈ X`.
41	Contains(u32),
42	/// `X ⊆ Y`.
43	Subset,
44	/// `X = Y`.
45	Equal,
46	/// `X = ∅`.
47	Empty,
48}
49
50impl Atom {
51	/// The number of arguments.
52	pub(crate) fn arguments(self) -> usize {
53		match self {
54			Atom::Singleton | Atom::Position(..) | Atom::Contains(_) | Atom::Empty => 1,
55			Atom::Distance(..) | Atom::Member(_) | Atom::Subset | Atom::Equal => 2,
56		}
57	}
58
59	fn initial(self) -> State {
60		match self {
61			Atom::Contains(k) => State::Wait(k),
62			_ => State::Start(0),
63		}
64	}
65
66	fn is_final(self, state: State) -> bool {
67		match self {
68			Atom::Subset | Atom::Equal | Atom::Empty => true,
69			_ => state == State::Done,
70		}
71	}
72
73	/// The state after reading `bits`, one per argument, or `None` if the word is rejected.
74	fn step(self, state: State, bits: [bool; 2]) -> Option<State> {
75		let [first, second] = bits;
76		match (self, state) {
77			(Atom::Subset, _) => (!first || second).then_some(state),
78			(Atom::Equal, _) => (first == second).then_some(state),
79			(Atom::Empty, _) => (!first).then_some(state),
80			// After the position, nothing but zeros on the first-order arguments.
81			(Atom::Singleton | Atom::Position(..) | Atom::Distance(..), State::Done) => {
82				(!first && !second).then_some(State::Done)
83			}
84			(Atom::Member(_), State::Done | State::Wait(_)) if first => None,
85			(Atom::Contains(_) | Atom::Member(_), State::Done) => Some(State::Done),
86			(Atom::Singleton, _) => Some(if first { State::Done } else { state }),
87			(Atom::Position(cmp, k), State::Start(n)) => {
88				if first {
89					cmp.holds(i64::from(n), k).then_some(State::Done)
90				} else {
91					// Positions from `saturation` on compare to `k` alike.
92					let saturation = u32::try_from((k + 1).max(0)).unwrap_or(u32::MAX);
93					Some(State::Start(n.saturating_add(1).min(saturation)))
94				}
95			}
96			(Atom::Distance(cmp, d), State::Start(_)) => match (first, second) {
97				(false, false) => Some(state),
98				(true, true) => cmp.holds(0, d).then_some(State::Done),
99				(x_first, _) => Some(State::Between { x_first, n: 0 }),
100			},
101			(Atom::Distance(cmp, d), State::Between { x_first, n }) => {
102				let (earlier, later) = if x_first {
103					(first, second)
104				} else {
105					(second, first)
106				};
107				if earlier {
108					return None;
109				}
110
111				if later {
112					let distance = i64::from(n) + 1;
113					let difference = if x_first { distance } else { -distance };
114					return cmp.holds(difference, d).then_some(State::Done);
115				}
116
117				// Distances above |d| + 1 compare to `d` alike.
118				let saturation = u32::try_from(d.unsigned_abs() + 1).unwrap_or(u32::MAX);
119				Some(State::Between {
120					x_first,
121					n: n.saturating_add(1).min(saturation),
122				})
123			}
124			(Atom::Member(k), State::Start(_)) => Some(match (first, k) {
125				(false, _) => state,
126				(true, 0) => return second.then_some(State::Done),
127				(true, k) => State::Wait(k - 1),
128			}),
129			(Atom::Member(_) | Atom::Contains(_), State::Wait(0)) => {
130				let member = if matches!(self, Atom::Contains(_)) {
131					first
132				} else {
133					second
134				};
135				member.then_some(State::Done)
136			}
137			(Atom::Member(_) | Atom::Contains(_), State::Wait(r)) => Some(State::Wait(r - 1)),
138			(_, _) => unreachable!("{self:?} never reaches {state:?}"),
139		}
140	}
141}

143/// A state of an [`Atom`].
144#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
145pub(crate) enum State {
146	/// No position read yet, after `n` columns (saturated).
147	Start(u32),
148	/// One of the two positions read, `n` columns ago (saturated); `x_first` if it was `x`.
149	Between { x_first: bool, n: u32 },
150	/// `r` more columns before the column whose position must be a member.
151	Wait(u32),
152	/// The atom holds; the positions have been read.
153	Done,
154}

156/// An [`Atom`] whose argument `i` is read on tape `tapes[i]`. Two arguments may share a tape, as in `x < x`.
157pub(crate) struct Leaf {
158	pub(crate) atom: Atom,
159	pub(crate) tapes: [usize; 2],
160	pub(crate) arity: usize,
161}
162
163impl Automaton for Leaf {
164	type State = State;
165
166	fn arity(&self) -> usize {
167		self.arity
168	}
169
170	fn initial_states(&self) -> impl IntoIterator<Item = State> {
171		[self.atom.initial()]
172	}
173
174	fn is_final(&self, state: State) -> bool {
175		self.atom.is_final(state)
176	}
177
178	fn for_each_move(&self, state: State, mut visit: impl FnMut(&[Symbol], State)) {
179		let mut symbols = vec![ZERO; self.arity];
180		for column in 0..1_usize << self.arity {
181			let bit = |tape: usize| column >> tape & 1 == 1;
182			for (tape, symbol) in symbols.iter_mut().enumerate() {
183				*symbol = if bit(tape) { ONE } else { ZERO };
184			}
185
186			let second = self.atom.arguments() == 2 && bit(self.tapes[1]);
187			if let Some(next) = self.atom.step(state, [bit(self.tapes[0]), second]) {
188				visit(&symbols, next);
189			}
190		}
191	}
192}
