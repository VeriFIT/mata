1//! WS1S, the weak monadic second-order logic of one successor: formulas over positions and finite sets of
2//! positions, parsed from a MONA-like syntax and decided by a [`Logic`] with [`Padding::Zero`].
3//!
4//! ```
5//! use mata_lazy::Options;
6//! use mata_logic::ws1s::{Value, Ws1s};
7//!
8//! let mut ws1s = Ws1s::new();
9//! let phi = ws1s.parse("(x = y + 1 & y = z + 1) => x = z + 2")?;
10//! assert!(ws1s.valid(&phi, &Options::new())?);
11//!
12//! let psi = ws1s.parse("x in X & 3 < x & all1 y: y in X => y <= x")?;
13//! let model = ws1s.model(&psi, &Options::new())?.expect("satisfiable");
14//! let Some(&Value::Position(x)) = model.get("x") else { panic!("x is a position") };
15//! assert!(x > 3);
16//! # Ok::<(), mata_logic::Error>(())
17//! ```
18//!
19//! # Semantics
20//!
21//! A first-order variable is a position, a natural number. A second-order variable is a finite set of positions.
22//! Quantifiers range over all of them, so `all1 x: ex1 y: x < y` is valid (there is no last position), and
23//! `ex2 X: 0 in X & all1 x: x in X => x + 1 in X` is not (no finite set is closed under successor). A formula
24//! with free variables is valid if every assignment satisfies it.
25//!
26//! # Syntax
27//!
28//! | Form | Meaning |
29//! |---|---|
30//! | `x`, `y1`, `pos_2` | a first-order variable: the name starts with a lowercase ASCII letter |
31//! | `X`, `Y1`, `Set_2` | a second-order variable: the name starts with an uppercase ASCII letter |
32//! | `3`, `x + 3` | a position: a constant, or a variable plus a constant |
33//! | `empty` | the empty set |
34//! | `t = u`, `t < u`, `t <= u` | positions compared |
35//! | `t in X` | membership |
36//! | `X = Y`, `X sub Y` | sets equal, subset |
37//! | `true`, `false` | constants |
38//! | `~φ`, `φ & ψ`, `φ \| ψ`, `φ => ψ`, `φ <=> ψ` | connectives, from the tightest to the loosest |
39//! | `ex1 x, y: φ`, `all1 x: φ` | first-order quantifiers |
40//! | `ex2 X, Y: φ`, `all2 X: φ` | second-order quantifiers |
41//!
42//! Names and digits are ASCII; a name continues with letters, digits and `_`. The keywords `ex1`, `ex2`, `all1`,
43//! `all2`, `in`, `sub`, `true`, `false` and `empty` are not names. The case of a name fixes its kind, so a
44//! quantifier binds names of its order only. `=>` is right associative, `<=>` left associative. A quantifier
45//! reaches as far right as possible: `a & ex1 x: b | c` is `a & (ex1 x: (b | c))`. A bound name hides the free
46//! variable and the outer bindings of the same name. Every free name is one variable of the [`Ws1s`]: formulas
47//! parsed by the same [`Ws1s`] share their free variables. The connectives and quantifiers nest at most 1000
48//! deep; a long chain of `&` or `|` counts as a balanced tree.
49//!
50//! # Encoding
51//!
52//! An assignment is a word with one tape per variable, the bit at step `i` telling whether position `i` is in the
53//! set; zero columns may follow. A position is a set with one element: each atom accepts only singletons on its
54//! first-order tapes, the first-order quantifiers conjoin this constraint to their body (`ex1 x: φ` is
55//! `∃x. sing(x) ∧ φ`, `all1 x: φ` is `¬∃x. sing(x) ∧ ¬φ`), and so do the queries for the free first-order
56//! variables. The negation of an atom thus holds words that are no assignment, but these never reach an answer.

70/// The value of a variable in an [`Assignment`].
71#[derive(Clone, Debug, PartialEq, Eq, Hash)]
72pub enum Value {
73	/// The position of a first-order variable.
74	Position(usize),
75	/// The positions of a second-order variable, ascending.
76	Set(Vec<usize>),
77}
78
79impl fmt::Display for Value {
80	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
81		match self {
82			Value::Position(position) => write!(f, "{position}"),
83			Value::Set(positions) => {
84				f.write_str("{")?;
85				for (index, position) in positions.iter().enumerate() {
86					if index > 0 {
87						f.write_str(", ")?;
88					}
89					write!(f, "{position}")?;
90				}
91				f.write_str("}")
92			}
93		}
94	}
95}

97/// Values of the free variables of a prop, displayed as `x = 3, X = {1, 4}`.
98#[derive(Clone, Debug, PartialEq, Eq, Hash)]
99pub struct Assignment {
100	values: Vec<(String, Value)>,
101}
102
103impl Assignment {
104	/// The value of the variable called `name`, or `None` if the assignment does not give it one.
105	#[must_use]
106	pub fn get(&self, name: &str) -> Option<&Value> {
107		self.values
108			.iter()
109			.find(|(known, _)| known == name)
110			.map(|(_, value)| value)
111	}
112
113	/// The variables by name with their values, in the order of [`Prop::vars`].
114	pub fn iter(&self) -> impl ExactSizeIterator<Item = (&str, &Value)> {
115		self.values
116			.iter()
117			.map(|(name, value)| (name.as_str(), value))
118	}
119}
120
121impl fmt::Display for Assignment {
122	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
123		for (index, (name, value)) in self.iter().enumerate() {
124			if index > 0 {
125				f.write_str(", ")?;
126			}
127			write!(f, "{name} = {value}")?;
128		}
129		Ok(())
130	}
131}

133/// Parses and decides WS1S formulas; see the [module](self) for the syntax and the semantics.
134#[derive(Debug)]
135pub struct Ws1s {
136	logic: Logic,
137	/// The free variables of the parsed formulas, by name.
138	free: FxHashMap<String, Var>,
139	/// The first-order variables, free and bound.
140	first_order: FxHashSet<Var>,
141	/// `sing(x)` per first-order variable `x`.
142	singletons: FxHashMap<Var, Prop>,
143}

163	/// The logic that holds the props, for its [`formula`](Logic::formula) and the names of the variables.
164	#[must_use]

166		&self.logic
167	}

169	/// The logic that holds the props, to combine them with its connectives. Its quantifiers do not constrain

171	pub fn logic_mut(&mut self) -> &mut Logic {
172		&mut self.logic

175	/// Parses `text` into a prop over its free variables.
176	///
177	/// A first-order variable is free in the prop even if no position satisfies its atoms, but an atom that the
178	/// parser decides, such as `x in empty` or `empty sub X`, drops its variables.
179	///
180	/// # Errors
181	///
182	/// [`Parse`](Error::Parse) if `text` is not a formula, with the byte offset of the error.
183	pub fn parse(&mut self, text: &str) -> Result<Prop, Error> {
184		let parsed = parse::parse(text)?;
185		self.build(parsed)
186	}
187
188	/// The prop of `parsed`, whose instructions are well nested.
189	fn build(&mut self, parsed: Parsed<'_>) -> Result<Prop, Error> {
190		let free: Vec<Var> = parsed
191			.free
192			.iter()
193			.map(|&name| self.free_var(name))
194			.collect();
195		let bound: Vec<Var> = parsed
196			.bound
197			.iter()
198			.map(|&name| self.new_var(name))
199			.collect();
200		let var = |name: Name| match name {
201			Name::Free(index) => free[index],
202			Name::Bound(index) => bound[index],
203		};
204
205		let mut stack: Vec<Prop> = Vec::new();
206		for instr in parsed.instrs {
207			let prop = match instr {
208				Instr::Constant(value) => self.logic.constant(value)?,
209				Instr::Atom { atom, args } => self.atom(atom, args.map(var))?,
210				Instr::Not => {
211					let operand = stack.pop().expect("an operand");
212					self.logic.not(&operand)?
213				}
214				Instr::And(operands) | Instr::Or(operands) => {
215					let conjunction = matches!(instr, Instr::And(_));
216					let operands = stack.split_off(stack.len() - operands);
217					self.balanced(operands, conjunction)?
218				}
219				Instr::Implies | Instr::Iff => {
220					let rhs = stack.pop().expect("an operand");
221					let lhs = stack.pop().expect("an operand");
222					if instr == Instr::Implies {
223						self.logic.implies(&lhs, &rhs)?
224					} else {
225						self.logic.iff(&lhs, &rhs)?
226					}
227				}
228				Instr::Quantifier {
229					universal,
230					first_order,
231					vars,
232				} => {
233					let body = stack.pop().expect("an operand");
234					self.quantify(universal, first_order, &bound[vars], &body)?
235				}
236			};
237			stack.push(prop);
238		}
239
240		let prop = stack.pop().expect("a formula");
241		debug_assert!(stack.is_empty(), "one formula");
242		Ok(prop)
243	}

245	/// An assignment that satisfies `prop`, or `None` if it is unsatisfiable.
246	///
247	/// # Errors
248	///
249	/// [`Lazy`](Error::Lazy): a limit of `options`, or `prop` belongs to another logic.
250	pub fn model(&mut self, prop: &Prop, options: &Options) -> Result<Option<Assignment>, Error> {
251		let mut guarded = prop.clone();
252		for &var in prop.vars() {
253			if self.first_order.contains(&var) {
254				let singleton = self.singleton(var)?;
255				guarded = self.logic.and(&guarded, &singleton)?;
256			}
257		}
258
259		Ok(self
260			.logic
261			.model(&guarded, options)?
262			.map(|model| self.decode(&model)))
263	}
264
265	/// An assignment that falsifies `prop`, or `None` if it is valid.
266	///
267	/// # Errors
268	///
269	/// As [`model`](Ws1s::model).
270	pub fn counterexample(
271		&mut self,
272		prop: &Prop,
273		options: &Options,
274	) -> Result<Option<Assignment>, Error> {
275		let not_prop = self.logic.not(prop)?;
276		self.model(&not_prop, options)
277	}

279	/// Whether some assignment satisfies `prop`.
280	///
281	/// # Errors
282	///
283	/// As [`model`](Ws1s::model).
284	pub fn satisfiable(&mut self, prop: &Prop, options: &Options) -> Result<bool, Error> {
285		Ok(self.model(prop, options)?.is_some())
286	}
287
288	/// Whether every assignment satisfies `prop`.
289	///
290	/// # Errors
291	///
292	/// As [`model`](Ws1s::model).
293	pub fn valid(&mut self, prop: &Prop, options: &Options) -> Result<bool, Error> {
294		Ok(self.counterexample(prop, options)?.is_none())
295	}

297	fn free_var(&mut self, name: &str) -> Var {
298		if let Some(&var) = self.free.get(name) {
299			return var;
300		}
301
302		let var = self.new_var(name);
303		self.free.insert(name.to_owned(), var);
304		var
305	}
306
307	fn new_var(&mut self, name: &str) -> Var {
308		let var = self.logic.var(name);
309		if is_first_order(name) {
310			self.first_order.insert(var);
311		}
312		var
313	}

315	fn atom(&mut self, atom: Atom, args: [Var; 2]) -> Result<Prop, Error> {
316		let vars: &[Var] = if atom.arguments() == 1 || args[0] == args[1] {
317			&args[..1]
318		} else {
319			&args
320		};
321		let leaf = Leaf {
322			atom,
323			tapes: args.map(|arg| usize::from(arg != vars[0])),
324			arity: vars.len(),
325		};
326		self.logic.atom(&leaf, vars)
327	}
328
329	fn singleton(&mut self, var: Var) -> Result<Prop, Error> {
330		if let Some(prop) = self.singletons.get(&var) {
331			return Ok(prop.clone());
332		}
333
334		let prop = self.atom(Atom::Singleton, [var, var])?;
335		self.singletons.insert(var, prop.clone());
336		Ok(prop)
337	}
338
339	/// The conjunction or disjunction of `operands`, as a balanced tree.
340	fn balanced(&mut self, mut operands: Vec<Prop>, conjunction: bool) -> Result<Prop, Error> {
341		while operands.len() > 1 {
342			let mut next = Vec::with_capacity(operands.len().div_ceil(2));
343			for pair in operands.chunks(2) {
344				next.push(match pair {
345					[lhs, rhs] if conjunction => self.logic.and(lhs, rhs)?,
346					[lhs, rhs] => self.logic.or(lhs, rhs)?,
347					[single] => single.clone(),
348					_ => unreachable!("chunks of at most 2"),
349				});
350			}
351			operands = next;
352		}
353
354		Ok(operands.pop().expect("at least one operand"))
355	}

357	fn quantify(
358		&mut self,
359		universal: bool,
360		first_order: bool,
361		vars: &[Var],
362		body: &Prop,
363	) -> Result<Prop, Error> {
364		if !first_order {
365			return if universal {
366				self.logic.forall(body, vars)
367			} else {
368				self.logic.exists(body, vars)
369			};
370		}
371
372		// ∃x. sing(x) ∧ φ, and ∀x. φ as ¬∃x. sing(x) ∧ ¬φ.
373		let mut guarded = if universal {
374			self.logic.not(body)?
375		} else {
376			body.clone()
377		};
378		for &var in vars {
379			let singleton = self.singleton(var)?;
380			guarded = self.logic.and(&guarded, &singleton)?;
381		}
382
383		let exists = self.logic.exists(&guarded, vars)?;
384		if universal {
385			self.logic.not(&exists)
386		} else {
387			Ok(exists)
388		}
389	}

391	fn decode(&self, model: &Model) -> Assignment {
392		let values = model
393			.vars()
394			.iter()
395			.map(|&var| {
396				let bits = model.bits(var).expect("a variable of the model");
397				let mut positions = bits
398					.iter()
399					.enumerate()
400					.filter(|&(_, &bit)| bit)
401					.map(|(position, _)| position);
402				let value = if self.first_order.contains(&var) {
403					let position = positions.next().expect("a position is a singleton");
404					Value::Position(position)
405				} else {
406					Value::Set(positions.collect())
407				};
408				let name = self.logic.name(var).expect("a variable of the logic");
409				(name.to_owned(), value)
410			})
411			.collect();
412
413		Assignment { values }
414	}
