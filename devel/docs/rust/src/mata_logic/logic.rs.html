11/// The bit 0, as the symbol of a tape.
12pub const ZERO: Symbol = Symbol::new(0);
13/// The bit 1, as the symbol of a tape.
14pub const ONE: Symbol = Symbol::new(1);
15/// The alphabet of every tape.
16pub(crate) const BITS: [Symbol; 2] = [ZERO, ONE];

18/// How a value has many encodings: the columns that can be appended to a word without changing the values it
19/// encodes.
20#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
21pub enum Padding {
22	/// Columns of zeros, as for natural numbers least significant bit first, or for finite sets of positions
23	/// (WS1S). The empty word encodes zeros (empty sets).
24	Zero,
25	/// The last column, again, as for integers in two's complement least significant bit first (LIA), whose last
26	/// bit is the sign. Every word has at least one column; the empty word encodes nothing.
27	Sign,
28}

48/// A formula of a [`Logic`]: a [`Term`] of its [`Formula`] with one tape per free variable, in the order of
49/// [`vars`](Prop::vars).
50///
51/// Its relation holds exactly the encodings of the assignments that satisfy it: every word, with every padding,
52/// of every satisfying assignment, and nothing else.
53#[derive(Clone, Debug, PartialEq, Eq, Hash)]
54pub struct Prop {
55	term: Term,
56	vars: Box<[Var]>,
57}

63		self.term
64	}

67	#[must_use]
68	pub fn vars(&self) -> &[Var] {
69		&self.vars
70	}
71}

73/// Builds formulas over variables from atoms given as automata, and decides them with [`mata_lazy`].
74///
75/// An assignment is encoded as a word of columns, one bit per variable, read on one tape per variable in lock step.
76/// An atom is an [`Automaton`] over the bits [`ZERO`] and [`ONE`] that accepts exactly the encodings of the
77/// assignments that satisfy it, with every [`Padding`]. The connectives keep this invariant: the complement drops
78/// the words that encode nothing, and a quantifier closes the relation under padding after it drops a tape.
79///
80/// ```
81/// use mata_logic::{Logic, Padding, ZERO, ONE};
82/// use mata_lazy::{Automaton, Options, Symbol};
83///
84/// /// `x = y` over natural numbers: both tapes read the same bits.
85/// struct Equal;
86///
87/// impl Automaton for Equal {
88///     type State = ();
89///
90///     fn arity(&self) -> usize { 2 }
91///     fn initial_states(&self) -> impl IntoIterator<Item = ()> { [()] }
92///     fn is_final(&self, (): ()) -> bool { true }
93///     fn for_each_move(&self, (): (), mut visit: impl FnMut(&[Symbol], ())) {
94///         visit(&[ZERO, ZERO], ());
95///         visit(&[ONE, ONE], ());
96///     }
97/// }
98///
99/// let mut logic = Logic::new(Padding::Zero);
100/// let (x, y) = (logic.var("x"), logic.var("y"));
101/// let equal = logic.atom(&Equal, &[x, y])?;
102/// // ∀x ∃y. x = y
103/// let every = logic.exists(&equal, &[y])?;
104/// let every = logic.forall(&every, &[x])?;
105/// assert!(logic.counterexample(&every, &Options::new())?.is_none());
106/// # Ok::<(), mata_logic::Error>(())
107/// ```
108#[derive(Debug)]
109pub struct Logic {
110	formula: Formula,
111	padding: Padding,
112	names: Vec<String>,
113	/// The valid encodings over `n` tapes, by `n`.
114	encodings: FxHashMap<usize, Term>,
115}

159	/// An atom: the relation of `automaton`, whose tape `i` encodes `vars[i]`.
160	///
161	/// The relation must hold exactly the encodings of the assignments that satisfy the atom, with every padding,
162	/// and read only [`ZERO`] and [`ONE`] (and [`ANY`](Symbol::ANY), which ranges over them); the other connectives
163	/// rely on it. An atom without variables is decided now: it is true if `automaton` accepts the empty word.
164	///
165	/// # Errors
166	///
167	/// [`AtomArity`](Error::AtomArity) if `automaton` does not have one tape per variable,
168	/// [`ForeignVar`](Error::ForeignVar), [`RepeatedVar`](Error::RepeatedVar), or [`Lazy`](Error::Lazy) from
169	/// [`Formula::leaf_over`].
170	pub fn atom(&mut self, automaton: &impl Automaton, vars: &[Var]) -> Result<Prop, Error> {
171		if automaton.arity() != vars.len() {
172			return Err(Error::AtomArity {
173				arity: automaton.arity(),
174				vars: vars.len(),
175			});
176		}
177
178		self.check_vars(vars)?;
179		let leaf = self.formula.leaf_over(automaton, [BITS])?;
180		if vars.is_empty() {
181			// A relation without tapes counts steps; a prop without variables holds every length or none.
182			let holds = !self.formula.is_empty(leaf)?;
183			return self.constant(holds);
184		}
185
186		let mut order: Vec<usize> = (0..vars.len()).collect();
187		order.sort_unstable_by_key(|&tape| vars[tape]);
188		let term = if order.is_sorted() {
189			leaf
190		} else {
191			self.formula.project(leaf, &order)?
192		};
193
194		Ok(Prop {
195			term,
196			vars: order.iter().map(|&tape| vars[tape]).collect(),
197		})
198	}

210			vars: Box::default(),

214	/// `lhs ∧ rhs`.
215	///
216	/// # Errors
217	///
218	/// [`Lazy`](Error::Lazy) if a prop belongs to another logic.
219	pub fn and(&mut self, lhs: &Prop, rhs: &Prop) -> Result<Prop, Error> {
220		let mut synchronized = Vec::new();
221		let mut layout = Vec::new();
222		let mut vars = Vec::new();
223		let (mut i, mut j) = (0, 0);
224		loop {
225			let (tape, var) = match (lhs.vars.get(i), rhs.vars.get(j)) {
226				(None, None) => break,
227				(Some(&left), Some(&right)) if left == right => {
228					synchronized.push((i, j));
229					i += 1;
230					j += 1;
231					(Tape::Lhs(i - 1), left)
232				}
233				(Some(&left), right) if right.is_none_or(|&right| left < right) => {
234					i += 1;
235					(Tape::Lhs(i - 1), left)
236				}
237				(_, Some(&right)) => {
238					j += 1;
239					(Tape::Rhs(j - 1), right)
240				}
241				(Some(_), None) => unreachable!("the guard above takes it"),
242			};
243			layout.push(tape);
244			vars.push(var);
245		}
246
247		let term = self
248			.formula
249			.sync_product(lhs.term, rhs.term, &synchronized, &layout)?;
250
251		Ok(Prop {
252			term,
253			vars: vars.into(),
254		})
255	}

257	/// `lhs ∨ rhs`.
258	///
259	/// # Errors
260	///
261	/// [`Lazy`](Error::Lazy) if a prop belongs to another logic.
262	pub fn or(&mut self, lhs: &Prop, rhs: &Prop) -> Result<Prop, Error> {
263		let lhs = self.cylindrify(lhs, &rhs.vars)?;
264		let rhs = self.cylindrify(rhs, &lhs.vars)?;
265		let term = self.formula.union(lhs.term, rhs.term)?;
266
267		Ok(Prop {
268			term,
269			vars: lhs.vars,
270		})
271	}

273	/// `¬prop`.
274	///
275	/// # Errors
276	///
277	/// [`Lazy`](Error::Lazy) if `prop` belongs to another logic.
278	pub fn not(&mut self, prop: &Prop) -> Result<Prop, Error> {
279		let complement = self.formula.complement(prop.term)?;
280		// The complement also holds the words that encode nothing.
281		let term = match self.padding {
282			Padding::Sign if !prop.vars.is_empty() => {
283				let encodings = self.encodings(prop.vars.len())?;
284				self.formula.intersection(complement, encodings)?
285			}
286			Padding::Sign | Padding::Zero => complement,
287		};
288
289		Ok(Prop {
290			term,
291			vars: prop.vars.clone(),
292		})
293	}

295	/// `lhs → rhs`.
296	///
297	/// # Errors
298	///
299	/// [`Lazy`](Error::Lazy) if a prop belongs to another logic.
300	pub fn implies(&mut self, lhs: &Prop, rhs: &Prop) -> Result<Prop, Error> {
301		let not_lhs = self.not(lhs)?;
302		self.or(&not_lhs, rhs)
303	}

305	/// `lhs ↔ rhs`.
306	///
307	/// # Errors
308	///
309	/// [`Lazy`](Error::Lazy) if a prop belongs to another logic.
310	pub fn iff(&mut self, lhs: &Prop, rhs: &Prop) -> Result<Prop, Error> {
311		let forward = self.implies(lhs, rhs)?;
312		let backward = self.implies(rhs, lhs)?;
313		self.and(&forward, &backward)
314	}

316	/// `∃vars. prop`. Variables of `vars` that are not free in `prop` are skipped.
317	///
318	/// A prop without free variables is decided now.
319	///
320	/// # Errors
321	///
322	/// [`ForeignVar`](Error::ForeignVar), [`RepeatedVar`](Error::RepeatedVar), or [`Lazy`](Error::Lazy) if
323	/// `prop` belongs to another logic.
324	pub fn exists(&mut self, prop: &Prop, vars: &[Var]) -> Result<Prop, Error> {
325		self.check_vars(vars)?;
326		let kept: Vec<usize> = (0..prop.vars.len())
327			.filter(|&tape| !vars.contains(&prop.vars[tape]))
328			.collect();
329		if kept.len() == prop.vars.len() {
330			return Ok(prop.clone());
331		}
332
333		if kept.is_empty() {
334			let holds = !self.formula.is_empty(prop.term)?;
335			return self.constant(holds);
336		}
337
338		let projected = self.formula.project(prop.term, &kept)?;
339		// The dropped tapes may have needed longer words than the kept ones: the kept words are then in the
340		// relation only with padding.
341		let pad = match self.padding {
342			Padding::Zero => Pad::Step(vec![ZERO; kept.len()].into()),
343			Padding::Sign => Pad::Last,
344		};
345		let term = self.formula.pad_closure(projected, pad)?;
346
347		Ok(Prop {
348			term,
349			vars: kept.iter().map(|&tape| prop.vars[tape]).collect(),
350		})
351	}

353	/// `∀vars. prop`, as `¬∃vars. ¬prop`.
354	///
355	/// # Errors
356	///
357	/// As [`exists`](Logic::exists).
358	pub fn forall(&mut self, prop: &Prop, vars: &[Var]) -> Result<Prop, Error> {
359		let not_prop = self.not(prop)?;
360		let exists = self.exists(&not_prop, vars)?;
361		self.not(&exists)
362	}

364	/// An assignment that satisfies `prop`, or `None` if it is unsatisfiable.
365	///
366	/// # Errors
367	///
368	/// [`Lazy`](Error::Lazy): a limit of `options`, or `prop` belongs to another logic.
369	pub fn model(&self, prop: &Prop, options: &Options) -> Result<Option<Model>, Error> {
370		let outcome = self.formula.check(prop.term, options)?;
371
372		Ok(outcome
373			.witness
374			.map(|witness| Model::new(prop.vars.clone(), &witness)))
375	}

377	/// An assignment that falsifies `prop`, or `None` if it is valid.
378	///
379	/// # Errors
380	///
381	/// As [`model`](Logic::model).
382	pub fn counterexample(
383		&mut self,
384		prop: &Prop,
385		options: &Options,
386	) -> Result<Option<Model>, Error> {
387		let not_prop = self.not(prop)?;
388		self.model(&not_prop, options)
389	}

391	fn check_vars(&self, vars: &[Var]) -> Result<(), Error> {
392		for (position, &var) in vars.iter().enumerate() {
393			if var.index() >= self.names.len() {
394				return Err(Error::ForeignVar { var });
395			}
396
397			if vars[..position].contains(&var) {
398				return Err(Error::RepeatedVar { var });
399			}
400		}
401
402		Ok(())
403	}

405	/// `prop` over the free variables of `prop` and `others`: the variables it lacks take any value.
406	fn cylindrify(&mut self, prop: &Prop, others: &[Var]) -> Result<Prop, Error> {
407		let missing: Box<[Var]> = others
408			.iter()
409			.filter(|var| prop.vars.binary_search(var).is_err())
410			.copied()
411			.collect();
412		if missing.is_empty() {
413			return Ok(prop.clone());
414		}
415
416		let any = Prop {
417			term: self.encodings(missing.len())?,
418			vars: missing,
419		};
420		self.and(prop, &any)
421	}

423	/// The words that encode something, over `tapes` tapes.
424	fn encodings(&mut self, tapes: usize) -> Result<Term, Error> {
425		if let Some(&term) = self.encodings.get(&tapes) {
426			return Ok(term);
427		}
428
429		let encodings = Encodings {
430			tapes,
431			nonempty: self.padding == Padding::Sign,
432		};
433		let term = self.formula.leaf_over(&encodings, [BITS])?;
434		self.encodings.insert(tapes, term);
435
436		Ok(term)
437	}
