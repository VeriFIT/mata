1//! The bridge to Mata's experimental transducers (`mata::nft`), over `cxx/nft.hh`. Every transducer it hands
2//! out is well formed: levels within range, initial and final states on level 0.
3
4/// The most levels a transducer may have (the shim unwinds jumps into one state per level).
5pub const MAX_LEVELS: usize = 1024;
6
7/// The library's default begin marker of a regex replace (`strings::replace::BEGIN_MARKER`).
8pub const BEGIN_MARKER: u32 = u32::MAX - 100;
9
10/// The library's default end marker of a literal replace (`strings::replace::END_MARKER`).
11pub const END_MARKER: u32 = u32::MAX - 99;
12
13/// The NFT bridge.
14#[cxx::bridge(namespace = "mata_sys")]
15pub mod ffi {

48		fn nft_adopt(nft: Pin<&mut Nft>) -> Result<()>;

52		/// Parses every section of a `.mata` text, which must all be well-formed explicit NFTs
53		/// (`@NFT-explicit`, with `%Levels` and `%LevelsNum`). Symbol names work as in
54		/// `crate::ffi::parse_mata`, with one alphabet for all levels.
55		fn nft_parse_mata(

59			named_symbols: &[u32],
60			next_symbol: u32,
61			new_names: &mut Vec<String>,
62			new_symbols: &mut Vec<u32>,
63		) -> Result<UniquePtr<ParsedNfts>>;
64		/// The number of parsed transducers.
65		fn nft_parsed_len(parsed: &ParsedNfts) -> usize;
66		/// Moves out the transducer at `index`.
67		fn nft_parsed_take(parsed: Pin<&mut ParsedNfts>, index: usize) -> Result<UniquePtr<Nft>>;

70
71		/// The structural view of a transducer (states, initial and final states, level-by-level edges).
72		fn nft_automaton(nft: &Nft) -> &Automaton;
73		/// The number of levels (tapes).
74		fn nft_num_levels(nft: &Nft) -> usize;
75		/// The level of `state`. Throws if `state` does not exist.
76		fn nft_level(nft: &Nft, state: u32) -> Result<usize>;
77		/// Adds a state with `level` and returns it. Throws if `level` does not exist.
78		fn nft_add_state(nft: Pin<&mut Nft>, level: usize) -> Result<u32>;
79		/// Makes `state` initial. Throws unless `state` exists and has level 0.
80		fn nft_add_initial(nft: Pin<&mut Nft>, state: u32) -> Result<()>;
81		/// Makes `state` non-initial; false if it was not initial.
82		fn nft_remove_initial(nft: Pin<&mut Nft>, state: u32) -> bool;
83		/// Makes `state` final. Throws unless `state` exists and has level 0.
84		fn nft_add_final(nft: Pin<&mut Nft>, state: u32) -> Result<()>;
85		/// Makes `state` non-final; false if it was not final.
86		fn nft_remove_final(nft: Pin<&mut Nft>, state: u32) -> bool;
87		/// Adds the transition reading `symbols[i]` on level `i`, between two level-0 states, through new
88		/// inner states. Throws unless there is one symbol per level and both states have level 0.
89		fn nft_add_transition(
90			nft: Pin<&mut Nft>,
91			source: u32,
92			symbols: &[u32],
93			target: u32,
94		) -> Result<()>;
95		/// Adds a path from `source` to `target` (both level 0) that reads one word per level: the word
96		/// for level `i` is the next `lengths[i]` symbols of `symbols`. Epsilon in the words is dropped.
97		fn nft_insert_words(
98			nft: Pin<&mut Nft>,
99			source: u32,
100			symbols: &[u32],
101			lengths: &[usize],
102			target: u32,
103		) -> Result<()>;
104		/// Whether the tuple of words (encoded as for `nft_insert_words`) is in the relation.
105		fn nft_accepts(nft: &Nft, symbols: &[u32], lengths: &[usize]) -> Result<bool>;

108
109		/// Removes the states that are not both reachable and co-reachable, renumbering the rest.
110		fn nft_trim(nft: Pin<&mut Nft>) -> Result<()>;
111		/// The union of the relations. Throws if the numbers of levels differ.
112		fn nft_union(lhs: &Nft, rhs: &Nft) -> Result<UniquePtr<Nft>>;
113		/// The intersection of the relations. Throws if the numbers of levels differ.
114		fn nft_intersection(lhs: &Nft, rhs: &Nft) -> Result<UniquePtr<Nft>>;
115		/// The concatenation of the relations. Throws if the numbers of levels differ.
116		fn nft_concatenate(lhs: &Nft, rhs: &Nft) -> Result<UniquePtr<Nft>>;
117		/// Composes, synchronizing level `lhs_sync_level` of `lhs` with `rhs_sync_level` of `rhs`. Throws

119		fn nft_compose(

123			rhs_sync_level: usize,
124			project_out_sync_level: bool,
125		) -> Result<UniquePtr<Nft>>;
126		/// Restricts level `level` to the language of `nfa`, optionally projecting that level out.

133		/// Removes the given levels (strictly increasing); at least one level must remain.
134		fn nft_project_out(nft: &Nft, levels: &[usize]) -> Result<UniquePtr<Nft>>;
135		/// Keeps only the given levels (strictly increasing, nonempty).
136		fn nft_project_to(nft: &Nft, levels: &[usize]) -> Result<UniquePtr<Nft>>;
137		/// Reverses the order of the levels (the inverse relation for two levels).
138		fn nft_invert(nft: &Nft) -> Result<UniquePtr<Nft>>;
139		/// Mata's `determinize`: the subset construction over the edges (epsilon counts as a symbol).

142		fn nft_remove_epsilon(nft: &Nft) -> Result<UniquePtr<Nft>>;
143		/// Every word reversed on its level (Mata's `revert`, then `invert_levels`).
144		fn nft_reverse(nft: &Nft) -> Result<UniquePtr<Nft>>;

148		fn nft_to_nfa(

156		/// Whether the relation of `smaller` is included in that of `bigger`.
157		fn nft_is_included(smaller: &Nft, bigger: &Nft) -> Result<bool>;

164		// Strings (`mata::applications::strings`). `alphabet` lists the symbols of the library's

277		/// Mata's `insert_level`. Throws if the result would have more than `MAX_LEVELS` levels.

279		/// Mata's `insert_levels`: `true` in `mask` is a new level. Throws unless `mask` has one `false`

282		/// Mata's `reduce` (simulation).
283		fn nft_reduce(nft: &Nft) -> Result<UniquePtr<Nft>>;

312impl std::fmt::Debug for ffi::Nft {
313	fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
314		f.debug_struct("Nft")
315			.field("num_levels", &ffi::nft_num_levels(self))
316			.field("automaton", ffi::nft_automaton(self))
317			.finish()
318	}
319}
320
321impl std::fmt::Debug for ffi::ParsedNfts {
322	fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
323		f.debug_struct("ParsedNfts")
324			.field("len", &ffi::nft_parsed_len(self))
325			.finish()
326	}
327}
