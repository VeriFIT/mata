1//! Native build of [Mata](https://github.com/VeriFIT/mata), the automata library of the VeriFIT group,
2//! and a [cxx] bridge to its NFA API ([`ffi`]) and its experimental transducer API ([`nft::ffi`]).
3//!
4//! The `mata` crate is the safe API on top. The bridge is public so that other crates can add their own C++
5//! code on the same `libmata`: build scripts get the headers as `DEP_MATA_INCLUDE`, and a bridge names the
6//! types as `type Nfa = mata_sys::ffi::Nfa;`.
7//!
8//! # Conventions of the bridge
9//!
10//! - States and symbols are `u32`; counts and indices are `usize`.
11//! - Functions that can throw return `Result<_, cxx::Exception>`. The `_within` functions return null when a
12//!   limit stopped them.
13//! - Every function is safe to call: the shim checks the preconditions that Mata checks only in debug builds.
14//! - Automata never carry an alphabet; alphabets are passed per call.
15//! - Language operations remove epsilon transitions first (on a copy), as Mata reads epsilon as a letter there.
16//!
17//! # Build
18//!
19//! The build script builds the Mata sources packaged with the crate (or, in the Mata repository, those of the
20//! repository), or uses an installed Mata.
21//! The crate README lists the requirements and options.
22

37	/// Complementation algorithm.
38	#[repr(u8)]
39	#[derive(Debug)]
40	enum Complementation {
41		/// Determinize, complete, swap final and non-final states.
42		Classical,

44		Brzozowski,
45	}

50	enum Determinization {

52		FourRussians,
53		/// The classical subset construction.
54		Classical,
55	}

57	/// Minimization algorithm.
58	#[repr(u8)]
59	#[derive(Debug)]
60	enum Minimization {
61		/// Determinize the reversal twice.
62		Brzozowski,
63		/// Partition refinement on a trimmed DFA.
64		Hopcroft,

488		/// Mata's `parse_mf` followed by `IntermediateAut::parse_from_mf`: every section of a `.mata`
489		/// text, bit-vector sections left as they are.
490		fn intermediate_parse_mata(text: &str) -> Result<UniquePtr<IntermediateAuts>>;
491		/// The number of intermediate automata.
492		fn intermediate_auts_len(automata: &IntermediateAuts) -> usize;
493		/// Moves out the intermediate automaton at `index`; taking one twice is an error.
494		fn intermediate_auts_take(
495			automata: Pin<&mut IntermediateAuts>,
496			index: usize,
497		) -> Result<UniquePtr<IntermediateAut>>;
498		/// An empty list of borrowed intermediate automata.
499		fn intermediate_refs_new<'a>() -> UniquePtr<IntermediateRefs<'a>>;
500		/// Appends `automaton`.
501		fn intermediate_refs_push<'a>(
502			refs: Pin<&mut IntermediateRefs<'a>>,
503			automaton: &'a IntermediateAut,
504		);
505		/// `Mintermization::mintermize` of the automata of `refs` together, so that they share their
506		/// minterms. Throws unless every one is an NFA or AFA with a bit-vector alphabet.
507		fn intermediate_mintermize(refs: &IntermediateRefs) -> Result<UniquePtr<IntermediateAuts>>;
508		/// `nfa::builder::construct` of an explicit NFA section, with symbol names as in
509		/// [`parse_mata`].
510		fn intermediate_construct_nfa(
511			automaton: &IntermediateAut,
512			names: &[String],
513			named_symbols: &[u32],
514			next_symbol: u32,
515			new_names: &mut Vec<String>,
516			new_symbols: &mut Vec<u32>,
517		) -> Result<UniquePtr<Nfa>>;
518		/// Mata's `operator<<` of an intermediate automaton.
519		fn intermediate_print(automaton: &IntermediateAut) -> Result<String>;
520	}
