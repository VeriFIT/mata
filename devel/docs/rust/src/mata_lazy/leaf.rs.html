40	pub(crate) fn push(&mut self, source: StateId, tuple: &[Symbol], target: StateId) {
41		debug_assert_eq!(tuple.len(), self.arity);
42		self.edges.push((source, target));
43		self.letters.extend_from_slice(tuple);
44	}

46	/// Pushes a move unless `pushed`, the edges of `source` so far, has it.
47	fn push_new(
48		&mut self,
49		pushed: &mut HashTable<usize>,
50		source: StateId,
51		tuple: &[Symbol],
52		target: StateId,
53	) {
54		let hash = FxBuildHasher.hash_one((tuple, target));
55		let same = |&edge: &usize| self.edges[edge].1 == target && self.tuple(edge) == tuple;
56		if pushed.find(hash, same).is_some() {
57			return;
58		}
59
60		let rehash = |&edge: &usize| FxBuildHasher.hash_one((self.tuple(edge), self.edges[edge].1));
61		pushed.insert_unique(hash, self.edges.len(), rehash);
62		self.push(source, tuple, target);
63	}

81	/// The tuple of every move, numbered in the order of first appearance, and the distinct tuples.
82	/// One linear pass, with a check per move, so cancellation never waits for a sort.
83	fn labels(
84		&self,
85		stop: &mut dyn FnMut() -> bool,
86	) -> Result<(Vec<usize>, Vec<&[Symbol]>), Stopped> {
87		let mut labels = Vec::with_capacity(self.edges.len());
88		let mut tuples = Vec::new();
89		let mut positions: FxHashMap<&[Symbol], usize> = FxHashMap::default();
90		for edge in 0..self.edges.len() {
91			if stop() {
92				return Err(Stopped);
93			}
94			let label = *positions.entry(self.tuple(edge)).or_insert_with_key(|_| {
95				tuples.push(self.tuple(edge));
96				tuples.len() - 1
97			});
98			labels.push(label);
99		}
100		Ok((labels, tuples))
101	}

103	/// Moves sorted by their dense `labels`: a counting sort, which checks `stop` per label and per move.
104	fn moves_by_label(
105		&self,
106		labels: &[usize],
107		letters: usize,
108		stop: &mut dyn FnMut() -> bool,
109	) -> Result<Vec<(usize, usize)>, Stopped> {
110		let mut starts = vec![0; letters + 1];
111		for &letter in labels {
112			if stop() {
113				return Err(Stopped);
114			}
115			starts[letter + 1] += 1;
116		}
117		for letter in 0..letters {
118			if stop() {
119				return Err(Stopped);
120			}
121			starts[letter + 1] += starts[letter];
122		}
123		let mut moves = Vec::new();
124		moves.resize(self.edges.len(), (0, 0));
125		for edge in 0..self.edges.len() {
126			if stop() {
127				return Err(Stopped);
128			}
129			moves[starts[labels[edge]]] = (labels[edge], edge);
130			starts[labels[edge]] += 1;
131		}
132		Ok(moves)
133	}

135	/// The quotient by the coarsest forward bisimulation, which keeps the language on every tape: the partition
136	/// refinement of Paige and Tarjan, by letter. Blocks are kept stable with respect to a coarser partition of
137	/// compound sets, counting for every state and letter its moves into each compound set; a compound set of
138	/// several blocks loses its smaller one, and only the moves into that block are visited, so every move is
139	/// visited O(log n) times. [`Stopped`] when `stop` says so (it is asked once per block taken out, and per move
140	/// visited).
141	pub(crate) fn quotient(&self, stop: &mut dyn FnMut() -> bool) -> Result<Raw, Stopped> {
142		let num_states = self.finals.len();
143		let num_edges = self.edges.len();
144		let (label, _) = self.labels(stop)?;
145		let tail = |edge: usize| self.edges[edge].0.index();
146		let mut incoming: Vec<Vec<usize>> = vec![Vec::new(); num_states];
147		for (edge, &(_, target)) in self.edges.iter().enumerate() {
148			if stop() {
149				return Err(Stopped);
150			}
151			incoming[target.index()].push(edge);
152		}
153		let mut blocks = self.blocks_by_letters(&label, stop)?;
154
155		// The moves of a state with a letter into a compound set, counted in one cell.
156		let mut counts: Vec<usize> = Vec::new();
157		let mut cells: FxHashMap<(usize, usize), usize> = FxHashMap::default();
158		let mut edge_cell: Vec<usize> = (0..num_edges)
159			.map(|edge| {
160				let cell = *cells.entry((tail(edge), label[edge])).or_insert_with(|| {
161					counts.push(0);
162					counts.len() - 1
163				});
164				counts[cell] += 1;
165				cell
166			})
167			.collect();
168
169		let mut compound_of: Vec<usize> = vec![0; num_states.max(1)];
170		let mut compounds: Vec<Vec<usize>> = vec![(0..blocks.sets).collect()];
171		let mut splittable: Vec<usize> = if blocks.sets > 1 { vec![0] } else { Vec::new() };
172		let size = |blocks: &Partition, block: usize| blocks.past[block] - blocks.first[block];
173		while let Some(&compound) = splittable.last() {
174			if stop() {
175				return Err(Stopped);
176			}
177			let last = compounds[compound].len() - 1;
178			let (one, other) = (compounds[compound][last], compounds[compound][last - 1]);
179			let splitter =
180				compounds[compound].swap_remove(if size(&blocks, one) <= size(&blocks, other) {
181					last
182				} else {
183					last - 1
184				});
185			if compounds[compound].len() < 2 {
186				splittable.pop();
187			}
188			compound_of[splitter] = compounds.len();
189			compounds.push(vec![splitter]);
190
191			// Moves into the splitter get new cells; per state and letter, the old cell then counts its moves
192			// into the rest of the compound set.
193			// A fresh map: clearing the first one, with a slot per move, would cost that much per splitter.
194			let mut cells: FxHashMap<(usize, usize), usize> = FxHashMap::default();
195			let mut entries: Vec<(usize, usize, usize)> = Vec::new();
196			for position in blocks.first[splitter]..blocks.past[splitter] {
197				for &edge in &incoming[blocks.elements[position]] {
198					if stop() {
199						return Err(Stopped);
200					}
201					let old = edge_cell[edge];
202					let cell = *cells.entry((tail(edge), label[edge])).or_insert_with(|| {
203						entries.push((tail(edge), label[edge], old));
204						counts.push(0);
205						counts.len() - 1
206					});
207					counts[old] -= 1;
208					counts[cell] += 1;
209					edge_cell[edge] = cell;
210				}
211			}
212
213			// A state with moves into the splitter is told apart by the letters of those moves and whether it
214			// has a move with the same letter into the rest; the states without such moves keep their block.
215			let by_state = canonical_cells(entries, &counts, stop)?;
216			let mut touched: FxHashMap<usize, KeyGroups> = FxHashMap::default();
217			for (state, key) in by_state {
218				touched
219					.entry(blocks.set[state])
220					.or_default()
221					.entry(key)
222					.or_default()
223					.push(state);
224			}
225			for (block, groups) in touched {
226				let all = groups.values().map(Vec::len).sum::<usize>() == size(&blocks, block);
227				for group in groups.into_values().skip(usize::from(all)) {
228					for state in group {
229						blocks.mark(state);
230					}
231					blocks.split();
232				}
233			}
234			for (new, parent) in blocks.created.drain(..) {
235				let compound = compound_of[parent];
236				compound_of[new] = compound;
237				compounds[compound].push(new);
238				if compounds[compound].len() == 2 {
239					splittable.push(compound);
240				}
241			}
242		}
243
244		Ok(self.merged(&blocks))
245	}

247	/// The states split by finality and the letters of their moves, so that the blocks are stable with respect to
248	/// the one compound set of all states.
249	fn blocks_by_letters(
250		&self,
251		label: &[usize],
252		stop: &mut dyn FnMut() -> bool,
253	) -> Result<Partition, Stopped> {
254		let num_states = self.finals.len();
255		let mut letters: Vec<Vec<usize>> = vec![Vec::new(); num_states];
256		for (edge, &(source, _)) in self.edges.iter().enumerate() {
257			if stop() {
258				return Err(Stopped);
259			}
260			letters[source.index()].push(label[edge]);
261		}
262		let mut blocks = Partition::new(num_states);
263		let mut groups: FxHashMap<(bool, LetterSet), Vec<usize>> = FxHashMap::default();
264		for (state, state_letters) in letters.into_iter().enumerate() {
265			if stop() {
266				return Err(Stopped);
267			}
268			let state_letters = LetterSet::new(state_letters);
269			groups
270				.entry((self.finals[state], state_letters))
271				.or_default()
272				.push(state);
273		}
274		for group in groups.into_values().skip(1) {
275			for state in group {
276				blocks.mark(state);
277			}
278			blocks.split();
279		}
280		blocks.created.clear();
281		Ok(blocks)
282	}

284	/// `self` with the states of every block of `blocks` merged.
285	fn merged(&self, blocks: &Partition) -> Raw {
286		let mut quotient = Raw::new(self.arity);
287		quotient.finals = vec![false; blocks.sets];
288		for state in (0..self.finals.len()).filter(|&state| self.finals[state]) {
289			quotient.finals[blocks.set[state]] = true;
290		}
291		quotient.initial = self
292			.initial
293			.iter()
294			.map(|state| StateId::new(blocks.set[state.index()]))
295			.collect();
296		quotient.initial.sort_unstable();
297		quotient.initial.dedup();
298		let mut moves: Vec<(usize, &[Symbol], usize)> = (0..self.edges.len())
299			.map(|edge| {
300				let (source, target) = self.edges[edge];
301				(
302					blocks.set[source.index()],
303					self.tuple(edge),
304					blocks.set[target.index()],
305				)
306			})
307			.collect();
308		moves.sort_unstable();
309		moves.dedup();
310		for (source, tuple, target) in moves {
311			quotient.push(StateId::new(source), tuple, StateId::new(target));
312		}
313		quotient
314	}

316	/// The subset construction over the tuples of `self`: `Ok(None)` once it has more than `max_states` states, its
317	/// subsets more than `max_members` members in all or it more than `max_moves` moves, [`Stopped`] when `stop`
318	/// says so (it is asked once per subset and per move).
319	pub(crate) fn determinize(
320		&self,
321		max_states: usize,
322		max_members: usize,
323		max_moves: usize,
324		stop: &mut dyn FnMut() -> bool,
325	) -> Result<Option<Raw>, Stopped> {
326		let mut outgoing: Vec<Vec<usize>> = vec![Vec::new(); self.finals.len()];
327		for (edge, &(source, _)) in self.edges.iter().enumerate() {
328			outgoing[source.index()].push(edge);
329		}
330
331		let mut start = self.initial.clone();
332		start.sort_unstable();
333		start.dedup();
334		let mut members = start.len();
335		let mut subsets: IndexSet<Vec<StateId>, FxBuildHasher> = IndexSet::default();
336		subsets.insert(start);
337		let mut dfa = Raw::new(self.arity);
338		dfa.initial.push(StateId::new(0));
339		let mut next = 0;
340		while next < subsets.len() {
341			if stop() {
342				return Err(Stopped);
343			}
344
345			let mut moves: FxHashMap<&[Symbol], Vec<StateId>> = FxHashMap::default();
346			for state in &subsets[next] {
347				for &edge in &outgoing[state.index()] {
348					if stop() {
349						return Err(Stopped);
350					}
351					moves
352						.entry(self.tuple(edge))
353						.or_default()
354						.push(self.edges[edge].1);
355				}
356			}
357			dfa.finals
358				.push(subsets[next].iter().any(|state| self.finals[state.index()]));
359			// Grouped without a sort. A subset is canonical: its members in state order, each once.
360			for (tuple, mut targets) in moves {
361				if stop() {
362					return Err(Stopped);
363				}
364				if dfa.edges.len() == max_moves {
365					return Ok(None);
366				}
367				// Only the target subset is sorted to get one canonical key, not all the moves.
368				targets.sort_unstable();
369				targets.dedup();
370				let target = targets;
371				let size = target.len();
372				let (index, new) = subsets.insert_full(target);
373				// Checked as each subset is found, so a state with many new targets does not emit them all first.
374				if new {
375					members += size;
376					if subsets.len() > max_states || members > max_members {
377						return Ok(None);
378					}
379				}
380				dfa.push(StateId::new(next), tuple, StateId::new(index));
381			}
382			next += 1;
383		}
384
385		Ok(Some(dfa))
386	}

388	/// The minimal automaton of `self`, which must be trimmed and deterministic over its tuples: the partition
389	/// refinement of Valmari and Lehtinen for partial DFAs (STACS 2008), in O(m log n) for m moves and n states.
390	/// [`Stopped`] when `stop` says so (it is asked once per class of moves, per move of it and per new block).
391	pub(crate) fn minimize(&self, stop: &mut dyn FnMut() -> bool) -> Result<Raw, Stopped> {
392		let num_states = self.finals.len();
393		let num_edges = self.edges.len();
394		let mut minimal = Raw::new(self.arity);
395		if num_states == 0 {
396			return Ok(minimal);
397		}
398		debug_assert_eq!(self.initial.len(), 1, "a deterministic automaton");
399
400		// Letters are the distinct tuples, numbered in their order; one linear pass per move.
401		let (label, tuples) = self.labels(stop)?;
402		let tail = |edge: usize| self.edges[edge].0.index();
403		let head = |edge: usize| self.edges[edge].1.index();
404
405		// The moves into each state.
406		let mut incoming_start = vec![0; num_states + 1];
407		for edge in 0..num_edges {
408			incoming_start[head(edge) + 1] += 1;
409		}
410		for state in 0..num_states {
411			incoming_start[state + 1] += incoming_start[state];
412		}
413		let mut incoming = vec![0; num_edges];
414		let mut fill = incoming_start.clone();
415		for edge in 0..num_edges {
416			incoming[fill[head(edge)]] = edge;
417			fill[head(edge)] += 1;
418		}
419
420		// Blocks of states, first split by finality.
421		let mut blocks = Partition::new(num_states);
422		for state in (0..num_states).filter(|&state| self.finals[state]) {
423			blocks.mark(state);
424		}
425		blocks.split();
426
427		// Cords: classes of moves, first by label; a linear counting sort.
428		let by_label = self.moves_by_label(&label, tuples.len(), stop)?;
429		let mut cords = Partition::new(num_edges);
430		cords.sets = 0;
431		for (position, &(letter, edge)) in by_label.iter().enumerate() {
432			if position == 0 || by_label[position - 1].0 != letter {
433				cords.first[cords.sets] = position;
434				cords.sets += 1;
435			}
436			cords.past[cords.sets - 1] = position + 1;
437			cords.elements[position] = edge;
438			cords.location[edge] = position;
439			cords.set[edge] = cords.sets - 1;
440		}
441
442		let mut block = 1;
443		let mut cord = 0;
444		while cord < cords.sets {
445			if stop() {
446				return Err(Stopped);
447			}
448			for &edge in &cords.elements[cords.first[cord]..cords.past[cord]] {
449				if stop() {
450					return Err(Stopped);
451				}
452				blocks.mark(tail(edge));
453			}
454			blocks.split();
455			cord += 1;
456
457			while block < blocks.sets {
458				if stop() {
459					return Err(Stopped);
460				}
461				for &state in &blocks.elements[blocks.first[block]..blocks.past[block]] {
462					for &edge in &incoming[incoming_start[state]..incoming_start[state + 1]] {
463						cords.mark(edge);
464					}
465				}
466				cords.split();
467				block += 1;
468			}
469		}
470
471		minimal.finals = vec![false; blocks.sets];
472		for state in (0..num_states).filter(|&state| self.finals[state]) {
473			minimal.finals[blocks.set[state]] = true;
474		}
475		minimal
476			.initial
477			.push(StateId::new(blocks.set[self.initial[0].index()]));
478		let mut moves: Vec<(usize, usize, usize)> = (0..num_edges)
479			.map(|edge| (blocks.set[tail(edge)], label[edge], blocks.set[head(edge)]))
480			.collect();
481		moves.sort_unstable();
482		moves.dedup();
483		for (source, letter, target) in moves {
484			minimal.push(StateId::new(source), tuples[letter], StateId::new(target));
485		}
486
487		Ok(minimal)
488	}

490	/// The useful part of `self`: states reachable from an initial one that reach a final one. [`Stopped`] when
491	/// `stop` says so (it is asked per state of the useful-state search, and per move).
492	pub(crate) fn trimmed(&self, stop: &mut dyn FnMut() -> bool) -> Result<Raw, Stopped> {
493		let renumbered = useful_states_until(self, stop)?;
494		let mut trimmed = Raw::new(self.arity);
495		trimmed.finals = vec![false; renumbered.iter().flatten().count()];
496		for (&new, &is_final) in renumbered.iter().zip(&self.finals) {
497			if let Some(new) = new {
498				trimmed.finals[new.index()] = is_final;
499			}
500		}
501		trimmed.initial = self
502			.initial
503			.iter()
504			.filter_map(|&state| renumbered[state.index()])
505			.collect();
506		for edge in 0..self.edges.len() {
507			if stop() {
508				return Err(Stopped);
509			}
510			let (source, target) = self.edges[edge];
511			if let (Some(source), Some(target)) =
512				(renumbered[source.index()], renumbered[target.index()])
513			{
514				trimmed.push(source, self.tuple(edge), target);
515			}
516		}
517
518		Ok(trimmed)
519	}
520}

526/// The states of one block with moves into a splitter, by their letters and whether they also reach the rest.
527type KeyGroups = FxHashMap<Vec<(usize, bool)>, Vec<usize>>;
528
529/// The `(letter, whether a move remains)` cells for every state, sorted within each state. The grouping is the same
530/// as sorting every entry together, but each sort covers only one state's cells and `stop` is asked per entry and state.
531fn canonical_cells(
532	entries: Vec<(usize, usize, usize)>,
533	counts: &[usize],
534	stop: &mut dyn FnMut() -> bool,
535) -> Result<FxHashMap<usize, Vec<(usize, bool)>>, Stopped> {
536	let mut by_state: FxHashMap<usize, Vec<(usize, bool)>> = FxHashMap::default();
537	for (state, letter, old) in entries {
538		if stop() {
539			return Err(Stopped);
540		}
541		by_state
542			.entry(state)
543			.or_default()
544			.push((letter, counts[old] > 0));
545	}
546	for key in by_state.values_mut() {
547		if stop() {
548			return Err(Stopped);
549		}
550		key.sort_unstable();
551	}
552	Ok(by_state)
553}

555/// A set of move labels, for grouping states by their outgoing letters without sorting them.
556#[derive(PartialEq, Eq)]
557struct LetterSet {
558	letters: FxHashSet<usize>,
559	/// The commutative hash: set equality is independent of insertion order.
560	fingerprint: u64,
561}
562
563impl LetterSet {
564	fn new(letters: Vec<usize>) -> LetterSet {
565		let mut fingerprint = 0;
566		let mut set = FxHashSet::default();
567		for letter in letters {
568			if set.insert(letter) {
569				let mut hash = FxHasher::default();
570				letter.hash(&mut hash);
571				fingerprint ^= hash.finish();
572			}
573		}
574		LetterSet {
575			letters: set,
576			fingerprint,
577		}
578	}
579}
580
581impl Hash for LetterSet {
582	fn hash<H: Hasher>(&self, state: &mut H) {
583		state.write_u64(self.fingerprint);
584	}
585}

587/// A partition of `0..n` whose sets are contiguous ranges of `elements`, refined by marking elements and splitting
588/// every touched set into its marked and unmarked part; the smaller part gets the new number.
589struct Partition {
590	sets: usize,
591	elements: Vec<usize>,
592	location: Vec<usize>,
593	set: Vec<usize>,
594	first: Vec<usize>,
595	past: Vec<usize>,
596	marked: Vec<usize>,
597	touched: Vec<usize>,
598	/// The sets that splits created, in order, each with the set it was split from.
599	created: Vec<(usize, usize)>,
600}
601
602impl Partition {
603	/// One set of all `n` elements, or none for `n == 0`.
604	fn new(n: usize) -> Partition {
605		let mut partition = Partition {
606			sets: usize::from(n > 0),
607			elements: (0..n).collect(),
608			location: (0..n).collect(),
609			set: vec![0; n],
610			first: vec![0; n + 1],
611			past: vec![0; n + 1],
612			marked: vec![0; n + 1],
613			touched: Vec::new(),
614			created: Vec::new(),
615		};
616		partition.past[0] = n;
617		partition
618	}
619
620	fn mark(&mut self, element: usize) {
621		let set = self.set[element];
622		let position = self.location[element];
623		let boundary = self.first[set] + self.marked[set];
624		if position < boundary {
625			return; // Marked already.
626		}
627		self.elements[position] = self.elements[boundary];
628		self.location[self.elements[position]] = position;
629		self.elements[boundary] = element;
630		self.location[element] = boundary;
631		if self.marked[set] == 0 {
632			self.touched.push(set);
633		}
634		self.marked[set] += 1;
635	}
636
637	fn split(&mut self) {
638		while let Some(set) = self.touched.pop() {
639			let boundary = self.first[set] + self.marked[set];
640			self.marked[set] = 0;
641			if boundary == self.past[set] {
642				continue;
643			}
644
645			let new = self.sets;
646			if boundary - self.first[set] <= self.past[set] - boundary {
647				self.first[new] = self.first[set];
648				self.past[new] = boundary;
649				self.first[set] = boundary;
650			} else {
651				self.past[new] = self.past[set];
652				self.first[new] = boundary;
653				self.past[set] = boundary;
654			}
655			for position in self.first[new]..self.past[new] {
656				self.set[self.elements[position]] = new;
657			}
658			self.created.push((new, set));
659			self.sets += 1;
660		}
661	}
662}

664/// How far the import of a source (a state that a non-silent move or the start reaches) is.
665#[derive(Clone, Copy)]
666enum Import {
667	Pending,
668	Walking,
669	/// Its finality and its moves, `raw.edges[start..end]`, with the silent moves removed.
670	Done {
671		is_final: bool,
672		start: usize,
673		end: usize,
674	},
675}
676
677/// The walk of the silent closure of a source.
678struct Walk<S> {
679	source: usize,
680	/// The states that silent moves reach from the source; also the worklist, expanded up to `next`.
681	closure: IndexSet<S, FxBuildHasher>,
682	next: usize,
683	is_final: bool,
684	/// The moves found so far; `pushed` holds them so that a move shared along the closure is pushed once.
685	moves: Raw,
686	pushed: HashTable<usize>,
687}
688
689impl<S: Copy + Eq + std::hash::Hash> Walk<S> {
690	/// What to do with the next member of the closure, given the sources and how far their imports are.
691	fn step(&self, states: &IndexSet<S, FxBuildHasher>, imports: &[Import]) -> Step {
692		let Some(state) = self.closure.get_index(self.next) else {
693			return Step::Finish;
694		};
695		if self.next == 0 {
696			return Step::Expand;
697		}
698
699		let Some(index) = states.get_index_of(state) else {
700			return Step::Expand;
701		};
702		match imports.get(index).copied().unwrap_or(Import::Pending) {
703			Import::Pending => Step::Descend(index),
704			// A silent cycle back to a source on the stack, whose closure is not complete yet.
705			Import::Walking => Step::Expand,
706			Import::Done {
707				is_final,
708				start,
709				end,
710			} => Step::Inherit {
711				is_final,
712				start,
713				end,
714			},
715		}
716	}
717}
718
719/// What the walk on top of the stack does with the next member of its closure.
720enum Step {
721	Expand,
722	/// Takes over the moves of a source that is done.
723	Inherit {
724		is_final: bool,
725		start: usize,
726		end: usize,
727	},
728	/// Walks this pending source first.
729	Descend(usize),
730	Finish,
731}

733/// A trimmed automaton over tuples of symbols with dense states and sorted moves.
734#[derive(Debug)]
735pub(crate) struct Leaf {
736	arity: usize,
737	initial: Vec<StateId>,
738	finals: Vec<bool>,
739	/// The moves of state `s` are `offsets[s]..offsets[s + 1]`.
740	offsets: Vec<usize>,
741	letters: Vec<Symbol>,
742	targets: Vec<StateId>,
743	/// The concrete symbols of every tape, the ones read before trimming and the declared ones.
744	pub(crate) tape_symbols: Vec<Vec<Symbol>>,
745	/// Whether a move reads the wildcard on each tape.
746	pub(crate) tape_has_any: Vec<bool>,
747	/// Whether a move reads padding on some tape.
748	pub(crate) has_padding: bool,
749	simulation: OnceLock<Simulation>,
750	/// Held while a query computes the simulation, so that others wait for it instead of computing it again.
751	computing: Mutex<()>,
752}
753
754impl Leaf {
755	/// Copies the reachable part of `automaton`, without silent moves; `declared[tape]` joins `tape`'s symbols.
756	///
757	/// A closure that silently reaches another source takes over that source's moves instead of walking past it, and
758	/// walks a pending source first, on a stack, so that a chain of `n` sources joined by silent moves calls
759	/// `for_each_move` `n` times rather than `n^2 / 2`.
760	pub(crate) fn import<A: Automaton>(
761		automaton: &A,
762		declared: Vec<Vec<Symbol>>,
763	) -> Result<Leaf, Error> {
764		let arity = automaton.arity();
765		let mut states: IndexSet<A::State, FxBuildHasher> = IndexSet::default();
766		let mut raw = Raw::new(arity);
767		for state in automaton.initial_states() {
768			raw.initial.push(StateId::new(states.insert_full(state).0));
769		}
770
771		let mut wrong_width = None;
772		let mut imports: Vec<Import> = Vec::new();
773		let mut walks: Vec<Walk<A::State>> = Vec::new();
774		// Finished walks, kept for their allocations.
775		let mut spare: Vec<Walk<A::State>> = Vec::new();
776		let mut first = 0;
777		while first < states.len() {
778			imports.resize(states.len(), Import::Pending);
779			if matches!(imports[first], Import::Pending) {
780				imports[first] = Import::Walking;
781				walks.push(start_walk(&mut spare, arity, first, states[first]));
782			}
783
784			first += 1;
785			while let Some(walk) = walks.last_mut() {
786				let step = walk.step(&states, &imports);
787
788				match step {
789					Step::Expand => {
790						let state = walk.closure[walk.next];
791						walk.next += 1;
792						walk.is_final |= automaton.is_final(state);
793						let source = StateId::new(walk.source);
794						automaton.for_each_move(state, |tuple, target| {
795							if tuple.len() != arity {
796								wrong_width.get_or_insert(tuple.len());
797							} else if arity > 0
798								&& tuple.iter().all(|&symbol| symbol == Symbol::EPSILON)
799							{
800								walk.closure.insert(target);
801							} else {
802								let target = StateId::new(states.insert_full(target).0);
803								walk.moves.push_new(&mut walk.pushed, source, tuple, target);
804							}
805						});
806
807						if let Some(found) = wrong_width {
808							return Err(Error::MoveWidth { arity, found });
809						}
810					}
811					Step::Inherit {
812						is_final,
813						start,
814						end,
815					} => {
816						walk.next += 1;
817						walk.is_final |= is_final;
818						let source = StateId::new(walk.source);
819						for edge in start..end {
820							walk.moves.push_new(
821								&mut walk.pushed,
822								source,
823								raw.tuple(edge),
824								raw.edges[edge].1,
825							);
826						}
827					}
828					Step::Descend(index) => {
829						imports.resize(states.len(), Import::Pending);
830						imports[index] = Import::Walking;
831						walks.push(start_walk(&mut spare, arity, index, states[index]));
832					}
833					Step::Finish => {
834						let mut walk = walks.pop().expect("the loop has a walk");
835						let (start, end) = raw.append(&mut walk.moves);
836						imports[walk.source] = Import::Done {
837							is_final: walk.is_final,
838							start,
839							end,
840						};
841						spare.push(walk);
842					}
843				}
844			}
845		}
846
847		raw.finals = imports
848			.iter()
849			.map(|import| match import {
850				Import::Done { is_final, .. } => *is_final,
851				Import::Pending | Import::Walking => unreachable!("every source is imported"),
852			})
853			.collect();
854
855		Ok(Leaf::build(&raw, declared))
856	}
857
858	/// Trims `raw` to the states that reach a final state, and sorts and deduplicates its moves.
859	pub(crate) fn build(raw: &Raw, declared: Vec<Vec<Symbol>>) -> Leaf {
860		let arity = raw.arity;
861		let mut tape_symbols = declared;
862		tape_symbols.resize(arity, Vec::new());
863		for (tape, symbols) in tape_symbols.iter_mut().enumerate() {
864			symbols.extend(column(&raw.letters, arity, tape).filter(|letter| letter.is_concrete()));
865			symbols.sort_unstable();
866			symbols.dedup();
867		}
868
869		let renumbered = useful_states(raw);
870		let num_states = renumbered.iter().flatten().count();
871		let mut initial: Vec<StateId> = raw
872			.initial
873			.iter()
874			.filter_map(|&state| renumbered[state.index()])
875			.collect();
876		initial.sort_unstable();
877		initial.dedup();
878
879		let mut finals = vec![false; num_states];
880		for (&new, &is_final) in renumbered.iter().zip(&raw.finals) {
881			if let Some(new) = new {
882				finals[new.index()] = is_final;
883			}
884		}
885
886		let mut edges: Vec<(StateId, usize, StateId)> = raw
887			.edges
888			.iter()
889			.enumerate()
890			.filter_map(|(edge, &(source, target))| {
891				Some((
892					renumbered[source.index()]?,
893					edge,
894					renumbered[target.index()]?,
895				))
896			})
897			.collect();
898		let key =
899			|&(source, edge, target): &(StateId, usize, StateId)| (source, raw.tuple(edge), target);
900		edges.sort_unstable_by_key(key);
901		edges.dedup_by_key(|edge| key(edge));
902
903		let mut offsets = vec![0; num_states + 1];
904		for &(source, _, _) in &edges {
905			offsets[source.index() + 1] += 1;
906		}
907
908		for state in 0..num_states {
909			offsets[state + 1] += offsets[state];
910		}
911
912		let mut letters = Vec::with_capacity(edges.len() * arity);
913		let mut targets = Vec::with_capacity(edges.len());
914		for &(_, edge, target) in &edges {
915			letters.extend_from_slice(raw.tuple(edge));
916			targets.push(target);
917		}
918
919		let tape_has_any = (0..arity)
920			.map(|tape| column(&letters, arity, tape).any(|letter| letter == Symbol::ANY))
921			.collect();
922		let has_padding = letters.contains(&Symbol::EPSILON);
923
924		Leaf {
925			arity,
926			initial,
927			finals,
928			offsets,
929			letters,
930			targets,
931			tape_symbols,
932			tape_has_any,
933			has_padding,
934			simulation: OnceLock::new(),
935			computing: Mutex::new(()),
936		}
937	}

959	/// Whether it has at most one initial state and no state has two moves on one tuple or a wildcard move, which
960	/// overlaps the others.
961	pub(crate) fn is_deterministic(&self) -> bool {
962		let tuple = |edge: usize| &self.letters[edge * self.arity..(edge + 1) * self.arity];
963
964		self.initial.len() <= 1
965			&& !self.letters.contains(&Symbol::ANY)
966			// The moves of a state are sorted by tuple, so equal tuples are adjacent.
967			&& (0..self.num_states()).all(|state| {
968				(self.offsets[state] + 1..self.offsets[state + 1])
969					.all(|edge| tuple(edge - 1) != tuple(edge))
970			})
971	}

973	/// The moves from `state`, sorted by tuple, then target.
974	pub(crate) fn moves(
975		&self,
976		state: StateId,
977	) -> impl ExactSizeIterator<Item = (&[Symbol], StateId)> {
978		let state = state.index();
979		let range = self.offsets[state]..self.offsets[state + 1];
980
981		range.map(|edge| {
982			let tuple = &self.letters[edge * self.arity..(edge + 1) * self.arity];
983
984			(tuple, self.targets[edge])
985		})
986	}

988	/// The simulation, computed on first use; [`TimedOut`] if computing or waiting passes `until` (nothing kept).
989	pub(crate) fn simulation(&self, until: &Until) -> Result<&Simulation, TimedOut> {
990		match self.simulation.get() {
991			Some(simulation) => Ok(simulation),
992			None => self.compute_simulation(until),
993		}
994	}
995
996	/// Computes the simulation or waits for the query computing it; apart, so that the lookup above inlines.
997	#[cold]
998	fn compute_simulation(&self, until: &Until) -> Result<&Simulation, TimedOut> {
999		let computing = if until.is_never() {
1000			self.computing.lock()
1001		} else {
1002			// Waits in slices, to see a cancellation that comes before the deadline.
1003			loop {
1004				let slice = Instant::now() + LOCK_SLICE;
1005				let wait = until.deadline.map_or(slice, |deadline| deadline.min(slice));
1006				if let Some(computing) = self.computing.try_lock_until(wait) {
1007					break computing;
1008				}
1009				if until.passed() {
1010					return Err(TimedOut);
1011				}
1012			}
1013		};
1014
1015		// The query that held the lock may have computed it.
1016		if let Some(simulation) = self.simulation.get() {
1017			return Ok(simulation);
1018		}
1019
1020		let computed = Simulation::compute(self, until)?;
1021		let simulation = self.simulation.get_or_init(|| computed);
1022		drop(computing);
1023
1024		Ok(simulation)
1025	}
1026
1027	/// The simulation, if a query computed it.
1028	pub(crate) fn computed_simulation(&self) -> Option<&Simulation> {
1029		self.simulation.get()
1030	}

1033/// A walk from `source`, whose state is `state`, reusing the allocations of a finished walk.
1034fn start_walk<S: Copy + Eq + std::hash::Hash>(
1035	spare: &mut Vec<Walk<S>>,
1036	arity: usize,
1037	source: usize,
1038	state: S,
1039) -> Walk<S> {
1040	let mut walk = spare.pop().unwrap_or_else(|| Walk {
1041		source,
1042		closure: IndexSet::default(),
1043		next: 0,
1044		is_final: false,
1045		moves: Raw::new(arity),
1046		pushed: HashTable::new(),
1047	});
1048	walk.source = source;
1049	walk.closure.clear();
1050	walk.closure.insert(state);
1051	walk.next = 0;
1052	walk.is_final = false;
1053	walk.pushed.clear();
1054	walk
1055}

1059	letters.chunks_exact(arity).map(move |tuple| tuple[tape])

1062/// The new number of every state that can reach a final state, in the old order.
1063fn useful_states(raw: &Raw) -> Vec<Option<StateId>> {
1064	useful_states_until(raw, &mut || false).expect("nothing stops")
1065}
1066
1067/// [`useful_states`], stopping once `stop` says so; asked per edge and per state visited.
1068fn useful_states_until(
1069	raw: &Raw,
1070	stop: &mut dyn FnMut() -> bool,
1071) -> Result<Vec<Option<StateId>>, Stopped> {
1072	let num_states = raw.finals.len();
1073	let mut predecessors = vec![Vec::new(); num_states];
1074	for &(source, target) in &raw.edges {
1075		if stop() {
1076			return Err(Stopped);
1077		}
1078		predecessors[target.index()].push(source);
1079	}
1080
1081	let mut useful = raw.finals.clone();
1082	let mut stack: Vec<StateId> = (0..num_states)
1083		.filter(|&state| useful[state])
1084		.map(StateId::new)
1085		.collect();
1086	while let Some(state) = stack.pop() {
1087		if stop() {
1088			return Err(Stopped);
1089		}
1090		for &source in &predecessors[state.index()] {
1091			if !useful[source.index()] {
1092				useful[source.index()] = true;
1093				stack.push(source);
1094			}
1095		}
1096	}
1097
1098	let mut renumbered = vec![None; num_states];
1099	for (new, old) in (0..num_states).filter(|&state| useful[state]).enumerate() {
1100		renumbered[old] = Some(StateId::new(new));
1101	}
1102
1103	Ok(renumbered)
