1//! Policies, which compute a [`Plan`] from the [`PlanView`] of a formula.
2
3use crate::{Annotation, NodeKind, Plan, PlanView};
4
5/// Computes the annotations of a query from the [view](crate::Formula::view) of its term.
6///
7/// Every plan keeps the answer of the query (see [`Plan`]), so a policy decides only what a query costs.
8pub trait Policy {
9	/// The plan for the query of `view.root`.
10	fn plan(&self, view: &PlanView) -> Plan;
11}
12
13/// Every operation lazy, even an [`explicit`](crate::Formula::explicit) node of the formula.
14///
15/// Without a plan, a query keeps the formula's explicit nodes; this policy is the purely lazy baseline.
16#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
17pub struct AllLazy;
18
19impl Policy for AllLazy {
20	fn plan(&self, view: &PlanView) -> Plan {
21		annotate_operations(view, Annotation::lazy())
22	}
23}
24
25/// Every operation built explicitly, without a budget: each node is explored completely, children first, as an
26/// eager library would. This is the baseline to compare the lazy engine with.
27#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
28pub struct AllEager;
29
30impl Policy for AllEager {
31	fn plan(&self, view: &PlanView) -> Plan {
32		annotate_operations(view, Annotation::eager())
33	}
34}

36/// Builds the largest subterms that are surely small explicitly, and keeps the rest lazy.
37///
38/// It bounds the macrostates of each term from the states of the leaves below it: a product multiplies the
39/// bounds of its operands, a union adds them, a complement raises 2 to the bound of its operand, and the other
40/// operations keep their operand's. A term other than the root whose bound is at most
41/// [`max_states`](Heuristic::max_states), used by the root or by a term over that bound, is built explicitly with
42/// that budget. The root stays lazy, so that the search stops at the first witness.
43///
44/// For `(A ∩ B) ∩ C` with small leaves `A` and `B` and a large `C`, `A ∩ B` is built explicitly and the outer
45/// intersection runs lazily.
46#[derive(Clone, Copy, Debug, PartialEq, Eq)]
47pub struct Heuristic {
48	max_states: usize,
49}
50
51impl Default for Heuristic {
52	fn default() -> Heuristic {
53		Heuristic::new()
54	}
55}
56
57impl Heuristic {
58	/// A bound of 4096 macrostates.
59	#[must_use]
60	pub fn new() -> Heuristic {
61		Heuristic { max_states: 4096 }
62	}
63
64	/// Builds explicitly the terms bounded by `max_states` macrostates.
65	#[must_use]
66	pub fn max_states(mut self, max_states: usize) -> Heuristic {
67		self.max_states = max_states;
68
69		self
70	}
71}
72
73impl Policy for Heuristic {
74	fn plan(&self, view: &PlanView) -> Plan {
75		// log2 of the bound of each node; children come first.
76		let mut bounds: Vec<f64> = Vec::with_capacity(view.nodes.len());
77		let bound = |bounds: &[f64], term| {
78			let index = view
79				.nodes
80				.binary_search_by_key(&term, |node| node.term)
81				.expect("a child is in the view");
82
83			bounds[index]
84		};
85
86		for node in &view.nodes {
87			let children = node.children.iter().map(|&child| bound(&bounds, child));
88			let log = match &node.kind {
89				NodeKind::Leaf => {
90					let states = node.leaf.as_ref().map_or(1, |leaf| leaf.states.max(1));
91
92					log2(states)
93				}
94				NodeKind::Union => {
95					let logs: Vec<f64> = children.collect();
96					let max = logs.iter().copied().fold(f64::NEG_INFINITY, f64::max);
97
98					max + logs
99						.iter()
100						.map(|log| (log - max).exp2())
101						.sum::<f64>()
102						.log2()
103				}
104				NodeKind::Complement => children.map(f64::exp2).sum(),
105				// Products multiply; the unary operations keep their operand's bound.
106				_ => children.sum(),
107			};
108
109			bounds.push(log);
110		}
111
112		let small_bound = log2(self.max_states);
113		let small = |index: usize| bounds[index] <= small_bound;
114		// Whether the root or a term over the bound uses the node.
115		let mut wanted = vec![false; view.nodes.len()];
116		for (index, node) in view.nodes.iter().enumerate() {
117			if node.term == view.root || !small(index) {
118				for &child in &node.children {
119					let child = view
120						.nodes
121						.binary_search_by_key(&child, |node| node.term)
122						.expect("a child is in the view");
123					wanted[child] = true;
124				}
125			}
126		}
127
128		view.nodes
129			.iter()
130			.enumerate()
131			.filter(|&(index, node)| {
132				let can_materialize =
133					!matches!(node.kind, NodeKind::Leaf | NodeKind::Explicit { .. });
134				wanted[index] && small(index) && can_materialize
135			})
136			.fold(Plan::new(), |plan, (_, node)| {
137				plan.annotate(node.term, Annotation::eager_within(self.max_states))
138			})
139	}
140}

142/// A plan given from outside, such as one that a person or a tool wrote after reading the [`PlanView`]; it does
143/// not depend on the view. With the feature `serde`, `Formula::plan_from_json` reads it from JSON.
144#[derive(Clone, Debug, Default, PartialEq, Eq)]
145pub struct External(pub Plan);
146
147impl Policy for External {
148	fn plan(&self, _: &PlanView) -> Plan {
149		self.0.clone()
150	}
151}
152
153/// `annotation` for each node of `view` other than leaves.
154fn annotate_operations(view: &PlanView, annotation: Annotation) -> Plan {
155	view.nodes
156		.iter()
157		.filter(|node| node.kind != NodeKind::Leaf)
158		.fold(Plan::new(), |plan, node| {
159			plan.annotate(node.term, annotation)
160		})
161}
162
163/// log2 of `count`, saturated at `u32::MAX`, as the engine bounds leaves.
164fn log2(count: usize) -> f64 {
165	f64::from(u32::try_from(count).unwrap_or(u32::MAX)).log2()
166}
