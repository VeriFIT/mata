72/// The step that [`pad_closure`](Formula::pad_closure) appends.
73#[derive(Clone, Debug, PartialEq, Eq, Hash)]
74pub enum Pad {
75	/// The same symbols every time, one per tape; [`ANY`](Symbol::ANY) reads any symbol of its tape's alphabet.

81#[derive(Debug)]
82pub(crate) enum Node {
83	Leaf(LeafId),
84	Union(NodeId, NodeId),
85	Intersection(NodeId, NodeId),
86	Complement(NodeId),
87	Identity(NodeId),
88	Project(NodeId, Arc<[usize]>),
89	PadClosure(NodeId, Arc<Pad>),
90	Sync(NodeId, NodeId, Arc<SyncSpec>),
91	Explicit(NodeId, ExplicitOptions),
92	Prefixes(NodeId),
93}
94
95impl Node {
96	/// The operands, in order.
97	pub(crate) fn children(&self) -> impl Iterator<Item = NodeId> {
98		let operands = match *self {
99			Node::Leaf(_) => [None, None],
100			Node::Complement(child)
101			| Node::Identity(child)
102			| Node::Project(child, _)
103			| Node::PadClosure(child, _)
104			| Node::Explicit(child, _)
105			| Node::Prefixes(child) => [Some(child), None],
106			Node::Union(a, b) | Node::Intersection(a, b) | Node::Sync(a, b, _) => {
107				[Some(a), Some(b)]
108			}
109		};
110
111		operands.into_iter().flatten()
112	}
113}

130/// A checked synchronized product.
131#[derive(Debug, PartialEq, Eq, Hash)]
132pub(crate) struct SyncSpec {
133	pub(crate) pairs: Box<[TapePair]>,
134	pub(crate) layout: Box<[ResultTape]>,

137impl SyncSpec {
138	/// The same product with its operands swapped.
139	pub(crate) fn swapped(&self) -> SyncSpec {
140		let swap = |tape| match tape {
141			Tape::Lhs(index) => Tape::Rhs(index),
142			Tape::Rhs(index) => Tape::Lhs(index),
143		};
144
145		SyncSpec {
146			pairs: self
147				.pairs
148				.iter()
149				.map(|pair| TapePair {
150					lhs: pair.rhs,
151					rhs: pair.lhs,
152				})
153				.collect(),
154			layout: self
155				.layout
156				.iter()
157				.map(|tape| ResultTape {
158					source: swap(tape.source),
159					peer: tape.peer,
160				})
161				.collect(),
162		}
163	}
164}

166/// A DAG of Boolean and relational operations over automata, checked for emptiness lazily.
167///
168/// Building a formula does no automata work. [`leaf`](Formula::leaf) copies an automaton, and the other builders add a node. [`is_empty`](Formula::is_empty) and [`check`](Formula::check) explore only the macrostates they need.
169///
170/// A term of arity *n* denotes a relation over *n*-tuples of words of equal length, read in lock step, where one
171/// step reads one symbol on every tape ([`EPSILON`](Symbol::EPSILON) pads a shorter word). The steps are what
172/// count: a step that pads every tape still is one (only a leaf's own moves that pad every tape are silent), and
173/// a term of arity 0 holds a number of steps. The complement is taken over the tuples of concrete symbols of each
174/// tape's alphabet, which a query computes from the leaves that share the tape.
175#[derive(Debug)]
176pub struct Formula {
177	id: u32,
178	pub(crate) nodes: TiVec<NodeId, Node>,
179	pub(crate) arities: TiVec<NodeId, usize>,
180	/// Whether a word of the node may read padding (a complement never does).
181	pub(crate) pads: TiVec<NodeId, bool>,
182	pub(crate) leaves: TiVec<LeafId, Leaf>,
183	pub(crate) explicit_cache: Mutex<engine::ExplicitCache>,
184}
185
186impl Default for Formula {
187	fn default() -> Formula {
188		Formula::new()
189	}
190}

193	/// An empty formula.
194	///
195	/// # Panics
196	///
197	/// If 2³² − 1 formulas have already been created in this process; reusing an id would let a term of an earlier
198	/// formula pass as one of the new formula.
199	#[must_use]
200	pub fn new() -> Formula {
201		let id = NEXT_FORMULA
202			.try_update(Ordering::Relaxed, Ordering::Relaxed, |id| id.checked_add(1))
203			.expect("formula ids exhausted");
204		Formula {
205			id,
206			nodes: TiVec::default(),
207			arities: TiVec::default(),
208			pads: TiVec::default(),
209			leaves: TiVec::default(),
210			explicit_cache: Mutex::new(engine::ExplicitCache::default()),
211		}
212	}

223	/// A leaf as [`leaf`](Formula::leaf), whose tape `i` also has the symbols of `alphabets[i]`, read or not, so
224	/// that a complement over the leaf includes them. A single alphabet applies to every tape. The symbols of
225	/// [`Automaton::alphabet`] join them, in `leaf` too.
226	///
227	/// ```
228	/// use mata_lazy::{Automaton, Formula, Symbol};
229	///
230	/// /// The words a*.
231	/// struct OnlyA;
232	///
233	/// impl Automaton for OnlyA {
234	///     type State = ();
235	///
236	///     fn arity(&self) -> usize { 1 }
237	///     fn initial_states(&self) -> impl IntoIterator<Item = ()> { [()] }
238	///     fn is_final(&self, (): ()) -> bool { true }
239	///     fn for_each_move(&self, (): (), mut visit: impl FnMut(&[Symbol], ())) {
240	///         visit(&[Symbol::from(b'a')], ());
241	///     }
242	/// }
243	///
244	/// let mut formula = Formula::new();
245	/// let leaf = formula.leaf(&OnlyA)?;
246	/// // Without b in the alphabet, the complement of a* is empty.
247	/// let not_a = formula.complement(leaf)?;
248	/// assert!(formula.is_empty(not_a)?);
249	/// let leaf = formula.leaf_over(&OnlyA, [*b"ab"])?;
250	/// let not_a = formula.complement(leaf)?;
251	/// assert!(!formula.is_empty(not_a)?);
252	/// # Ok::<(), Box<dyn std::error::Error>>(())
253	/// ```
254	///
255	/// # Errors
256	///
257	/// As [`leaf`](Formula::leaf), [`AlphabetCount`](Error::AlphabetCount) if `alphabets` has neither one alphabet
258	/// nor one per tape, or [`ReservedSymbol`](Error::ReservedSymbol).
259	pub fn leaf_over<I>(
260		&mut self,
261		automaton: &impl Automaton,
262		alphabets: impl IntoIterator<Item = I>,
263	) -> Result<Term, Error>
264	where
265		I: IntoIterator<Item: Into<Symbol>>,
266	{
267		let tapes = automaton.arity();
268		let mut declared: Vec<Vec<Symbol>> = alphabets
269			.into_iter()
270			.map(|alphabet| alphabet.into_iter().map(Into::into).collect())
271			.collect();
272
273		match declared.len() {
274			0 => {}
275			1 => declared = vec![declared.remove(0); tapes],
276			found if found == tapes => {}
277			found => {
278				return Err(Error::AlphabetCount {
279					arity: tapes,
280					found,
281				});
282			}
283		}
284
285		declared.resize(tapes, Vec::new());
286		for (tape, symbols) in declared.iter_mut().enumerate() {
287			symbols.extend(automaton.alphabet(tape));
288		}
289
290		if let Some(&symbol) = declared
291			.iter()
292			.flatten()
293			.find(|symbol| !symbol.is_concrete())
294		{
295			return Err(Error::ReservedSymbol { symbol });
296		}
297
298		let leaf = Leaf::import(automaton, declared)?;
299		let arity = leaf.arity();
300		let leaf = self.leaves.push_and_get_key(leaf);
301
302		Ok(self.push(Node::Leaf(leaf), arity))
303	}

349	/// The projection to `tapes`, in that order; the other tapes are dropped (existentially).
350	///
351	/// # Errors
352	///
353	/// [`TapeOutOfRange`](Error::TapeOutOfRange), [`RepeatedTape`](Error::RepeatedTape), or
354	/// [`ForeignTerm`](Error::ForeignTerm).
355	pub fn project(&mut self, term: Term, tapes: &[usize]) -> Result<Term, Error> {
356		let node = self.node(term)?;
357		let arity = self.arities[node];
358
359		let mut seen = FxHashSet::default();
360		for &tape in tapes {
361			if tape >= arity {
362				return Err(Error::TapeOutOfRange { tape, arity });
363			}
364
365			if !seen.insert(tape) {
366				return Err(Error::RepeatedTape { tape });
367			}
368		}
369
370		Ok(self.push(Node::Project(node, tapes.into()), tapes.len()))
371	}

373	/// The words of `term` with padding steps cut off the end: `{w | w·p^k ∈ term for some k ≥ 0}`, where `p`
374	/// is `pad`. Logics that encode numbers as words need it after a projection, as the dropped tapes may have
375	/// needed longer words than the kept ones. Symbols of a [`Pad::Step`] join the alphabets of their tapes.
376	///
377	/// # Errors
378	///
379	/// [`Arity`](Error::Arity) if a [`Pad::Step`] does not have one symbol per tape, or
380	/// [`ForeignTerm`](Error::ForeignTerm).
381	pub fn pad_closure(&mut self, term: Term, pad: Pad) -> Result<Term, Error> {
382		let node = self.node(term)?;
383		let arity = self.arities[node];
384		if let Pad::Step(step) = &pad
385			&& step.len() != arity
386		{
387			return Err(Error::Arity {
388				operation: "pad_closure",
389				expected: arity,
390				found: step.len(),
391			});
392		}
393
394		Ok(self.push(Node::PadClosure(node, Arc::new(pad)), arity))
395	}

397	/// The synchronized product, which holds the tuples of both operands whose `synchronized` tape pairs read
398	/// the same words, arranged by `layout`. Tapes missing from `layout` are dropped.
399	///
400	/// ```
401	/// use mata_lazy::{Automaton, Formula, Symbol, Tape};
402	///
403	/// /// The single pair (x, y).
404	/// struct Pair(u8, u8);
405	///
406	/// impl Automaton for Pair {
407	///     type State = bool;
408	///
409	///     fn arity(&self) -> usize { 2 }
410	///     fn initial_states(&self) -> impl IntoIterator<Item = bool> { [false] }
411	///     fn is_final(&self, read: bool) -> bool { read }
412	///     fn for_each_move(&self, read: bool, mut visit: impl FnMut(&[Symbol], bool)) {
413	///         if !read {
414	///             visit(&[self.0.into(), self.1.into()], true);
415	///         }
416	///     }
417	/// }
418	///
419	/// let mut formula = Formula::new();
420	/// let ab = formula.leaf(&Pair(b'a', b'b'))?;
421	/// let bc = formula.leaf(&Pair(b'b', b'c'))?;
422	/// // {(a, b, c)}: tape 1 of `ab` meets tape 0 of `bc`.
423	/// let layout = [Tape::Lhs(0), Tape::Lhs(1), Tape::Rhs(1)];
424	/// let abc = formula.sync_product(ab, bc, &[(1, 0)], &layout)?;
425	/// assert_eq!(formula.arity(abc)?, 3);
426	/// assert!(!formula.is_empty(abc)?);
427	/// // Tape 0 of `ab` reads a, tape 0 of `bc` reads b.
428	/// let none = formula.sync_product(ab, bc, &[(0, 0)], &[])?;
429	/// assert!(formula.is_empty(none)?);
430	/// # Ok::<(), mata_lazy::Error>(())
431	/// ```
432	///
433	/// # Errors
434	///
435	/// [`SyncTapeOutOfRange`](Error::SyncTapeOutOfRange), [`RepeatedSyncTape`](Error::RepeatedSyncTape) if a tape is
436	/// synchronized twice or a result tape (or its synchronized peer) appears twice in `layout`, or
437	/// [`ForeignTerm`](Error::ForeignTerm).
438	pub fn sync_product(
439		&mut self,
440		lhs: Term,
441		rhs: Term,
442		synchronized: &[(usize, usize)],
443		layout: &[Tape],
444	) -> Result<Term, Error> {
445		let (lhs, rhs) = (self.node(lhs)?, self.node(rhs)?);
446		let checked = |tape: Tape| {
447			let arity = match tape {
448				Tape::Lhs(_) => self.arities[lhs],
449				Tape::Rhs(_) => self.arities[rhs],
450			};
451
452			if tape.index() < arity {
453				Ok(tape)
454			} else {
455				Err(Error::SyncTapeOutOfRange { tape, arity })
456			}
457		};
458
459		let mut peers: FxHashMap<Tape, Tape> = FxHashMap::default();
460		for &(i, j) in synchronized {
461			let (lhs_tape, rhs_tape) = (checked(Tape::Lhs(i))?, checked(Tape::Rhs(j))?);
462			for (tape, peer) in [(lhs_tape, rhs_tape), (rhs_tape, lhs_tape)] {
463				if peers.insert(tape, peer).is_some() {
464					return Err(Error::RepeatedSyncTape { tape });
465				}
466			}
467		}
468
469		// A synchronized pair gives at most one result tape.
470		let mut used = FxHashSet::default();
471		let mut result_tapes = Vec::with_capacity(layout.len());
472		for &tape in layout {
473			let tape = checked(tape)?;
474			if !used.insert(tape) {
475				return Err(Error::RepeatedSyncTape { tape });
476			}
477
478			let peer = peers.get(&tape).copied();
479			if let Some(peer) = peer
480				&& !used.insert(peer)
481			{
482				return Err(Error::RepeatedSyncTape { tape });
483			}
484
485			result_tapes.push(ResultTape {
486				source: tape,
487				peer: peer.map(Tape::index),
488			});
489		}
490
491		let arity = result_tapes.len();
492		let spec = SyncSpec {
493			pairs: synchronized
494				.iter()
495				.map(|&(lhs, rhs)| TapePair { lhs, rhs })
496				.collect(),
497			layout: result_tapes.into(),
498		};
499
500		Ok(self.push(Node::Sync(lhs, rhs, Arc::new(spec)), arity))
501	}

503	/// The composition on the `synchronized` tape pairs, which are dropped. The result has the other tapes of
504	/// `lhs`, then the other tapes of `rhs`.
505	///
506	/// # Errors
507	///
508	/// As [`sync_product`](Formula::sync_product).
509	pub fn compose_on(
510		&mut self,
511		lhs: Term,
512		rhs: Term,
513		synchronized: &[(usize, usize)],
514	) -> Result<Term, Error> {
515		let (lhs_arity, rhs_arity) = (self.arity(lhs)?, self.arity(rhs)?);
516
517		let kept_lhs = (0..lhs_arity)
518			.filter(|&i| !synchronized.iter().any(|&(l, _)| l == i))
519			.map(Tape::Lhs);
520		let kept_rhs = (0..rhs_arity)
521			.filter(|&j| !synchronized.iter().any(|&(_, r)| r == j))
522			.map(Tape::Rhs);
523		let layout: Vec<Tape> = kept_lhs.chain(kept_rhs).collect();
524
525		self.sync_product(lhs, rhs, synchronized, &layout)
526	}

565	/// The prefix closure `{u | ∃v: u·v ∈ term}`, taken over steps: a prefix keeps the first steps of all tapes
566	/// together, padding included. It is empty when `term` is, and holds the empty word otherwise.
567	///
568	/// A query decides whether a macrostate of `term` reaches an accepting one with a search from it, whose answers
569	/// it keeps; leaves and explicit nodes are trimmed, so for them it needs none. With this, `term`'s macrostates
570	/// and their subsumption carry over unchanged.
571	///
572	/// ```
573	/// # #[cfg(feature = "regex")] {
574	/// use mata_lazy::Formula;
575	/// use mata_lazy::regex::Regex;
576	///
577	/// let mut formula = Formula::new();
578	/// let abc = formula.leaf(&Regex::new("abc")?)?;
579	/// let ab = formula.leaf(&Regex::new("ab")?)?;
580	/// let ac = formula.leaf(&Regex::new("ac")?)?;
581	/// let prefixes = formula.prefixes(abc)?;
582	/// let has_ab = formula.intersection(prefixes, ab)?;
583	/// let has_ac = formula.intersection(prefixes, ac)?;
584	/// assert!(!formula.is_empty(has_ab)? && formula.is_empty(has_ac)?);
585	/// # }
586	/// # Ok::<(), Box<dyn std::error::Error>>(())
587	/// ```
588	///
589	/// # Errors
590	///
591	/// [`ForeignTerm`](Error::ForeignTerm).
592	pub fn prefixes(&mut self, term: Term) -> Result<Term, Error> {
593		let node = self.node(term)?;
594
595		Ok(self.push(Node::Prefixes(node), self.arities[node]))
596	}

598	/// The same relation, built as an explicit automaton when a query starts; see

608	/// The same relation, built as an explicit automaton when a query starts.
609	///
610	/// The query explores `term` completely, numbers its macrostates densely, trims the result and takes its
611	/// quotient by bisimulation. When the subset construction over tuples of the quotient stays within four times
612	/// its states, sixteen times its states in members of all subsets and sixteen times its moves, the result is
613	/// minimized; otherwise the quotient is kept. Then the query reads it like a leaf, with a simulation of its own.
614	/// Explicit nodes stacked on each other therefore keep the size of their languages, not of their products. The
615	/// query's limits hold through this work, and an automaton built in full is kept even if the query then passes
616	/// a limit. Rewrites stop at this node. Materialized automata are reused by later queries over the same
617	/// alphabets (see [`clear_cache`](Formula::clear_cache)). When the exploration passes `options.max_states`, the
618	/// node stays lazy.
619	///
620	/// # Errors
621	///
622	/// [`ForeignTerm`](Error::ForeignTerm).
623	pub fn explicit_with(&mut self, term: Term, options: ExplicitOptions) -> Result<Term, Error> {
624		let node = self.node(term)?;
625
626		Ok(self.push(Node::Explicit(node, options), self.arities[node]))
627	}

656	/// Checks `term` for emptiness with `options`, and reports a witness and statistics.
657	///
658	/// # Errors
659	///
660	/// [`StateLimit`](Error::StateLimit) or [`TimeLimit`](Error::TimeLimit) (with the statistics so far) if the query
661	/// passes a limit of `options`, [`TapeOutOfRange`](Error::TapeOutOfRange) or
662	/// [`ReservedSymbol`](Error::ReservedSymbol) for an alphabet of `options`,
663	/// [`ComplementTooLarge`](Error::ComplementTooLarge), or [`ForeignTerm`](Error::ForeignTerm) for `term` or an
664	/// explicit term of `options`.
665	pub fn check(&self, term: Term, options: &Options) -> Result<Outcome, Error> {
666		let mut engine = engine::Engine::new(self, self.node(term)?, options, None)?;
667		let (witness, bounded) = engine.search()?;
668
669		Ok(Outcome {
670			witness,
671			bounded,
672			stats: engine.stats(),
673		})
674	}

676	/// Checks `term` under each of `plans` at once, one thread per plan, and returns the first answer with the
677	/// index of its plan; the other queries stop. Every plan keeps the answer (see [`Plan`]), so only the time to
678	/// it depends on the plans. `options` holds for every query, its own plan replaced.
679	///
680	/// A plan whose thread cannot be started (as on WebAssembly without threads) runs on the calling thread, before
681	/// the plans after it are started.
682	///
683	/// # Errors
684	///
685	/// [`InvalidPlan`](Error::InvalidPlan) for no plans, [`ForeignTerm`](Error::ForeignTerm) for `term`, and when
686	/// every query fails, the error of the first plan (as of [`check`](Formula::check)).
687	pub fn check_portfolio(
688		&self,
689		term: Term,
690		options: &Options,
691		plans: &[Plan],
692	) -> Result<(usize, Outcome), Error> {
693		let node = self.node(term)?;
694		if plans.is_empty() {
695			return Err(Error::InvalidPlan {
696				reason: "a portfolio needs at least one plan".into(),
697			});
698		}
699
700		let cancel = Arc::new(AtomicBool::new(false));
701		let run = |index: usize| {
702			let options = options.clone().plan(plans[index].clone());
703			let result = engine::Engine::new(self, node, &options, Some(Arc::clone(&cancel)))
704				.and_then(|mut engine| {
705					let (witness, bounded) = engine.search()?;
706					Ok(Outcome {
707						witness,
708						bounded,
709						stats: engine.stats(),
710					})
711				});
712			if result.is_ok() {
713				cancel.store(true, Ordering::Relaxed);
714			}
715			result
716		};
717
718		thread::scope(|scope| {
719			let (sender, receiver) = mpsc::channel();
720			for index in 0..plans.len() {
721				// A plan run on this thread may have answered already.
722				if cancel.load(Ordering::Relaxed) {
723					break;
724				}
725				let thread_sender = sender.clone();
726				let run = &run;
727				let spawned = thread::Builder::new().spawn_scoped(scope, move || {
728					// The receiver is gone only after an answer, when this one is not needed.
729					let _ = thread_sender.send((index, run(index)));
730				});
731				if spawned.is_err() {
732					let _ = sender.send((index, run(index)));
733				}
734			}
735			drop(sender);
736
737			let mut first_error = None;
738			for (index, result) in receiver {
739				match result {
740					Ok(outcome) => return Ok((index, outcome)),
741					Err(error) => {
742						if first_error.as_ref().is_none_or(|&(first, _)| index < first) {
743							first_error = Some((index, error));
744						}
745					}
746				}
747			}
748
749			// Every plan sends its result, and without an answer none was cancelled.
750			Err(first_error.map_or_else(
751				|| Error::InvalidPlan {
752					reason: "no plan of the portfolio answered".into(),
753				},
754				|(_, error)| error,
755			))
756		})
757	}

759	/// Builds the whole relation of `term` as an explicit automaton (trimmed).
760	///
761	/// Other libraries copy it into their own automata; with Mata, `Nft::try_from(&explicit)` (in `mata::lazy`)
762	/// gives a transducer with one level per tape. [`leaf`](Formula::leaf) reads it back into a formula.
763	///
764	/// # Errors
765	///
766	/// As [`check`](Formula::check).
767	pub fn materialize(&self, term: Term, options: &Options) -> Result<ExplicitAutomaton, Error> {
768		let node = self.node(term)?;
769		let mut engine = engine::Engine::new(self, node, options, None)?;
770
771		Ok(ExplicitAutomaton::new(engine.materialize_root()?))
772	}

794	fn push(&mut self, node: Node, arity: usize) -> Term {
795		let pads = match node {
796			Node::Leaf(leaf) => self.leaves[leaf].has_padding,
797			Node::Complement(_) => false,
798			Node::Identity(child)
799			| Node::Project(child, _)
800			| Node::PadClosure(child, _)
801			| Node::Explicit(child, _)
802			| Node::Prefixes(child) => self.pads[child],
803			Node::Union(a, b) | Node::Intersection(a, b) | Node::Sync(a, b, _) => {
804				self.pads[a] || self.pads[b]
805			}
806		};
807
808		self.pads.push(pads);
809		let id = self.nodes.push_and_get_key(node);
810		self.arities.push(arity);
811
812		self.term(id)
813	}

815	fn node_of_arity(
816		&self,
817		operation: &'static str,
818		term: Term,
819		expected: usize,
820	) -> Result<NodeId, Error> {
821		let node = self.node(term)?;
822		let found = self.arities[node];
823		if found != expected {
824			return Err(Error::Arity {
825				operation,
826				expected,
827				found,
828			});
829		}
830
831		Ok(node)
832	}

834	/// The nodes of two operands of equal arity, and that arity.
835	fn same_arity(
836		&self,
837		operation: &'static str,
838		lhs: Term,
839		rhs: Term,
840	) -> Result<(NodeId, NodeId, usize), Error> {
841		let lhs = self.node(lhs)?;
842		let arity = self.arities[lhs];
843		let rhs = self.node_of_arity(operation, rhs, arity)?;
844
845		Ok((lhs, rhs, arity))
846	}
