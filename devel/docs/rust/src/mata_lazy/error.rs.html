1//! Errors of building and querying a formula.
2
3use std::time::Duration;
4
5use crate::{Stats, Symbol, Tape};
6
7/// Why a builder or a query of a [`Formula`](crate::Formula) failed.
8#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
9#[non_exhaustive]
10pub enum Error {
11	/// The term belongs to another formula.
12	#[error("the term belongs to another formula")]
13	ForeignTerm,
14	/// An operand of the wrong arity.
15	#[error("{operation} expects arity {expected}, found {found}")]
16	Arity {
17		/// The builder.
18		operation: &'static str,
19		/// The arity it needs.
20		expected: usize,
21		/// The arity it got.
22		found: usize,
23	},
24	/// A tape at or above the arity of its term.
25	#[error("tape {tape} is out of range for arity {arity}")]
26	TapeOutOfRange {
27		/// The tape.
28		tape: usize,
29		/// The arity of the term.
30		arity: usize,
31	},
32	/// A tape listed twice in a projection.
33	#[error("tape {tape} is listed twice")]
34	RepeatedTape {
35		/// The tape.
36		tape: usize,
37	},
38	/// A tape of a synchronized product at or above the arity of its operand.
39	#[error("{tape:?} is out of range for arity {arity}")]
40	SyncTapeOutOfRange {
41		/// The tape.
42		tape: Tape,
43		/// The arity of its operand.
44		arity: usize,
45	},
46	/// A tape of a synchronized product that is synchronized twice, or laid out twice (itself or its
47	/// synchronized peer).
48	#[error("{tape:?} is used twice")]
49	RepeatedSyncTape {
50		/// The tape.
51		tape: Tape,
52	},
53	/// A move of a leaf that does not read one symbol per tape.
54	#[error("a move reads {found} symbols, but the automaton has {arity} tapes")]
55	MoveWidth {
56		/// The arity of the automaton.
57		arity: usize,
58		/// The symbols of the move.
59		found: usize,
60	},
61	/// [`EPSILON`](Symbol::EPSILON) or [`ANY`](Symbol::ANY) in an alphabet.
62	#[error("the reserved symbol {symbol} cannot be in an alphabet")]
63	ReservedSymbol {
64		/// The symbol.
65		symbol: Symbol,
66	},
67	/// A complement whose alphabets have more tuples than `usize` can count.
68	#[error("a complement over {arity} tapes has too many tuples to enumerate")]
69	ComplementTooLarge {
70		/// The arity of the complement.
71		arity: usize,
72	},
73	/// Neither one alphabet nor one per tape.
74	#[error("{found} alphabets for {arity} tapes")]
75	AlphabetCount {
76		/// The arity of the automaton.
77		arity: usize,
78		/// The alphabets given.
79		found: usize,
80	},
81	/// A [`Plan`](crate::Plan) that annotates an operand order of a term that has none, or JSON that is no plan of
82	/// the formula (a missing term, a term twice, bad syntax); the whole plan is rejected.
83	#[error("invalid plan: {reason}")]
84	InvalidPlan {
85		/// What is wrong.
86		reason: String,
87	},
88	/// The query held more macrostates than [`max_states`](crate::Options::max_states).
89	#[error("the query holds more than {max_states} macrostates")]
90	#[non_exhaustive]
91	StateLimit {
92		/// The limit.
93		max_states: usize,
94		/// What the query did until it stopped.
95		stats: Box<Stats>,
96	},
97	/// The query ran longer than [`time_limit`](crate::Options::time_limit).
98	#[error("the query ran longer than {limit:?}")]
99	#[non_exhaustive]
100	TimeLimit {
101		/// The limit.
102		limit: Duration,
103		/// What the query did until it stopped.
104		stats: Box<Stats>,
105	},
106	/// A query of a [portfolio](crate::Formula::check_portfolio) stopped because another plan answered first. The
107	/// portfolio returns that answer, so this error stays inside it.
108	#[error("another plan of the portfolio answered first")]
109	#[non_exhaustive]
110	Cancelled {
111		/// What the query did until it stopped.
112		stats: Box<Stats>,
113	},
114}
