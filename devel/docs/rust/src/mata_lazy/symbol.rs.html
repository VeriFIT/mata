1//! Symbols, numbered as in Mata.
2
3use std::fmt;
4
5/// A symbol, a number that all leaves of a formula share, as Mata numbers them. Two values are reserved.
6///
7/// A byte is the symbol of the same number, as in Mata's `from_regex`.
8#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
9pub struct Symbol(u32);
10
11impl Symbol {
12	/// Padding, where the tape reads nothing while the other tapes read (Mata's `EPSILON`, the largest `u32`).
13	pub const EPSILON: Symbol = Symbol(u32::MAX);
14	/// Any one symbol of the tape's alphabet (Mata's `DONT_CARE`, the `u32` below `EPSILON`).
15	pub const ANY: Symbol = Symbol(u32::MAX - 1);
16
17	/// The symbol with number `value`.
18	#[must_use]
19	pub const fn new(value: u32) -> Symbol {
20		Symbol(value)
21	}
22
23	/// The number of this symbol.
24	#[must_use]
25	pub const fn value(self) -> u32 {
26		self.0
27	}
28
29	/// Neither padding nor the wildcard.
30	#[must_use]
31	pub const fn is_concrete(self) -> bool {
32		self.0 < Symbol::ANY.0
33	}
34}
35
36/// The symbol that both symbols read, if they agree. [`ANY`](Symbol::ANY) takes the other symbol.
37pub(crate) fn merge(lhs: Symbol, rhs: Symbol) -> Option<Symbol> {
38	if lhs == rhs {
39		return Some(lhs);
40	}
41
42	match (lhs, rhs) {
43		(Symbol::EPSILON, _) | (_, Symbol::EPSILON) => None,
44		(Symbol::ANY, other) | (other, Symbol::ANY) => Some(other),
45		_ => None,
46	}
47}
48
49impl From<u8> for Symbol {
50	fn from(byte: u8) -> Symbol {
51		Symbol(byte.into())
52	}
53}
54
55impl From<u32> for Symbol {
56	fn from(value: u32) -> Symbol {
57		Symbol(value)
58	}
59}
60
61impl From<Symbol> for u32 {
62	fn from(symbol: Symbol) -> u32 {
63		symbol.0
64	}
65}
66
67/// Prints the number, `ε` for [`EPSILON`](Symbol::EPSILON) or `*` for [`ANY`](Symbol::ANY).
68impl fmt::Display for Symbol {
69	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
70		match *self {
71			Symbol::EPSILON => f.write_str("ε"),
72			Symbol::ANY => f.write_str("*"),
73			Symbol(value) => write!(f, "{value}"),
74		}
75	}
76}
