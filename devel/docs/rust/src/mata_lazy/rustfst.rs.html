1//! [rustfst](https://docs.rs/rustfst) transducers as leaves.
2//!
3//! A [`VectorFst`] or a [`ConstFst`] is a leaf of arity 2, where tape 0 reads the input labels and tape 1 the
4//! output labels. Label 0 ([`EPS_LABEL`]) is padding, and label `n > 0` is the symbol `n`, so the largest
5//! label is [`EPSILON`](Symbol::EPSILON) and the one below it [`ANY`](Symbol::ANY). An acceptor is the identity
6//! relation of its language; `formula.project(leaf, &[0])` gives the language.
7//!
8//! Weights only decide whether a transition or a final weight exists. Zero ones are dropped, so the leaf holds
9//! the paths whose weights are all nonzero. That is the support of the weighted relation when nonzero weights
10//! neither multiply nor add up to zero (no zero divisors, zero-sum-free addition). Like a Mata transducer, an
11//! FST is read in lock step, while rustfst aligns epsilon asynchronously.
12//!
13//! ```
14//! use mata_lazy::Formula;
15//! use rustfst::fst_impls::VectorFst;
16//! use rustfst::fst_traits::MutableFst;
17//! use rustfst::semirings::{Semiring, TropicalWeight};
18//! use rustfst::Tr;
19//!
20//! // a ↦ b, any number of times.
21//! let mut fst = VectorFst::<TropicalWeight>::new();
22//! let q = fst.add_state();
23//! fst.set_start(q)?;
24//! fst.set_final(q, TropicalWeight::one())?;
25//! fst.add_tr(q, Tr::new(1, 2, TropicalWeight::new(0.5), q))?;
26//!
27//! let mut formula = Formula::new();
28//! let relation = formula.leaf(&fst)?;
29//! let outputs = formula.project(relation, &[1])?;
30//! assert!(!formula.is_empty(outputs)?);
31//! # Ok::<(), Box<dyn std::error::Error>>(())
32//! ```
33
34use rustfst::fst_impls::{ConstFst, VectorFst};
35use rustfst::fst_traits::CoreFst;
36use rustfst::semirings::Semiring;
37use rustfst::{EPS_LABEL, Label, StateId};
38
39use crate::{Automaton, Symbol};
40
41fn symbol(label: Label) -> Symbol {
42	if label == EPS_LABEL {
43		Symbol::EPSILON
44	} else {
45		Symbol::new(label)
46	}
47}
48
49macro_rules! fst_leaf {
50	($($fst:ident),*) => {$(
51		/// Arity 2, with input labels on tape 0 and output labels on tape 1.
52		impl<W: Semiring> Automaton for $fst<W> {
53			type State = StateId;
54
55			fn arity(&self) -> usize {
56				2
57			}
58
59			fn initial_states(&self) -> impl IntoIterator<Item = StateId> {
60				self.start()
61			}
62
63			fn is_final(&self, state: StateId) -> bool {
64				matches!(self.final_weight(state), Ok(Some(weight)) if !weight.is_zero())
65			}
66
67			fn for_each_move(&self, state: StateId, mut visit: impl FnMut(&[Symbol], StateId)) {
68				let Ok(trs) = self.get_trs(state) else {
69					return;
70				};
71
72				for tr in trs.iter() {
73					if !tr.weight.is_zero() {
74						visit(&[symbol(tr.ilabel), symbol(tr.olabel)], tr.nextstate);
75					}
76				}
77			}
78		}
79	)*};
80}
81
82fst_leaf!(VectorFst, ConstFst);
