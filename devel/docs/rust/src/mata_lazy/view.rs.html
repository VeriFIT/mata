1//! A read-only view of a formula's DAG, for policies and tools that plan queries.

7/// The terms that one term depends on, children first, as [`Formula::view`] gives them.
8#[derive(Clone, Debug, PartialEq, Eq)]
9#[cfg_attr(feature = "serde", derive(serde::Serialize))]
10#[non_exhaustive]
11pub struct PlanView {
12	/// The viewed term.
13	pub root: Term,
14	/// The terms that `root` depends on and `root` itself, by [`id`](Term::id), so children come first.
15	pub nodes: Vec<NodeView>,
16}
17
18impl PlanView {
19	/// The node of `term`, if `root` depends on it.
20	#[must_use]
21	pub fn node(&self, term: Term) -> Option<&NodeView> {
22		self.nodes
23			.binary_search_by_key(&term, |node| node.term)
24			.ok()
25			.map(|index| &self.nodes[index])
26	}
27
28	/// The view as JSON, with terms written as their [`id`](Term::id)s.
29	///
30	/// # Panics
31	///
32	/// Never: a view is numbers and names, which JSON always holds.
33	#[cfg(feature = "serde")]
34	#[must_use]
35	pub fn to_json(&self) -> String {
36		serde_json::to_string_pretty(self).expect("a view is plain data")
37	}
38}

40/// One term of a [`PlanView`].
41#[derive(Clone, Debug, PartialEq, Eq)]
42#[cfg_attr(feature = "serde", derive(serde::Serialize))]
43#[non_exhaustive]
44pub struct NodeView {
45	/// The term.
46	pub term: Term,
47	/// The builder that made it.
48	pub kind: NodeKind,
49	/// The number of tapes.
50	pub arity: usize,
51	/// The operands, in the order passed to the builder.
52	pub children: Vec<Term>,
53	/// How many times the terms of the view use it as an operand.
54	pub fan_out: usize,
55	/// Whether a word of the term may read padding.
56	pub pads: bool,
57	/// The automaton of a leaf.
58	pub leaf: Option<LeafInfo>,
59}
60
61/// The builder of a [`NodeView`].
62#[derive(Clone, Debug, PartialEq, Eq)]
63#[cfg_attr(
64	feature = "serde",
65	derive(serde::Serialize),
66	serde(rename_all = "snake_case")
67)]
68#[non_exhaustive]
69pub enum NodeKind {
70	/// [`leaf`](Formula::leaf) or [`leaf_over`](Formula::leaf_over).
71	Leaf,
72	/// [`union`](Formula::union).
73	Union,
74	/// [`intersection`](Formula::intersection).
75	Intersection,
76	/// [`complement`](Formula::complement).
77	Complement,
78	/// [`identity`](Formula::identity).
79	Identity,
80	/// [`project`](Formula::project).
81	Project {
82		/// The kept tapes, in order.
83		tapes: Vec<usize>,
84	},
85	/// [`pad_closure`](Formula::pad_closure).
86	PadClosure,
87	/// [`sync_product`](Formula::sync_product), which also builds compositions and images.
88	Sync {
89		/// The synchronized pairs of a left and a right tape.
90		pairs: Vec<(usize, usize)>,
91		/// The tapes of the result.
92		layout: Vec<Tape>,
93	},
94	/// [`explicit`](Formula::explicit) or [`explicit_with`](Formula::explicit_with).
95	Explicit {
96		/// Its budget.
97		max_states: Option<usize>,
98	},
99	/// [`prefixes`](Formula::prefixes).
100	Prefixes,
101}
102
103/// The automaton of a leaf, as the formula holds it (trimmed, without silent moves).
104#[derive(Clone, Debug, PartialEq, Eq)]
105#[cfg_attr(feature = "serde", derive(serde::Serialize))]
106#[non_exhaustive]
107pub struct LeafInfo {
108	/// The number of states.
109	pub states: usize,
110	/// The number of moves.
111	pub transitions: usize,
112	/// Whether it has at most one initial state and no state has two moves on one tuple or a move on
113	/// [`ANY`](crate::Symbol::ANY).
114	pub deterministic: bool,
115	/// The number of concrete symbols of each tape: those it reads and those it was given.
116	pub tape_alphabet_sizes: Vec<usize>,
117}

119impl Formula {
120	/// The terms that `term` depends on, with their kinds, operands and leaf sizes.
121	///
122	/// # Errors
123	///
124	/// [`ForeignTerm`](Error::ForeignTerm).
125	pub fn view(&self, term: Term) -> Result<PlanView, Error> {
126		let root = self.node(term)?;
127		let mut seen = vec![false; root.index() + 1];
128		seen[root.index()] = true;
129		let mut stack = vec![root];
130		while let Some(node) = stack.pop() {
131			for child in self.nodes[node].children() {
132				if !std::mem::replace(&mut seen[child.index()], true) {
133					stack.push(child);
134				}
135			}
136		}
137
138		let mut fan_out = vec![0; seen.len()];
139		for (index, _) in seen.iter().enumerate().filter(|&(_, &seen)| seen) {
140			for child in self.nodes[NodeId::new(index)].children() {
141				fan_out[child.index()] += 1;
142			}
143		}
144
145		let nodes = seen
146			.iter()
147			.enumerate()
148			.filter(|&(_, &seen)| seen)
149			.map(|(index, _)| self.node_view(NodeId::new(index), fan_out[index]))
150			.collect();
151
152		Ok(PlanView { root: term, nodes })
153	}
154
155	/// The plan of `policy` for the query of `term`.
156	///
157	/// # Errors
158	///
159	/// [`ForeignTerm`](Error::ForeignTerm).
160	pub fn plan_with(&self, term: Term, policy: &(impl Policy + ?Sized)) -> Result<Plan, Error> {
161		Ok(policy.plan(&self.view(term)?))
162	}
163
164	fn node_view(&self, node: NodeId, fan_out: usize) -> NodeView {
165		let (kind, leaf) = match &self.nodes[node] {
166			Node::Leaf(leaf) => {
167				let leaf = &self.leaves[*leaf];
168				let info = LeafInfo {
169					states: leaf.num_states(),
170					transitions: leaf.num_moves(),
171					deterministic: leaf.is_deterministic(),
172					tape_alphabet_sizes: leaf.tape_symbols.iter().map(Vec::len).collect(),
173				};
174
175				(NodeKind::Leaf, Some(info))
176			}
177			Node::Union(..) => (NodeKind::Union, None),
178			Node::Intersection(..) => (NodeKind::Intersection, None),
179			Node::Complement(_) => (NodeKind::Complement, None),
180			Node::Identity(_) => (NodeKind::Identity, None),
181			Node::Project(_, tapes) => (
182				NodeKind::Project {
183					tapes: tapes.to_vec(),
184				},
185				None,
186			),
187			Node::PadClosure(..) => (NodeKind::PadClosure, None),
188			Node::Sync(_, _, spec) => (
189				NodeKind::Sync {
190					pairs: spec.pairs.iter().map(|pair| (pair.lhs, pair.rhs)).collect(),
191					layout: spec.layout.iter().map(|tape| tape.source).collect(),
192				},
193				None,
194			),
195			Node::Explicit(_, options) => (
196				NodeKind::Explicit {
197					max_states: options.max_states,
198				},
199				None,
200			),
201			Node::Prefixes(_) => (NodeKind::Prefixes, None),
202		};
203
204		NodeView {
205			term: self.term(node),
206			kind,
207			arity: self.arities[node],
208			children: self.nodes[node]
209				.children()
210				.map(|child| self.term(child))
211				.collect(),
212			fan_out,
213			pads: self.pads[node],
214			leaf,
215		}
216	}
