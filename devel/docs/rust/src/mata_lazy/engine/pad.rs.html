1//! Pad closures: the words of the child that reach a final state when padding steps follow.
2//!
3//! A macrostate of a pad closure is a [`Tagged`] state of the child, whose branch numbers the step that padding
4//! repeats: always 0 for [`Pad::Step`]; for [`Pad::Last`], 0 before the first step and `i + 1` for the `i`-th
5//! step interned in [`PadSteps`]. A target accepts if the child's accepts, or if a final state is reachable from
6//! it on the padding step alone.
7

19/// The last steps that the macrostates of a [`Pad::Last`] closure remember, and what padding reaches.
20#[derive(Debug, Default)]
21pub(crate) struct PadSteps {
22	steps: IndexSet<Box<[Symbol]>, FxBuildHasher>,
23	/// Whether a final state of the child is reachable from a state on the padding step of a branch.
24	reaches: FxHashMap<(StateId, u32), bool>,
25}
26
27/// Whether a symbolic move label includes a tape reading `step`.
28fn reads(tuple: &[TapeLabel], step: &[Symbol]) -> bool {
29	tuple
30		.iter()
31		.zip(step)
32		.all(|(read, &padded)| match (read, padded) {
33			(TapeLabel::Simple(Symbol::EPSILON), Symbol::ANY)
34			| (TapeLabel::Set(_), Symbol::EPSILON) => false,
35			(_, Symbol::ANY) => true,
36			(TapeLabel::Simple(read), padded) => *read == padded || *read == Symbol::ANY,
37			(TapeLabel::Set(symbols), padded) => symbols.binary_search(&padded).is_ok(),
38		})
39}
40
41impl Engine<'_> {
42	/// The padding step of `branch` of the pad closure `node`, if it has one.
43	fn pad_step(&self, node: ExecId, pad: &Pad, branch: u32) -> Option<Box<[Symbol]>> {
44		match pad {
45			Pad::Step(step) => Some(step.clone()),
46			Pad::Last => {
47				let index = usize::try_from(branch.checked_sub(1)?).expect("a u32 fits in usize");
48
49				Some(self.pad_steps[node].steps[index].clone())
50			}
51		}
52	}
53
54	/// Whether a final state of `child` is reachable from `state` on the padding step of `branch`, which takes
55	/// at least one step.
56	fn pad_reaches(
57		&mut self,
58		node: ExecId,
59		child: ExecId,
60		step: &[Symbol],
61		branch: u32,
62		state: StateId,
63	) -> Result<bool, Limit> {
64		if let Some(&known) = self.pad_steps[node].reaches.get(&(state, branch)) {
65			return Ok(known);
66		}
67
68		let mut seen = FxHashSet::default();
69		seen.insert(state);
70		let mut stack = vec![state];
71		while let Some(from) = stack.pop() {
72			for (tuple, target) in self.post(child, from)?.iter() {
73				if !reads(tuple, step) {
74					continue;
75				}
76
77				if target.accepting {
78					self.pad_steps[node].reaches.insert((state, branch), true);
79
80					return Ok(true);
81				}
82
83				if seen.insert(target.state) {
84					stack.push(target.state);
85				}
86			}
87		}
88
89		// Nothing reachable from the states seen accepts on the step either.
90		let reaches = &mut self.pad_steps[node].reaches;
91		reaches.extend(seen.into_iter().map(|seen| ((seen, branch), false)));
92
93		Ok(false)
94	}
95
96	/// The macrostate of `node` for the child's `target` in `branch`.
97	pub(super) fn pad_target(
98		&mut self,
99		node: ExecId,
100		child: ExecId,
101		pad: &Pad,
102		branch: u32,
103		target: Target,
104	) -> Result<Target, Limit> {
105		let accepting = target.accepting
106			|| match self.pad_step(node, pad, branch) {
107				Some(step) => self.pad_reaches(node, child, &step, branch, target.state)?,
108				None => false,
109			};
110		let state = self.stores[node].intern_tagged(Tagged {
111			branch,
112			state: target.state,
113		});
114
115		Ok(Target { state, accepting })
116	}
117
118	pub(super) fn pad_closure_post(
119		&mut self,
120		node: ExecId,
121		child: ExecId,
122		pad: &Pad,
123		state: StateId,
124	) -> Result<Post, Limit> {
125		let Tagged { branch, state } = self.stores[node].tagged(state);
126		let child_post = self.post(child, state)?;
127		let arity = self.exec.nodes[node].arity;
128		let start = self.stores[node].len();
129		let mut post = Post::with_capacity(arity, child_post.len());
130		match pad {
131			Pad::Step(_) => {
132				for (tuple, target) in child_post.iter() {
133					post.push_labels(tuple, self.pad_target(node, child, pad, branch, target)?);
134				}
135			}
136			Pad::Last => {
137				// Last-step reachability branches on exact steps, so symbolic dimensions expand incrementally.
138				let universes: Vec<_> = (0..arity)
139					.map(|tape| self.alphabets.universe(node, tape).clone())
140					.collect();
141				for (tuple, target) in child_post.iter() {
142					for_each_concrete(&universes, tuple, |step| {
143						self.budget.step(self.stores[node].len() - start);
144						self.stopped()?;
145						let index = self.pad_steps[node].steps.insert_full(step.into()).0;
146						let branch = u32::try_from(index + 1).expect("fewer than 2^32 steps");
147						post.push(step, self.pad_target(node, child, pad, branch, target)?);
148
149						Ok(())
150					})?;
151				}
152
153				post.dedup();
154			}
155		}
156
157		self.budget.add(self.stores[node].len() - start);
158
159		Ok(post)
160	}
161}
