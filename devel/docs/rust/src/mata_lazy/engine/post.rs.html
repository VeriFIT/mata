41/// One tape of a move. A set denotes every concrete symbol in it, sorted without repetitions.
42#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
43pub(crate) enum TapeLabel {
44	Simple(Symbol),
45	Set(Rc<[Symbol]>),
46}
47
48impl TapeLabel {
49	fn set(mut symbols: Vec<Symbol>) -> Option<TapeLabel> {
50		symbols.sort_unstable();
51		symbols.dedup();
52		debug_assert!(symbols.iter().all(|symbol| symbol.is_concrete()));
53
54		(!symbols.is_empty()).then(|| TapeLabel::Set(symbols.into()))
55	}
56
57	pub(super) fn symbol(&self) -> Symbol {
58		match self {
59			TapeLabel::Simple(symbol) => *symbol,
60			TapeLabel::Set(_) => panic!("a concrete-label join received a set"),
61		}
62	}
63
64	/// The concrete `position`-th symbol, where padding and the wildcard each have at most one expansion.
65	pub(super) fn symbol_at(&self, universe: &[Symbol], position: usize) -> Option<Symbol> {
66		match self {
67			TapeLabel::Simple(Symbol::ANY) => universe.get(position).copied(),
68			TapeLabel::Simple(symbol) => (position == 0).then_some(*symbol),
69			TapeLabel::Set(symbols) => symbols.get(position).copied(),
70		}
71	}
72
73	fn intersects(&self, other: &TapeLabel) -> bool {
74		match (self, other) {
75			(TapeLabel::Simple(a), TapeLabel::Simple(b)) => merge(*a, *b).is_some(),
76			(TapeLabel::Simple(Symbol::ANY), TapeLabel::Set(_))
77			| (TapeLabel::Set(_), TapeLabel::Simple(Symbol::ANY)) => true,
78			(TapeLabel::Simple(symbol), TapeLabel::Set(symbols))
79			| (TapeLabel::Set(symbols), TapeLabel::Simple(symbol)) => {
80				symbol.is_concrete() && symbols.binary_search(symbol).is_ok()
81			}
82			(TapeLabel::Set(a), TapeLabel::Set(b)) => {
83				let (short, long) = if a.len() <= b.len() { (a, b) } else { (b, a) };
84				short
85					.iter()
86					.any(|symbol| long.binary_search(symbol).is_ok())
87			}
88		}
89	}
90
91	fn intersection(&self, other: &TapeLabel) -> Option<TapeLabel> {
92		match (self, other) {
93			(TapeLabel::Simple(a), TapeLabel::Simple(b)) => merge(*a, *b).map(TapeLabel::Simple),
94			(TapeLabel::Simple(Symbol::ANY), TapeLabel::Set(symbols))
95			| (TapeLabel::Set(symbols), TapeLabel::Simple(Symbol::ANY)) => {
96				Some(TapeLabel::Set(Rc::clone(symbols)))
97			}
98			(TapeLabel::Simple(symbol), TapeLabel::Set(symbols))
99			| (TapeLabel::Set(symbols), TapeLabel::Simple(symbol)) => (symbol.is_concrete()
100				&& symbols.binary_search(symbol).is_ok())
101			.then_some(TapeLabel::Simple(*symbol)),
102			(TapeLabel::Set(a), TapeLabel::Set(b)) => {
103				let mut symbols = Vec::new();
104				let (mut i, mut j) = (0, 0);
105				while i < a.len() && j < b.len() {
106					match a[i].cmp(&b[j]) {
107						std::cmp::Ordering::Less => i += 1,
108						std::cmp::Ordering::Greater => j += 1,
109						std::cmp::Ordering::Equal => {
110							symbols.push(a[i]);
111							i += 1;
112							j += 1;
113						}
114					}
115				}
116
117				TapeLabel::set(symbols)
118			}
119		}
120	}
121
122	/// Splits `part`, which this label overlaps, into the covered and uncovered concrete symbols.
123	fn split(&self, part: &[Symbol]) -> (Vec<Symbol>, Vec<Symbol>) {
124		match self {
125			TapeLabel::Simple(Symbol::ANY) => (part.to_vec(), Vec::new()),
126			TapeLabel::Simple(Symbol::EPSILON) => (Vec::new(), part.to_vec()),
127			TapeLabel::Simple(symbol) => {
128				let mut covered = Vec::new();
129				let mut uncovered = Vec::new();
130				for &letter in part {
131					if letter == *symbol {
132						covered.push(letter);
133					} else {
134						uncovered.push(letter);
135					}
136				}
137
138				(covered, uncovered)
139			}
140			TapeLabel::Set(symbols) => {
141				let mut covered = Vec::new();
142				let mut uncovered = Vec::new();
143				let mut next = 0;
144				for &letter in part {
145					while next < symbols.len() && symbols[next] < letter {
146						next += 1;
147					}
148					if next < symbols.len() && symbols[next] == letter {
149						covered.push(letter);
150					} else {
151						uncovered.push(letter);
152					}
153				}
154
155				(covered, uncovered)
156			}
157		}
158	}
159}

161/// Calls `visit` for each concrete tuple of `labels`, without collecting their Cartesian product.
162pub(super) fn for_each_concrete(
163	universes: &[Rc<[Symbol]>],
164	labels: &[TapeLabel],
165	mut visit: impl FnMut(&[Symbol]) -> Result<(), Limit>,
166) -> Result<(), Limit> {
167	debug_assert_eq!(universes.len(), labels.len());
168	let mut positions = vec![0; labels.len()];
169	let mut tuple: Vec<Symbol> = match labels
170		.iter()
171		.zip(universes)
172		.map(|(label, universe)| label.symbol_at(universe, 0))
173		.collect()
174	{
175		Some(tuple) => tuple,
176		None => return Ok(()),
177	};
178
179	loop {
180		visit(&tuple)?;
181		let mut wrapped = true;
182		for tape in (0..labels.len()).rev() {
183			positions[tape] += 1;
184			if let Some(symbol) = labels[tape].symbol_at(&universes[tape], positions[tape]) {
185				tuple[tape] = symbol;
186				wrapped = false;
187				break;
188			}
189
190			positions[tape] = 0;
191			tuple[tape] = labels[tape]
192				.symbol_at(&universes[tape], 0)
193				.expect("the first concrete tuple exists");
194		}
195
196		if wrapped {
197			return Ok(());
198		}
199	}
200}

245	/// The same moves, each target accepting as `accepting` decides, or the limit that it passed.
246	pub(super) fn reflagged(
247		&self,
248		mut accepting: impl FnMut(Target) -> Result<bool, Limit>,
249	) -> Result<Post, Limit> {
250		let targets = (0..self.len())
251			.map(|index| {

254				Ok(target.state.pack(accepting(target)?))
255			})

301	/// Pushes a move to `pair` unless the post has it; `seen` holds the index and pair of each move.
302	fn push_new(
303		&mut self,
304		seen: &mut HashTable<(u32, Pair)>,
305		store: &mut Store,
306		tuple: &[TapeLabel],
307		pair: Pair,
308		accepting: bool,
309	) {
310		let hash = FxBuildHasher.hash_one((tuple, pair));
311		let same =
312			|&(index, other): &(u32, Pair)| other == pair && self.tuple(index as usize) == tuple;
313		if seen.find(hash, same).is_some() {
314			return;
315		}
316
317		let rehash = |&(index, other): &(u32, Pair)| {
318			FxBuildHasher.hash_one((self.tuple(index as usize), other))
319		};
320		let index = index_u32(self.len());
321		seen.insert_unique(hash, (index, pair), rehash);
322
323		let state = store.intern_pair(pair);
324		self.push_labels(tuple, Target { state, accepting });
325	}

339	/// Sorts the moves by exact structural label and target, without repeats.
340	pub(super) fn dedup(&mut self) {
341		let key = |index: usize| (self.tuple(index), self.state(index));
342		let mut order: Vec<usize> = (0..self.len()).collect();
343		order.sort_unstable_by_key(|&index| key(index));
344		order.dedup_by_key(|index| key(*index));
345
346		let mut unique = Post::with_capacity(self.arity, order.len());
347		for index in order {
348			unique.push_labels(self.tuple(index), self.target(index));
349		}
350
351		*self = unique;
352	}

358/// The moves of a post grouped by their letters on some tapes, to join without trying every pair.
359#[derive(Debug)]
360struct ByKey {
361	/// The keys, sorted; the moves of `keys[i]` are `moves[starts[i]..starts[i + 1]]`, in increasing order.
362	keys: Vec<u64>,
363	starts: Vec<u32>,
364	moves: Vec<u32>,
365	/// The moves that read the wildcard on a key tape.
366	any: Vec<u32>,
367	len: usize,
368}
369
370impl ByKey {
371	fn new(post: &Post, tapes: &[usize]) -> ByKey {
372		let mut keyed: Vec<(u64, u32)> = Vec::with_capacity(post.len());
373		let mut any = Vec::new();
374		for (index, (tuple, _)) in (0_u32..).zip(post.iter()) {
375			match key(tapes.iter().map(|&tape| tuple[tape].symbol())) {
376				Some(key) => keyed.push((key, index)),
377				None => any.push(index),
378			}
379		}
380
381		keyed.sort_unstable();
382		let mut keys = Vec::new();
383		let mut starts = Vec::new();
384		for (at, &(key, _)) in (0_u32..).zip(&keyed) {
385			if keys.last() != Some(&key) {
386				keys.push(key);
387				starts.push(at);
388			}
389		}
390
391		starts.push(index_u32(keyed.len()));
392
393		ByKey {
394			keys,
395			starts,
396			moves: keyed.into_iter().map(|(_, index)| index).collect(),
397			any,
398			len: post.len(),
399		}
400	}
401
402	fn candidates(&self, letters: impl Iterator<Item = Symbol>, mut visit: impl FnMut(usize)) {
403		let Some(key) = key(letters) else {
404			(0..self.len).for_each(visit);
405
406			return;
407		};
408
409		if let Ok(at) = self.keys.binary_search(&key) {
410			let range = self.starts[at] as usize..self.starts[at + 1] as usize;
411			for &index in &self.moves[range] {
412				visit(index as usize);
413			}
414		}
415
416		for &index in &self.any {
417			visit(index as usize);
418		}
419	}
420}

422/// `letters` folded into one number, `None` if one is the wildcard; past two letters, keys may collide.
423fn key(mut letters: impl Iterator<Item = Symbol>) -> Option<u64> {
424	letters.try_fold(0_u64, |key, letter| {
425		(letter != Symbol::ANY).then(|| key.rotate_left(32) ^ u64::from(letter.value()))
426	})
427}

429#[derive(Clone, Copy)]
430enum Product<'s> {
431	Intersection,
432	Sync(&'s SyncSpec),
433	Diagonal,
434}
435
436impl Product<'_> {
437	/// Writes the symbolic tuple that `lhs` and `rhs` read together into `out`, if they overlap.
438	fn merge(self, lhs: &[TapeLabel], rhs: &[TapeLabel], out: &mut Vec<TapeLabel>) -> bool {
439		out.clear();
440		match self {
441			Product::Intersection => {
442				for (a, b) in lhs.iter().zip(rhs) {
443					let Some(label) = a.intersection(b) else {
444						return false;
445					};
446
447					out.push(label);
448				}
449			}
450			Product::Sync(spec) => {
451				if !spec
452					.pairs
453					.iter()
454					.all(|pair| lhs[pair.lhs].intersects(&rhs[pair.rhs]))
455				{
456					return false;
457				}
458
459				for tape in &spec.layout {
460					let label = match (tape.source, tape.peer) {
461						(Tape::Lhs(i), None) => lhs[i].clone(),
462						(Tape::Rhs(j), None) => rhs[j].clone(),
463						(Tape::Lhs(i), Some(j)) | (Tape::Rhs(j), Some(i)) => lhs[i]
464							.intersection(&rhs[j])
465							.expect("every synchronized pair overlaps"),
466					};
467
468					out.push(label);
469				}
470			}
471			Product::Diagonal => {
472				let Some(label) = lhs[0]
473					.intersection(&rhs[0])
474					.and_then(|label| label.intersection(&rhs[1]))
475				else {
476					return false;
477				};
478
479				out.push(label);
480			}
481		}
482
483		true
484	}
485}

487/// How a product finds the moves of its right operand that may merge with a left move.
488enum Candidates {
489	ByKey(ByKey),
490	/// Every move of the right operand.
491	All(usize),
492}
493
494impl Candidates {
495	/// The candidates of `right` on `tapes`, indexed by key if `indexed` and wholly simple.
496	fn new(right: &Post, tapes: &[usize], indexed: bool) -> Candidates {
497		if indexed {
498			Candidates::ByKey(ByKey::new(right, tapes))
499		} else {
500			Candidates::All(right.len())
501		}
502	}
503
504	/// Visits the moves that may merge with `letters`, and maybe others; the caller merges them.
505	fn visit(&mut self, letters: impl Iterator<Item = Symbol>, visit: impl FnMut(usize)) {
506		match self {
507			Candidates::ByKey(by_key) => by_key.candidates(letters, visit),
508			Candidates::All(len) => (0..*len).for_each(visit),
509		}
510	}
511}

522impl Engine<'_> {
523	/// The moves from `state` of `node`, or the limit that the query passed while computing them.
524	pub(crate) fn post(&mut self, node: ExecId, state: StateId) -> Result<Rc<Post>, Limit> {
525		if let Op::Transparent(child) = self.exec.nodes[node].op {
526			return self.post(child, state);
527		}
528
529		let index = state.index();
530		if let Some(Some(post)) = self.posts[node].get(index) {
531			if let Some(budget) = &mut self.cache_budget {
532				budget.touch(node.index(), state);
533			}
534
535			return Ok(Rc::clone(post));
536		}
537
538		let (post, nanos) = self.timed(|engine| deep(|| engine.compute_post(node, state)));
539		self.post_counts[node] += 1;
540		// Past a limit the post may be partial, so it is neither kept nor used.
541		let mut post = post?;
542		self.stopped()?;
543		if !self.cached[node] {
544			return Ok(Rc::new(post));
545		}
546
547		post.shrink_to_fit();
548		let post = Rc::new(post);
549		let slots = &mut self.posts[node];
550		if slots.len() <= index {
551			slots.resize(index + 1, None);
552		}
553
554		slots[index] = Some(Rc::clone(&post));
555		if let Some(budget) = &mut self.cache_budget {
556			let dropped = budget.admit(node.index(), state, nanos, &post);
557			self.evict(dropped);
558		}
559
560		Ok(post)
561	}
562
563	/// Drops the posts that the cache budget evicted.
564	pub(super) fn evict(&mut self, dropped: Vec<(usize, StateId)>) {
565		for (key, state) in dropped {
566			let removed = self.posts[ExecId::new(key)][state.index()].take();
567			if let (Some(budget), Some(post)) = (&mut self.cache_budget, &removed) {
568				budget.retire(post);
569			}
570		}
571		if let Some(budget) = &mut self.cache_budget {
572			budget.reclaim();
573		}
574	}
575
576	/// Runs `work` and, under a cache budget, times it without the posts it computed, which are cached on their own.
577	pub(super) fn timed<T>(&mut self, work: impl FnOnce(&mut Self) -> T) -> (T, u64) {
578		if self.cache_budget.is_none() {
579			return (work(self), 0);
580		}
581
582		let start = Instant::now();
583		let outer = self.nested_nanos;
584		self.nested_nanos = 0;
585		let result = work(self);
586		let total = u64::try_from(start.elapsed().as_nanos()).unwrap_or(u64::MAX);
587		let own = total.saturating_sub(self.nested_nanos);
588		self.nested_nanos = outer.saturating_add(total);
589
590		(result, own)
591	}

593	fn compute_post(&mut self, node: ExecId, state: StateId) -> Result<Post, Limit> {
594		match self.exec.nodes[node].op.clone() {
595			Op::Leaf(leaf) => Ok(self.leaf_post(node, &leaf, state)),
596			Op::Union(branches) => self.union_post(node, &branches, state),
597			Op::Intersection(a, b) => self.product_post(node, a, b, state, Product::Intersection),
598			Op::Sync(a, b, spec) => self.product_post(node, a, b, state, Product::Sync(&spec)),
599			Op::Diagonal(u, x) => self.product_post(node, u, x, state, Product::Diagonal),
600			Op::Complement(child) => self.complement_post(node, child, state),
601			Op::Identity(child) => self.identity_post(node, child, state),
602			Op::Project(child, tapes) => self.project_post(node, child, &tapes, state),
603			Op::PadClosure(child, pad) => self.pad_closure_post(node, child, &pad, state),
604			Op::Prefixes(child) => self.prefixes_post(node, child, state),
605			Op::Explicit { .. } | Op::Transparent(_) => {
606				unreachable!("explicit nodes are replaced before the search")
607			}
608		}
609	}

611	/// The leaf's moves, except those that read the wildcard on a tape with an empty universe.
612	fn leaf_post(&self, node: ExecId, leaf: &Leaf, state: StateId) -> Post {
613		let arity = self.exec.nodes[node].arity;
614		let dead: Vec<usize> = (0..arity)
615			.filter(|&tape| {
616				leaf.tape_has_any[tape] && self.alphabets.universe(node, tape).is_empty()
617			})
618			.collect();
619
620		let mut post = Post::with_capacity(arity, leaf.moves(state).len());
621		for (tuple, target) in leaf.moves(state) {
622			if dead.iter().any(|&tape| tuple[tape] == Symbol::ANY) {
623				continue;
624			}
625
626			post.push(tuple, Target::of_leaf(leaf, target));
627		}
628
629		post
630	}

632	fn union_post(
633		&mut self,
634		node: ExecId,
635		branches: &[ExecId],
636		state: StateId,
637	) -> Result<Post, Limit> {
638		let Tagged { branch, state } = self.stores[node].tagged(state);
639		let child = self.post(branches[branch as usize], state)?;
640		let start = self.stores[node].len();
641		let mut post = Post::with_capacity(self.exec.nodes[node].arity, child.len());
642		for (tuple, target) in child.iter() {
643			post.push_labels(tuple, self.tagged_target(node, branch, target));
644		}
645
646		self.budget.add(self.stores[node].len() - start);
647
648		Ok(post)
649	}

651	pub(super) fn tagged_target(&mut self, node: ExecId, branch: u32, target: Target) -> Target {
652		let state = self.stores[node].intern_tagged(Tagged {
653			branch,
654			state: target.state,
655		});
656
657		Target { state, ..target }
658	}

660	/// The child's moves with each concrete letter on both tapes. Sets expand on one tape, never as a product.
661	fn identity_post(
662		&mut self,
663		node: ExecId,
664		child: ExecId,
665		state: StateId,
666	) -> Result<Post, Limit> {
667		let child_post = self.post(child, state)?;
668		let universe = self.alphabets.universe(node, 0);
669		let mut post = Post::with_capacity(2, child_post.len());
670		for (tuple, target) in child_post.iter() {
671			match &tuple[0] {
672				TapeLabel::Simple(Symbol::ANY) => {
673					for &symbol in universe.iter() {
674						post.push(&[symbol, symbol], target);
675					}
676				}
677				TapeLabel::Set(symbols) => {
678					for &symbol in symbols.iter() {
679						post.push(&[symbol, symbol], target);
680					}
681				}
682				TapeLabel::Simple(symbol) => post.push(&[*symbol, *symbol], target),
683			}
684		}
685
686		Ok(post)
687	}

689	fn project_post(
690		&mut self,
691		node: ExecId,
692		child: ExecId,
693		tapes: &[usize],
694		state: StateId,
695	) -> Result<Post, Limit> {
696		let arity = self.exec.nodes[node].arity;
697		let child_post = self.post(child, state)?;
698		let mut post = Post::with_capacity(arity, child_post.len());
699		let mut projected = Vec::with_capacity(arity);
700		for (tuple, target) in child_post.iter() {
701			projected.clear();
702			projected.extend(tapes.iter().map(|&tape| tuple[tape].clone()));
703			post.push_labels(&projected, target);
704		}
705
706		post.dedup();
707
708		Ok(post)
709	}

711	fn product_post(
712		&mut self,
713		node: ExecId,
714		lhs: ExecId,
715		rhs: ExecId,
716		state: StateId,
717		product: Product<'_>,
718	) -> Result<Post, Limit> {
719		let pair = self.stores[node].pair(state);
720		let (left, right) = (self.post(lhs, pair.lhs)?, self.post(rhs, pair.rhs)?);
721		let plan = &self.plans[node];
722		let simple = left.wholly_simple() && right.wholly_simple();
723		let indexed =
724			simple && !plan.keys.is_empty() && left.len() > 1 && right.len() > INDEX_THRESHOLD;
725
726		if simple
727			&& let Product::Sync(spec) = product
728			&& let Some(post) = self.group_join(node, lhs, &left, &right, spec, indexed)
729		{
730			return Ok(post);
731		}
732
733		Ok(self.pair_join(node, &left, &right, product, indexed))
734	}

736	/// The post by the group join, where the plan allows it and it looks cheaper than the pair join.
737	fn group_join(
738		&mut self,
739		node: ExecId,
740		lhs: ExecId,
741		left: &Post,
742		right: &Rc<Post>,
743		spec: &SyncSpec,
744		indexed: bool,
745	) -> Option<Post> {
746		let plan = self.plans[node].group.as_ref()?;
747		let index = plan
748			.index
749			.get_or_init(|| LetterIndex::new(self.alphabets.universe(lhs, plan.hidden.lhs)))
750			.as_ref()?;
751		let join = GroupJoin {
752			left,
753			right,
754			spec,
755			plan,
756			index,
757			indexed,
758		};
759
760		join.run(
761			&mut self.stores[node],
762			&mut self.budget,
763			&mut self.join_scratch,
764		)
765	}

767	/// Merges each left move with the right moves that may merge with it.
768	fn pair_join(
769		&mut self,
770		node: ExecId,
771		left: &Post,
772		right: &Post,
773		product: Product<'_>,
774		indexed: bool,
775	) -> Post {
776		let plan = &self.plans[node];
777		// Sized for one move per left move, as a join that drops tapes has at least as many.
778		let mut seen = plan
779			.dedup
780			.then(|| HashTable::with_capacity(left.len().min(SEEN_CAPACITY)));
781		let mut post = Post::new(self.exec.nodes[node].arity);
782		let mut candidates = Candidates::new(right, &plan.right_tapes, indexed);
783		let store = &mut self.stores[node];
784		let start = store.len();
785
786		for (left_tuple, left_target) in left.iter() {
787			self.budget.step(store.len() - start);
788			if self.budget.passed() {
789				break;
790			}
791
792			let letters = plan.keys.iter().map(|pair| left_tuple[pair.lhs].symbol());
793			candidates.visit(letters, |index| {
794				if !product.merge(left_tuple, right.tuple(index), &mut self.join_buffers.tuple) {
795					return;
796				}
797
798				let right_target = right.target(index);
799				let pair = Pair {
800					lhs: left_target.state,
801					rhs: right_target.state,
802				};
803				let accepting = left_target.accepting && right_target.accepting;
804				if let Some(seen) = &mut seen {
805					post.push_new(seen, store, &self.join_buffers.tuple, pair, accepting);
806				} else {
807					let state = store.intern_pair(pair);
808					post.push_labels(&self.join_buffers.tuple, Target { state, accepting });
809				}
810			});
811		}
812
813		self.budget.add(store.len() - start);
814
815		post
816	}

818	/// Partitions the concrete tuples into symbolic Cartesian regions, each to its minimized successor set.
819	fn complement_post(
820		&mut self,
821		node: ExecId,
822		child: ExecId,
823		state: StateId,
824	) -> Result<Post, Limit> {
825		let arity = self.exec.nodes[node].arity;
826		let members = self.stores[node].set(state).to_vec();
827		let universes: Vec<Rc<[Symbol]>> = (0..arity)
828			.map(|tape| Rc::clone(self.alphabets.universe(node, tape)))
829			.collect();
830		if universes.iter().any(|universe| universe.is_empty()) {
831			return Ok(Post::new(arity));
832		}
833
834		let posts = members
835			.iter()
836			.map(|&member| self.post(child, member))
837			.collect::<Result<Vec<Rc<Post>>, Limit>>()?;
838		let mut region: Vec<TapeLabel> = universes.into_iter().map(TapeLabel::Set).collect();
839		let active: Vec<ActiveMove> = posts
840			.iter()
841			.flat_map(|post| post.iter())
842			.filter(|(tuple, _)| !tuple.contains(&TapeLabel::Simple(Symbol::EPSILON)))
843			.filter(|(tuple, _)| {
844				tuple
845					.iter()
846					.zip(&region)
847					.all(|(label, region)| label.intersects(region))
848			})
849			.map(|(tuple, target)| ActiveMove { tuple, target })
850			.collect();
851
852		let mut post = Post::new(arity);
853		let start = self.stores[node].len();
854		self.complement_region(node, &mut region, &active, start, &mut post, 0)?;
855		self.budget.add(self.stores[node].len() - start);
856
857		Ok(post)
858	}

860	/// Splits one tape at a time into the minterms of the active moves; no product of tape minterms is built.
861	fn complement_region(
862		&mut self,
863		node: ExecId,
864		region: &mut [TapeLabel],
865		active: &[ActiveMove],
866		start: usize,
867		post: &mut Post,
868		next_tape: usize,
869	) -> Result<(), Limit> {
870		self.budget.step(self.stores[node].len() - start);
871		if self.budget.passed() {
872			return Ok(());
873		}
874		let Op::Complement(child) = self.exec.nodes[node].op else {
875			unreachable!("only complements partition concrete regions")
876		};
877
878		// Earlier tapes are already disjoint minterms of every active move, so they cannot split again.
879		if !active.is_empty() {
880			for tape in next_tape..region.len() {
881				let TapeLabel::Set(current) = &region[tape] else {
882					unreachable!("a complement region is a concrete set on every tape")
883				};
884				let mut parts = vec![Rc::clone(current)];
885				for active_move in active {
886					let mut next = Vec::with_capacity(parts.len() + 1);
887					for part in &parts {
888						let (covered, uncovered) = active_move.tuple[tape].split(part);
889						if covered.is_empty() || uncovered.is_empty() {
890							next.push(Rc::clone(part));
891							continue;
892						}
893
894						next.push(covered.into());
895						next.push(uncovered.into());
896					}
897					parts = next;
898				}
899
900				if parts.len() == 1 {
901					continue;
902				}
903
904				let previous = region[tape].clone();
905				for part in parts {
906					region[tape] = TapeLabel::Set(part);
907					let next: Vec<ActiveMove> = active
908						.iter()
909						.filter(|active_move| active_move.tuple[tape].intersects(&region[tape]))
910						.copied()
911						.collect();
912					deep(|| self.complement_region(node, region, &next, start, post, tape + 1))?;
913					if self.budget.passed() {
914						break;
915					}
916				}
917				region[tape] = previous;
918
919				return Ok(());
920			}
921		}
922
923		let mut successors = Vec::with_capacity(active.len());
924		let mut accepting = true;
925		for active_move in active {
926			successors.push(active_move.target.state);
927			accepting &= !active_move.target.accepting;
928		}
929		self.subsumption
930			.minimize(&self.exec, &self.stores, child, &mut successors);
931		let state = self.stores[node].intern_set(&successors);
932		post.push_labels(region, Target { state, accepting });
933
934		Ok(())
935	}
