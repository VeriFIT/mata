66	/// The digit that each tape reads.
67	tapes: Box<[usize]>,
68	len: usize,

71/// One digit of a [`Table`], a position in a universe.
72#[derive(Clone, Debug)]
73struct Digit {
74	universe: Rc<[Symbol]>,
75	/// How many tuples one step of this digit skips.
76	stride: usize,
77	/// The value of the first symbol of the universe.
78	first: u32,
79	/// The position of each symbol by its value minus `first` (the largest `u32` if absent); empty if too sparse.
80	positions: Box<[u32]>,
81}
82
83impl Digit {
84	fn new(universe: &Rc<[Symbol]>, stride: usize) -> Digit {
85		let (first, last) = match (universe.first(), universe.last()) {
86			(Some(first), Some(last)) => (first.value(), last.value()),
87			_ => (0, 0),
88		};
89
90		let span = (last - first) as usize + 1;
91
92		let positions = if universe.len() > 1 && span <= 4 * universe.len() {
93			let mut positions = vec![u32::MAX; span];
94			for (position, letter) in (0_u32..).zip(universe.iter()) {
95				positions[(letter.value() - first) as usize] = position;
96			}
97
98			positions.into()
99		} else {
100			Box::default()
101		};
102
103		Digit {
104			universe: Rc::clone(universe),
105			stride,
106			first,
107			positions,
108		}
109	}
110
111	fn position(&self, letter: Symbol) -> Option<usize> {
112		if self.positions.is_empty() {
113			return self.universe.binary_search(&letter).ok();
114		}
115
116		let offset = letter.value().checked_sub(self.first)? as usize;
117		let position = *self.positions.get(offset)?;
118
119		(position != u32::MAX).then_some(position as usize)
120	}
121}

123/// Tuples of a table built between two checks of the query's limits.
124const TABLE_CHUNK: usize = 1 << 10;
125
126/// The most tuples of a table allocated up front, so that a table the limits cut short is not allocated whole.
127const TABLE_RESERVED: usize = 1 << 16;
128
129impl Table {
130	/// The table of `len` tuples over `universes`; past a limit of `budget` it stops, short.
131	fn new(universes: &[Rc<[Symbol]>], len: usize, budget: &mut Budget) -> Table {
132		let mut letters = Vec::with_capacity(len.min(TABLE_RESERVED) * universes.len());
133		let mut positions = vec![0; universes.len()];
134		for start in (0..len).step_by(TABLE_CHUNK) {
135			budget.step(0);
136			if budget.passed() {
137				break;
138			}
139
140			for _ in start..len.min(start + TABLE_CHUNK) {
141				letters.extend(
142					positions
143						.iter()
144						.zip(universes)
145						.map(|(&position, universe)| universe[position]),
146				);
147				for (position, universe) in positions.iter_mut().zip(universes).rev() {
148					*position += 1;
149					if *position < universe.len() {
150						break;
151					}
152
153					*position = 0;
154				}
155			}
156		}
157
158		let mut strides = vec![1_usize; universes.len()];
159		for tape in (1..universes.len()).rev() {
160			// Saturates only for a table without tuples, which never reads its strides.
161			strides[tape - 1] = strides[tape].saturating_mul(universes[tape].len());
162		}
163
164		Table {
165			letters: letters.into(),
166			digits: universes
167				.iter()
168				.zip(strides)
169				.map(|(universe, stride)| Digit::new(universe, stride))
170				.collect(),
171			tapes: (0..universes.len()).collect(),
172			len,
173		}
174	}
175
176	/// The table of the identity of a post over this table, with each letter on both tapes.
177	fn diagonal(&self) -> Table {
178		debug_assert_eq!(self.tapes.len(), 1);
179
180		Table {
181			letters: self
182				.letters
183				.iter()
184				.flat_map(|&letter| [letter, letter])
185				.collect(),
186			digits: self.digits.clone(),
187			tapes: [0, 0].into(),
188			len: self.len,
189		}
190	}
191
192	/// The index of a concrete tuple of a complement's table, if each letter is in its universe.
193	// Called for every move of the members' posts, so kept inline in the complement's post.
194	#[inline]
195	fn index(&self, tuple: &[Symbol]) -> Option<usize> {
196		debug_assert_eq!(tuple.len(), self.digits.len());
197
198		tuple
199			.iter()
200			.zip(&self.digits)
201			.try_fold(0, |index, (&letter, digit)| {
202				Some(index + digit.position(letter)? * digit.stride)
203			})
204	}

206	/// Visits, in increasing order, the tuples whose letters on `tapes` merge with `letters`.
207	pub(super) fn matching(
208		&self,
209		tapes: &[usize],
210		letters: impl Iterator<Item = Symbol>,
211		odometer: &mut Odometer,
212		mut visit: impl FnMut(usize),
213	) {
214		const FREE: usize = usize::MAX;
215
216		if self.len == 0 {
217			return;
218		}
219
220		odometer.positions.clear();
221		odometer.positions.resize(self.digits.len(), FREE);
222		for (&tape, letter) in tapes.iter().zip(letters) {
223			if letter == Symbol::ANY {
224				continue;
225			}
226
227			let digit = self.tapes[tape];
228			let Some(position) = self.digits[digit].position(letter) else {
229				return;
230			};
231
232			if odometer.positions[digit] != FREE && odometer.positions[digit] != position {
233				return;
234			}
235
236			odometer.positions[digit] = position;
237		}
238
239		// The free digits after the last fixed one vary fastest, so their tuples are consecutive.
240		let fixed_end = odometer
241			.positions
242			.iter()
243			.rposition(|&position| position != FREE)
244			.map_or(0, |digit| digit + 1);
245		let run = match fixed_end {
246			0 => self.len,
247			end => self.digits[end - 1].stride,
248		};
249
250		let mut start: usize = (0..fixed_end)
251			.filter(|&digit| odometer.positions[digit] != FREE)
252			.map(|digit| odometer.positions[digit] * self.digits[digit].stride)
253			.sum();
254
255		// The other free digits count in `positions`, from 0.
256		odometer.free.clear();
257		odometer
258			.free
259			.extend((0..fixed_end).filter(|&digit| odometer.positions[digit] == FREE));
260		for &digit in &odometer.free {
261			odometer.positions[digit] = 0;
262		}
263
264		loop {
265			(start..start + run).for_each(&mut visit);
266
267			// The free digits step like an odometer, and the tuples end when they all wrap.
268			let mut wrapped = true;
269			for &digit in odometer.free.iter().rev() {
270				odometer.positions[digit] += 1;
271				start += self.digits[digit].stride;
272				if odometer.positions[digit] < self.digits[digit].universe.len() {
273					wrapped = false;
274
275					break;
276				}
277
278				start -= odometer.positions[digit] * self.digits[digit].stride;
279				odometer.positions[digit] = 0;
280			}
281
282			if wrapped {
283				return;
284			}
285		}
286	}
287}

405	/// Pushes a move to `pair` unless the post has it; `seen` holds the index and pair of each move.
406	fn push_new(
407		&mut self,
408		seen: &mut HashTable<(u32, Pair)>,
409		store: &mut Store,
410		tuple: &[Symbol],
411		pair: Pair,
412		accepting: bool,
413	) {
414		let hash = FxBuildHasher.hash_one((tuple, pair));
415		let same =
416			|&(index, other): &(u32, Pair)| other == pair && self.tuple(index as usize) == tuple;
417		if seen.find(hash, same).is_some() {
418			return;
419		}
420
421		let rehash = |&(index, other): &(u32, Pair)| {
422			FxBuildHasher.hash_one((self.tuple(index as usize), other))
423		};
424		let index = index_u32(self.len());
425		seen.insert_unique(hash, (index, pair), rehash);
426
427		let state = store.intern_pair(pair);
428		self.push(tuple, Target { state, accepting });
429	}

444	/// Sorts the moves by tuple and target, without repeats.
445	pub(super) fn dedup(&mut self) {
446		let key = |index: usize| (self.tuple(index), self.state(index));
447		let mut order: Vec<usize> = (0..self.len()).collect();
448		order.sort_unstable_by_key(|&index| key(index));
449		order.dedup_by_key(|index| key(*index));
450
451		let mut unique = Post::with_capacity(self.arity, order.len());
452		for index in order {
453			unique.push(self.tuple(index), self.target(index));
454		}
455
456		*self = unique;
457	}

463/// The moves of a post grouped by their letters on some tapes, to join without trying every pair.
464#[derive(Debug)]
465struct ByKey {
466	/// The keys, sorted; the moves of `keys[i]` are `moves[starts[i]..starts[i + 1]]`, in increasing order.
467	keys: Vec<u64>,
468	starts: Vec<u32>,
469	moves: Vec<u32>,
470	/// The moves that read the wildcard on a key tape.
471	any: Vec<u32>,
472	len: usize,
473}
474
475impl ByKey {
476	fn new(post: &Post, tapes: &[usize]) -> ByKey {
477		let mut keyed: Vec<(u64, u32)> = Vec::with_capacity(post.len());
478		let mut any = Vec::new();
479		for (index, (tuple, _)) in (0_u32..).zip(post.iter()) {
480			match key(tapes.iter().map(|&tape| tuple[tape])) {
481				Some(key) => keyed.push((key, index)),
482				None => any.push(index),
483			}
484		}
485
486		keyed.sort_unstable();
487		let mut keys = Vec::new();
488		let mut starts = Vec::new();
489		for (at, &(key, _)) in (0_u32..).zip(&keyed) {
490			if keys.last() != Some(&key) {
491				keys.push(key);
492				starts.push(at);
493			}
494		}
495
496		starts.push(index_u32(keyed.len()));
497
498		ByKey {
499			keys,
500			starts,
501			moves: keyed.into_iter().map(|(_, index)| index).collect(),
502			any,
503			len: post.len(),
504		}
505	}
506
507	fn candidates(&self, letters: impl Iterator<Item = Symbol>, mut visit: impl FnMut(usize)) {
508		let Some(key) = key(letters) else {
509			(0..self.len).for_each(visit);
510
511			return;
512		};
513
514		if let Ok(at) = self.keys.binary_search(&key) {
515			let range = self.starts[at] as usize..self.starts[at + 1] as usize;
516			for &index in &self.moves[range] {
517				visit(index as usize);
518			}
519		}
520
521		for &index in &self.any {
522			visit(index as usize);
523		}
524	}
525}

527/// `letters` folded into one number, `None` if one is the wildcard; past two letters, keys may collide.
528fn key(mut letters: impl Iterator<Item = Symbol>) -> Option<u64> {
529	letters.try_fold(0_u64, |key, letter| {
530		(letter != Symbol::ANY).then(|| key.rotate_left(32) ^ u64::from(letter.value()))
531	})
532}

534#[derive(Clone, Copy)]
535enum Product<'s> {
536	Intersection,
537	Sync(&'s SyncSpec),
538	Diagonal,
539}
540
541impl Product<'_> {
542	/// Writes the tuple that `lhs` and `rhs` read together into `out`, if they agree.
543	fn merge(self, lhs: &[Symbol], rhs: &[Symbol], out: &mut Vec<Symbol>) -> bool {
544		out.clear();
545		match self {
546			Product::Intersection => {
547				for (&a, &b) in lhs.iter().zip(rhs) {
548					let Some(letter) = merge(a, b) else {
549						return false;
550					};
551
552					out.push(letter);
553				}
554			}
555			Product::Sync(spec) => {
556				if !spec
557					.pairs
558					.iter()
559					.all(|pair| merge(lhs[pair.lhs], rhs[pair.rhs]).is_some())
560				{
561					return false;
562				}
563
564				for tape in &spec.layout {
565					let letter = match (tape.source, tape.peer) {
566						(Tape::Lhs(i), None) => lhs[i],
567						(Tape::Rhs(j), None) => rhs[j],
568						(Tape::Lhs(i), Some(j)) | (Tape::Rhs(j), Some(i)) => {
569							merge(lhs[i], rhs[j]).unwrap_or(Symbol::ANY)
570						}
571					};
572
573					out.push(letter);
574				}
575			}
576			Product::Diagonal => {
577				let Some(letter) = merge(lhs[0], rhs[0]).and_then(|letter| merge(letter, rhs[1]))
578				else {
579					return false;
580				};
581
582				out.push(letter);
583			}
584		}
585
586		true
587	}
588}

590/// How a product finds the moves of its right operand that may merge with a left move.
591enum Candidates<'a> {
592	/// The right operand's shared table, read on `tapes`; it needs no index.
593	Table {
594		table: &'a Table,
595		tapes: &'a [usize],
596		scratch: &'a mut Odometer,
597	},
598	ByKey(ByKey),
599	/// Every move of the right operand.
600	All(usize),
601}
602
603impl<'a> Candidates<'a> {
604	/// The candidates of `right` on `tapes`, indexed by key if `indexed` and it has no table.
605	fn new(
606		right: &'a Post,
607		tapes: &'a [usize],
608		indexed: bool,
609		scratch: &'a mut Odometer,
610	) -> Candidates<'a> {
611		match &right.letters {
612			Letters::Shared(table) => Candidates::Table {
613				table,
614				tapes,
615				scratch,
616			},
617			Letters::Owned(_) if indexed => Candidates::ByKey(ByKey::new(right, tapes)),
618			Letters::Owned(_) => Candidates::All(right.len()),
619		}
620	}
621
622	/// Visits the moves that may merge with `letters`, and maybe others; the caller merges them.
623	fn visit(&mut self, letters: impl Iterator<Item = Symbol>, visit: impl FnMut(usize)) {
624		match self {
625			Candidates::Table {
626				table,
627				tapes,
628				scratch,
629			} => table.matching(tapes, letters, scratch, visit),
630			Candidates::ByKey(by_key) => by_key.candidates(letters, visit),
631			Candidates::All(len) => (0..*len).for_each(visit),
632		}
633	}
634}

646impl Engine<'_> {
647	/// The moves from `state` of `node`, or the limit that the query passed while computing them.
648	pub(crate) fn post(&mut self, node: ExecId, state: StateId) -> Result<Rc<Post>, Limit> {
649		if let Op::Transparent(child) = self.exec.nodes[node].op {
650			return self.post(child, state);
651		}
652
653		let index = state.index();
654		if let Some(Some(post)) = self.posts[node].get(index) {
655			if let Some(budget) = &mut self.cache_budget {
656				budget.touch(node.index(), state);
657			}
658
659			return Ok(Rc::clone(post));
660		}
661
662		let (post, nanos) = self.timed(|engine| deep(|| engine.compute_post(node, state)));
663		self.post_counts[node] += 1;
664		// Past a limit the post may be partial, so it is neither kept nor used.
665		let mut post = post?;
666		self.stopped()?;
667		if !self.cached[node] {
668			return Ok(Rc::new(post));
669		}
670
671		post.shrink_to_fit();
672		let post = Rc::new(post);
673		let slots = &mut self.posts[node];
674		if slots.len() <= index {
675			slots.resize(index + 1, None);
676		}
677
678		slots[index] = Some(Rc::clone(&post));
679		if let Some(budget) = &mut self.cache_budget {
680			let dropped = budget.admit(node.index(), state, nanos, post.bytes());
681			self.evict(dropped);
682		}
683
684		Ok(post)
685	}

687	/// Drops the posts and the rows of dense complements (keyed after the nodes) that the cache budget evicted.
688	pub(super) fn evict(&mut self, dropped: Vec<(usize, StateId)>) {
689		let num_nodes = self.exec.nodes.len();
690		for (key, state) in dropped {
691			if key < num_nodes {
692				self.posts[ExecId::new(key)][state.index()] = None;
693			} else {
694				self.evict_row(ExecId::new(key - num_nodes), state);
695			}
696		}
697	}

699	/// Runs `work` and, under a cache budget, times it without the posts it computed, which are cached on their own.
700	pub(super) fn timed<T>(&mut self, work: impl FnOnce(&mut Self) -> T) -> (T, u64) {
701		if self.cache_budget.is_none() {
702			return (work(self), 0);
703		}
704
705		let start = Instant::now();
706		let outer = self.nested_nanos;
707		self.nested_nanos = 0;
708		let result = work(self);
709		let total = u64::try_from(start.elapsed().as_nanos()).unwrap_or(u64::MAX);
710		let own = total.saturating_sub(self.nested_nanos);
711		self.nested_nanos = outer.saturating_add(total);
712
713		(result, own)
714	}

716	fn compute_post(&mut self, node: ExecId, state: StateId) -> Result<Post, Limit> {
717		match self.exec.nodes[node].op.clone() {
718			Op::Leaf(leaf) => Ok(self.leaf_post(node, &leaf, state)),
719			Op::Union(branches) => self.union_post(node, &branches, state),
720			Op::Intersection(a, b) => self.product_post(node, a, b, state, Product::Intersection),
721			Op::Sync(a, b, spec) => self.product_post(node, a, b, state, Product::Sync(&spec)),
722			Op::Diagonal(u, x) => self.product_post(node, u, x, state, Product::Diagonal),
723			Op::Complement(child) => self.complement_post(node, child, state),
724			Op::Identity(child) => self.identity_post(node, child, state),
725			Op::Project(child, tapes) => self.project_post(node, child, &tapes, state),
726			Op::PadClosure(child, pad) => self.pad_closure_post(node, child, &pad, state),
727			Op::Prefixes(child) => self.prefixes_post(node, child, state),
728			Op::Explicit { .. } | Op::Transparent(_) => {
729				unreachable!("explicit nodes are replaced before the search")
730			}
731		}
732	}

734	/// The leaf's moves, except those that read the wildcard on a tape with an empty universe.
735	fn leaf_post(&self, node: ExecId, leaf: &Leaf, state: StateId) -> Post {
736		let arity = self.exec.nodes[node].arity;
737		let dead: Vec<usize> = (0..arity)
738			.filter(|&tape| {
739				leaf.tape_has_any[tape] && self.alphabets.universe(node, tape).is_empty()
740			})
741			.collect();
742
743		let mut post = Post::with_capacity(arity, leaf.moves(state).len());
744		for (tuple, target) in leaf.moves(state) {
745			if dead.iter().any(|&tape| tuple[tape] == Symbol::ANY) {
746				continue;
747			}
748
749			post.push(tuple, Target::of_leaf(leaf, target));
750		}
751
752		post
753	}

755	fn union_post(
756		&mut self,
757		node: ExecId,
758		branches: &[ExecId],
759		state: StateId,
760	) -> Result<Post, Limit> {
761		let Tagged { branch, state } = self.stores[node].tagged(state);
762		let child = self.post(branches[branch as usize], state)?;
763		let start = self.stores[node].len();
764		let mut post = Post::with_capacity(self.exec.nodes[node].arity, child.len());
765		for (tuple, target) in child.iter() {
766			post.push(tuple, self.tagged_target(node, branch, target));
767		}
768
769		self.budget.add(self.stores[node].len() - start);
770
771		Ok(post)
772	}

774	pub(super) fn tagged_target(&mut self, node: ExecId, branch: u32, target: Target) -> Target {
775		let state = self.stores[node].intern_tagged(Tagged {
776			branch,
777			state: target.state,
778		});
779
780		Target { state, ..target }
781	}

783	/// The child's moves with each letter on both tapes; a wildcard stands for every letter of the universe.
784	fn identity_post(
785		&mut self,
786		node: ExecId,
787		child: ExecId,
788		state: StateId,
789	) -> Result<Post, Limit> {
790		let child_post = self.post(child, state)?;
791		if let Letters::Shared(table) = &child_post.letters {
792			let diagonal = self.tables[node].get_or_insert_with(|| Rc::new(table.diagonal()));
793
794			return Ok(Post {
795				arity: 2,
796				letters: Letters::Shared(Rc::clone(diagonal)),
797				targets: child_post.targets.clone(),
798			});
799		}
800
801		let universe = self.alphabets.universe(node, 0);
802		let mut post = Post::with_capacity(2, child_post.len());
803		for (tuple, target) in child_post.iter() {
804			if tuple[0] == Symbol::ANY {
805				for &letter in universe.iter() {
806					post.push(&[letter, letter], target);
807				}
808			} else {
809				post.push(&[tuple[0], tuple[0]], target);
810			}
811		}
812
813		Ok(post)
814	}

816	fn project_post(
817		&mut self,
818		node: ExecId,
819		child: ExecId,
820		tapes: &[usize],
821		state: StateId,
822	) -> Result<Post, Limit> {
823		let arity = self.exec.nodes[node].arity;
824		let child_post = self.post(child, state)?;
825		let mut post = Post::with_capacity(arity, child_post.len());
826		let mut projected = Vec::with_capacity(arity);
827		for (tuple, target) in child_post.iter() {
828			projected.clear();
829			projected.extend(tapes.iter().map(|&tape| tuple[tape]));
830			post.push(&projected, target);
831		}
832
833		post.dedup();
834
835		Ok(post)
836	}

838	fn product_post(
839		&mut self,
840		node: ExecId,
841		lhs: ExecId,
842		rhs: ExecId,
843		state: StateId,
844		product: Product<'_>,
845	) -> Result<Post, Limit> {
846		let pair = self.stores[node].pair(state);
847		let (left, right) = (self.post(lhs, pair.lhs)?, self.post(rhs, pair.rhs)?);
848		let plan = &self.plans[node];
849		let indexed = !plan.keys.is_empty() && left.len() > 1 && right.len() > INDEX_THRESHOLD;
850
851		if let Product::Sync(spec) = product
852			&& let Some(post) = self.group_join(node, lhs, &left, &right, spec, indexed)
853		{
854			return Ok(post);
855		}
856
857		Ok(self.pair_join(node, &left, &right, product, indexed))
858	}

860	/// The post by the group join, where the plan allows it and it looks cheaper than the pair join.
861	fn group_join(
862		&mut self,
863		node: ExecId,
864		lhs: ExecId,
865		left: &Post,
866		right: &Rc<Post>,
867		spec: &SyncSpec,
868		indexed: bool,
869	) -> Option<Post> {
870		let plan = self.plans[node].group.as_ref()?;
871		let index = plan
872			.index
873			.get_or_init(|| LetterIndex::new(self.alphabets.universe(lhs, plan.hidden.lhs)))
874			.as_ref()?;
875		let join = GroupJoin {
876			left,
877			right,
878			spec,
879			plan,
880			index,
881			indexed,
882		};
883
884		join.run(
885			&mut self.stores[node],
886			&mut self.budget,
887			&mut self.join_scratch,
888		)
889	}

891	/// Merges each left move with the right moves that may merge with it.
892	fn pair_join(
893		&mut self,
894		node: ExecId,
895		left: &Post,
896		right: &Post,
897		product: Product<'_>,
898		indexed: bool,
899	) -> Post {
900		let plan = &self.plans[node];
901		// Sized for one move per left move, as a join that drops tapes has at least as many.
902		let mut seen = plan
903			.dedup
904			.then(|| HashTable::with_capacity(left.len().min(SEEN_CAPACITY)));
905		let mut post = Post::new(self.exec.nodes[node].arity);
906		let mut candidates = Candidates::new(
907			right,
908			&plan.right_tapes,
909			indexed,
910			&mut self.join_buffers.odometer,
911		);
912		let store = &mut self.stores[node];
913		let start = store.len();
914
915		for (left_tuple, left_target) in left.iter() {
916			self.budget.step(store.len() - start);
917			if self.budget.passed() {
918				break;
919			}
920
921			let letters = plan.keys.iter().map(|pair| left_tuple[pair.lhs]);
922			candidates.visit(letters, |index| {
923				if !product.merge(left_tuple, right.tuple(index), &mut self.join_buffers.tuple) {
924					return;
925				}
926
927				let right_target = right.target(index);
928				let pair = Pair {
929					lhs: left_target.state,
930					rhs: right_target.state,
931				};
932				let accepting = left_target.accepting && right_target.accepting;
933				if let Some(seen) = &mut seen {
934					post.push_new(seen, store, &self.join_buffers.tuple, pair, accepting);
935				} else {
936					let state = store.intern_pair(pair);
937					post.push(&self.join_buffers.tuple, Target { state, accepting });
938				}
939			});
940		}
941
942		self.budget.add(store.len() - start);
943
944		post
945	}

947	/// One move per tuple of the tapes' alphabets, to the minimized set of the members' successors.
948	fn complement_post(
949		&mut self,
950		node: ExecId,
951		child: ExecId,
952		state: StateId,
953	) -> Result<Post, Limit> {
954		let arity = self.exec.nodes[node].arity;
955		let members = self.stores[node].set(state).to_vec();
956		let universes: Vec<Rc<[Symbol]>> = (0..arity)
957			.map(|tape| Rc::clone(self.alphabets.universe(node, tape)))
958			.collect();
959		let num_tuples = self
960			.alphabets
961			.tuple_count(node)
962			.expect("the tuple count is checked when the query starts");
963		if let Some(post) = self.dense_post(node, child, &members, &universes, num_tuples)? {
964			return Ok(post);
965		}
966
967		let posts = members
968			.iter()
969			.map(|&member| self.post(child, member))
970			.collect::<Result<Vec<Rc<Post>>, Limit>>()?;
971		let table = self.tuple_table(node, &universes, num_tuples)?;
972		let mut exact: FxHashMap<usize, Vec<Target>> = FxHashMap::default();
973		let mut wildcards: Vec<(&[Symbol], Target)> = Vec::new();
974		for (tuple, target) in posts.iter().flat_map(|post| post.iter()) {
975			if tuple.contains(&Symbol::EPSILON) {
976				continue;
977			}
978
979			if tuple.contains(&Symbol::ANY) {
980				wildcards.push((tuple, target));
981			} else if let Some(index) = table.index(tuple) {
982				exact.entry(index).or_default().push(target);
983			}
984		}
985
986		let mut post = Post::with_table(arity, Rc::clone(&table));
987		let mut successors = Vec::new();
988		let start = self.stores[node].len();
989		for index in 0..num_tuples {
990			self.budget.step(self.stores[node].len() - start);
991			if self.budget.passed() {
992				break;
993			}
994
995			let tuple = &table.letters[index * arity..(index + 1) * arity];
996			let wildcard_targets = wildcards
997				.iter()
998				.filter(|(letters, _)| {
999					letters
1000						.iter()
1001						.zip(tuple)
1002						.all(|(&letter, &concrete)| letter == Symbol::ANY || letter == concrete)
1003				})
1004				.map(|(_, target)| target);
1005			let targets = exact
1006				.get(&index)
1007				.into_iter()
1008				.flatten()
1009				.chain(wildcard_targets);
1010
1011			successors.clear();
1012			let mut accepting = true;
1013			for target in targets {
1014				successors.push(target.state);
1015				accepting &= !target.accepting;
1016			}
1017
1018			self.subsumption
1019				.minimize(&self.exec, &self.stores, child, &mut successors);
1020			let state = self.stores[node].intern_set(&successors);
1021			post.push_target(Target { state, accepting });
1022		}
1023
1024		self.budget.add(self.stores[node].len() - start);
1025
1026		Ok(post)
1027	}

1029	/// Every tuple of the universes of the complement `node`, in odometer order (last tape fastest), built once;
1030	/// or the limit that the query passed while building it.
1031	pub(super) fn tuple_table(
1032		&mut self,
1033		node: ExecId,
1034		universes: &[Rc<[Symbol]>],
1035		num_tuples: usize,
1036	) -> Result<Rc<Table>, Limit> {
1037		if let Some(table) = &self.tables[node] {
1038			return Ok(Rc::clone(table));
1039		}
1040
1041		let table = Rc::new(Table::new(universes, num_tuples, &mut self.budget));
1042		// A table that a limit cut short is not kept.
1043		self.stopped()?;
1044		self.tables[node] = Some(Rc::clone(&table));
1045
1046		Ok(table)
1047	}
