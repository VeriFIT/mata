1//! Which cached posts to drop over [`cache_budget`](crate::Options::cache_budget), by GreedyDual-Size with
2//! frequency (Cherkasova, 1998). Priority is the floor plus compute time per byte times reads since cached; the
3//! lowest goes and becomes the floor.
4
5use std::cmp::Reverse;
6use std::collections::BinaryHeap;
7
8use rustc_hash::FxHashMap;
9
10use crate::ids::StateId;
11
12#[derive(Clone, Copy, Debug)]
13struct Entry {
14	priority: u64,
15	/// Nanoseconds of computing per 1024 bytes.
16	density: u64,
17	bytes: usize,
18	/// Which push of the entry into the queue is current.
19	version: u64,
20	/// Requests since it was cached, the first included.
21	reads: u64,
22}
23
24/// The cached entries by key and state; a key is a node, or the rows of a dense complement.
25#[derive(Debug)]
26pub(crate) struct CacheBudget {
27	limit: usize,
28	bytes: usize,
29	floor: u64,
30	entries: Vec<FxHashMap<StateId, Entry>>,
31	/// How many entries are cached.
32	live: usize,
33	/// `(priority, key, state, version)` of every push, lowest priority first; a replaced push is skipped.
34	queue: BinaryHeap<Reverse<(u64, usize, StateId, u64)>>,
35	version: u64,
36	pub(crate) dropped: usize,
37}
38
39impl CacheBudget {
40	pub(crate) fn new(limit: usize, num_keys: usize) -> CacheBudget {
41		CacheBudget {
42			limit,
43			bytes: 0,
44			floor: 0,
45			entries: vec![FxHashMap::default(); num_keys],
46			live: 0,
47			queue: BinaryHeap::new(),
48			version: 0,
49			dropped: 0,
50		}
51	}
52
53	/// Renews the priority of a cached entry that is read again.
54	pub(crate) fn touch(&mut self, key: usize, state: StateId) {
55		let Some(entry) = self.entries[key].get_mut(&state) else {
56			return;
57		};
58
59		entry.reads += 1;
60		entry.priority = self
61			.floor
62			.saturating_add(entry.density.saturating_mul(entry.reads));
63		self.version += 1;
64		entry.version = self.version;
65		self.queue
66			.push(Reverse((entry.priority, key, state, entry.version)));
67		self.compact();
68	}
69
70	/// Records an entry just cached; returns the cached entries to drop, which may include it.
71	pub(crate) fn admit(
72		&mut self,
73		key: usize,
74		state: StateId,
75		nanos: u64,
76		bytes: usize,
77	) -> Vec<(usize, StateId)> {
78		let density = u128::from(nanos) * 1024 / u128::try_from(bytes.max(1)).unwrap_or(u128::MAX);
79		let density = u64::try_from(density).unwrap_or(u64::MAX);
80		let priority = self.floor.saturating_add(density);
81
82		self.version += 1;
83		let entry = Entry {
84			priority,
85			density,
86			bytes,
87			version: self.version,
88			reads: 1,
89		};
90		let replaced = self.entries[key].insert(state, entry);
91		if replaced.is_none() {
92			self.live += 1;
93		}
94
95		self.bytes += bytes;
96		self.queue
97			.push(Reverse((priority, key, state, self.version)));
98		self.compact();
99
100		let mut dropped = Vec::new();
101		while self.bytes > self.limit
102			&& let Some(Reverse((priority, key, state, version))) = self.queue.pop()
103		{
104			if let Some(entry) = self.entries[key].get(&state)
105				&& entry.version == version
106			{
107				self.bytes -= entry.bytes;
108				self.entries[key].remove(&state);
109				self.live -= 1;
110				self.floor = priority;
111				dropped.push((key, state));
112			}
113		}
114
115		self.dropped += dropped.len();
116
117		dropped
118	}
119
120	/// Forgets an entry that its owner dropped by itself.
121	pub(crate) fn remove(&mut self, key: usize, state: StateId) {
122		if let Some(entry) = self.entries[key].remove(&state) {
123			self.bytes -= entry.bytes;
124			self.live -= 1;
125		}
126	}
127
128	/// Rebuilds the queue from the entries once most of its pushes are replaced ones.
129	fn compact(&mut self) {
130		if self.queue.len() > 4 * self.live + 1024 {
131			self.queue = self
132				.entries
133				.iter()
134				.enumerate()
135				.flat_map(|(key, entries)| {
136					entries.iter().map(move |(&state, entry)| {
137						Reverse((entry.priority, key, state, entry.version))
138					})
139				})
140				.collect();
141		}
142	}
143}

145#[cfg(test)]
146mod tests {
147	use super::CacheBudget;
148	use crate::ids::{Id, StateId};
149
150	#[test]
151	fn reads_without_new_entries_keep_the_queue_small() {
152		let mut budget = CacheBudget::new(1 << 20, 1);
153		let state = StateId::new(0);
154		budget.admit(0, state, 1000, 64);
155		for _ in 0..100_000 {
156			budget.touch(0, state);
157		}
158
159		assert!(budget.queue.len() <= 4 * budget.live + 1024);
160	}
