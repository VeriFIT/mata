1//! The execution DAG of one query, which is the formula after rewrites, with shared subterms merged.
2//!
3//! The rewrites keep the relation.
4//! - complements are pushed down to leaves and non-Boolean nodes (De Morgan, and `¬¬X = X` for an `X`
5//!   without padding, as a complement has no padded words);
6//! - `op(id(A), id(B)) = id(op(A, B))` for union and intersection;
7//! - projections fold, `project(id(L), [i]) = L`, `project(id(L), [0, 1] or [1, 0]) = id(L)`, and a projection
8//!   to all tapes in order is dropped;
9//! - `project(id(U) ∩ X, [i])` becomes one diagonal node, or `U ∩ X'` when `X` is a sync product of a tape of
10//!   each operand, where `X'` also synchronizes those two tapes, so it builds only the moves that the diagonal keeps;
11//! - `op(X, X) = X`, `prefixes(prefixes(X)) = prefixes(X)`, and structurally equal nodes are built once;
12//! - the cheaper operand of a union or intersection goes left;
13//! - a composition chain is re-associated to minimize its estimated intermediate macrostate bounds, without
14//!   changing the order of the relations.
15//!
16//! Explicit nodes and nodes with a fixed operand order stop composition-chain flattening.
17//!
18//! Explicit nodes stop the rewrites from looking into them. The query's annotations make nodes explicit, keep
19//! explicit nodes of the formula lazy, and fix the operand order of an intersection (not one that De Morgan
20//! turns into a union) or of a synchronized product.
21

55#[derive(Clone, Debug)]
56pub(crate) enum Op<'f> {
57	Leaf(LeafRef<'f>),
58	/// Nested unions flattened into one node, cheaper branches first.
59	Union(Arc<[ExecId]>),
60	Intersection(ExecId, ExecId),
61	Complement(ExecId),
62	Identity(ExecId),
63	Project(ExecId, Arc<[usize]>),
64	/// `{w | w·p^k ∈ child}`; its macrostates are a state of the child and the number of the last step read
65	/// (see [`PadSteps`](super::pad::PadSteps)).
66	PadClosure(ExecId, Arc<Pad>),
67	Sync(ExecId, ExecId, Arc<SyncSpec>),
68	/// `{w | w ∈ lhs, (w, w) ∈ rhs}`, with pair macrostates.
69	Diagonal(ExecId, ExecId),
70	/// Replaced by a leaf, or by [`Transparent`](Op::Transparent), before the search.
71	Explicit {
72		child: ExecId,
73		options: ExplicitOptions,
74		origin: NodeId,
75	},
76	/// An explicit node that stayed lazy, whose macrostates are its child's.
77	Transparent(ExecId),
78	/// The prefix closure, whose macrostates are its child's, accepting when they reach an accepting one.
79	Prefixes(ExecId),
80}
81
82impl Op<'_> {
83	pub(crate) fn children(&self) -> impl Iterator<Item = ExecId> {
84		let (operands, branches): ([Option<ExecId>; 2], &[ExecId]) = match self {
85			Op::Leaf(_) => ([None, None], &[]),
86			Op::Union(branches) => ([None, None], branches),
87			Op::Intersection(a, b) | Op::Sync(a, b, _) | Op::Diagonal(a, b) => {
88				([Some(*a), Some(*b)], &[])
89			}
90			Op::Complement(a)
91			| Op::Identity(a)
92			| Op::Project(a, _)
93			| Op::PadClosure(a, _)
94			| Op::Explicit { child: a, .. }
95			| Op::Transparent(a)
96			| Op::Prefixes(a) => ([Some(*a), None], &[]),
97		};
98
99		operands
100			.into_iter()
101			.flatten()
102			.chain(branches.iter().copied())
103	}
104
105	/// The child whose macrostates this node shares, if any.
106	pub(crate) fn transparent_child(&self) -> Option<ExecId> {
107		match *self {
108			Op::Identity(child)
109			| Op::Project(child, _)
110			| Op::Transparent(child)
111			| Op::Prefixes(child) => Some(child),
112			_ => None,
113		}
114	}
115}

133impl Exec<'_> {
134	/// `node` and the children whose macrostates it shares, down to their owner.
135	pub(crate) fn sharing(&self, node: ExecId) -> impl Iterator<Item = ExecId> {
136		std::iter::successors(Some(node), |&node| self.nodes[node].op.transparent_child())
137	}

141		self.sharing(node).last().unwrap_or(node)

175/// What a source node in one form becomes in the DAG.
176#[derive(Debug)]
177enum Decision {
178	Alias(Src),
179	Leaf(LeafId),
180	Complement(Src),
181	/// The operation, its operands, and whether their order is fixed.
182	Boolean(Boolean, Src, Src, bool),
183	IdentityOf(Boolean, Src, Src, bool),
184	Identity(Src),
185	Project(Src, Arc<[usize]>),
186	PadClosure(Src, Arc<Pad>),
187	Diagonal(Src, Src),
188	/// `U ∩ X'` for the diagonal of `U` and a sync product, with the product's two tapes synchronized.
189	DiagonalSync(Src, Src, Src, Arc<SyncSpec>),
190	Sync(Src, Src, Arc<SyncSpec>),
191	/// Binary compositions in their original order; explicit and fixed-order nodes are opaque.
192	CompositionChain(Box<[Src]>, Arc<SyncSpec>),
193	Explicit(Src, ExplicitOptions, NodeId),
194	Prefixes(Src),
195}
196
197impl Decision {
198	fn deps(&self) -> Vec<Src> {
199		match self {
200			Decision::Leaf(_) => Vec::new(),
201			Decision::Alias(s)
202			| Decision::Complement(s)
203			| Decision::Identity(s)
204			| Decision::Project(s, _)
205			| Decision::PadClosure(s, _)
206			| Decision::Explicit(s, ..)
207			| Decision::Prefixes(s) => vec![*s],
208			Decision::Boolean(_, a, b, _)
209			| Decision::IdentityOf(_, a, b, _)
210			| Decision::Diagonal(a, b)
211			| Decision::Sync(a, b, _) => vec![*a, *b],
212			Decision::CompositionChain(leaves, _) => leaves.to_vec(),
213			Decision::DiagonalSync(u, a, b, _) => vec![*u, *a, *b],
214		}
215	}
216}

218struct Rewriter<'a> {
219	formula: &'a Formula,
220	annotations: &'a Annotations,
221}
222
223impl Rewriter<'_> {
224	fn is_marked(&self, node: NodeId) -> bool {
225		self.annotations.explicit.contains_key(&node)
226			&& !matches!(self.formula.nodes[node], Node::Explicit(..))
227	}
228
229	fn order(&self, node: NodeId) -> Order {
230		self.annotations
231			.order
232			.get(&node)
233			.copied()
234			.unwrap_or_default()
235	}
236
237	fn identity_child(&self, node: NodeId) -> Option<NodeId> {
238		match self.formula.nodes[node] {
239			Node::Identity(child) if !self.is_marked(node) => Some(child),
240			_ => None,
241		}
242	}
243
244	/// The decision of every source node that `root` depends on, in an order where dependencies come first (a
245	/// smaller node, or the same node in an earlier form).
246	fn decide_all(&self, root: NodeId) -> BTreeMap<Src, Decision> {
247		let mut decisions = BTreeMap::new();
248		let mut stack = vec![plain(root)];
249		while let Some(src) = stack.pop() {
250			if let btree_map::Entry::Vacant(entry) = decisions.entry(src) {
251				let decision = self.decide(src);
252				stack.extend(decision.deps());
253				entry.insert(decision);
254			}
255		}
256
257		decisions
258	}
259
260	fn decide(&self, (node, form): Src) -> Decision {
261		if self.is_marked(node) {
262			match form {
263				Form::Plain => {
264					let options = self.annotations.explicit[&node];
265
266					return Decision::Explicit((node, Form::Inner), options, node);
267				}
268				Form::Negated => return Decision::Complement(plain(node)),
269				Form::Inner => {}
270			}
271		}
272
273		let negated = form == Form::Negated;
274
275		match &self.formula.nodes[node] {
276			Node::Explicit(child, _) if self.annotations.lazy.contains(&node) => {
277				Decision::Alias((*child, form))
278			}
279			Node::Complement(child) if negated && self.formula.pads[*child] => {
280				Decision::Complement(plain(node))
281			}
282			Node::Complement(child) => {
283				Decision::Alias((*child, if negated { Form::Plain } else { Form::Negated }))
284			}
285			Node::Union(a, b) => self.decide_boolean(Boolean::Union, *a, *b, negated, Order::Auto),
286			Node::Intersection(a, b) => {
287				let order = self.order(node);
288
289				self.decide_boolean(Boolean::Intersection, *a, *b, negated, order)
290			}
291			_ if negated => Decision::Complement(plain(node)),
292			Node::Leaf(leaf) => Decision::Leaf(*leaf),
293			Node::Identity(child) => Decision::Identity(plain(*child)),
294			Node::Project(child, tapes) => self.decide_project(*child, tapes),
295			Node::PadClosure(child, pad) => Decision::PadClosure(plain(*child), Arc::clone(pad)),
296			Node::Sync(a, b, spec) if self.order(node) == Order::Auto && spec.is_composition() => {
297				self.decide_composition(*a, *b, spec)
298			}
299			Node::Sync(a, b, spec) if self.order(node) == Order::Swapped => {
300				Decision::Sync(plain(*b), plain(*a), Arc::new(spec.swapped()))
301			}
302			Node::Sync(a, b, spec) => Decision::Sync(plain(*a), plain(*b), Arc::clone(spec)),
303			Node::Explicit(child, options) => {
304				let options = self.annotations.explicit.get(&node).unwrap_or(options);
305
306				Decision::Explicit(plain(*child), *options, node)
307			}
308			Node::Prefixes(child) => Decision::Prefixes(plain(*child)),
309		}
310	}
311
312	fn composition_parts(&self, node: NodeId) -> Option<(NodeId, NodeId, &Arc<SyncSpec>)> {
313		if self.is_marked(node) || self.order(node) != Order::Auto {
314			return None;
315		}
316
317		match &self.formula.nodes[node] {
318			Node::Sync(a, b, spec)
319				if spec.is_composition()
320					&& self.formula.arities[*a] == 2
321					&& self.formula.arities[*b] == 2 =>
322			{
323				Some((*a, *b, spec))
324			}
325			_ => None,
326		}
327	}
328
329	/// Flattens a chain without changing the order of its relations. Only a canonical composition may be opened;
330	/// all other operands are opaque leaves.
331	fn decide_composition(&self, a: NodeId, b: NodeId, spec: &Arc<SyncSpec>) -> Decision {
332		let mut pending = vec![b, a];
333		let mut leaves = Vec::new();
334		while let Some(node) = pending.pop() {
335			if let Some((a, b, _)) = self.composition_parts(node) {
336				pending.extend([b, a]);
337			} else {
338				leaves.push(plain(node));
339			}
340		}
341
342		if leaves.len() > 2 {
343			Decision::CompositionChain(leaves.into(), Arc::clone(spec))
344		} else {
345			Decision::Sync(plain(a), plain(b), Arc::clone(spec))
346		}
347	}
348
349	fn decide_boolean(
350		&self,
351		op: Boolean,
352		a: NodeId,
353		b: NodeId,
354		negated: bool,
355		order: Order,
356	) -> Decision {
357		if negated {
358			return Decision::Boolean(op.dual(), (a, Form::Negated), (b, Form::Negated), false);
359		}
360
361		let (a, b) = if order == Order::Swapped {
362			(b, a)
363		} else {
364			(a, b)
365		};
366		let fixed = order != Order::Auto;
367
368		match (self.identity_child(a), self.identity_child(b)) {
369			(Some(lhs), Some(rhs)) => Decision::IdentityOf(op, plain(lhs), plain(rhs), fixed),
370			_ => Decision::Boolean(op, plain(a), plain(b), fixed),
371		}
372	}
373
374	fn decide_project(&self, child: NodeId, tapes: &Arc<[usize]>) -> Decision {
375		let mut base = child;
376		let mut kept = Arc::clone(tapes);
377		while !self.is_marked(base)
378			&& let Node::Project(inner, inner_tapes) = &self.formula.nodes[base]
379		{
380			kept = kept.iter().map(|&tape| inner_tapes[tape]).collect();
381			base = *inner;
382		}
383
384		if let Some(language) = self.identity_child(base) {
385			match kept.len() {
386				1 => return Decision::Alias(plain(language)),
387				2 => return Decision::Alias(plain(base)),
388				_ => {}
389			}
390		}
391
392		if kept.iter().copied().eq(0..self.formula.arities[base]) {
393			return Decision::Alias(plain(base));
394		}
395
396		if kept.len() == 1
397			&& !self.is_marked(base)
398			&& self.order(base) == Order::Auto
399			&& let Node::Intersection(p, q) = self.formula.nodes[base]
400			&& self.formula.arities[base] == 2
401		{
402			if let Some(u) = self.identity_child(p) {
403				return self.decide_diagonal(u, q);
404			}
405
406			if let Some(u) = self.identity_child(q) {
407				return self.decide_diagonal(u, p);
408			}
409		}
410
411		Decision::Project(plain(base), kept)
412	}
413
414	/// `{w | w ∈ u, (w, w) ∈ x}`.
415	fn decide_diagonal(&self, u: NodeId, x: NodeId) -> Decision {
416		if !self.is_marked(x)
417			&& self.order(x) == Order::Auto
418			&& let Node::Sync(a, b, spec) = &self.formula.nodes[x]
419			&& let [first, second] = *spec.layout
420			&& first.peer.is_none()
421			&& second.peer.is_none()
422			&& let (Tape::Lhs(lhs), Tape::Rhs(rhs)) | (Tape::Rhs(rhs), Tape::Lhs(lhs)) =
423				(first.source, second.source)
424		{
425			let spec = SyncSpec {
426				pairs: spec
427					.pairs
428					.iter()
429					.copied()
430					.chain([TapePair { lhs, rhs }])
431					.collect(),
432				layout: [ResultTape {
433					source: Tape::Lhs(lhs),
434					peer: Some(rhs),
435				}]
436				.into(),
437			};
438
439			return Decision::DiagonalSync(plain(u), plain(*a), plain(*b), Arc::new(spec));
440		}
441
442		Decision::Diagonal(plain(u), plain(x))
443	}
444}

464/// Depth and log2 of a bound on the number of macrostates, to order operands.
465#[derive(Clone, Copy)]
466struct Cost {
467	depth: u32,
468	log_states: f64,
469}
470
471impl Ord for Cost {
472	fn cmp(&self, other: &Cost) -> Ordering {
473		self.depth
474			.cmp(&other.depth)
475			.then(self.log_states.total_cmp(&other.log_states))
476	}
477}
478
479impl PartialOrd for Cost {
480	fn partial_cmp(&self, other: &Cost) -> Option<Ordering> {
481		Some(self.cmp(other))
482	}
483}
484
485impl PartialEq for Cost {
486	fn eq(&self, other: &Cost) -> bool {
487		self.cmp(other).is_eq()
488	}
489}

493#[derive(Default)]
494struct Emitter<'f> {
495	nodes: TiVec<ExecId, ExecNode<'f>>,
496	costs: TiVec<ExecId, Cost>,
497	pads: TiVec<ExecId, bool>,
498	interned: FxHashMap<Key, ExecId>,
499}
500
501impl<'f> Emitter<'f> {
502	fn emit(&mut self, key: Key, op: Op<'f>, arity: usize, cost: Cost) -> ExecId {
503		match self.interned.entry(key) {
504			Entry::Occupied(entry) => *entry.get(),
505			Entry::Vacant(entry) => {
506				let pads = match &op {
507					Op::Leaf(leaf) => leaf.has_padding,
508					Op::Complement(_) => false,
509					op => op.children().any(|child| self.pads[child]),
510				};
511
512				self.pads.push(pads);
513				let id = self.nodes.push_and_get_key(ExecNode { op, arity });
514				self.costs.push(cost);
515
516				*entry.insert(id)
517			}
518		}
519	}
520
521	/// The node of `decision`, whose dependencies are `built`.
522	fn emit_decision(
523		&mut self,
524		formula: &'f Formula,
525		built: &FxHashMap<Src, ExecId>,
526		decision: &Decision,
527	) -> ExecId {
528		match decision {
529			Decision::Alias(src) => built[src],
530			Decision::Leaf(index) => self.leaf(*index, &formula.leaves[*index]),
531			Decision::Complement(src) => self.complement(built[src]),
532			Decision::Boolean(op, a, b, fixed) => self.boolean(*op, built[a], built[b], *fixed),
533			Decision::IdentityOf(op, lhs, rhs, fixed) => {
534				let inner = self.boolean(*op, built[lhs], built[rhs], *fixed);
535
536				self.identity(inner)
537			}
538			Decision::Identity(src) => self.identity(built[src]),
539			Decision::Project(src, tapes) => self.project(built[src], tapes),
540			Decision::PadClosure(src, pad) => self.pad_closure(built[src], pad),
541			Decision::Diagonal(u, x) => self.diagonal(built[u], built[x]),
542			Decision::DiagonalSync(u, a, b, spec) => {
543				let synced = self.sync(built[a], built[b], spec);
544
545				self.intersection(built[u], synced, false)
546			}
547			Decision::Sync(a, b, spec) => self.sync(built[a], built[b], spec),
548			Decision::CompositionChain(leaves, spec) => {
549				let leaves: Vec<ExecId> = leaves.iter().map(|src| built[src]).collect();
550
551				self.compose_chain(&leaves, spec)
552			}
553			Decision::Prefixes(src) => self.prefixes(built[src]),
554			Decision::Explicit(src, options, origin) => {
555				self.explicit(built[src], *options, *origin)
556			}
557		}
558	}

560	fn unary_cost(&self, child: ExecId) -> Cost {
561		let cost = self.costs[child];
562
563		Cost {
564			depth: cost.depth + 1,
565			log_states: cost.log_states,
566		}
567	}
568
569	fn product_cost(&self, a: ExecId, b: ExecId) -> Cost {
570		let (a, b) = (self.costs[a], self.costs[b]);
571
572		Cost {
573			depth: a.depth.max(b.depth) + 1,
574			log_states: a.log_states + b.log_states,
575		}
576	}
577
578	fn leaf(&mut self, index: LeafId, leaf: &'f Leaf) -> ExecId {
579		let states = u32::try_from(leaf.num_states().max(1)).unwrap_or(u32::MAX);
580		let cost = Cost {
581			depth: 0,
582			log_states: f64::from(states).log2(),
583		};
584		let op = Op::Leaf(LeafRef::Borrowed(leaf));
585
586		self.emit(Key::Leaf(index), op, leaf.arity(), cost)
587	}
588
589	/// `fixed` keeps the order of the operands of an intersection.
590	fn boolean(&mut self, op: Boolean, a: ExecId, b: ExecId, fixed: bool) -> ExecId {
591		match op {
592			Boolean::Union => self.union(a, b),
593			Boolean::Intersection => self.intersection(a, b, fixed),
594		}
595	}
596
597	/// `fixed` keeps the order of the operands, which otherwise puts the cheaper one left.
598	fn intersection(&mut self, a: ExecId, b: ExecId, fixed: bool) -> ExecId {
599		if a == b {
600			return a;
601		}
602
603		let (a, b, key) = if fixed {
604			(a, b, Key::OrderedIntersection(a, b))
605		} else if self.costs[a] <= self.costs[b] {
606			(a, b, Key::Intersection(a.min(b), a.max(b)))
607		} else {
608			(b, a, Key::Intersection(a.min(b), a.max(b)))
609		};
610
611		let cost = self.product_cost(a, b);
612		let arity = self.nodes[a].arity;
613
614		self.emit(key, Op::Intersection(a, b), arity, cost)
615	}
616
617	/// The branches that `node` adds to a union above it.
618	fn push_branches(&self, node: ExecId, branches: &mut Vec<ExecId>) {
619		match &self.nodes[node].op {
620			Op::Union(inner) => branches.extend_from_slice(inner),
621			_ => branches.push(node),
622		}
623	}
624
625	fn union(&mut self, a: ExecId, b: ExecId) -> ExecId {
626		let mut branches = Vec::new();
627		self.push_branches(a, &mut branches);
628		self.push_branches(b, &mut branches);
629
630		self.union_of(branches)
631	}
632
633	/// One union node over `branches`, as a chain of binary unions would copy every macrostate into each union
634	/// above it.
635	fn union_of(&mut self, mut branches: Vec<ExecId>) -> ExecId {
636		branches.sort_unstable();
637		branches.dedup();
638		if let [only] = branches[..] {
639			return only;
640		}
641
642		let key = Key::Union(branches.as_slice().into());
643		let costs = branches.iter().map(|&node| self.costs[node]);
644		let cost = Cost {
645			depth: costs.clone().map(|cost| cost.depth).max().unwrap_or(0) + 1,
646			log_states: log2_sum(costs.map(|cost| cost.log_states)),
647		};
648		branches.sort_by_key(|&branch| self.costs[branch]);
649		let arity = self.nodes[branches[0]].arity;
650
651		self.emit(key, Op::Union(branches.into()), arity, cost)
652	}

654	fn identity(&mut self, child: ExecId) -> ExecId {
655		let cost = self.unary_cost(child);
656
657		self.emit(Key::Identity(child), Op::Identity(child), 2, cost)
658	}

660	fn prefixes(&mut self, child: ExecId) -> ExecId {
661		if matches!(self.nodes[child].op, Op::Prefixes(_)) {
662			return child;
663		}
664
665		let cost = self.unary_cost(child);
666		let arity = self.nodes[child].arity;
667
668		self.emit(Key::Prefixes(child), Op::Prefixes(child), arity, cost)
669	}

671	fn complement(&mut self, child: ExecId) -> ExecId {
672		if let Op::Complement(inner) = self.nodes[child].op
673			&& !self.pads[inner]
674		{
675			return inner;
676		}
677
678		let mut cost = self.unary_cost(child);
679		// Up to one macrostate per set of the child's states.
680		cost.log_states = cost.log_states.exp2();
681		let arity = self.nodes[child].arity;
682
683		self.emit(Key::Complement(child), Op::Complement(child), arity, cost)
684	}

709	/// Composes the relations in `leaves`, from left to right, minimizing the sum of estimated macrostate bounds of
710	/// every composition below the root. This is optimal under [`Cost`], not necessarily for actual lazy exploration.
711	fn compose_chain(&mut self, leaves: &[ExecId], spec: &Arc<SyncSpec>) -> ExecId {
712		#[derive(Clone, Copy)]
713		struct Choice {
714			work: f64,
715			depth: u32,
716			split: usize,
717		}
718
719		debug_assert!(leaves.len() >= 3);
720		let mut trees: Vec<Vec<ExecId>> = (0..leaves.len())
721			.map(|index| vec![leaves[index]; leaves.len()])
722			.collect();
723		let mut best = vec![
724			vec![
725				Choice {
726					work: f64::INFINITY,
727					depth: 0,
728					split: 0,
729				};
730				leaves.len()
731			];
732			leaves.len()
733		];
734		for (index, &leaf) in leaves.iter().enumerate() {
735			best[index][index].depth = self.costs[leaf].depth;
736		}
737
738		// `work[t]` is log2(sum of internal-node bounds in t); it omits only t's own root. A leaf has none.
739		for len in 2..=leaves.len() {
740			for start in 0..=leaves.len() - len {
741				let end = start + len - 1;
742				for split in start + 1..=end {
743					let (left, right) = (best[start][split - 1], best[split][end]);
744					let candidate = Choice {
745						work: log2_sum([left.work, right.work].into_iter()),
746						depth: left.depth.max(right.depth) + 1,
747						split,
748					};
749					let current = best[start][end];
750					if current.split == 0
751						|| (
752							candidate.work.total_cmp(&current.work),
753							candidate.depth.cmp(&current.depth),
754						) < (Ordering::Equal, Ordering::Equal)
755					{
756						best[start][end] = candidate;
757					}
758				}
759
760				let split = best[start][end].split;
761				trees[start][end] = self.sync(trees[start][split - 1], trees[split][end], spec);
762			}
763		}
764
765		trees[0][leaves.len() - 1]
766	}

768	fn sync(&mut self, a: ExecId, b: ExecId, spec: &Arc<SyncSpec>) -> ExecId {
769		let cost = self.product_cost(a, b);
770		let key = Key::Sync(a, b, Arc::clone(spec));
771		let op = Op::Sync(a, b, Arc::clone(spec));
772
773		self.emit(key, op, spec.layout.len(), cost)
774	}

776	fn explicit(&mut self, child: ExecId, options: ExplicitOptions, origin: NodeId) -> ExecId {
777		// The outer node adds nothing, unless the inner one may stay lazy at fewer states than it allows.
778		if let Op::Explicit { options: inner, .. } = self.nodes[child].op
779			&& inner
780				.max_states
781				.is_none_or(|inner| options.max_states.is_some_and(|outer| outer <= inner))
782		{
783			return child;
784		}
785
786		let cost = self.unary_cost(child);
787		let op = Op::Explicit {
788			child,
789			options,
790			origin,
791		};
792		let arity = self.nodes[child].arity;
793
794		self.emit(Key::Explicit(child, origin), op, arity, cost)
795	}
796}

798/// `log2(Σ 2^x)` over `logs`, taken relative to the largest so that it does not overflow.
799fn log2_sum(logs: impl Iterator<Item = f64> + Clone) -> f64 {
800	let max = logs.clone().fold(f64::NEG_INFINITY, f64::max);
801	if max.is_infinite() {
802		return max;
803	}
804
805	max + logs.map(|log| (log - max).exp2()).sum::<f64>().log2()
806}

808/// Builds the execution DAG of `root`, with the query's `annotations`.
809pub(crate) fn build<'f>(formula: &'f Formula, root: NodeId, annotations: &Annotations) -> Exec<'f> {
810	let decisions = Rewriter {
811		formula,
812		annotations,
813	}
814	.decide_all(root);
815	let aliases = aliases(&decisions);
816	let resolve = |src: Src| aliases.get(&src).copied().unwrap_or(src);
817	let inner_unions = inner_unions(&decisions, &aliases, resolve(plain(root)));
818	let mut emitter = Emitter::default();
819	let mut built = FxHashMap::default();
820	// The branches of the unions that only one union above uses, which get no node of their own.
821	let mut pending: FxHashMap<Src, Vec<ExecId>> = FxHashMap::default();
822	for (&src, decision) in &decisions {
823		let Decision::Boolean(Boolean::Union, a, b, _) = decision else {
824			// The union above an alias of a pending union takes its branches.
825			if !(matches!(decision, Decision::Alias(_)) && inner_unions.contains(&resolve(src))) {
826				let id = emitter.emit_decision(formula, &built, decision);
827				built.insert(src, id);
828			}
829
830			continue;
831		};
832
833		let [first, second] = [a, b].map(|operand| {
834			pending.remove(&resolve(*operand)).unwrap_or_else(|| {
835				let mut branches = Vec::new();
836				emitter.push_branches(built[operand], &mut branches);
837
838				branches
839			})
840		});
841
842		// The larger list takes the smaller, so that a chain of unions costs no more than sorting its branches.
843		let (mut branches, smaller) = if first.len() >= second.len() {
844			(first, second)
845		} else {
846			(second, first)
847		};
848
849		branches.extend(smaller);
850		if inner_unions.contains(&src) {
851			pending.insert(src, branches);
852		} else {
853			built.insert(src, emitter.union_of(branches));
854		}
855	}
856
857	let root = built[&plain(root)];
858	let mut nodes = emitter.nodes;
859	nodes.truncate(root.index() + 1);
860	let reachable = reachable(&nodes, root);
861	let mut sources: Vec<(NodeId, ExecId)> = built
862		.into_iter()
863		.filter(|&((_, form), id)| form == Form::Plain && id <= root && reachable[id])
864		.map(|((node, _), id)| (node, id))
865		.collect();
866	sources.sort_unstable();
867
868	Exec {
869		nodes,
870		root,
871		reachable,
872		sources,
873	}
874}

876/// The node that each alias stands for, past chains of aliases.
877fn aliases(decisions: &BTreeMap<Src, Decision>) -> FxHashMap<Src, Src> {
878	let mut aliases = FxHashMap::default();
879	for (&src, decision) in decisions {
880		if let Decision::Alias(target) = decision {
881			let target = aliases.get(target).copied().unwrap_or(*target);
882			aliases.insert(src, target);
883		}
884	}
885
886	aliases
887}
888
889/// The unions, other than `root`, whose only user is one union, counting the users of their aliases.
890fn inner_unions(
891	decisions: &BTreeMap<Src, Decision>,
892	aliases: &FxHashMap<Src, Src>,
893	root: Src,
894) -> FxHashSet<Src> {
895	let mut users: FxHashMap<Src, (usize, bool)> = FxHashMap::default();
896	for decision in decisions.values() {
897		if matches!(decision, Decision::Alias(_)) {
898			continue;
899		}
900
901		let by_union = matches!(decision, Decision::Boolean(Boolean::Union, ..));
902		for dep in decision.deps() {
903			let dep = aliases.get(&dep).copied().unwrap_or(dep);
904			let (count, all_unions) = users.entry(dep).or_insert((0, true));
905			*count += 1;
906			*all_unions &= by_union;
907		}
908	}
909
910	decisions
911		.iter()
912		.filter(|&(&src, decision)| {
913			src != root
914				&& matches!(decision, Decision::Boolean(Boolean::Union, ..))
915				&& users.get(&src) == Some(&(1, true))
916		})
917		.map(|(&src, _)| src)
918		.collect()
919}

921/// Whether `from` reaches each of `nodes[..=from]`, where children come before their parents.
922pub(crate) fn reachable(nodes: &TiVec<ExecId, ExecNode<'_>>, from: ExecId) -> TiVec<ExecId, bool> {
923	let mut reachable = TiVec::from(vec![false; from.index() + 1]);
924	reachable[from] = true;
925	for node in reachable.keys().rev() {
926		if reachable[node] {
927			for child in nodes[node].op.children() {
928				reachable[child] = true;
929			}
930		}
931	}
932
933	reachable
934}
