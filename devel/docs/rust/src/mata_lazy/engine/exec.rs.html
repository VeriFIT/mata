1//! The execution DAG of one query, which is the formula after rewrites, with shared subterms merged.
2//!
3//! The rewrites keep the relation.
4//! - complements are pushed down to leaves and non-Boolean nodes (De Morgan, and `¬¬X = X` for an `X`
5//!   without padding, as a complement has no padded words);
6//! - `op(id(A), id(B)) = id(op(A, B))` for union and intersection;
7//! - projections fold, `project(id(L), [i]) = L`, `project(id(L), [0, 1] or [1, 0]) = id(L)`, and a projection
8//!   to all tapes in order is dropped;
9//! - `project(id(U) ∩ X, [i])` becomes one diagonal node, or `U ∩ X'` when `X` is a sync product of a tape of
10//!   each operand, where `X'` also synchronizes those two tapes, so it builds only the moves that the diagonal keeps;
11//! - `op(X, X) = X`, `prefixes(prefixes(X)) = prefixes(X)`, and structurally equal nodes are built once;
12//! - the operand of the smaller bound on its macrostates goes left, and an automatic intersection reassociates,
13//!   `(A ∩ B) ∩ C` becoming the cheapest of the three pairwise shapes;
14//! - a composition chain is re-associated to minimize its estimated intermediate macrostate bounds, without
15//!   changing the order of the relations.
16//!
17//! Explicit nodes and nodes with a fixed operand order stop composition-chain flattening.
18//!
19//! Explicit nodes stop the rewrites from looking into them. The query's annotations make nodes explicit, keep
20//! explicit nodes of the formula lazy, and fix the operand order of an intersection (not one that De Morgan
21//! turns into a union) or of a synchronized product.
22

56#[derive(Clone, Debug)]
57pub(crate) enum Op<'f> {
58	Leaf(LeafRef<'f>),
59	/// Nested unions flattened into one node, cheaper branches first.
60	Union(Arc<[ExecId]>),
61	Intersection(ExecId, ExecId),
62	Complement(ExecId),
63	Identity(ExecId),
64	Project(ExecId, Arc<[usize]>),
65	/// `{w | w·p^k ∈ child}`; its macrostates are a state of the child and the number of the last step read
66	/// (see [`PadSteps`](super::pad::PadSteps)).
67	PadClosure(ExecId, Arc<Pad>),
68	Sync(ExecId, ExecId, Arc<SyncSpec>),
69	/// `{w | w ∈ lhs, (w, w) ∈ rhs}`, with pair macrostates.
70	Diagonal(ExecId, ExecId),
71	/// Replaced by a leaf, or by [`Transparent`](Op::Transparent), before the search.
72	Explicit {
73		child: ExecId,
74		options: ExplicitOptions,
75		origin: NodeId,
76	},
77	/// An explicit node that stayed lazy, whose macrostates are its child's.
78	Transparent(ExecId),
79	/// The prefix closure, whose macrostates are its child's, accepting when they reach an accepting one.
80	Prefixes(ExecId),
81}

83impl Op<'_> {
84	pub(crate) fn children(&self) -> impl Iterator<Item = ExecId> {
85		let (operands, branches): ([Option<ExecId>; 2], &[ExecId]) = match self {
86			Op::Leaf(_) => ([None, None], &[]),
87			Op::Union(branches) => ([None, None], branches),
88			Op::Intersection(a, b) | Op::Sync(a, b, _) | Op::Diagonal(a, b) => {
89				([Some(*a), Some(*b)], &[])
90			}
91			Op::Complement(a)
92			| Op::Identity(a)
93			| Op::Project(a, _)
94			| Op::PadClosure(a, _)
95			| Op::Explicit { child: a, .. }
96			| Op::Transparent(a)
97			| Op::Prefixes(a) => ([Some(*a), None], &[]),
98		};
99
100		operands
101			.into_iter()
102			.flatten()
103			.chain(branches.iter().copied())
104	}
105
106	/// The child whose macrostates this node shares, if any.
107	pub(crate) fn transparent_child(&self) -> Option<ExecId> {
108		match *self {
109			Op::Identity(child)
110			| Op::Project(child, _)
111			| Op::Transparent(child)
112			| Op::Prefixes(child) => Some(child),
113			_ => None,
114		}
115	}
116}

136	pub(crate) fn sharing(&self, node: ExecId) -> impl Iterator<Item = ExecId> {
137		std::iter::successors(Some(node), |&node| self.nodes[node].op.transparent_child())

176/// What a source node in one form becomes in the DAG.
177#[derive(Debug)]
178enum Decision {
179	Alias(Src),
180	Leaf(LeafId),
181	Complement(Src),
182	/// The operation, its operands, and whether their order is fixed.
183	Boolean(Boolean, Src, Src, bool),
184	IdentityOf(Boolean, Src, Src, bool),
185	Identity(Src),
186	Project(Src, Arc<[usize]>),
187	PadClosure(Src, Arc<Pad>),
188	Diagonal(Src, Src),
189	/// `U ∩ X'` for the diagonal of `U` and a sync product, with the product's two tapes synchronized.
190	DiagonalSync(Src, Src, Src, Arc<SyncSpec>),
191	Sync(Src, Src, Arc<SyncSpec>),
192	/// Binary compositions in their original order; explicit and fixed-order nodes are opaque.
193	CompositionChain(Box<[Src]>, Arc<SyncSpec>),
194	Explicit(Src, ExplicitOptions, NodeId),
195	Prefixes(Src),
196}
197
198impl Decision {
199	fn deps(&self) -> Vec<Src> {
200		match self {
201			Decision::Leaf(_) => Vec::new(),
202			Decision::Alias(s)
203			| Decision::Complement(s)
204			| Decision::Identity(s)
205			| Decision::Project(s, _)
206			| Decision::PadClosure(s, _)
207			| Decision::Explicit(s, ..)
208			| Decision::Prefixes(s) => vec![*s],
209			Decision::Boolean(_, a, b, _)
210			| Decision::IdentityOf(_, a, b, _)
211			| Decision::Diagonal(a, b)
212			| Decision::Sync(a, b, _) => vec![*a, *b],
213			Decision::CompositionChain(leaves, _) => leaves.to_vec(),
214			Decision::DiagonalSync(u, a, b, _) => vec![*u, *a, *b],
215		}
216	}
217}

219struct Rewriter<'a> {
220	formula: &'a Formula,
221	annotations: &'a Annotations,
222}
223
224impl Rewriter<'_> {
225	fn is_marked(&self, node: NodeId) -> bool {
226		self.annotations.explicit.contains_key(&node)
227			&& !matches!(self.formula.nodes[node], Node::Explicit(..))
228	}
229
230	fn order(&self, node: NodeId) -> Order {
231		self.annotations
232			.order
233			.get(&node)
234			.copied()
235			.unwrap_or_default()
236	}
237
238	fn identity_child(&self, node: NodeId) -> Option<NodeId> {
239		match self.formula.nodes[node] {
240			Node::Identity(child) if !self.is_marked(node) => Some(child),
241			_ => None,
242		}
243	}
244
245	/// The decision of every source node that `root` depends on, in an order where dependencies come first (a
246	/// smaller node, or the same node in an earlier form).
247	fn decide_all(&self, root: NodeId) -> BTreeMap<Src, Decision> {
248		let mut decisions = BTreeMap::new();
249		let mut stack = vec![plain(root)];
250		while let Some(src) = stack.pop() {
251			if let btree_map::Entry::Vacant(entry) = decisions.entry(src) {
252				let decision = self.decide(src);
253				stack.extend(decision.deps());
254				entry.insert(decision);
255			}
256		}
257
258		decisions
259	}
260
261	fn decide(&self, (node, form): Src) -> Decision {
262		if self.is_marked(node) {
263			match form {
264				Form::Plain => {
265					let options = self.annotations.explicit[&node];
266
267					return Decision::Explicit((node, Form::Inner), options, node);
268				}
269				Form::Negated => return Decision::Complement(plain(node)),
270				Form::Inner => {}
271			}
272		}
273
274		let negated = form == Form::Negated;
275
276		match &self.formula.nodes[node] {
277			Node::Explicit(child, _) if self.annotations.lazy.contains(&node) => {
278				Decision::Alias((*child, form))
279			}
280			Node::Complement(child) if negated && self.formula.pads[*child] => {
281				Decision::Complement(plain(node))
282			}
283			Node::Complement(child) => {
284				Decision::Alias((*child, if negated { Form::Plain } else { Form::Negated }))
285			}
286			Node::Union(a, b) => self.decide_boolean(Boolean::Union, *a, *b, negated, Order::Auto),
287			Node::Intersection(a, b) => {
288				let order = self.order(node);
289
290				self.decide_boolean(Boolean::Intersection, *a, *b, negated, order)
291			}
292			_ if negated => Decision::Complement(plain(node)),
293			Node::Leaf(leaf) => Decision::Leaf(*leaf),
294			Node::Identity(child) => Decision::Identity(plain(*child)),
295			Node::Project(child, tapes) => self.decide_project(*child, tapes),
296			Node::PadClosure(child, pad) => Decision::PadClosure(plain(*child), Arc::clone(pad)),
297			Node::Sync(a, b, spec) if self.order(node) == Order::Auto && spec.is_composition() => {
298				self.decide_composition(*a, *b, spec)
299			}
300			Node::Sync(a, b, spec) if self.order(node) == Order::Swapped => {
301				Decision::Sync(plain(*b), plain(*a), Arc::new(spec.swapped()))
302			}
303			Node::Sync(a, b, spec) => Decision::Sync(plain(*a), plain(*b), Arc::clone(spec)),
304			Node::Explicit(child, options) => {
305				let options = self.annotations.explicit.get(&node).unwrap_or(options);
306
307				Decision::Explicit(plain(*child), *options, node)
308			}
309			Node::Prefixes(child) => Decision::Prefixes(plain(*child)),
310		}
311	}
312
313	fn composition_parts(&self, node: NodeId) -> Option<(NodeId, NodeId, &Arc<SyncSpec>)> {
314		if self.is_marked(node) || self.order(node) != Order::Auto {
315			return None;
316		}
317
318		match &self.formula.nodes[node] {
319			Node::Sync(a, b, spec)
320				if spec.is_composition()
321					&& self.formula.arities[*a] == 2
322					&& self.formula.arities[*b] == 2 =>
323			{
324				Some((*a, *b, spec))
325			}
326			_ => None,
327		}
328	}
329
330	/// Flattens a chain without changing the order of its relations. Only a canonical composition may be opened;
331	/// all other operands are opaque leaves.
332	fn decide_composition(&self, a: NodeId, b: NodeId, spec: &Arc<SyncSpec>) -> Decision {
333		let mut pending = vec![b, a];
334		let mut leaves = Vec::new();
335		while let Some(node) = pending.pop() {
336			if let Some((a, b, _)) = self.composition_parts(node) {
337				pending.extend([b, a]);
338			} else {
339				leaves.push(plain(node));
340			}
341		}
342
343		if leaves.len() > 2 {
344			Decision::CompositionChain(leaves.into(), Arc::clone(spec))
345		} else {
346			Decision::Sync(plain(a), plain(b), Arc::clone(spec))
347		}
348	}
349
350	fn decide_boolean(
351		&self,
352		op: Boolean,
353		a: NodeId,
354		b: NodeId,
355		negated: bool,
356		order: Order,
357	) -> Decision {
358		if negated {
359			return Decision::Boolean(op.dual(), (a, Form::Negated), (b, Form::Negated), false);
360		}
361
362		let (a, b) = if order == Order::Swapped {
363			(b, a)
364		} else {
365			(a, b)
366		};
367		let fixed = order != Order::Auto;
368
369		match (self.identity_child(a), self.identity_child(b)) {
370			(Some(lhs), Some(rhs)) => Decision::IdentityOf(op, plain(lhs), plain(rhs), fixed),
371			_ => Decision::Boolean(op, plain(a), plain(b), fixed),
372		}
373	}
374
375	fn decide_project(&self, child: NodeId, tapes: &Arc<[usize]>) -> Decision {
376		let mut base = child;
377		let mut kept = Arc::clone(tapes);
378		while !self.is_marked(base)
379			&& let Node::Project(inner, inner_tapes) = &self.formula.nodes[base]
380		{
381			kept = kept.iter().map(|&tape| inner_tapes[tape]).collect();
382			base = *inner;
383		}
384
385		if let Some(language) = self.identity_child(base) {
386			match kept.len() {
387				1 => return Decision::Alias(plain(language)),
388				2 => return Decision::Alias(plain(base)),
389				_ => {}
390			}
391		}
392
393		if kept.iter().copied().eq(0..self.formula.arities[base]) {
394			return Decision::Alias(plain(base));
395		}
396
397		if kept.len() == 1
398			&& !self.is_marked(base)
399			&& self.order(base) == Order::Auto
400			&& let Node::Intersection(p, q) = self.formula.nodes[base]
401			&& self.formula.arities[base] == 2
402		{
403			if let Some(u) = self.identity_child(p) {
404				return self.decide_diagonal(u, q);
405			}
406
407			if let Some(u) = self.identity_child(q) {
408				return self.decide_diagonal(u, p);
409			}
410		}
411
412		Decision::Project(plain(base), kept)
413	}
414
415	/// `{w | w ∈ u, (w, w) ∈ x}`.
416	fn decide_diagonal(&self, u: NodeId, x: NodeId) -> Decision {
417		if !self.is_marked(x)
418			&& self.order(x) == Order::Auto
419			&& let Node::Sync(a, b, spec) = &self.formula.nodes[x]
420			&& let [first, second] = *spec.layout
421			&& first.peer.is_none()
422			&& second.peer.is_none()
423			&& let (Tape::Lhs(lhs), Tape::Rhs(rhs)) | (Tape::Rhs(rhs), Tape::Lhs(lhs)) =
424				(first.source, second.source)
425		{
426			let spec = SyncSpec {
427				pairs: spec
428					.pairs
429					.iter()
430					.copied()
431					.chain([TapePair { lhs, rhs }])
432					.collect(),
433				layout: [ResultTape {
434					source: Tape::Lhs(lhs),
435					peer: Some(rhs),
436				}]
437				.into(),
438			};
439
440			return Decision::DiagonalSync(plain(u), plain(*a), plain(*b), Arc::new(spec));
441		}
442
443		Decision::Diagonal(plain(u), plain(x))
444	}
445}

447/// The structure of a node, by which equal nodes are built once.

449enum Key {

451	Union(Arc<[ExecId]>),
452	Intersection(ExecId, ExecId),
453	/// An intersection whose operands an annotation ordered.
454	OrderedIntersection(ExecId, ExecId),

457	Project(ExecId, Arc<[usize]>),
458	PadClosure(ExecId, Arc<Pad>),

460	Diagonal(ExecId, ExecId),
461	Explicit(ExecId, NodeId),
462	Prefixes(ExecId),

465/// A bound on the number of macrostates, and the depth, to order operands: the smaller bound goes left, depth only
466/// breaking ties, as a shallower operand can still be the larger one in a product join.
467#[derive(Clone, Copy)]
468struct Cost {
469	depth: u32,
470	log_states: f64,
471}
472
473impl Ord for Cost {
474	fn cmp(&self, other: &Cost) -> Ordering {
475		self.log_states
476			.total_cmp(&other.log_states)
477			.then(self.depth.cmp(&other.depth))
478	}
479}
480
481impl PartialOrd for Cost {
482	fn partial_cmp(&self, other: &Cost) -> Option<Ordering> {
483		Some(self.cmp(other))
484	}
485}
486
487impl PartialEq for Cost {
488	fn eq(&self, other: &Cost) -> bool {
489		self.cmp(other).is_eq()
490	}
491}

495#[derive(Default)]
496struct Emitter<'f> {
497	nodes: TiVec<ExecId, ExecNode<'f>>,
498	costs: TiVec<ExecId, Cost>,
499	pads: TiVec<ExecId, bool>,
500	interned: FxHashMap<Key, ExecId>,
501	/// Intersections already reassociated, so that the pairwise shape they chose is not reassociated again.
502	reassociated: FxHashSet<ExecId>,
503	/// Intersections whose operands an annotation ordered, which keep the shape the plan chose in a reassociation.
504	ordered: FxHashSet<ExecId>,
505}
506
507impl<'f> Emitter<'f> {
508	fn emit(&mut self, key: Key, op: Op<'f>, arity: usize, cost: Cost) -> ExecId {
509		match self.interned.entry(key) {
510			Entry::Occupied(entry) => *entry.get(),
511			Entry::Vacant(entry) => {
512				let pads = match &op {
513					Op::Leaf(leaf) => leaf.has_padding,
514					Op::Complement(_) => false,
515					op => op.children().any(|child| self.pads[child]),
516				};
517
518				self.pads.push(pads);
519				let id = self.nodes.push_and_get_key(ExecNode { op, arity });
520				self.costs.push(cost);
521
522				*entry.insert(id)
523			}
524		}
525	}
526
527	/// The node of `decision`, whose dependencies are `built`.
528	fn emit_decision(
529		&mut self,
530		formula: &'f Formula,
531		built: &FxHashMap<Src, ExecId>,
532		decision: &Decision,
533	) -> ExecId {
534		match decision {
535			Decision::Alias(src) => built[src],
536			Decision::Leaf(index) => self.leaf(*index, &formula.leaves[*index]),
537			Decision::Complement(src) => self.complement(built[src]),
538			Decision::Boolean(op, a, b, fixed) => self.boolean(*op, built[a], built[b], *fixed),
539			Decision::IdentityOf(op, lhs, rhs, fixed) => {
540				let inner = self.boolean(*op, built[lhs], built[rhs], *fixed);
541
542				self.identity(inner)
543			}
544			Decision::Identity(src) => self.identity(built[src]),
545			Decision::Project(src, tapes) => self.project(built[src], tapes),
546			Decision::PadClosure(src, pad) => self.pad_closure(built[src], pad),
547			Decision::Diagonal(u, x) => self.diagonal(built[u], built[x]),
548			Decision::DiagonalSync(u, a, b, spec) => {
549				let synced = self.sync(built[a], built[b], spec);
550
551				self.intersection(built[u], synced, false)
552			}
553			Decision::Sync(a, b, spec) => self.sync(built[a], built[b], spec),
554			Decision::CompositionChain(leaves, spec) => {
555				let leaves: Vec<ExecId> = leaves.iter().map(|src| built[src]).collect();
556
557				self.compose_chain(&leaves, spec)
558			}
559			Decision::Prefixes(src) => self.prefixes(built[src]),
560			Decision::Explicit(src, options, origin) => {
561				self.explicit(built[src], *options, *origin)
562			}
563		}
564	}

566	fn unary_cost(&self, child: ExecId) -> Cost {
567		let cost = self.costs[child];
568
569		Cost {
570			depth: cost.depth + 1,
571			log_states: cost.log_states,
572		}
573	}
574
575	fn product_cost(&self, a: ExecId, b: ExecId) -> Cost {
576		let (a, b) = (self.costs[a], self.costs[b]);
577
578		Cost {
579			depth: a.depth.max(b.depth) + 1,
580			log_states: a.log_states + b.log_states,
581		}
582	}

584	fn leaf(&mut self, index: LeafId, leaf: &'f Leaf) -> ExecId {
585		let states = u32::try_from(leaf.num_states().max(1)).unwrap_or(u32::MAX);
586		let cost = Cost {
587			depth: 0,
588			log_states: f64::from(states).log2(),
589		};
590		let op = Op::Leaf(LeafRef::Borrowed(leaf));
591
592		self.emit(Key::Leaf(index), op, leaf.arity(), cost)
593	}

595	/// `fixed` keeps the order of the operands of an intersection. An automatic one reassociates over an
596	/// intersection on the left, `(A ∩ B) ∩ C` becoming the cheapest of the three pairwise shapes, below.
597	fn boolean(&mut self, op: Boolean, a: ExecId, b: ExecId, fixed: bool) -> ExecId {
598		match op {
599			Boolean::Union => self.union(a, b),
600			Boolean::Intersection if fixed => self.intersection(a, b, true),
601			Boolean::Intersection => self.associative_intersection(a, b),
602		}
603	}
604
605	/// The intersection of `a` and `b`, the cheaper operand left. An intersection on the left reassociates:
606	/// `(A ∩ B) ∩ C` becomes the cheapest of `(A ∩ B) ∩ C`, `(A ∩ C) ∩ B` and `(B ∩ C) ∩ A`, by the bound on the
607	/// macrostates of the outer product, as the join of a product explores each left move against the right ones,
608	/// so the pairwise shape decides what gets built, not the nesting of the formula. An annotated operand keeps
609	/// the shape the plan chose, an intersection whose operands the plan ordered keeps its grouping, and a
610	/// reassociated subterm is not reassociated again (which would undo it).
611	fn associative_intersection(&mut self, a: ExecId, b: ExecId) -> ExecId {
612		if let Op::Intersection(c, d) = self.nodes[a].op.clone()
613			&& !self.reassociated.contains(&a)
614			&& !self.ordered.contains(&a)
615			&& let Some((left, right)) = [
616				(a, b),
617				(self.intersection(c, b, false), d),
618				(self.intersection(d, b, false), c),
619			]
620			.into_iter()
621			.min_by_key(|&(outer, inner)| self.product_cost(outer, inner))
622			&& !self.annotated(left)
623			&& !self.annotated(right)
624		{
625			self.reassociated.insert(left);
626
627			return self.intersection(left, right, false);
628		}
629
630		self.intersection(a, b, false)
631	}
632
633	/// `fixed` keeps the order of the operands, which otherwise puts the cheaper one left.
634	fn intersection(&mut self, a: ExecId, b: ExecId, fixed: bool) -> ExecId {
635		if a == b {
636			return a;
637		}
638
639		let (a, b, key) = if fixed {
640			(a, b, Key::OrderedIntersection(a, b))
641		} else if self.costs[a] <= self.costs[b] {
642			(a, b, Key::Intersection(a.min(b), a.max(b)))
643		} else {
644			(b, a, Key::Intersection(a.min(b), a.max(b)))
645		};
646
647		let cost = self.product_cost(a, b);
648		let arity = self.nodes[a].arity;
649		let id = self.emit(key, Op::Intersection(a, b), arity, cost);
650
651		if fixed {
652			self.ordered.insert(id);
653		}
654
655		id
656	}
657
658	/// Whether any annotation of the query touches `node`, so that its shape is the plan's choice: a leaf and an
659	/// intersection are fine to reassociate around, an inner node is not (its plan may rely on its position, as
660	/// `Order` does, and its cost feeds the outer one).
661	fn annotated(&self, node: ExecId) -> bool {
662		!matches!(&self.nodes[node].op, Op::Leaf(_) | Op::Intersection(..))
663	}

665	/// The branches that `node` adds to a union above it.
666	fn push_branches(&self, node: ExecId, branches: &mut Vec<ExecId>) {
667		match &self.nodes[node].op {
668			Op::Union(inner) => branches.extend_from_slice(inner),
669			_ => branches.push(node),
670		}
671	}

673	fn union(&mut self, a: ExecId, b: ExecId) -> ExecId {
674		let mut branches = Vec::new();
675		self.push_branches(a, &mut branches);
676		self.push_branches(b, &mut branches);
677
678		self.union_of(branches)
679	}
680
681	/// One union node over `branches`, as a chain of binary unions would copy every macrostate into each union
682	/// above it.
683	fn union_of(&mut self, mut branches: Vec<ExecId>) -> ExecId {
684		branches.sort_unstable();
685		branches.dedup();
686		if let [only] = branches[..] {
687			return only;
688		}
689
690		let key = Key::Union(branches.as_slice().into());
691		let costs = branches.iter().map(|&node| self.costs[node]);
692		let cost = Cost {
693			depth: costs.clone().map(|cost| cost.depth).max().unwrap_or(0) + 1,
694			log_states: log2_sum(costs.map(|cost| cost.log_states)),
695		};
696		branches.sort_by_key(|&branch| self.costs[branch]);
697		let arity = self.nodes[branches[0]].arity;
698
699		self.emit(key, Op::Union(branches.into()), arity, cost)
700	}

719	fn complement(&mut self, child: ExecId) -> ExecId {
720		if let Op::Complement(inner) = self.nodes[child].op
721			&& !self.pads[inner]
722		{
723			return inner;
724		}
725
726		let mut cost = self.unary_cost(child);
727		// Up to one macrostate per set of the child's states.
728		cost.log_states = cost.log_states.exp2();
729		let arity = self.nodes[child].arity;
730
731		self.emit(Key::Complement(child), Op::Complement(child), arity, cost)
732	}

757	/// Composes the relations in `leaves`, from left to right, minimizing the sum of estimated macrostate bounds of
758	/// every composition below the root. This is optimal under [`Cost`], not necessarily for actual lazy exploration.
759	fn compose_chain(&mut self, leaves: &[ExecId], spec: &Arc<SyncSpec>) -> ExecId {
760		#[derive(Clone, Copy)]
761		struct Choice {
762			work: f64,
763			depth: u32,
764			split: usize,
765		}
766
767		debug_assert!(leaves.len() >= 3);
768		let mut trees: Vec<Vec<ExecId>> = (0..leaves.len())
769			.map(|index| vec![leaves[index]; leaves.len()])
770			.collect();
771		let mut best = vec![
772			vec![
773				Choice {
774					work: f64::INFINITY,
775					depth: 0,
776					split: 0,
777				};
778				leaves.len()
779			];
780			leaves.len()
781		];
782		for (index, &leaf) in leaves.iter().enumerate() {
783			best[index][index].depth = self.costs[leaf].depth;
784		}
785
786		// `work[t]` is log2(sum of internal-node bounds in t); it omits only t's own root. A leaf has none.
787		for len in 2..=leaves.len() {
788			for start in 0..=leaves.len() - len {
789				let end = start + len - 1;
790				for split in start + 1..=end {
791					let (left, right) = (best[start][split - 1], best[split][end]);
792					let candidate = Choice {
793						work: log2_sum([left.work, right.work].into_iter()),
794						depth: left.depth.max(right.depth) + 1,
795						split,
796					};
797					let current = best[start][end];
798					if current.split == 0
799						|| (
800							candidate.work.total_cmp(&current.work),
801							candidate.depth.cmp(&current.depth),
802						) < (Ordering::Equal, Ordering::Equal)
803					{
804						best[start][end] = candidate;
805					}
806				}
807
808				let split = best[start][end].split;
809				trees[start][end] = self.sync(trees[start][split - 1], trees[split][end], spec);
810			}
811		}
812
813		trees[0][leaves.len() - 1]
814	}

817		let cost = self.product_cost(a, b);
818		let key = Key::Sync(a, b, Arc::clone(spec));
819		let op = Op::Sync(a, b, Arc::clone(spec));

822	}

824	fn explicit(&mut self, child: ExecId, options: ExplicitOptions, origin: NodeId) -> ExecId {
825		// The outer node adds nothing, unless the inner one may stay lazy at fewer states than it allows.
826		if let Op::Explicit { options: inner, .. } = self.nodes[child].op
827			&& inner
828				.max_states
829				.is_none_or(|inner| options.max_states.is_some_and(|outer| outer <= inner))
830		{
831			return child;
832		}
833
834		let cost = self.unary_cost(child);
835		let op = Op::Explicit {
836			child,
837			options,
838			origin,
839		};
840		let arity = self.nodes[child].arity;
841
842		self.emit(Key::Explicit(child, origin), op, arity, cost)
843	}

846/// `log2(Σ 2^x)` over `logs`, taken relative to the largest so that it does not overflow.
847fn log2_sum(logs: impl Iterator<Item = f64> + Clone) -> f64 {
848	let max = logs.clone().fold(f64::NEG_INFINITY, f64::max);
849	if max.is_infinite() {
850		return max;
851	}
852
853	max + logs.map(|log| (log - max).exp2()).sum::<f64>().log2()
854}

856/// Builds the execution DAG of `root`, with the query's `annotations`.
857pub(crate) fn build<'f>(formula: &'f Formula, root: NodeId, annotations: &Annotations) -> Exec<'f> {
858	let decisions = Rewriter {
859		formula,
860		annotations,
861	}
862	.decide_all(root);
863	let aliases = aliases(&decisions);
864	let resolve = |src: Src| aliases.get(&src).copied().unwrap_or(src);
865	let inner_unions = inner_unions(&decisions, &aliases, resolve(plain(root)));
866	let mut emitter = Emitter::default();
867	let mut built = FxHashMap::default();
868	// The branches of the unions that only one union above uses, which get no node of their own.
869	let mut pending: FxHashMap<Src, Vec<ExecId>> = FxHashMap::default();
870	for (&src, decision) in &decisions {
871		let Decision::Boolean(Boolean::Union, a, b, _) = decision else {
872			// The union above an alias of a pending union takes its branches.
873			if !(matches!(decision, Decision::Alias(_)) && inner_unions.contains(&resolve(src))) {
874				let id = emitter.emit_decision(formula, &built, decision);
875				built.insert(src, id);
876			}
877
878			continue;
879		};
880
881		let [first, second] = [a, b].map(|operand| {
882			pending.remove(&resolve(*operand)).unwrap_or_else(|| {
883				let mut branches = Vec::new();
884				emitter.push_branches(built[operand], &mut branches);
885
886				branches
887			})
888		});
889
890		// The larger list takes the smaller, so that a chain of unions costs no more than sorting its branches.
891		let (mut branches, smaller) = if first.len() >= second.len() {
892			(first, second)
893		} else {
894			(second, first)
895		};
896
897		branches.extend(smaller);
898		if inner_unions.contains(&src) {
899			pending.insert(src, branches);
900		} else {
901			built.insert(src, emitter.union_of(branches));
902		}
903	}
904
905	let root = built[&plain(root)];
906	let mut nodes = emitter.nodes;
907	nodes.truncate(root.index() + 1);
908	let reachable = reachable(&nodes, root);
909	let mut sources: Vec<(NodeId, ExecId)> = built
910		.into_iter()
911		.filter(|&((_, form), id)| form == Form::Plain && id <= root && reachable[id])
912		.map(|((node, _), id)| (node, id))
913		.collect();
914	sources.sort_unstable();
915
916	Exec {
917		nodes,
918		root,
919		reachable,
920		sources,
921	}
922}

924/// The node that each alias stands for, past chains of aliases.
925fn aliases(decisions: &BTreeMap<Src, Decision>) -> FxHashMap<Src, Src> {
926	let mut aliases = FxHashMap::default();
927	for (&src, decision) in decisions {
928		if let Decision::Alias(target) = decision {
929			let target = aliases.get(target).copied().unwrap_or(*target);
930			aliases.insert(src, target);
931		}
932	}
933
934	aliases
935}

937/// The unions, other than `root`, whose only user is one union, counting the users of their aliases.
938fn inner_unions(
939	decisions: &BTreeMap<Src, Decision>,
940	aliases: &FxHashMap<Src, Src>,
941	root: Src,
942) -> FxHashSet<Src> {
943	let mut users: FxHashMap<Src, (usize, bool)> = FxHashMap::default();
944	for decision in decisions.values() {
945		if matches!(decision, Decision::Alias(_)) {
946			continue;
947		}
948
949		let by_union = matches!(decision, Decision::Boolean(Boolean::Union, ..));
950		for dep in decision.deps() {
951			let dep = aliases.get(&dep).copied().unwrap_or(dep);
952			let (count, all_unions) = users.entry(dep).or_insert((0, true));
953			*count += 1;
954			*all_unions &= by_union;
955		}
956	}
957
958	decisions
959		.iter()
960		.filter(|&(&src, decision)| {
961			src != root
962				&& matches!(decision, Decision::Boolean(Boolean::Union, ..))
963				&& users.get(&src) == Some(&(1, true))
964		})
965		.map(|(&src, _)| src)
966		.collect()
967}

969/// Whether `from` reaches each of `nodes[..=from]`, where children come before their parents.
970pub(crate) fn reachable(nodes: &TiVec<ExecId, ExecNode<'_>>, from: ExecId) -> TiVec<ExecId, bool> {
971	let mut reachable = TiVec::from(vec![false; from.index() + 1]);
972	reachable[from] = true;
973	for node in reachable.keys().rev() {
974		if reachable[node] {
975			for child in nodes[node].op.children() {
976				reachable[child] = true;
977			}
978		}
979	}
980
981	reachable
982}
