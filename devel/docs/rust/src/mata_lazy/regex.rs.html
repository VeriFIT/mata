1//! Leaves from [regex-automata](https://docs.rs/regex-automata) DFAs.
2//!
3//! A DFA reads bytes, and a byte is the symbol of the same number, as in Mata's `from_regex`. So a regex-automata
4//! leaf and a Mata leaf of the same pattern read the same symbols.
5//!
6//! ```
7//! use mata_lazy::Formula;
8//! use mata_lazy::regex::Regex;
9//!
10//! let mut formula = Formula::new();
11//! let ab = formula.leaf(&Regex::new("(ab)*")?)?;
12//! let a_or_b = formula.leaf(&Regex::new("[ab]*")?)?;
13//! let not_a_or_b = formula.complement(a_or_b)?;
14//! let difference = formula.intersection(ab, not_a_or_b)?;
15//! assert!(formula.is_empty(difference)?);
16//! # Ok::<(), Box<dyn std::error::Error>>(())
17//! ```
18

30type Dfa = dense::DFA<Vec<u32>>;
31
32/// Why a [`Regex`] could not be built.
33#[derive(Debug, thiserror::Error)]
34#[non_exhaustive]
35pub enum Error {
36	/// regex-automata could not build the DFA, for example for a syntax error.
37	#[error(transparent)]
38	Build(#[from] Box<dense::BuildError>),
39	/// The DFA has no anchored start state.
40	#[error(transparent)]
41	Start(#[from] StartError),
42	/// The DFA gives up on some input it can reach (a quit byte), so its language is not known.
43	#[error("the DFA quits on some input, so its language is not known")]
44	Quit,
45}

47/// The language of a regular expression, as a dense DFA that must match the whole word.
48#[derive(Clone, Debug)]
49pub struct Regex {
50	dfa: Dfa,
51	start: StateID,
52	/// The states that lead to a match. The others include the states that only report a match one byte
53	/// late, which would otherwise put every byte into the alphabet.
54	live: FxHashSet<StateID>,
55}
56
57impl Regex {
58	/// The DFA of `pattern`, over bytes, without Unicode classes (`.` is any byte but `\n`).
59	///
60	/// # Errors
61	///
62	/// [`Build`](Error::Build) if regex-automata cannot build the DFA, for example for a syntax error.
63	pub fn new(pattern: &str) -> Result<Regex, Error> {
64		let dfa = dense::Builder::new()
65			.configure(
66				dense::Config::new()
67					.start_kind(StartKind::Anchored)
68					.match_kind(MatchKind::All),
69			)
70			.syntax(syntax::Config::new().unicode(false).utf8(false))
71			.build(pattern)
72			.map_err(Box::new)?;
73
74		Regex::from_dfa(dfa)
75	}
76
77	/// A DFA built elsewhere. It must report all matches ([`All`](MatchKind::All)); otherwise words that a longer
78	/// match passes through are lost.
79	///
80	/// It must be a forward DFA: words are read in the order its transitions are followed, so a reverse DFA
81	/// (built with [`thompson::Config::reverse`](regex_automata::nfa::thompson::Config::reverse)) yields every
82	/// word reversed, such as `ba` for the pattern `ab`.
83	///
84	/// # Errors
85	///
86	/// [`Start`](Error::Start) if the DFA does not support anchored searches, or [`Quit`](Error::Quit) if it
87	/// can reach a quit byte, such as a non-ASCII byte under a heuristic Unicode word boundary.
88	pub fn from_dfa(dfa: dense::DFA<Vec<u32>>) -> Result<Regex, Error> {
89		let start = dfa.start_state(&start::Config::new().anchored(Anchored::Yes))?;
90		let live = live_states(&dfa, start)?;
91
92		Ok(Regex { dfa, start, live })
93	}
94}
95
96/// The moves of `state` that do not lead to the dead state.
97fn targets(dfa: &Dfa, state: StateID) -> impl Iterator<Item = (u8, StateID)> {
98	(u8::MIN..=u8::MAX)
99		.map(move |byte| (byte, dfa.next_state(state, byte)))
100		.filter(|&(_, target)| !dfa.is_dead_state(target))
101}
102
103/// A DFA reports a match one byte late, so a state accepts if the end of input leads to a match.
104fn accepts(dfa: &Dfa, state: StateID) -> bool {
105	dfa.is_match_state(dfa.next_eoi_state(state))
106}
107
108/// The states reachable from `start` that reach an accepting state; [`Quit`](Error::Quit) if a quit state is
109/// reachable.
110fn live_states(dfa: &Dfa, start: StateID) -> Result<FxHashSet<StateID>, Error> {
111	let mut predecessors: FxHashMap<StateID, Vec<StateID>> = FxHashMap::default();
112	let mut reached = FxHashSet::from_iter([start]);
113	let mut stack = vec![start];
114	while let Some(state) = stack.pop() {
115		for (_, target) in targets(dfa, state) {
116			if dfa.is_quit_state(target) {
117				return Err(Error::Quit);
118			}
119
120			predecessors.entry(target).or_default().push(state);
121			if reached.insert(target) {
122				stack.push(target);
123			}
124		}
125	}
126
127	let mut live: FxHashSet<StateID> = reached
128		.into_iter()
129		.filter(|&state| accepts(dfa, state))
130		.collect();
131	let mut stack: Vec<StateID> = live.iter().copied().collect();
132	while let Some(state) = stack.pop() {
133		for &source in predecessors.get(&state).into_iter().flatten() {
134			if live.insert(source) {
135				stack.push(source);
136			}
137		}
138	}
139
140	Ok(live)
141}

143impl Automaton for Regex {
144	type State = StateID;
145
146	fn arity(&self) -> usize {
147		1
148	}
149
150	fn initial_states(&self) -> impl IntoIterator<Item = StateID> {
151		[self.start]
152	}
153
154	fn is_final(&self, state: StateID) -> bool {
155		accepts(&self.dfa, state)
156	}
157
158	fn for_each_move(&self, state: StateID, mut visit: impl FnMut(&[Symbol], StateID)) {
159		for (byte, target) in targets(&self.dfa, state) {
160			if self.live.contains(&target) {
161				visit(&[Symbol::from(byte)], target);
162			}
163		}
164	}
165}

167impl FromStr for Regex {
168	type Err = Error;
169
170	fn from_str(pattern: &str) -> Result<Regex, Error> {
171		Regex::new(pattern)
172	}
