7/// How [`Nfa::determinize_with`](crate::Nfa::determinize_with) determinizes.
8/// Both give the same DFA, but they may number its states differently.
9#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
10#[non_exhaustive]
11pub enum Determinization {
12	/// The "Four Russians" subset construction: the post of each part of a macrostate inside a block of
13	/// states is computed once and kept in a table. Usually much faster on large automata.
14	#[default]
15	FourRussians,
16	/// The classical subset construction.
17	Classical,
18}
19
20/// How [`Nfa::minimize_with`](crate::Nfa::minimize_with) minimizes. The minimizing strategies give a trimmed DFA.
21#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
22#[non_exhaustive]
23pub enum Minimization {
24	/// Brzozowski's algorithm: determinize the reversal twice. Works on any automaton, but can blow up
25	/// exponentially even on a DFA.
26	Brzozowski,
27	/// Hopcroft's partition refinement on a trimmed DFA, without input normalization.
28	Hopcroft,
29	/// Normalize an arbitrary automaton, then apply Hopcroft's partition refinement.
30	#[default]
31	HopcroftNondet,
32}
33
34/// How [`Nfa::complement_with`](crate::Nfa::complement_with) complements.
35#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
36#[non_exhaustive]
37pub enum Complementation {
38	/// Determinize, complete with a sink state, and swap final and non-final states.
39	#[default]
40	Classical,
41	/// Minimize with Brzozowski's algorithm, complete, and swap final and non-final states.
42	Brzozowski,
43}

45/// How [`Nfa::reduce_with`](crate::Nfa::reduce_with) shrinks an automaton. The result keeps the language
46/// and may stay nondeterministic.
47#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
48#[non_exhaustive]
49pub enum Reduction {
50	/// Merge states related by forward simulation.
51	#[default]
52	Simulation,
53	/// Build the residual automaton (the canonical RFSA).
54	Residual {
55		/// When the residual states are computed.
56		construction: ResidualConstruction,
57		/// Forward or backward residuals.
58		direction: Direction,
59	},
60}
61
62/// When a residual reduction computes the residual states.
63#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
64#[non_exhaustive]
65pub enum ResidualConstruction {
66	/// After a full determinization (Mata's `"after"`).
67	#[default]
68	After,
69	/// During the determinization (Mata's `"with"`).
70	During,
71}
72
73/// Direction of a residual reduction.
74#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
75pub enum Direction {
76	/// Residuals of the automaton.
77	#[default]
78	Forward,
79	/// Residuals of the reversed automaton.
80	Backward,
81}

83/// Algorithm of the universality, inclusion and equivalence checks.
84#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
85#[non_exhaustive]
86pub enum CheckAlgorithm {
87	/// Complement one side and search the product for an accepted word.
88	Naive,
89	/// Search the subset construction lazily, pruned by antichains. Usually much faster.
90	#[default]
91	Antichains,
92}

94/// What [`Nfa::regex`](crate::Nfa::regex) optimizes for.
95#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
96#[non_exhaustive]
97pub enum RegexPrintMode {
98	/// Conversion speed: a regex for a program to read.
99	#[default]
100	Fast,
101	/// The length and the legibility of the regex: a regex for a human to read. Takes longer, and the
102	/// result is never longer than the [`Fast`](RegexPrintMode::Fast) one.
103	Readable,
104}

106impl From<Determinization> for ffi::Determinization {
107	fn from(algorithm: Determinization) -> ffi::Determinization {
108		match algorithm {
109			Determinization::FourRussians => ffi::Determinization::FourRussians,
110			Determinization::Classical => ffi::Determinization::Classical,
111		}
112	}
113}
114
115impl From<Minimization> for ffi::Minimization {
116	fn from(algorithm: Minimization) -> ffi::Minimization {
117		match algorithm {
118			Minimization::Brzozowski => ffi::Minimization::Brzozowski,
119			Minimization::Hopcroft => ffi::Minimization::Hopcroft,
120			Minimization::HopcroftNondet => ffi::Minimization::HopcroftNondet,
121		}
122	}
123}
124
125impl From<Complementation> for ffi::Complementation {
126	fn from(algorithm: Complementation) -> ffi::Complementation {
127		match algorithm {
128			Complementation::Classical => ffi::Complementation::Classical,
129			Complementation::Brzozowski => ffi::Complementation::Brzozowski,
130		}
131	}
132}
133
134impl From<ResidualConstruction> for ffi::ResidualConstruction {
135	fn from(construction: ResidualConstruction) -> ffi::ResidualConstruction {
136		match construction {
137			ResidualConstruction::After => ffi::ResidualConstruction::After,
138			ResidualConstruction::During => ffi::ResidualConstruction::With,
139		}
140	}
141}
142
143impl From<Direction> for ffi::Direction {
144	fn from(direction: Direction) -> ffi::Direction {
145		match direction {
146			Direction::Forward => ffi::Direction::Forward,
147			Direction::Backward => ffi::Direction::Backward,
148		}
149	}
150}

152impl From<CheckAlgorithm> for ffi::CheckAlgorithm {
153	fn from(algorithm: CheckAlgorithm) -> ffi::CheckAlgorithm {
154		match algorithm {
155			CheckAlgorithm::Naive => ffi::CheckAlgorithm::Naive,
156			CheckAlgorithm::Antichains => ffi::CheckAlgorithm::Antichains,

158	}
159}
