1//! Lazy emptiness checking of Boolean and relational combinations of automata and transducers: the
2//! [`mata-lazy`](https://github.com/VeriFIT/mata/tree/devel/bindings/rust/mata-lazy) crate, with Mata's [`Nfa`]s
3//! and [`Nft`]s as leaves.
4//!
5//! Everything of `mata-lazy` is here. This module adds the [`Automaton`] implementations of [`Nfa`] (arity 1) and
6//! [`Nft`] (one tape per level), the conversion of an [`ExplicitAutomaton`] into an [`Nft`], and the conversions
7//! between Mata's [`Symbol`](crate::Symbol) and this module's [`Symbol`], which keep the number. Mata's
8//! `EPSILON` is [`EPSILON`](Symbol::EPSILON) and its `DONT_CARE` is [`ANY`](Symbol::ANY), also in an [`Nfa`].
9//!
10//! ```
11//! use mata::Nfa;
12//! use mata::lazy::Formula;
13//!
14//! let mut formula = Formula::new();
15//! let a = formula.leaf(&Nfa::from_regex("a(a|b)*")?)?;
16//! let b = formula.leaf(&Nfa::from_regex("(a|b)*b")?)?;
17//! // L(a) ⊆ L(b) iff L(a) ∩ ¬L(b) = ∅.
18//! let not_b = formula.complement(b)?;
19//! let difference = formula.intersection(a, not_b)?;
20//! let witness = formula.find_word(difference)?.expect("`a` is not included");
21//! assert_eq!(witness.tape(0), Some(vec![b'a'.into()]));
22//! # Ok::<(), Box<dyn std::error::Error>>(())
23//! ```
24

32/// Arity 1. Epsilon transitions are silent.
33impl Automaton for Nfa {
34	type State = State;
35
36	fn arity(&self) -> usize {
37		1
38	}
39
40	fn initial_states(&self) -> impl IntoIterator<Item = State> {
41		Nfa::initial_states(self).iter().copied()
42	}
43
44	fn is_final(&self, state: State) -> bool {
45		Nfa::is_final(self, state)
46	}
47
48	fn for_each_move(&self, state: State, mut visit: impl FnMut(&[Symbol], State)) {
49		for transition in self.transitions_from(state) {
50			visit(&[transition.symbol.into()], transition.target);
51		}
52	}
53}

55/// One tape per level. A transition (a chain of one edge per level) is one move.
56impl Automaton for Nft {
57	type State = State;
58
59	fn arity(&self) -> usize {
60		self.num_levels()
61	}
62
63	fn initial_states(&self) -> impl IntoIterator<Item = State> {
64		Nft::initial_states(self).iter().copied()
65	}
66
67	fn is_final(&self, state: State) -> bool {
68		Nft::is_final(self, state)
69	}
70
71	fn for_each_move(&self, state: State, mut visit: impl FnMut(&[Symbol], State)) {
72		let mut tuple = Vec::with_capacity(self.num_levels());
73		walk_levels(self, state, &mut tuple, &mut HashSet::new(), &mut visit);
74	}
75}
76
77/// Follows the edges from a state of level `tuple.len()` until they return to level 0. An edge that skips
78/// levels reads its symbol on each of them, as Mata's jumps do.
79///
80/// `seen` holds the inner states reached so far with their prefix, each walked once: paths that meet again with
81/// the same prefix give the same moves, and there can be exponentially many of them.
82fn walk_levels(
83	nft: &Nft,
84	state: State,
85	tuple: &mut Vec<Symbol>,
86	seen: &mut HashSet<(State, Vec<Symbol>)>,
87	visit: &mut dyn FnMut(&[Symbol], State),
88) {
89	let levels = nft.num_levels();
90	let level = tuple.len();
91
92	for edge in nft.edges_from(state) {
93		let next = match nft.level(edge.target) {
94			Some(0) => levels,
95			Some(next) => next,
96			// An `Nft` is well formed, so its edges lead to its states.
97			None => continue,
98		};
99
100		if next <= level {
101			continue;
102		}
103
104		tuple.resize(next, edge.symbol.into());
105		if next == levels {
106			visit(tuple, edge.target);
107		} else if seen.insert((edge.target, tuple.clone())) {
108			walk_levels(nft, edge.target, tuple, seen, visit);
109		}
110
111		tuple.truncate(level);
112	}
113}

115/// One level per tape, trimmed as the automaton is. [`ANY`](Symbol::ANY) stays as
116/// [`DONT_CARE`](crate::nft::DONT_CARE), and [`to_nfa_over`](Nft::to_nfa_over) expands it for arity 1.
117///
118/// Mata reads padding asynchronously, so for a relation that is not length-preserving the Mata relation drops it:
119/// `(a, ε)` then `(ε, b)` is the pair `(a, b)` there. [`leaf_over`](Formula::leaf_over) reads the transducer back
120/// in lock step, given the alphabets.
121///
122/// # Errors
123///
124/// If Mata refuses the number of levels: 0 or more than [`MAX_LEVELS`](crate::nft::MAX_LEVELS).
125impl TryFrom<&ExplicitAutomaton> for Nft {
126	type Error = NftError;
127
128	fn try_from(automaton: &ExplicitAutomaton) -> Result<Nft, NftError> {
129		let mut nft = Nft::new(automaton.arity())?;
130		let states: Vec<State> = (0..automaton.num_states())
131			.map(|_| nft.add_state())
132			.collect();
133		for state in automaton.initial_states() {
134			nft.add_initial(states[state])?;
135		}
136
137		let mut symbols = Vec::with_capacity(automaton.arity());
138		for (state, &source) in states.iter().enumerate() {
139			if automaton.is_final(state) {
140				nft.add_final(source)?;
141			}
142
143			for (tuple, target) in automaton.moves(state) {
144				symbols.clear();
145				symbols.extend(tuple.iter().copied().map(crate::Symbol::from));
146				nft.add_transition(source, &symbols, states[target])?;
147			}
148		}
149
150		Ok(nft)
151	}
152}
