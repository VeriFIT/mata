3use std::fmt;
4use std::iter::FusedIterator;
5use std::ops::Range;
6
7use mata_sys::ffi;
8
9use crate::types::states_from_raw;
10use crate::{State, Symbol, Transition, check};
11
12/// Iterator over the transitions of an automaton, ordered by source, then symbol, then target.
13///
14/// Returned by [`Nfa::transitions`](crate::Nfa::transitions) and
15/// [`Nfa::transitions_from`](crate::Nfa::transitions_from), and by the transducer counterparts.
16#[derive(Clone)]
17pub struct Transitions<'a> {
18	automaton: &'a ffi::Automaton,
19	/// The source states still to visit; the current one is `sources.start - 1`.
20	sources: Range<u32>,
21	/// The symbol posts of the current source still to visit.
22	posts: Range<usize>,
23	symbol: Symbol,
24	targets: std::slice::Iter<'a, State>,
25}
26
27impl<'a> Transitions<'a> {
28	pub(crate) fn new(automaton: &'a ffi::Automaton, sources: Range<u32>) -> Transitions<'a> {
29		Transitions {
30			automaton,
31			sources,
32			posts: 0..0,
33			symbol: Symbol::new(0),
34			targets: [].iter(),
35		}
36	}
37}
38
39impl Iterator for Transitions<'_> {
40	type Item = Transition;
41
42	fn next(&mut self) -> Option<Transition> {
43		loop {
44			if let Some(&target) = self.targets.next() {
45				let source = State::new(self.sources.start - 1);
46				return Some(Transition {
47					source,
48					symbol: self.symbol,
49					target,
50				});
51			}
52			if let Some(post) = self.posts.next() {
53				let (automaton, source) = (self.automaton, self.sources.start - 1);
54				self.symbol = Symbol::new(check(ffi::symbol_post_symbol(automaton, source, post)));
55				self.targets =
56					states_from_raw(check(ffi::symbol_post_targets(automaton, source, post)))
57						.iter();
58				continue;
59			}
60			let source = self.sources.next()?;
61			self.posts = 0..ffi::num_symbol_posts(self.automaton, source);
62		}
63	}
64}
65
66impl FusedIterator for Transitions<'_> {}
67
68// SAFETY: The iterator only reads the borrowed automaton, through `const` Mata calls. `Nfa` and `Nft` are
69// `Sync` (see the safety comments there), so the automaton may be read from any thread while it is borrowed.
70unsafe impl Send for Transitions<'_> {}
71
72// SAFETY: Through `&Transitions`, only `Clone` and `Debug` are reachable, which read the same shared borrow.
73unsafe impl Sync for Transitions<'_> {}
74
75impl fmt::Debug for Transitions<'_> {
76	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
77		f.debug_struct("Transitions")
78			.field("sources", &self.sources)
79			.finish_non_exhaustive()
80	}
81}
