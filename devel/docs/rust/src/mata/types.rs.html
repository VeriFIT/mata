6/// A state of an automaton.
7///
8/// The states of an automaton with `n` states are `0..n`. States are 32-bit numbers, as in Mata, so an
9/// automaton has fewer than `u32::MAX` states. Creating a `State` never fails; the methods of
10/// [`Nfa`](crate::Nfa) check that the state exists.
11#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
12#[cfg_attr(
13	feature = "serde",
14	derive(serde::Serialize, serde::Deserialize),
15	serde(transparent)
16)]
17#[repr(transparent)]
18pub struct State(u32);
19
20impl State {
21	/// The state with number `index`.
22	#[must_use]
23	pub const fn new(index: u32) -> State {
24		State(index)
25	}
26
27	/// The number of this state, for indexing.
28	#[must_use]
29	pub const fn index(self) -> usize {
30		self.0 as usize
31	}
32
33	/// The number of this state as the bridge passes it.
34	pub(crate) const fn raw(self) -> u32 {
35		self.0
36	}
37}
38
39impl From<u32> for State {
40	fn from(index: u32) -> State {
41		State(index)
42	}
43}
44
45/// Fails for an index that does not fit in 32 bits.
46impl TryFrom<usize> for State {
47	type Error = TryFromIntError;
48
49	fn try_from(index: usize) -> Result<State, TryFromIntError> {
50		u32::try_from(index).map(State)
51	}
52}
53
54impl From<State> for u32 {
55	fn from(state: State) -> u32 {
56		state.0
57	}
58}
59
60impl From<State> for usize {
61	fn from(state: State) -> usize {
62		state.index()
63	}
64}
65
66/// Prints `q3`, as the `.mata` format does.
67impl fmt::Display for State {
68	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
69		write!(f, "q{}", self.0)
70	}
71}

73/// A transition label.
74///
75/// Symbols are plain numbers; an [`Alphabet`](crate::Alphabet) can give them names. Automata built from
76/// regular expressions read bytes, so `Symbol::from(b'a')` is the symbol of the letter `a`.
77#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
78#[cfg_attr(
79	feature = "serde",
80	derive(serde::Serialize, serde::Deserialize),
81	serde(transparent)
82)]
83#[repr(transparent)]
84pub struct Symbol(u32);
85
86impl Symbol {
87	/// The epsilon symbol, `u32::MAX` (Mata's `EPSILON`). A transition under it reads no input.
88	pub const EPSILON: Symbol = Symbol(mata_sys::EPSILON);
89
90	/// The symbol with number `value`.
91	#[must_use]
92	pub const fn new(value: u32) -> Symbol {
93		Symbol(value)
94	}
95
96	/// The number of this symbol.
97	#[must_use]
98	pub const fn value(self) -> u32 {
99		self.0
100	}
101
102	/// Whether this is [`Symbol::EPSILON`].
103	#[must_use]
104	pub const fn is_epsilon(self) -> bool {
105		self.0 == mata_sys::EPSILON
106	}
107}
108
109impl From<u8> for Symbol {
110	fn from(byte: u8) -> Symbol {
111		Symbol(u32::from(byte))
112	}
113}
114
115impl From<u32> for Symbol {
116	fn from(value: u32) -> Symbol {
117		Symbol(value)
118	}
119}
120
121impl From<Symbol> for u32 {
122	fn from(symbol: Symbol) -> u32 {
123		symbol.0
124	}
125}
126
127/// Prints the number, or `<eps>` for [`Symbol::EPSILON`], as the `.mata` format does.
128impl fmt::Display for Symbol {
129	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
130		if self.is_epsilon() {
131			f.write_str("<eps>")
132		} else {
133			write!(f, "{}", self.0)
134		}
135	}
136}

138/// A finite word. Functions take words as `&[Symbol]`.
139///
140/// For a byte string, [`word_from_bytes`] gives the word over its bytes.
141pub type Word = Vec<Symbol>;
142
143/// The word of the bytes of `text`, one symbol per byte, as automata built by
144/// [`Nfa::from_regex`](crate::Nfa::from_regex) read text.
145///
146/// ```
147/// use mata::{Nfa, word_from_bytes};
148///
149/// let nfa = Nfa::from_regex("a(b|c)*")?;
150/// assert!(nfa.accepts(&word_from_bytes("abcb")));
151/// # Ok::<(), mata::RegexError>(())
152/// ```
153#[must_use]
154pub fn word_from_bytes(text: impl AsRef<[u8]>) -> Word {
155	text.as_ref().iter().copied().map(Symbol::from).collect()
156}

158/// The transition `source --symbol--> target`.
159#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
160#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
161pub struct Transition {
162	/// The state the transition leaves.
163	pub source: State,
164	/// The symbol it reads.
165	pub symbol: Symbol,
166	/// The state it enters.
167	pub target: State,
168}
169
170impl Transition {
171	/// The transition `source --symbol--> target`.
172	#[must_use]
173	pub const fn new(source: State, symbol: Symbol, target: State) -> Transition {
174		Transition {
175			source,
176			symbol,
177			target,
178		}
179	}
180}
181
182impl From<(State, Symbol, State)> for Transition {
183	fn from((source, symbol, target): (State, Symbol, State)) -> Transition {
184		Transition {
185			source,
186			symbol,
187			target,
188		}
189	}
190}

192/// Prints `q0 97 q1`, a transition line of the `.mata` format.
193impl fmt::Display for Transition {
194	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
195		write!(f, "{} {} {}", self.source, self.symbol, self.target)
196	}
197}

199/// An accepting run: a word together with the states it visits.
200///
201/// `path.len() == word.len() + 1`. The word lists every transition taken, so it contains
202/// [`Symbol::EPSILON`] where the run follows an epsilon transition.
203#[derive(Clone, Debug, PartialEq, Eq, Hash)]
204#[non_exhaustive]
205pub struct Run {
206	/// The symbols read.
207	pub word: Word,
208	/// The states visited, starting in an initial state and ending in a final one.
209	pub path: Vec<State>,
210}

212/// A word accepted by exactly one of two automata, from
213/// [`Nfa::find_distinguishing_word`](crate::Nfa::find_distinguishing_word).
214#[derive(Clone, Debug, PartialEq, Eq, Hash)]
215#[non_exhaustive]
216pub struct Distinction {
217	/// The word.
218	pub word: Word,
219	/// The automaton that accepts it.
220	pub accepted_by: Side,
221}

223/// One of the two automata of a binary check.
224#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
225pub enum Side {
226	/// The receiver, `a` in `a.find_distinguishing_word(&b)`.
227	Left,
228	/// The argument, `b` in `a.find_distinguishing_word(&b)`.
229	Right,
230}

232/// The states, sorted (the bridge gives initial and final states in no particular order).
233pub(crate) fn sorted(states: &[State]) -> Vec<State> {
234	let mut states = states.to_vec();
235	states.sort_unstable();
236	states
237}

239/// Debug-prints the items of an iterator as a list, each in its `Display` form (such as `q0 97 q1` for a
240/// transition).
241pub(crate) struct Lines<I>(pub(crate) I);
242
243impl<I: Clone + Iterator<Item: fmt::Display>> fmt::Debug for Lines<I> {
244	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
245		/// Debug-prints an item with its `Display` form.
246		struct Shown<T>(T);
247		impl<T: fmt::Display> fmt::Debug for Shown<T> {
248			fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
249				fmt::Display::fmt(&self.0, f)
250			}
251		}
252		f.debug_list().entries(self.0.clone().map(Shown)).finish()
253	}
254}

256/// Views raw states from the bridge as [`State`]s.
257pub(crate) fn states_from_raw(raw: &[u32]) -> &[State] {
258	// SAFETY: `State` is `repr(transparent)` over `u32`, so both slices have the same layout.
259	unsafe { std::slice::from_raw_parts(raw.as_ptr().cast::<State>(), raw.len()) }
260}
261
262/// Views [`Symbol`]s as the raw symbols of the bridge.
263pub(crate) fn symbols_to_raw(symbols: &[Symbol]) -> &[u32] {
264	// SAFETY: `Symbol` is `repr(transparent)` over `u32`, so both slices have the same layout.
265	unsafe { std::slice::from_raw_parts(symbols.as_ptr().cast::<u32>(), symbols.len()) }
266}

268/// Converts raw symbols from the bridge into a [`Word`].
269pub(crate) fn word_from_raw(raw: Vec<u32>) -> Word {
270	raw.into_iter().map(Symbol).collect()
271}

273/// A number of states as a 32-bit state number (Mata keeps counts below `u32::MAX`).
274pub(crate) fn raw_state_count(num_states: usize) -> u32 {
275	u32::try_from(num_states).expect("Mata keeps the number of states below 2^32")
276}
