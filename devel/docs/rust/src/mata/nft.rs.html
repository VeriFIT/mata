1//! Finite transducers: Mata's experimental `mata::nft::Nft`.
2//!
3//! A transducer with *n* levels (tapes) accepts a relation over *n*-tuples of words. Mata encodes one
4//! transducer transition as a chain of single-symbol edges through states of levels 0, 1, …, *n* − 1 and
5//! back to level 0: the edge leaving a level-*i* state reads the symbol of tape *i*. [`Symbol::EPSILON`]
6//! on a tape reads nothing there, and [`DONT_CARE`] reads any one symbol. Only the states of level 0 are
7//! the "real" states of the transducer; initial and final states always have level 0. Where an [`Nfa`]
8//! turns into a tape ([`Nft::identity`], [`Nft::apply`], [`Nft::restrict`]), a symbol of the automaton
9//! equal to [`DONT_CARE`] becomes the wildcard.
10//!
11//! Mata's transducer interface is experimental and moving fast, so expect breaking changes in this API.
12//!
13//! ```
14//! use mata::nft::Nft;
15//! use mata::{Nfa, Symbol};
16//!
17//! let (a, b) = (Symbol::from(b'a'), Symbol::from(b'b'));
18//! // Replace every `a` by `b`, keep every `b`.
19//! let mut t = Nft::new(2)?;
20//! let q = t.add_state();
21//! t.add_initial(q)?;
22//! t.add_final(q)?;
23//! t.add_transition(q, &[a, b], q)?;
24//! t.add_transition(q, &[b, b], q)?;
25//!
26//! assert!(t.accepts(&[[a, b, a], [b, b, b]]));
27//! assert!(!t.accepts(&[[a], [a]]));
28//!
29//! // The image of a(a|b)* is b+.
30//! let input = Nfa::from_regex("a(a|b)*")?;
31//! let image = t.apply(&input, 0)?.to_nfa();
32//! assert!(image.is_equivalent_to(&Nfa::from_regex("b+")?));
33//! # Ok::<(), mata::Error>(())
34//! ```

51/// The most levels a transducer may have (parsed jumps are unwound into one edge per level).
52pub const MAX_LEVELS: usize = mata_sys::nft::MAX_LEVELS;
53
54/// The symbol that matches any one symbol on a tape (Mata's `DONT_CARE`, `u32::MAX - 1`). It does not
55/// match [`Symbol::EPSILON`], which reads no symbol.
56pub const DONT_CARE: Symbol = Symbol::new(u32::MAX - 1);

58/// A finite transducer: Mata's `mata::nft::Nft`.
59///
60/// See the [module documentation](self) for how transitions are encoded. An `Nft` is always well formed:
61/// every state has a level below [`Nft::num_levels`], initial and final states have level 0, and every
62/// edge leads to a later level or back to level 0.
63///
64/// # Thread safety
65///
66/// `Nft` is `Send` and `Sync`, as [`Nfa`] is.
67pub struct Nft {
68	/// Never null.
69	inner: UniquePtr<ffi::Nft>,
70}
71
72// SAFETY: As for `Nfa`: an `Nft` owns its C++ object, the shim never attaches alphabets, and mata-sys links
73// only a Mata without the static buffers shared between automata, so Mata keeps no shared state.
74unsafe impl Send for Nft {}
75
76// SAFETY: The `&self` methods call `const` Mata functions, which do not mutate the transducer.
77unsafe impl Sync for Nft {}

94/// Unwraps a call whose arguments were checked already: a failure is a broken precondition, so it panics.
95fn expect<T>(result: Result<T, mata_sys::cxx::Exception>) -> T {

99fn check_num_levels(num_levels: usize) -> Result<(), NftError> {
100	if (1..=MAX_LEVELS).contains(&num_levels) {
101		Ok(())
102	} else {
103		Err(NftError::InvalidNumLevels { num_levels })
104	}
105}

107/// The words as the bridge takes them: all symbols concatenated, and the length of each word.
108fn flatten<W: AsRef<[Symbol]>>(words: &[W]) -> (Vec<u32>, Vec<usize>) {
109	let symbols = words
110		.iter()
111		.flat_map(|word| symbols_to_raw(word.as_ref()).iter().copied())
112		.collect();
113	let lengths = words.iter().map(|word| word.as_ref().len()).collect();
114	(symbols, lengths)
115}

136	/// Checks that `found` (levels, symbols or words) matches the number of levels.
137	fn check_arity(&self, found: usize) -> Result<(), NftError> {
138		let expected = self.num_levels();
139		if found == expected {
140			Ok(())
141		} else {
142			Err(NftError::LevelMismatch { expected, found })
143		}
144	}
145
146	fn check_level(&self, level: usize) -> Result<(), NftError> {
147		let num_levels = self.num_levels();
148		if level < num_levels {
149			Ok(())
150		} else {
151			Err(NftError::NoSuchLevel { level, num_levels })
152		}
153	}
154
155	/// Checks that `levels` are levels in strictly increasing order.
156	fn check_levels(&self, levels: &[usize]) -> Result<(), NftError> {
157		for &level in levels {
158			self.check_level(level)?;
159		}
160		if levels.windows(2).all(|pair| pair[0] < pair[1]) {
161			Ok(())
162		} else {
163			Err(NftError::LevelsNotIncreasing)
164		}
165	}
166
167	/// Checks that `state` is a state of level 0, where transitions start and end.
168	fn check_level_zero(&self, state: State) -> Result<(), NftError> {
169		let level = self.level(state).ok_or_else(|| NoSuchState {
170			state,
171			num_states: self.num_states(),
172		})?;
173		if level == 0 {
174			Ok(())
175		} else {
176			Err(NftError::NotLevelZero { state, level })
177		}
178	}

215	/// Parses a transducer in the `.mata` format (`@NFT-explicit`, with `%Levels` and `%LevelsNum`).
216	/// Symbols are read as in [`Nfa::parse_mata`], with one alphabet for all levels.
217	///
218	/// # Errors
219	///
220	/// [`ParseError`] if the text is not exactly one valid, well-formed NFT section with at most
221	/// [`MAX_LEVELS`] levels.
222	pub fn parse_mata(text: &str, alphabet: &mut Alphabet) -> Result<Nft, ParseError> {
223		exactly_one(Nft::parse_mata_all(text, alphabet)?, "transducer")
224	}

226	/// Parses every transducer section of a `.mata` text, as [`Nft::parse_mata`].
227	///
228	/// # Errors
229	///
230	/// [`ParseError`] if the text is not valid or a section is not a well-formed NFT.
231	pub fn parse_mata_all(text: &str, alphabet: &mut Alphabet) -> Result<Vec<Nft>, ParseError> {
232		let (names, symbols) = alphabet.raw_parse_names();
233		let mut new_names = Vec::new();
234		let mut new_symbols = Vec::new();
235		let parsed = ffi::nft_parse_mata(
236			text,
237			false,
238			&names,
239			&symbols,
240			alphabet.raw_next_symbol(),
241			&mut new_names,
242			&mut new_symbols,
243		)
244		.map_err(|error| ParseError::new(error.what()))?;
245		alphabet.add_parsed_names(new_names, &new_symbols);
246		Ok(take_parsed(parsed))
247	}

261	/// Takes over a transducer built by other C++ code through [`mata_sys`], dropping its alphabets and
262	/// unwinding its jumps.
263	///
264	/// # Errors
265	///
266	/// [`InvalidAutomaton`] if `ffi` is null, its transitions are not kept as Mata keeps them (see
267	/// [`Nfa::from_ffi`]), or it is not well formed (see [`Nft`]).
268	pub fn from_ffi(mut ffi: UniquePtr<ffi::Nft>) -> Result<Nft, InvalidAutomaton> {
269		if ffi.is_null() {
270			return Err(InvalidAutomaton::new("the transducer is null"));
271		}
272		ffi::nft_adopt(ffi.pin_mut()).map_err(|error| InvalidAutomaton::new(error.what()))?;
273		Ok(Nft { inner: ffi })
274	}

380	/// Adds the transition from `source` to `target` that reads `symbols[i]` on level `i`, through new
381	/// inner states. A symbol may be [`Symbol::EPSILON`] (read nothing on that tape) or [`DONT_CARE`].
382	///
383	/// # Errors
384	///
385	/// [`NftError::NoSuchState`] or [`NftError::NotLevelZero`] if `source` or `target` is not a state of
386	/// level 0, and [`NftError::LevelMismatch`] if `symbols` does not have one symbol per level.
387	pub fn add_transition(
388		&mut self,
389		source: State,
390		symbols: &[Symbol],
391		target: State,
392	) -> Result<(), NftError> {
393		self.check_level_zero(source)?;
394		self.check_level_zero(target)?;
395		self.check_arity(symbols.len())?;
396		expect(ffi::nft_add_transition(
397			self.raw_mut(),
398			source.raw(),
399			symbols_to_raw(symbols),
400			target.raw(),
401		));
402		Ok(())
403	}

405	/// Adds a path from `source` to `target` that reads `words[i]` on level `i`. Shorter words are padded
406	/// with epsilon, so the path has as many transitions as the longest word. [`Symbol::EPSILON`] in a
407	/// word reads nothing, so a word of epsilons only counts as empty.
408	///
409	/// ```
410	/// use mata::nft::Nft;
411	/// use mata::Symbol;
412	///
413	/// let mut t = Nft::new(2)?;
414	/// let (p, q) = (t.add_state(), t.add_state());
415	/// t.add_initial(p)?;
416	/// t.add_final(q)?;
417	/// t.add_path(p, &[&b"abc".map(Symbol::from)[..], &b"x".map(Symbol::from)[..]], q)?;
418	/// assert!(t.accepts(&[&b"abc".map(Symbol::from)[..], &b"x".map(Symbol::from)[..]]));
419	/// # Ok::<(), mata::NftError>(())
420	/// ```
421	///
422	/// # Errors
423	///
424	/// [`NftError::NoSuchState`] or [`NftError::NotLevelZero`] if `source` or `target` is not a state of
425	/// level 0, [`NftError::LevelMismatch`] if `words` does not have one word per level, and
426	/// [`NftError::EmptyPath`] if all words are empty (or epsilon only).
427	#[doc(alias = "insert_word_by_levels")]
428	pub fn add_path<W: AsRef<[Symbol]>>(
429		&mut self,
430		source: State,
431		words: &[W],
432		target: State,
433	) -> Result<(), NftError> {
434		self.check_level_zero(source)?;
435		self.check_level_zero(target)?;
436		self.check_arity(words.len())?;
437		if words
438			.iter()
439			.all(|word| word.as_ref().iter().all(|symbol| symbol.is_epsilon()))
440		{
441			return Err(NftError::EmptyPath);
442		}
443		let (symbols, lengths) = flatten(words);
444		expect(ffi::nft_insert_words(
445			self.raw_mut(),
446			source.raw(),
447			&symbols,
448			&lengths,
449			target.raw(),
450		));
451		Ok(())
452	}

468	/// The single-level edges leaving `source`, ordered by symbol, then target. Empty for a state that
469	/// does not exist.
470	#[must_use]
471	pub fn edges_from(&self, source: State) -> Transitions<'_> {
472		let num_states = raw_state_count(self.num_states());
473		let start = source.raw().min(num_states);
474		Transitions::new(
475			self.automaton(),
476			start..num_states.min(start.saturating_add(1)),

482	/// Whether the tuple of words, `words[i]` on level `i`, is in the relation. [`Symbol::EPSILON`] in a
483	/// word is ignored, and [`DONT_CARE`] matches any symbol. A tuple with another number of words than
484	/// levels is not in the relation.
485	#[must_use]

487	pub fn accepts<W: AsRef<[Symbol]>>(&self, words: &[W]) -> bool {
488		if words.len() != self.num_levels() {
489			return false;
490		}
491		let (symbols, lengths) = flatten(words);
492		expect(ffi::nft_accepts(self.raw(), &symbols, &lengths))

502	/// Whether the relation of `self` is included in that of `other`, decided on the encodings.
503	///
504	/// `true` is always right; `false` only for transducers without epsilon. Mata compares the interleaved
505	/// encodings, so reading (*a*, ε) and then (ε, *b*) differs from reading (*a*, *b*) at once, although both
506	/// accept the pair (*a*, *b*). [`DONT_CARE`] counts as any symbol.
507	///
508	/// # Errors

515		Ok(check(ffi::nft_is_included(self.raw(), other.raw())))

589	/// The composition: the last level of `self` is matched with level 0 of `other`, and both are
590	/// projected out. For two transducers with two levels each, this is the relational composition
591	/// {(*u*, *w*) | (*u*, *v*) ∈ `self`, (*v*, *w*) ∈ `other`}.
592	///
593	/// # Errors
594	///
595	/// [`NftError::NoLevelLeft`] if both transducers have a single level, and
596	/// [`NftError::InvalidNumLevels`] if the result would have more than [`MAX_LEVELS`].
597	#[must_use = "returns a new transducer and leaves `self` unchanged"]
598	pub fn compose(&self, other: &Nft) -> Result<Nft, NftError> {
599		self.compose_on(other, self.num_levels() - 1, 0, SyncLevel::ProjectOut)
600	}
601
602	/// The composition synchronizing level `self_level` of `self` with level `other_level` of `other`,
603	/// which `sync_level` keeps or projects out. The result has the levels of `self` before `self_level`,
604	/// then those of `other` before `other_level`, then the synchronized level (if kept), then the
605	/// remaining levels of `self` and of `other`.
606	///
607	/// Mata can also synchronize several levels at once; this binding does not offer that yet.
608	///
609	/// # Errors
610	///
611	/// [`NftError::NoSuchLevel`] if a level does not exist, [`NftError::NoLevelLeft`] if the result
612	/// would have no level, and [`NftError::InvalidNumLevels`] if it would have more than [`MAX_LEVELS`].
613	#[must_use = "returns a new transducer and leaves `self` unchanged"]
614	pub fn compose_on(
615		&self,
616		other: &Nft,
617		self_level: usize,
618		other_level: usize,
619		sync_level: SyncLevel,
620	) -> Result<Nft, NftError> {
621		self.check_level(self_level)?;
622		other.check_level(other_level)?;
623		let removed = match sync_level {
624			SyncLevel::Keep => 1,
625			SyncLevel::ProjectOut => 2,
626		};
627		if self.num_levels() + other.num_levels() <= removed {
628			return Err(NftError::NoLevelLeft);
629		}
630		check_num_levels(self.num_levels() + other.num_levels() - removed)?;
631		Ok(Nft::from_owned(expect(ffi::nft_compose(
632			self.raw(),
633			other.raw(),
634			self_level,
635			other_level,
636			sync_level == SyncLevel::ProjectOut,
637		))))
638	}

640	/// Restricts level `level` to the words of `nfa` and projects that level out: for a two-level
641	/// transducer and `level` 0, the image of the language of `nfa` (use [`Nft::to_nfa`] on the result).
642	///
643	/// # Errors

645	/// [`NftError::NoSuchLevel`] if `level` does not exist, and [`NftError::NoLevelLeft`] if the
646	/// transducer has only one level.

678	///
679	/// # Errors
680	///

767	/// The transducer whose interleaved encoding (see [`Nft::to_nfa`]) is `nfa`: Mata's
768	/// `from_nfa_with_levels_advancing`, which gives the initial states level 0 and every successor of a
769	/// level-*i* state level *i* + 1 modulo `num_levels`. States unreachable from an initial state get

785	/// Mata's `complement` (the classical algorithm) over the symbols of `alphabet`: determinizes, makes
786	/// the transducer complete on every level and swaps the final and non-final states of level 0. A
787	/// [`DONT_CARE`] edge is read as one edge per symbol of `alphabet`.
788	///
789	/// # Errors
790	///
791	/// [`NftError::EpsilonTransitions`] if the transducer has an epsilon edge, which Mata's complement
792	/// does not support.
793	///
794	/// # Panics
795	///
796	/// If `alphabet` contains [`DONT_CARE`], which is not a letter.
797	///
798	/// # Complexity
799	///
800	/// Determinizes: exponential in the worst case.
801	pub fn complement(&self, alphabet: &Alphabet) -> Result<Nft, NftError> {
802		if mata_sys::ffi::has_epsilon_transitions(self.automaton()) {
803			return Err(NftError::EpsilonTransitions);
804		}
805		Ok(new_nft(ffi::nft_complement(
806			self.raw(),
807			&alphabet.raw_symbols(),
808		)))
809	}

811	/// Mata's `Nft::is_complete`: whether every reachable state has an edge under every symbol of
812	/// `alphabet`; a [`DONT_CARE`] edge covers them all.
813	///
814	/// # Errors
815	///
816	/// [`NftError::NotInAlphabet`] if a reachable state has an edge under a symbol outside `alphabet`.
817	///
818	/// # Panics
819	///
820	/// If `alphabet` contains [`DONT_CARE`], which is not a letter.
821	pub fn is_complete(&self, alphabet: &Alphabet) -> Result<bool, NftError> {
822		assert!(
823			!alphabet.contains(DONT_CARE),
824			"DONT_CARE is not a letter of an alphabet"
825		);

827			.map_err(|_| NftError::NotInAlphabet)

866	/// Mata's `Nft::is_in_lang`: whether the interleaved `word` (see [`Nft::to_nfa`]) is accepted, its
867	/// symbol *i* read on level *i* modulo the number of levels. With `match_prefix`, accepting a prefix
868	/// suffices. `has_epsilon_cycles` `false` promises [`Nft::has_epsilon_cycle`] is `false`, which lets
869	/// Mata use a faster search.
870	///
871	/// # Errors
872	///
873	/// [`NftError::InterleavedLength`] if the length of `word` is not a multiple of the number of levels,
874	/// and [`NftError::EpsilonCycle`] if `has_epsilon_cycles` is `false` but there is one.
875	#[doc(alias = "is_in_lang")]
876	pub fn accepts_interleaved(
877		&self,
878		word: &[Symbol],
879		match_prefix: bool,
880		has_epsilon_cycles: bool,
881	) -> Result<bool, NftError> {
882		let num_levels = self.num_levels();
883		if !word.len().is_multiple_of(num_levels) {
884			return Err(NftError::InterleavedLength {
885				length: word.len(),
886				num_levels,
887			});
888		}
889		if !has_epsilon_cycles && self.has_epsilon_cycle() {
890			return Err(NftError::EpsilonCycle);
891		}
892		Ok(expect(ffi::nft_is_in_lang(
893			self.raw(),
894			symbols_to_raw(word),
895			match_prefix,
896			has_epsilon_cycles,
897		)))
898	}

1005			.field("final", &Lines(sorted(self.final_states()).into_iter()))
1006			.field("edges", &Lines(self.edges()))

1008	}
