1//! Parsed `.mata` sections that are not constructed yet: Mata's `mata::IntermediateAut`.
2
3use std::fmt;
4
5use mata_sys::cxx::UniquePtr;
6use mata_sys::ffi;
7
8use crate::nft::Nft;
9use crate::{Alphabet, IntermediateError, Nfa, ParseError, check};

11/// One section of a `.mata` text as Mata's parser reads it, before an automaton is constructed from it:
12/// Mata's `mata::IntermediateAut`.
13///
14/// [`Nfa::parse_mata`] does the three steps of loading in one call; they are available separately as
15/// [`IntermediateAutomaton::parse_mata_all`], [`IntermediateAutomaton::mintermize`] for sections with a
16/// bit-vector alphabet, and [`IntermediateAutomaton::construct_nfa`] or
17/// [`IntermediateAutomaton::construct_nft`].
18///
19/// The [`Display`](fmt::Display) form is Mata's debugging dump of the section, not the `.mata` format.
20///
21/// # Examples
22///
23/// ```
24/// use mata::{Alphabet, IntermediateAutomaton};
25///
26/// let text = "@NFA-bits\n%Alphabet-auto\n%Initial q0\n%Final q1\nq0 (a0 & !a1) q1\n";
27/// let parsed = IntermediateAutomaton::parse_mata_all(text)?;
28/// let explicit = IntermediateAutomaton::mintermize(&[&parsed[0]])?;
29/// let mut sigma = Alphabet::new();
30/// let nfa = explicit[0].construct_nfa(&mut sigma)?;
31/// assert_eq!(nfa.num_transitions(), 1);
32/// # Ok::<(), mata::Error>(())
33/// ```
34pub struct IntermediateAutomaton {
35	/// Never null.
36	inner: UniquePtr<ffi::IntermediateAut>,
37}
38
39// SAFETY: An `IntermediateAutomaton` owns its C++ object, which is plain data (strings and vectors) that
40// no other object refers to; mintermization, the one step that uses process-wide state (CUDD), is
41// serialized by the shim.
42unsafe impl Send for IntermediateAutomaton {}
43
44// SAFETY: The `&self` methods read the object through `const` references only.
45unsafe impl Sync for IntermediateAutomaton {}
46
47impl IntermediateAutomaton {
48	/// Parses every section of a `.mata` text (Mata's `parser::parse_mf` followed by
49	/// `IntermediateAut::parse_from_mf`). Sections with a bit-vector alphabet keep their formulas.
50	///
51	/// # Errors
52	///
53	/// [`ParseError`] if the text is not valid.
54	pub fn parse_mata_all(text: &str) -> Result<Vec<IntermediateAutomaton>, ParseError> {
55		let parsed =
56			ffi::intermediate_parse_mata(text).map_err(|error| ParseError::new(error.what()))?;
57		Ok(take_all(parsed))
58	}
59
60	/// Mintermizes `automata` together (Mata's `Mintermization::mintermize`): the formulas over bit
61	/// vectors become explicit symbols `0`, `1`, … that stand for the same minterms in every result, in
62	/// the order of `automata`. Automata mintermized in separate calls do not share their minterms.
63	///
64	/// # Errors
65	///
66	/// [`IntermediateError::NotMintermizable`] unless every automaton is an NFA with a bit-vector
67	/// alphabet.
68	pub fn mintermize(
69		automata: &[&IntermediateAutomaton],
70	) -> Result<Vec<IntermediateAutomaton>, IntermediateError> {
71		let mut refs = ffi::intermediate_refs_new();
72		for (index, automaton) in automata.iter().enumerate() {
73			let raw = automaton.raw();
74			if !raw.is_nfa() || !raw.is_bitvector() {
75				return Err(IntermediateError::NotMintermizable { index });
76			}
77			ffi::intermediate_refs_push(refs.pin_mut(), raw);
78		}
79		let mintermized =
80			ffi::intermediate_mintermize(&refs).map_err(|error| ParseError::new(error.what()))?;
81		Ok(take_all(mintermized))
82	}
83
84	/// Constructs the NFA of an explicit NFA section (Mata's `nfa::builder::construct`). Symbol names
85	/// are read through `alphabet` as in [`Nfa::parse_mata`], which adds the new ones; the numbers that
86	/// mintermization gives the minterms of a section stay names of their symbols.
87	///
88	/// # Errors
89	///
90	/// [`IntermediateError::WrongKind`] if the section is not an NFA,
91	/// [`IntermediateError::NeedsMintermization`] if it still has a bit-vector alphabet, and
92	/// [`IntermediateError::Parse`] if it has a transition Mata cannot construct or `alphabet` has no
93	/// symbol left for a new name.
94	pub fn construct_nfa(&self, alphabet: &mut Alphabet) -> Result<Nfa, IntermediateError> {
95		self.require("NFA", self.raw().is_nfa())?;
96		let (names, symbols) = alphabet.raw_parse_names();
97		let mut new_names = Vec::new();
98		let mut new_symbols = Vec::new();
99		let nfa = ffi::intermediate_construct_nfa(
100			self.raw(),
101			&names,
102			&symbols,
103			alphabet.raw_next_symbol(),
104			&mut new_names,
105			&mut new_symbols,
106		)
107		.map_err(|error| ParseError::new(error.what()))?;
108		alphabet.add_minterm_names(new_names, &new_symbols);
109		Ok(Nfa::from_owned(nfa))
110	}
111
112	/// Constructs the transducer of an explicit NFT section (Mata's `nft::builder::construct`), with
113	/// symbol names as in [`IntermediateAutomaton::construct_nfa`]. Mata reads no `%Levels` or
114	/// `%LevelsNum` on this path: every state is on level 0 of a two-level transducer, so each
115	/// transition reads its symbol on both levels. [`Nft::parse_mata`] reads the levels.
116	///
117	/// # Errors
118	///
119	/// [`IntermediateError::WrongKind`] if the section is not an NFT,
120	/// [`IntermediateError::NeedsMintermization`] if it has a bit-vector alphabet, and
121	/// [`IntermediateError::Parse`] if it has a transition Mata cannot construct or `alphabet` has no
122	/// symbol left for a new name.
123	pub fn construct_nft(&self, alphabet: &mut Alphabet) -> Result<Nft, IntermediateError> {
124		self.require("NFT", self.raw().is_nft())?;
125		let (names, symbols) = alphabet.raw_parse_names();
126		let mut new_names = Vec::new();
127		let mut new_symbols = Vec::new();
128		let nft = mata_sys::nft::ffi::intermediate_construct_nft(
129			self.raw(),
130			&names,
131			&symbols,
132			alphabet.raw_next_symbol(),
133			&mut new_names,
134			&mut new_symbols,
135		)
136		.map_err(|error| ParseError::new(error.what()))?;
137		alphabet.add_minterm_names(new_names, &new_symbols);
138		Ok(Nft::from_owned(nft))
139	}
140
141	/// The underlying C++ object, for passing to other C++ code through [`mata_sys`].
142	#[must_use]
143	pub fn as_ffi(&self) -> &ffi::IntermediateAut {
144		self.raw()
145	}
146
147	fn raw(&self) -> &ffi::IntermediateAut {
148		&self.inner
149	}
150
151	/// Checks that the section is of the kind a construction builds, with an explicit alphabet.
152	fn require(&self, expected: &'static str, kind: bool) -> Result<(), IntermediateError> {
153		if !kind {
154			return Err(IntermediateError::WrongKind { expected });
155		}
156		if self.raw().is_bitvector() {
157			return Err(IntermediateError::NeedsMintermization);
158		}
159		Ok(())
160	}
161}
162
163fn take_all(mut automata: UniquePtr<ffi::IntermediateAuts>) -> Vec<IntermediateAutomaton> {
164	(0..ffi::intermediate_auts_len(&automata))
165		.map(|index| {
166			let inner = check(ffi::intermediate_auts_take(automata.pin_mut(), index));
167			assert!(
168				!inner.is_null(),
169				"mata-sys returned a null intermediate automaton"
170			);
171			IntermediateAutomaton { inner }
172		})
173		.collect()
174}

176/// Mata's `operator<<`: the type, naming, alphabet, initial and final states, and transitions.
177impl fmt::Display for IntermediateAutomaton {
178	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
179		f.write_str(&check(ffi::intermediate_print(self.raw())))
180	}
181}
182
183impl fmt::Debug for IntermediateAutomaton {
184	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
185		f.debug_struct("IntermediateAutomaton")
186			.finish_non_exhaustive()
187	}
