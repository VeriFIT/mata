17/// One script operation: its script name, the handler that serves it, the argument counts it accepts,
18/// the sorts of its operands and of its result where they are fixed, and the language version that
19/// added it.
20pub(crate) struct Operation {
21	/// Name in a script, such as `nfa:union-nondeterministic`.
22	pub name: &'static str,
23	/// Handler in [`dispatch`].
24	pub handler: &'static str,
25	/// Smallest number of arguments the operation accepts.
26	pub minimum: usize,
27	/// Largest number of arguments the operation accepts, `None` for no limit.
28	pub maximum: Option<usize>,
29	/// Sorts of the positional operands, comma-separated. An empty entry leaves that operand unchecked;
30	/// an empty string leaves all of them unchecked, as for operations that accept a prefix of them.
31	pub operands: &'static str,
32	/// Sort of the result, empty when it depends on the operands.
33	pub result: &'static str,
34	/// The language version that added the operation; a script declaring an older one does not see it.
35	pub since: Version,
36}
37
38const fn op(
39	name: &'static str,
40	handler: &'static str,
41	minimum: usize,
42	maximum: Option<usize>,
43	operands: &'static str,
44	result: &'static str,
45) -> Operation {
46	Operation {
47		name,
48		handler,
49		minimum,
50		maximum,
51		operands,
52		result,
53		since: FIRST,
54	}
55}

575/// Checks keyword argument `name` of `operation` written as `value`, a hint or an optional parameter in
576/// the text of the command line: `true`/`false` for `Bool`, decimal for `Int`.
577///
578/// # Errors
579///
580/// The operation takes no such keyword, or the value does not fit it.
581pub(crate) fn check_keyword(operation: &str, name: &str, value: &str) -> Result<(), String> {
582	let Some(parameter) = optional_parameter(operation, name) else {
583		return check_hint(operation, name, value);
584	};
585	let fits = match parameter.sort {
586		"Bool" => matches!(value, "true" | "false"),
587		"Int" => value.parse::<i64>().is_ok(),
588		_ => true,
589	};
590	if fits {
591		Ok(())
592	} else {
593		Err(format!(
594			":{name} of {operation} must be of sort {}, not \"{value}\"",
595			parameter.sort
596		))
597	}
598}

600/// The command-line text of keyword argument `name` of `operation` given as `value` in a script.
601///
602/// # Errors
603///
604/// `value` is not of the sort the keyword takes.
605pub(crate) fn keyword_text(operation: &str, name: &str, value: &Value) -> Result<String, String> {
606	match optional_parameter(operation, name).map(OptionalParameter::script_sort) {
607		Some("Bool") => as_bool(value)
608			.map(|value| value.to_string())
609			.map_err(|_| format!("the value of :{name} must be a Bool")),
610		Some("Int") => as_int(value)
611			.map(|value| value.to_string())
612			.map_err(|_| format!("the value of :{name} must be an Int")),
613		Some(_) => as_string(value).map_err(|_| format!("the value of :{name} must be a String")),
614		None => as_string(value).map_err(|_| format!("the value of hint :{name} must be a String")),
615	}
616}

618/// The value of optional parameter `parameter` written as `text`, which [`check_keyword`] accepted.
619fn optional_value(
620	parameter: &OptionalParameter,
621	text: &str,
622	engine: &mut Engine,
623) -> Result<Value, Error> {
624	Ok(match parameter.sort {
625		"Bool" => Value::Bool(text == "true"),
626		"Int" => Value::Int(text.parse().map_err(|_| {
627			Error::plain(format!(
628				":{} must be an Int, not \"{text}\"",
629				parameter.name
630			))
631		})?),
632		"Symbol" => Value::Symbol(engine.intern_name(text).map_err(Error::plain)?),
633		_ => Value::String(text.to_owned()),
634	})
635}

637/// The value of an optional parameter, of the Rust type of its sort.
638pub(crate) trait OptionalValue: Sized {

1809/// Whether an operation accepts a call with `given` arguments.
1810pub(crate) fn accepts(operation: &Operation, given: usize) -> bool {
1811	given >= operation.minimum && operation.maximum.is_none_or(|maximum| given <= maximum)

1814/// The sort operand `index` of `operation` must have, if it is fixed. A last entry ending in `*`
1815/// applies to every later operand too.
1816pub(crate) fn operand_sort(operation: &Operation, index: usize) -> Option<&'static str> {
1817	if operation.operands.is_empty() {
1818		return None;
1819	}
1820	let entries: Vec<&'static str> = operation.operands.split(',').collect();
1821	let entry = match entries.get(index) {
1822		Some(entry) => *entry,
1823		None => entries.last().filter(|entry| entry.ends_with('*'))?,
1824	};
1825	let entry = entry.trim_end_matches('*');
1826	(!entry.is_empty()).then_some(entry)
1827}

1829/// The error for a call whose argument count the operation does not accept.
1830pub(crate) fn arity_message(operation: &Operation, given: usize) -> String {
1831	match operation.maximum {
1832		Some(maximum) if maximum == operation.minimum => format!(
1833			"'{}' takes {maximum} arguments, {given} given",
1834			operation.name
1835		),
1836		Some(maximum) if operation.minimum == 0 => format!(
1837			"'{}' takes at most {maximum} arguments, {given} given",
1838			operation.name
1839		),
1840		Some(maximum) => format!(
1841			"'{}' takes {} to {maximum} arguments, {given} given",
1842			operation.name, operation.minimum
1843		),
1844		None => format!(
1845			"'{}' takes at least {} arguments, {given} given",
1846			operation.name, operation.minimum
1847		),
1848	}
1849}

1851/// Checks the hints of one call against the operation's [`HINTS`].
1852///
1853/// # Errors
1854///
1855/// The operation takes no such hint, or the value is not one the hint accepts.
1856pub(crate) fn check_hint(operation: &str, name: &str, value: &str) -> Result<(), String> {
1857	let Some(hint) = hint(operation, name) else {
1858		let keywords = hints_of(operation)
1859			.map(|hint| format!(":{}", hint.name))
1860			.chain(
1861				optional_parameters_of(operation).map(|parameter| format!(":{}", parameter.name)),
1862			)
1863			.collect::<Vec<_>>();
1864		return Err(if keywords.is_empty() {
1865			format!("operation {operation} takes no hints")
1866		} else {
1867			format!(
1868				"operation {operation} has no hint ':{name}'; it takes {}",
1869				keywords.join(", ")
1870			)
1871		});
1872	};
1873	if hint.values.contains(&value) {
1874		Ok(())
1875	} else {
1876		Err(format!(
1877			"hint :{name} of {operation} is one of {}, not \"{value}\"",
1878			hint.values
1879				.iter()
1880				.map(|value| format!("\"{value}\""))
1881				.collect::<Vec<_>>()
1882				.join(", ")
1883		))
1884	}
1885}

1887/// Calls operation `name` with positional `arguments` and the `hints` of this call, which override
1888/// the defaults set with `(set-option :default …)`.
1889pub(crate) fn call(
1890	name: &str,
1891	arguments: Vec<Value>,
1892	hints: &[(String, String)],
1893	engine: &mut crate::eval::Engine,
1894) -> Result<Value, Error> {
1895	let operation =
1896		operation(name).ok_or_else(|| Error::plain(format!("unknown operation '{name}'")))?;
1897	let given = arguments.len();
1898	if !accepts(operation, given) {
1899		return Err(Error::plain(arity_message(operation, given)));
1900	}
1901	let mut chosen = engine.defaults_of(name);
1902	for (key, value) in hints {
1903		check_keyword(name, key, value).map_err(Error::plain)?;
1904		chosen.retain(|(known, _)| known != key);
1905		chosen.push((key.clone(), value.clone()));
1906	}
1907	let mut parameters = Vec::new();
1908	for (key, text) in &chosen {
1909		if let Some(parameter) = optional_parameter(name, key) {
1910			parameters.push((key.clone(), optional_value(parameter, text, engine)?));
1911		}
1912	}
1913	let saved = std::mem::replace(&mut engine.hints, chosen);
1914	let saved_parameters = std::mem::replace(&mut engine.call_parameters, parameters);
1915	let result = if folds(name) && given > 2 {
1916		let mut operands = arguments.into_iter();
1917		let first = operands.next().expect("at least three operands");
1918		operands.try_fold(first, |folded, next| {
1919			dispatch(operation.handler, name, vec![folded, next], engine)
1920		})
1921	} else {
1922		dispatch(operation.handler, name, arguments, engine)
1923	};
1924	engine.hints = saved;
1925	engine.call_parameters = saved_parameters;
1926	result
1927}

1929/// The value of hint `name` for the call being made, or its first (default) value.
1930fn hint_value(engine: &Engine, operation: &str, name: &str) -> &'static str {
1931	let hint = hint(operation, name).expect("every hint a handler reads is in HINTS");
1932	let chosen = engine
1933		.hints
1934		.iter()
1935		.find(|(key, _)| key == name)
1936		.map(|(_, value)| value.as_str());
1937	hint.values
1938		.iter()
1939		.copied()
1940		.find(|value| Some(*value) == chosen)
1941		.unwrap_or(hint.values[0])
1942}

1952fn dispatch(
1953	handler: &str,
1954	name: &str,
1955	arguments: Vec<Value>,
1956	engine: &mut crate::eval::Engine,
1957) -> Result<Value, Error> {
1958	match handler {
1959		"not" => {
1960			let [value] = take(arguments, name)?;
1961			Ok(Value::Bool(!as_bool(&value).map_err(Error::plain)?))
1962		}
1963		"eq" => {
1964			let [lhs, rhs] = take(arguments, name)?;
1965			let equal = values_equal(&lhs, &rhs).map_err(Error::plain)?;
1966			Ok(Value::Bool(equal))
1967		}
1968		"lt" => {
1969			let [lhs, rhs] = take(arguments, name)?;
1970			match (&lhs, &rhs) {
1971				(Value::Int(lhs), Value::Int(rhs)) => Ok(Value::Bool(lhs < rhs)),
1972				(Value::Word(lhs), Value::Word(rhs)) => Ok(Value::Bool(lhs < rhs)),
1973				_ => Err(Error::plain("'< needs Int or Word operands".into())),
1974			}
1975		}
1976		"le" => {
1977			let [lhs, rhs] = take(arguments, name)?;
1978			match (&lhs, &rhs) {
1979				(Value::Int(lhs), Value::Int(rhs)) => Ok(Value::Bool(lhs <= rhs)),
1980				(Value::Word(lhs), Value::Word(rhs)) => Ok(Value::Bool(lhs <= rhs)),
1981				_ => Err(Error::plain("'<=' needs Int or Word operands".into())),
1982			}
1983		}
1984		"add" => {
1985			let [lhs, rhs] = take(arguments, name)?;
1986			match (&lhs, &rhs) {
1987				(Value::Int(lhs), Value::Int(rhs)) => Ok(Value::Int(
1988					lhs.checked_add(*rhs)
1989						.ok_or_else(|| Error::plain("integer overflow".into()))?,
1990				)),
1991				_ => Err(Error::plain("'+' needs Int operands".into())),
1992			}
1993		}
1994		"sub" => {
1995			let [lhs, rhs] = take(arguments, name)?;
1996			match (&lhs, &rhs) {
1997				(Value::Int(lhs), Value::Int(rhs)) => Ok(Value::Int(
1998					lhs.checked_sub(*rhs)
1999						.ok_or_else(|| Error::plain("integer overflow".into()))?,
2000				)),
2001				_ => Err(Error::plain("'-' needs Int operands".into())),
2002			}
2003		}
2004		"concat" => {
2005			let mut text = String::new();
2006			for argument in &arguments {
2007				text.push_str(&as_string(argument).map_err(Error::plain)?);
2008			}
2009			Ok(Value::String(text))
2010		}
2011		"to_string" => {
2012			let [value] = take(arguments, name)?;
2013			Ok(Value::String(match &value {
2014				Value::String(text) => text.clone(),
2015				Value::List(_) => list_text(&value, engine),
2016				other => crate::display_value(other, engine)
2017					.trim_end_matches('\n')
2018					.to_owned(),
2019			}))
2020		}
2021		"string_to_int" => {
2022			let [value] = take(arguments, name)?;
2023			let text = as_string(&value).map_err(Error::plain)?;
2024			text.trim()
2025				.parse::<i64>()
2026				.map(Value::Int)
2027				.map_err(|_| Error::plain(format!("'{text}' is not an Int")))
2028		}
2029		"is_none" => {
2030			let [value] = take(arguments, name)?;
2031			Ok(Value::Bool(matches!(value, Value::None)))
2032		}
2033		"get" => {
2034			let [value] = take(arguments, name)?;
2035			match value {
2036				Value::None => Ok(Value::None),
2037				other => Ok(other),
2038			}
2039		}
2040		"word" => {
2041			let mut symbols = Vec::new();
2042			for argument in &arguments {
2043				let name = as_string(argument).map_err(Error::plain)?;
2044				symbols.push(engine.intern_name(&name).map_err(Error::plain)?);
2045			}
2046			Ok(Value::Word(Word::from(symbols)))
2047		}
2048		"symbol" => {
2049			let [name] = take(arguments, name)?;
2050			let name = as_string(&name).map_err(Error::plain)?;
2051			let symbol = engine.intern_name(&name).map_err(Error::plain)?;
2052			Ok(Value::Symbol(symbol))
2053		}
2054		_ => dispatch_list_structure(handler, name, arguments, engine),
2055	}
2056}

2058/// Serves `list` and the operations that read a list by position: `list:length`, `list:nth`, `list:take`,
2059/// `list:drop`, and `list:range`.
2060fn dispatch_list_structure(
2061	handler: &str,
2062	name: &str,
2063	arguments: Vec<Value>,
2064	engine: &mut Engine,
2065) -> Result<Value, Error> {
2066	match handler {
2067		"list" => Ok(Value::List(Rc::new(arguments))),
2068		"list_length" => {
2069			let [list] = take(arguments, name)?;
2070			integer(list_len(&list)?.len())
2071		}
2072		"list_nth" => {
2073			let [list, index] = take(arguments, name)?;
2074			let list = list_len(&list)?;
2075			let index = length(
2076				as_int(&index)?,
2077				"list:nth needs a non-negative index".into(),
2078			)?;
2079			list.get(index)
2080				.cloned()
2081				.ok_or_else(|| Error::plain("list:nth index out of bounds".into()))
2082		}
2083		"list_take" | "list_drop" => {
2084			let [list, count] = take(arguments, name)?;
2085			let list = list_len(&list)?;
2086			let count = length(
2087				as_int(&count)?,
2088				format!("{name} needs a non-negative count"),
2089			)?;
2090			let count = count.min(list.len());
2091			let taken = if handler == "list_take" {
2092				list[..count].to_vec()
2093			} else {
2094				list[count..].to_vec()
2095			};
2096			Ok(Value::List(Rc::new(taken)))
2097		}
2098		"list_range" => {
2099			let [start, end] = take(arguments, name)?;
2100			let start = as_int(&start)?;
2101			let end = as_int(&end)?;
2102			if start < 0 {
2103				return Err(Error::plain(format!(
2104					"{name} needs a start of at least 0, not {start}"
2105				)));
2106			}
2107			if start > end {
2108				return Ok(Value::List(Rc::new(Vec::new())));
2109			}
2110			// A range is meant for a number of rounds that is known in advance, so one beyond
2111			// millions is a script bug; rejecting it keeps a mistaken bound from consuming the
2112			// memory of the run.
2113			if end - start > MAX_RANGE_SPAN {
2114				return Err(Error::plain(format!(
2115					"{name} cannot hold {start} to {end}; the span must be at most {MAX_RANGE_SPAN}"
2116				)));
2117			}
2118			Ok(Value::List(Rc::new((start..end).map(Value::Int).collect())))
2119		}
2120		_ => dispatch_list_iteration(handler, name, arguments, engine),
2121	}
2122}

2124/// The elements of a list value, which the callers may clone one element at a time instead of the

2126fn list_len(value: &Value) -> Result<Rc<Vec<Value>>, String> {

2129		_ => Err("expected a list".into()),

2133/// The most elements one `list:range` may hold: a range is meant for a number of rounds that is
2134/// known in advance, so one beyond millions is a script bug, and a smaller bound keeps a mistaken
2135/// one from consuming the memory of the run.
2136const MAX_RANGE_SPAN: i64 = 1 << 24;
2137
2138/// A list as one line for `to-string`: the elements spaced out, nested lists in parentheses, as
2139/// `script-language.md` "Core functions" says: any value on one line. A list nested deeper than
2140/// [`MAX_NESTING`] is shown as `…` without a walk, because the walk itself must not follow the
2141/// nesting off the stack.
2142fn list_text(value: &Value, engine: &Engine) -> String {
2143	if nesting_of(value, 0).is_none() {
2144		return "…".into();
2145	}
2146	let mut text = String::new();
2147	// A depth-first walk with an explicit stack: the `bool` says whether the element is done, so
2148	// that a nested list prints its closing parenthesis in order, and whether the element is the
2149	// first of its list, which is what earns it the leading space of a separator. The text an
2150	// element renders to is its business, so the separator may not be read back off it.
2151	let mut stack: Vec<(&Value, bool, bool)> = vec![(value, false, true)];
2152	while let Some((element, done, first)) = stack.pop() {
2153		if done {
2154			text.push(')');
2155			continue;
2156		}
2157		if !first && !text.is_empty() {
2158			text.push(' ');
2159		}
2160		match element {
2161			Value::List(list) => {
2162				text.push('(');
2163				if list.is_empty() {
2164					text.push(')');
2165				} else {
2166					stack.push((element, true, false));
2167					for (index, item) in list.as_ref().iter().enumerate().rev() {
2168						stack.push((item, false, index == 0));
2169					}
2170				}
2171			}
2172			Value::Word(word) => {
2173				text.push_str(&engine.script_alphabet.display_word(word).to_string());
2174			}
2175			other => {
2176				// A string (or an automaton) whose text runs onto more lines is shown on the one
2177				// line of the list, with a line break as `\n`. Only the terminator the display
2178				// adds is dropped; a line break the string itself ends in is escaped like any other.
2179				let element = crate::display_value(other, engine);
2180				let element = element.strip_suffix('\n').unwrap_or(&element);
2181				text.push_str(&element.replace('\n', "\\n"));
2182			}
2183		}
2184	}
2185	text
2186}

2188/// One step of the walk: entering a value at a level, or a list whose children have run and whose
2189/// height can now be read back from the map.
2190enum Step<'a> {
2191	Enter(&'a Value, usize),
2192	Exit(&'a Rc<Vec<Value>>, usize),
2193}
2194
2195/// The deepest nesting of `value`, or `None` when it nests deeper than [`MAX_NESTING`]. The walk
2196/// itself is iterative, so it cannot overflow on the value it measures, and the height of a list
2197/// is remembered by identity, so a list the value holds at many places is measured once, not once
2198/// per place.
2199fn nesting_of(value: &Value, depth: usize) -> Option<usize> {
2200	if depth > crate::eval::MAX_NESTING {
2201		return None;
2202	}
2203	let mut heights: std::collections::HashMap<usize, usize> = std::collections::HashMap::new();
2204	let mut deepest = depth;
2205	let mut stack = vec![Step::Enter(value, depth)];
2206	while let Some(step) = stack.pop() {
2207		match step {
2208			Step::Enter(element, level) => {
2209				if level > crate::eval::MAX_NESTING {
2210					return None;
2211				}
2212				let Value::List(list) = element else {
2213					deepest = deepest.max(level);
2214					continue;
2215				};
2216				if let Some(&height) = heights.get(&(Rc::as_ptr(list) as usize)) {
2217					deepest = deepest.max(level + height - 1);
2218					if deepest > crate::eval::MAX_NESTING {
2219						return None;
2220					}
2221				} else {
2222					stack.push(Step::Exit(list, level));
2223					for item in list.iter() {
2224						stack.push(Step::Enter(item, level + 1));
2225					}
2226				}
2227			}
2228			Step::Exit(list, level) => {
2229				// The children ran before this step, so every list among them has its height. A
2230				// child nests one level below the list, whether it is a list of that height or a
2231				// scalar, so both count as one level plus the child's own depth.
2232				let height = 1 + list
2233					.iter()
2234					.map(|item| match item {
2235						Value::List(inner) => heights[&(Rc::as_ptr(inner) as usize)],
2236						_ => 1,
2237					})
2238					.max()
2239					.unwrap_or(0);
2240				if level + height - 1 > crate::eval::MAX_NESTING {
2241					return None;
2242				}
2243				heights.insert(Rc::as_ptr(list) as usize, height);
2244				deepest = deepest.max(level + height - 1);
2245			}
2246		}
2247	}
2248	Some(deepest)
2249}

2714/// The verdict of a decision that `holds` makes without a counterexample, or, when the script asks for
2715/// witnesses, that `find` makes by searching one; the counterexample becomes the witness.
2716fn decide(
2717	engine: &mut Engine,
2718	holds: impl FnOnce(&Engine) -> bool,
2719	find: impl FnOnce(&Engine) -> Option<Word>,
2720) -> Value {
2721	if !(engine.options.produce_witnesses.on() || engine.witness_requested.on()) {
2722		engine.last_witness = None;
2723		return Value::Bool(holds(engine));
2724	}
2725	let word = find(engine);
2726	let result = word.is_none();
2727	engine.last_witness = word.map(Value::Word);
2728	Value::Bool(result)
2729}

3495/// Serves the `automaton:` operations, which call the `nfa:` or `nft:` operation of the same name by
3496/// the kind of their operands, and the operations on intermediate automata: `intermediate:` and
3497/// `nfa:construct` / `nft:construct`.
3498fn dispatch_automaton(
3499	handler: &str,
3500	name: &str,
3501	arguments: Vec<Value>,
3502	engine: &mut Engine,
3503) -> Result<Value, Error> {
3504	match handler {
3505		"automaton" => {
3506			let rest = name
3507				.strip_prefix("automaton:")
3508				.expect("an automaton: operation");
3509			let target = format!("{}:{rest}", automaton_namespace(name, &arguments)?);
3510			// A default set for the `automaton:` operation does not apply to a kind without the hint.
3511			let defaults = engine.defaults_of(name);
3512			let mut hints = Vec::new();
3513			for (key, value) in &engine.hints {
3514				match check_hint(&target, key, value) {
3515					Ok(()) => hints.push((key.clone(), value.clone())),
3516					Err(_) if defaults.contains(&(key.clone(), value.clone())) => {}
3517					Err(message) => return Err(Error::plain(message)),
3518				}
3519			}
3520			Ok(match call(&target, arguments, &hints, engine)? {
3521				Value::Nfa(nfa) => Value::Automaton(Rc::new(AutomatonKind::Nfa(nfa))),
3522				Value::Nft(nft) => Value::Automaton(Rc::new(AutomatonKind::Nft(nft))),
3523				other => other,
3524			})
3525		}
3526		"automaton_load" => {
3527			let [paths] = take(arguments, name)?;
3528			engine.load(&paths, "automaton")
3529		}
3530		"automaton_parse" => {
3531			let [text] = take(arguments, name)?;
3532			let text = as_string(&text).map_err(Error::plain)?;
3533			engine.parse_text(&text, "automaton")
3534		}
3535		"intermediate_parse" => {
3536			let [text] = take(arguments, name)?;
3537			let text = as_string(&text).map_err(Error::plain)?;
3538			let parsed = IntermediateAutomaton::parse_mata_all(&text)
3539				.map_err(|e| Error::plain(e.to_string()))?;
3540			Ok(intermediate_list(parsed))
3541		}
3542		"intermediate_mintermize" => {
3543			let [value] = take(arguments, name)?;
3544			if let Value::List(items) = &value {
3545				let automata = items
3546					.iter()
3547					.map(intermediate_of)
3548					.collect::<Result<Vec<_>, _>>()?;
3549				let mintermized = IntermediateAutomaton::mintermize(&automata)
3550					.map_err(|e| Error::plain(e.to_string()))?;
3551				return Ok(intermediate_list(mintermized));
3552			}
3553			let mintermized = IntermediateAutomaton::mintermize(&[intermediate_of(&value)?])
3554				.map_err(|e| Error::plain(e.to_string()))?
3555				.into_iter()
3556				.next()
3557				.expect("one mintermized automaton per input");
3558			Ok(Value::Intermediate(Rc::new(mintermized)))
3559		}
3560		"nfa_construct" => {
3561			let [value] = take(arguments, name)?;
3562			let nfa = intermediate_of(&value)?
3563				.construct_nfa(&mut engine.script_alphabet)
3564				.map_err(|e| Error::plain(e.to_string()))?;
3565			Ok(Value::Nfa(Rc::new(nfa)))
3566		}
3567		"nft_construct" => {
3568			let [value] = take(arguments, name)?;
3569			let nft = intermediate_of(&value)?
3570				.construct_nft(&mut engine.script_alphabet)
3571				.map_err(|e| Error::plain(e.to_string()))?;
3572			Ok(Value::Nft(Rc::new(nft)))
3573		}
3574		_ if handler.starts_with("lazy_") => {
3575			crate::lazy::dispatch(handler, name, arguments, engine)
3576		}
3577		_ if handler.starts_with("ws1s_") => {
3578			crate::logic::dispatch(handler, name, arguments, engine)
3579		}
3580		_ => Err(Error::plain(format!(
3581			"operation '{name}' is not available in the Rust bindings"
3582		))),
3583	}

3733/// an error rather than a verdict, because a wrong `false` would hide the reason. The pairs left to
3734/// compare wait on an explicit stack, so the nesting never grows the call stack, which the supported
3735/// WebAssembly targets cannot grow past their default. Equal verdicts for list pairs are memoized by
3736/// identity with the deepest level they were walked at, so shared lists compare in one walk instead
3737/// of exponentially many, and a pair met deeper than before is walked again against the limit.
3738fn values_equal(lhs: &Value, rhs: &Value) -> Result<bool, String> {
3739	let mut equal_lists = std::collections::HashMap::new();
3740	let mut pending = vec![(lhs, rhs, 0)];
3741	while let Some((lhs, rhs, depth)) = pending.pop() {
3742		if depth > crate::eval::MAX_NESTING {
3743			return Err(format!(
3744				"the values compared nest deeper than {}",
3745				crate::eval::MAX_NESTING
3746			));
3747		}
3748		match (lhs, rhs) {
3749			(Value::List(lhs), Value::List(rhs)) => {
3750				// A pair whose verdict an earlier one left behind is never recorded wrongly: a
3751				// differing element or a too deep pair ends the whole comparison instead of being
3752				// recorded, so the pairs stored here are the ones whose walk finished. Their walk
3753				// covers this one only from a level at least as deep, where the limit was closer.
3754				let key = (std::ptr::from_ref(&**lhs), std::ptr::from_ref(&**rhs));
3755				match equal_lists.get(&key) {
3756					Some(&walked) if walked >= depth => continue,
3757					_ => {
3758						equal_lists.insert(key, depth);
3759					}
3760				}
3761				if lhs.len() != rhs.len() {
3762					return Ok(false);
3763				}
3764				for (lhs, rhs) in lhs.iter().zip(rhs.iter()) {
3765					pending.push((lhs, rhs, depth + 1));
3766				}
3767			}
3768			(Value::Bool(lhs), Value::Bool(rhs)) if lhs != rhs => return Ok(false),
3769			(Value::Int(lhs), Value::Int(rhs)) if lhs != rhs => return Ok(false),
3770			(Value::String(lhs), Value::String(rhs)) if lhs != rhs => return Ok(false),
3771			(Value::Word(lhs), Value::Word(rhs)) if lhs != rhs => return Ok(false),
3772			(Value::Symbol(lhs), Value::Symbol(rhs)) if lhs != rhs => return Ok(false),
3773			// The same node of the formula, as a leaf of one value is; not the same relation.
3774			(Value::Term(lhs), Value::Term(rhs)) if lhs != rhs => return Ok(false),
3775			(Value::Bool(_), Value::Bool(_))
3776			| (Value::Int(_), Value::Int(_))
3777			| (Value::String(_), Value::String(_))
3778			| (Value::Word(_), Value::Word(_))
3779			| (Value::Symbol(_), Value::Symbol(_))
3780			| (Value::Term(_), Value::Term(_))
3781			| (Value::None, Value::None) => {}
3782			_ => return Ok(false),
3783		}
3784	}
3785	Ok(true)
3786}
