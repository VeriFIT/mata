13/// Where a script comes from: a file, standard input, or the text of `--expression`.
14pub(crate) enum Source {
15	File(String),
16	Stdin,
17	Expression(String),
18}
19
20impl Source {
21	/// The name `check` reports and the text of the script, with the directory its paths are relative to.
22	fn read(&self) -> Result<(String, String, Option<std::path::PathBuf>), Error> {
23		match self {
24			Source::Expression(text) => Ok(("<expression>".into(), text.clone(), None)),
25			Source::Stdin => Ok(("<stdin>".into(), crate::eval::read_text("-")?, None)),
26			Source::File(path) => {
27				let text = std::fs::read_to_string(path)
28					.map_err(|e| Error::plain(format!("cannot read '{path}': {e}")))?;
29				let base = std::path::Path::new(path)
30					.parent()
31					.map(std::path::Path::to_path_buf)
32					.filter(|parent| !parent.as_os_str().is_empty());
33				Ok((path.clone(), text, base))
34			}
35		}
36	}
37}

39/// What one `mata script run` is given besides the script.
40pub(crate) struct Inputs {
41	pub(crate) positional: Vec<String>,
42	pub(crate) named: HashMap<String, String>,
43	pub(crate) parameters: HashMap<String, String>,
44	/// How long the run may take; see [`Engine::deadline`](crate::eval::Engine::deadline).
45	pub(crate) time_limit: Option<std::time::Duration>,
46}
47
48/// Parses a script after reading the language version of its header, and checks the version first, so
49/// a script for another version is rejected before syntax this interpreter does not know is read.
50fn parse(source: &str) -> Result<(Vec<Spanned<FormKind>>, crate::version::Version), Error> {
51	let version = crate::version::declared(source)?.unwrap_or(crate::version::LANGUAGE);
52	Ok((ast::parse(source)?, version))
53}

55/// Runs a script; `checked` runs the static check of `mata script check` first, so nothing runs
56/// when the check fails.
57pub(crate) fn run(
58	source: &Source,
59	inputs: Inputs,
60	output: Option<&str>,
61	checked: bool,
62) -> Result<Exit, Error> {
63	let (_, text, base_dir) = source.read()?;
64	if matches!(source, Source::Stdin) && inputs.positional.iter().any(|input| input == "-") {
65		return Err(Error::plain(
66			"standard input holds the script, so no input can be '-'".into(),
67		));
68	}
69	let (forms, version) = parse(&text)?;
70	let mut engine = Engine::for_language(version);
71	if checked {
72		crate::check::check(&forms, &engine, Some(&inputs))?;
73	}
74	// The script resolves the relative paths it loads against its own directory, but a path given on
75	// the command line is relative to the current one.
76	let anchor = |path: &String| anchored(path, base_dir.as_deref());
77	let positional = inputs
78		.positional
79		.iter()
80		.map(anchor)
81		.collect::<Result<Vec<_>, _>>()?;
82	let named = inputs
83		.named
84		.iter()
85		.map(|(name, path)| Ok((name.clone(), anchor(path)?)))
86		.collect::<Result<HashMap<_, _>, Error>>()?;
87	engine.set_inputs(positional.clone(), named);
88	engine.parameters = inputs.parameters;
89	engine.set_base_dir(base_dir);
90	engine.deadline = inputs
91		.time_limit
92		.and_then(|limit| Some((std::time::Instant::now().checked_add(limit)?, limit)));
93	if let Some(path) = output.filter(|path| *path != "-") {
94		// The script was read already, but writing over its file would still destroy it.
95		let script = match source {
96			Source::File(script) => Some(("script", script.as_str())),
97			Source::Stdin | Source::Expression(_) => None,
98		};
99		let given = inputs.positional.iter().chain(inputs.named.values());
100		let files = script
101			.into_iter()
102			.chain(given.map(|input| ("input", input.as_str())));
103		engine.sink = create_output(path, files)?;
104	}
105	let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
106		let bound = bind_declarations(&mut engine, &forms, &positional)?;
107		engine.execute(&forms[bound..])
108	}));
109	match result {
110		Ok(Ok(())) => {}
111		Ok(Err(error)) => return Err(error),
112		Err(_) => {
113			return Err(Error::plain(
114				"the Mata library aborted the operation".into(),
115			));
116		}
117	}
118	Ok(if engine.assertions_failed.on() {
119		Exit::Assertion
120	} else {
121		Exit::Success
122	})
123}

125/// A command-line path made absolute when it is relative and the script is in another directory.
126fn anchored(path: &str, base_dir: Option<&std::path::Path>) -> Result<String, Error> {
127	if base_dir.is_none() || path == "-" || std::path::Path::new(path).is_absolute() {
128		return Ok(path.to_owned());
129	}
130	std::path::absolute(path)
131		.map_err(|e| Error::plain(format!("cannot resolve '{path}': {e}")))?
132		.into_os_string()
133		.into_string()
134		.map_err(|_| {
135			Error::plain(format!(
136				"cannot resolve '{path}': the current directory is not UTF-8"
137			))
138		})
139}

141pub(crate) fn check(source: &Source) -> Result<Exit, Error> {
142	let (name, text, _) = source.read()?;
143	let (forms, version) = parse(&text)?;
144	let engine = Engine::for_language(version);
145	let report = crate::check::check(&forms, &engine, None)?;
146	crate::write_stdout(&format!("checked: {name}\n"))?;
147	eprintln!(
148		"termination: {}",
149		if report.termination {
150			"guaranteed"
151		} else {
152			"not-guaranteed"
153		}
154	);
155	for warning in &report.warnings {
156		eprintln!("{warning}");
157	}
158	Ok(Exit::Success)
159}

161pub(crate) fn repl(output: Option<&str>) -> Result<Exit, Error> {
162	let mut engine = Engine::new();
163	if let Some(path) = output.filter(|path| *path != "-") {
164		engine.sink = create_output(path, [])?;
165	}
166	let interactive = std::io::stdin().is_terminal();
167	let mut failed = false;
168	let mut buffer = String::new();
169	let mut stdin = std::io::stdin().lock();
170	loop {
171		if interactive {
172			crate::write_stdout("mata> ")?;
173		}
174		let mut line = String::new();
175		let read = stdin
176			.read_line(&mut line)
177			.map_err(|e| Error::plain(format!("cannot read standard input: {e}")))?;
178		if read == 0 {
179			if !buffer.trim().is_empty() {
180				eprintln!("mata: the input ends inside a form");
181				failed = true;
182			}
183			break;
184		}
185		buffer.push_str(&line);
186		let forms = match ast::parse_complete(&buffer) {
187			Ok(Some(forms)) => forms,
188			Ok(None) => continue,
189			Err(error) => {
190				buffer.clear();
191				eprintln!("mata: {}", error.render());
192				failed = true;
193				continue;
194			}
195		};
196		buffer.clear();
197		for form in forms {
198			let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
199				engine.execute_form(&form)
200			}));
201			match result {
202				Ok(Ok(value)) => {
203					if !matches!(value, Value::None) {
204						let text = summary(&value, &engine);
205						engine.emit(&text)?;
206					}
207				}
208				Ok(Err(error)) => {
209					eprintln!("mata: {}", error.render());
210					failed = true;
211				}
212				Err(_) => {
213					eprintln!("mata: the Mata library aborted the operation");
214					failed = true;
215				}
216			}
217			if engine.exit.on() {
218				break;
219			}
220		}
221		if engine.exit.on() {
222			break;
223		}
224	}
225	Ok(if failed {
226		Exit::Usage
227	} else if engine.assertions_failed.on() {
228		Exit::Assertion
229	} else {
230		Exit::Success
231	})
232}

234/// Opens the `--output` file at once, so every value is written as it is printed. The file must not
235/// be one of `files`, the script and its inputs, each with what it is, which opening it would empty.
236fn create_output<'a>(
237	path: &str,
238	files: impl IntoIterator<Item = (&'a str, &'a str)>,
239) -> Result<Sink, Error> {
240	if let Ok(output) = std::fs::canonicalize(path)
241		&& let Some((what, file)) = files.into_iter().find(|(_, file)| {
242			*file != "-" && std::fs::canonicalize(file).is_ok_and(|file| file == output)
243		}) {
244		return Err(Error::plain(format!(
245			"the output '{path}' is the {what} '{file}'"
246		)));
247	}
248	let file = std::fs::File::create(path)
249		.map_err(|e| Error::plain(format!("cannot write '{path}': {e}")))?;
250	Ok(Sink::File(std::io::BufWriter::new(file)))
251}

253/// The value of a REPL form: an automaton by its kind and size, so that a large one does not flood the
254/// session; `print` shows it in full.
255fn summary(value: &Value, engine: &Engine) -> String {
256	let automaton = match value {
257		Value::Nfa(nfa) => Some(("Nfa", nfa.num_states(), nfa.num_transitions())),
258		Value::Nft(nft) => Some(("Nft", nft.num_states(), nft.num_edges())),
259		Value::Automaton(automaton) => Some(match &**automaton {
260			AutomatonKind::Nfa(nfa) => ("Nfa", nfa.num_states(), nfa.num_transitions()),
261			AutomatonKind::Nft(nft) => ("Nft", nft.num_states(), nft.num_edges()),
262		}),
263		_ => None,
264	};
265	match automaton {
266		Some((kind, states, transitions)) => {
267			format!("#<{kind} with {states} states and {transitions} transitions>\n")
268		}
269		None => display_value(value, engine),
270	}
271}

273/// Binds the declared inputs and parameters, which come first in a script, interleaved only with
274/// `set-option` and `set-info`. Those forms run in order with the declarations, so an option such as
275/// `:alphabet` applies to the automata the declarations load. A positional input file with several
276/// automata gives one automaton to each declaration in turn, and every input given must be taken. The
277/// inputs are read ahead at the first declaration, after the forms before it, so that bitvector
278/// automata of different files share their minterms. Returns how many forms it ran.
279fn bind_declarations(
280	engine: &mut Engine,
281	forms: &[Spanned<FormKind>],
282	inputs: &[String],
283) -> Result<usize, Error> {
284	let named = declared_named_inputs(engine, forms)?;
285	let declares = named.is_some();
286	let mut unread = named;
287	let mut files = inputs.iter();
288	let mut pending: std::collections::VecDeque<Value> = std::collections::VecDeque::new();
289	let mut bound = 0;
290	for form in forms {
291		match &form.node {
292			FormKind::Declare {
293				name,
294				sort,
295				variadic,
296			} => {
297				if let Some(named) = unread.take() {
298					for path in inputs.iter().chain(&named) {
299						engine.read_ahead(path)?;
300					}
301				}
302				let at = |mut error: Error| {
303					error.at.get_or_insert((form.line, form.column));
304					error
305				};
306				let kind = kind_of(sort);
307				let narrowed = |value| {
308					narrow_input(kind, value)
309						.map_err(|message| form.at(format!("input '{name}': {message}")))
310				};
311				let value = if *variadic {
312					let mut values = pending
313						.drain(..)
314						.map(narrowed)
315						.collect::<Result<Vec<_>, _>>()?;
316					for path in files.by_ref() {
317						values.extend(engine.load_inputs(path, kind).map_err(at)?);
318					}
319					Value::List(std::rc::Rc::new(values))
320				} else if let Some(path) = engine.named_inputs.get(name).cloned() {
321					engine.load_input(&path, kind).map_err(at)?
322				} else {
323					// Every section of a file is parsed as the kind it is, then checked against the
324					// declaration that takes it.
325					if pending.is_empty() {
326						let path = files
327							.next()
328							.ok_or_else(|| form.at(format!("no input for '{name}'")))?;
329						pending.extend(engine.load_inputs(path, "automaton").map_err(at)?);
330					}
331					narrowed(
332						pending
333							.pop_front()
334							.expect("a loaded input holds an automaton"),
335					)?
336				};
337				engine.scope.define(name.clone(), value);
338			}
339			FormKind::Header { .. }
340			| FormKind::SetOption { .. }
341			| FormKind::SetInfo { .. }
342			| FormKind::Parameter { .. } => {
343				engine.execute_form(form)?;
344			}
345			_ => break,
346		}
347		bound += 1;
348	}
349	let left = files.count();
350	if declares && left > 0 {
351		return Err(Error::plain(format!(
352			"{left} positional inputs are left over; declare an input for each, or use declare-inputs"
353		)));
354	}
355	if !pending.is_empty() {
356		return Err(Error::plain(format!(
357			"{} automata of the inputs are left over; declare an input for each, or use declare-inputs",
358			pending.len()
359		)));
360	}
361	Ok(bound)
362}

364/// Rejects a named input that no `declare-input` takes. Returns the paths of the named inputs in the
365/// order of their declarations, or `None` when the script declares no input.
366fn declared_named_inputs(
367	engine: &Engine,
368	forms: &[Spanned<FormKind>],
369) -> Result<Option<Vec<String>>, Error> {
370	let declared: Vec<(&str, bool)> = forms
371		.iter()
372		.take_while(|form| {
373			matches!(
374				form.node,
375				FormKind::Declare { .. }
376					| FormKind::Header { .. }
377					| FormKind::SetOption { .. }
378					| FormKind::SetInfo { .. }
379					| FormKind::Parameter { .. }
380			)
381		})
382		.filter_map(|form| match &form.node {
383			FormKind::Declare { name, variadic, .. } => Some((name.as_str(), *variadic)),
384			_ => None,
385		})
386		.collect();
387	if declared.is_empty() {
388		return Ok(None);
389	}
390	let mut named: Vec<&String> = engine.named_inputs.keys().collect();
391	named.sort();
392	if let Some(name) = named
393		.iter()
394		.find(|name| !declared.contains(&(name.as_str(), false)))
395	{
396		return Err(Error::plain(format!(
397			"--input {name}: the script declares no input '{name}'"
398		)));
399	}
400	Ok(Some(
401		declared
402			.iter()
403			.filter_map(|(name, _)| engine.named_inputs.get(*name).cloned())
404			.collect(),
405	))
