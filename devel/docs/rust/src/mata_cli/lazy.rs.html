21/// The operations that run one check of a term, with the statistics `lazy:stats` reports.
22pub(crate) const CHECKS: &[&str] = &[
23	"lazy:empty?",
24	"lazy:empty-within?",
25	"lazy:witness",
26	"lazy:included?",
27	"lazy:equivalent?",
28];

30/// Whether `name` is a query, which a run measures and traces as a library call; a builder only adds a node.
31pub(crate) fn is_query(name: &str) -> bool {
32	CHECKS.contains(&name) || name == "lazy:materialize"
33}

35/// Whether `expression` is an application of a check, the operand of `lazy:stats`.
36pub(crate) fn applies_check(expression: &Expr) -> bool {
37	matches!(&expression.node, Kind::Call { function, .. }
38		if matches!(&function.node, Kind::Name(name) if CHECKS.contains(&name.as_str())))
39}

41/// The error for a `lazy:stats` whose operand is not an application of a check.
42pub(crate) fn stats_message() -> String {
43	format!(
44		"lazy:stats applies to an application of {}",
45		CHECKS.join(", ")
46	)
47}

49/// The statistics of a query as `lazy:stats` prints them: a record name under the label and its value.
50pub(crate) fn stats_records(stats: &Stats) -> [(&'static str, usize); 7] {
51	let (mut nodes, mut node_states, mut node_transitions) = (0, 0, 0);
52	for node in stats.explicit.iter().filter(|node| !node.lazy) {
53		nodes += 1;
54		node_states += node.states;
55		node_transitions += node.transitions;
56	}
57	[
58		("explored", stats.explored),
59		("pruned", stats.pruned),
60		("macrostates", stats.macrostates),
61		("dropped", stats.dropped),
62		("explicit", nodes),
63		("explicit-states", node_states),
64		("explicit-transitions", node_transitions),
65	]
66}

68/// An automaton imported as a leaf, held weakly: the entry keeps the address of the value from being reused
69/// without keeping the automaton alive.
70enum Imported {
71	Nfa(#[expect(dead_code, reason = "held only to keep the address")] Weak<Nfa>),
72	Nft(#[expect(dead_code, reason = "held only to keep the address")] Weak<Nft>),
73}
74
75/// The leaf of a value is found by the address of the value and the symbols `:alphabet` adds to it.
76type LeafKey = (usize, Option<Vec<lazy::Symbol>>);
77
78/// The formula of a run. Every `Term` of the run is a node of it, so terms of different queries combine; it
79/// only grows while the run lasts.
80pub(crate) struct Lazy {
81	formula: Formula,
82	/// The leaf of every automaton value imported so far, so that a value used in many terms is copied once.
83	leaves: HashMap<LeafKey, (Imported, Term)>,
84	/// The annotations used by every query that has no `:plan` or `:policy`.
85	annotations: lazy::Plan,
86	/// The statistics of the last query, for `lazy:stats`, and the index of the plan that answered a portfolio.
87	pub(crate) stats: Option<(Stats, Option<usize>)>,
88}

90impl Lazy {
91	pub(crate) fn new() -> Lazy {
92		Lazy {
93			formula: Formula::new(),
94			leaves: HashMap::new(),
95			annotations: lazy::Plan::new(),
96			stats: None,
97		}
98	}
99
100	/// The number of tapes of `term`.
101	pub(crate) fn arity(&self, term: Term) -> usize {
102		self.formula
103			.arity(term)
104			.expect("every term of a run belongs to its formula")
105	}
106
107	/// The leaf of an automaton value, imported at its first use with `alphabet`.
108	fn leaf(&mut self, value: &Value, alphabet: Option<Vec<lazy::Symbol>>) -> Result<Term, Error> {
109		enum Automaton<'a> {
110			Nfa(&'a Rc<Nfa>),
111			Nft(&'a Rc<Nft>),
112		}
113		let automaton = match value {
114			Value::Nfa(nfa) => Automaton::Nfa(nfa),
115			Value::Nft(nft) => Automaton::Nft(nft),
116			Value::Automaton(kind) => match &**kind {
117				AutomatonKind::Nfa(nfa) => Automaton::Nfa(nfa),
118				AutomatonKind::Nft(nft) => Automaton::Nft(nft),
119			},
120			other => {
121				return Err(Error::plain(format!(
122					"expected an Nfa or an Nft, found {}",
123					value_sort(other)
124				)));
125			}
126		};
127		let address = match automaton {
128			Automaton::Nfa(nfa) => Rc::as_ptr(nfa).addr(),
129			Automaton::Nft(nft) => Rc::as_ptr(nft).addr(),
130		};
131		let key = (address, alphabet);
132		if let Some((_, term)) = self.leaves.get(&key) {
133			return Ok(*term);
134		}
135		let formula = &mut self.formula;
136		let (imported, term) = match (automaton, &key.1) {
137			(Automaton::Nfa(nfa), None) => {
138				(Imported::Nfa(Rc::downgrade(nfa)), formula.leaf(&**nfa))
139			}
140			(Automaton::Nfa(nfa), Some(symbols)) => (
141				Imported::Nfa(Rc::downgrade(nfa)),
142				formula.leaf_over(&**nfa, [symbols.iter().copied()]),
143			),
144			(Automaton::Nft(nft), None) => {
145				(Imported::Nft(Rc::downgrade(nft)), formula.leaf(&**nft))
146			}
147			(Automaton::Nft(nft), Some(symbols)) => (
148				Imported::Nft(Rc::downgrade(nft)),
149				formula.leaf_over(&**nft, [symbols.iter().copied()]),
150			),
151		};
152		let term = term?;
153		self.leaves.insert(key, (imported, term));
154		Ok(term)
155	}
156
157	/// Runs one check of `term`, or a portfolio of its plans, and keeps its statistics for `lazy:stats`.
158	fn check(&mut self, term: Term, query: &Query) -> Result<Outcome, Error> {
159		self.stats = None;
160		let (plan, outcome) = if query.portfolio.is_empty() {
161			(None, self.formula.check(term, &query.options)?)
162		} else {
163			let (plan, outcome) =
164				self.formula
165					.check_portfolio(term, &query.options, &query.portfolio)?;
166			(Some(plan), outcome)
167		};
168		self.stats = Some((outcome.stats.clone(), plan));
169		Ok(outcome)
170	}
171
172	/// `lhs ∩ ¬rhs`, the words of `lhs` that `rhs` lacks.
173	fn difference(&mut self, lhs: Term, rhs: Term) -> Result<Term, Error> {
174		let complement = self.formula.complement(rhs)?;
175		Ok(self.formula.intersection(lhs, complement)?)
176	}
177
178	/// The term of an operand of the optimizer, whose leaves are automata.
179	fn term_of(&mut self, tree: &Tree<Value>) -> Result<Term, Error> {
180		type Join = fn(&mut Formula, Term, Term) -> Result<Term, lazy::Error>;
181		let fold = |lazy: &mut Lazy, trees: &[Tree<Value>], join: Join| -> Result<Term, Error> {
182			let mut joined = None;
183			for tree in trees {
184				let term = lazy.term_of(tree)?;
185				joined = Some(match joined {
186					None => term,
187					Some(joined) => join(&mut lazy.formula, joined, term)?,
188				});
189			}
190			Ok(joined.expect("an operation has operands"))
191		};
192		match tree {
193			Tree::Leaf(value) => self.leaf(value, None),
194			Tree::Union(trees) => fold(self, trees, Formula::union),
195			Tree::Intersection(trees) => fold(self, trees, Formula::intersection),
196			Tree::Difference(lhs, rhs) => {
197				let (lhs, rhs) = (self.term_of(lhs)?, self.term_of(rhs)?);
198				self.difference(lhs, rhs)
199			}
200			Tree::Complement(tree) => {
201				let term = self.term_of(tree)?;
202				Ok(self.formula.complement(term)?)
203			}
204		}
205	}
206}

208/// Serves plan inspection and `lazy:annotate`; these do no automata search.
209fn dispatch_planning(
210	handler: &str,
211	arguments: Vec<Value>,
212	engine: &mut Engine,
213) -> Result<Value, Error> {
214	match handler {
215		"lazy_plan" => {
216			let [] = take(arguments, "lazy:plan")?;
217			Ok(Value::String(engine.lazy.annotations.to_json()))
218		}
219		"lazy_view" => {
220			let [value] = take(arguments, "lazy:view")?;
221			let view = engine
222				.lazy
223				.formula
224				.view(term(&value)?)
225				.map_err(Error::from)?;
226			Ok(Value::String(view.to_json()))
227		}
228		"lazy_annotate" => {
229			let [value] = take(arguments, "lazy:annotate")?;
230			annotate(engine, &value)
231		}
232		_ => unreachable!("the caller names only planning operations"),
233	}
234}

236/// Records one operation's evaluation and operand order for later queries.
237fn annotate(engine: &mut Engine, value: &Value) -> Result<Value, Error> {
238	let operand = term(value)?;
239	let previous = engine
240		.lazy
241		.annotations
242		.annotation(operand)
243		.copied()
244		.unwrap_or_default();
245	let eval = optional::<String>(engine, "eval");
246	let max_states = optional::<i64>(engine, "max-states")
247		.map(|limit| count(limit, ":max-states"))
248		.transpose()?;
249	let mut annotation = previous;
250	annotation.eval = match eval.as_deref() {
251		Some("keep") if max_states.is_none() => lazy::Eval::Keep,
252		None if max_states.is_none() => previous.eval,
253		Some("lazy") if max_states.is_none() => lazy::Eval::Lazy,
254		Some("eager") | None => lazy::Eval::Eager { max_states },
255		Some("keep" | "lazy") => {
256			return Err(Error::plain(
257				":max-states of lazy:annotate requires :eval \"eager\"".into(),
258			));
259		}
260		Some(other) => {
261			return Err(Error::plain(format!(
262				":eval of lazy:annotate must be keep, lazy or eager, not {other}"
263			)));
264		}
265	};
266	if let Some(order) = optional::<String>(engine, "order") {
267		annotation.order = match order.as_str() {
268			"auto" => lazy::Order::Auto,
269			"as-written" => lazy::Order::AsWritten,
270			"swapped" => lazy::Order::Swapped,
271			_ => {
272				return Err(Error::plain(format!(
273					":order of lazy:annotate must be auto, as-written or swapped, not {order}"
274				)));
275			}
276		};
277	}
278	engine.lazy.annotations = engine
279		.lazy
280		.annotations
281		.clone()
282		.annotate(operand, annotation);
283	Ok(Value::Term(operand))
284}

286/// Serves the `lazy:` operations.
287pub(crate) fn dispatch(
288	handler: &str,
289	name: &str,
290	arguments: Vec<Value>,
291	engine: &mut Engine,
292) -> Result<Value, Error> {
293	match handler {
294		"lazy_plan" | "lazy_view" | "lazy_annotate" => {
295			dispatch_planning(handler, arguments, engine)
296		}
297		"lazy_leaf" => {
298			let [value] = take(arguments, name)?;
299			let alphabet = alphabet(engine)?;
300			engine.lazy.leaf(&value, alphabet).map(Value::Term)
301		}
302		"lazy_union" | "lazy_intersection" | "lazy_compose" | "lazy_post_image"
303		| "lazy_pre_image" => {
304			let [lhs, rhs] = take(arguments, name)?;
305			let (lhs, rhs) = (term(&lhs)?, term(&rhs)?);
306			let formula = &mut engine.lazy.formula;
307			match handler {
308				"lazy_union" => formula.union(lhs, rhs),
309				"lazy_intersection" => formula.intersection(lhs, rhs),
310				"lazy_compose" => formula.compose(lhs, rhs),
311				"lazy_post_image" => formula.post_image(lhs, rhs),
312				_ => formula.pre_image(lhs, rhs),
313			}
314			.map(Value::Term)
315			.map_err(Error::from)
316		}
317		"lazy_complement" | "lazy_identity" | "lazy_prefixes" => {
318			let [value] = take(arguments, name)?;
319			let operand = term(&value)?;
320			let formula = &mut engine.lazy.formula;
321			match handler {
322				"lazy_complement" => formula.complement(operand),
323				"lazy_identity" => formula.identity(operand),
324				_ => formula.prefixes(operand),
325			}
326			.map(Value::Term)
327			.map_err(Error::from)
328		}
329		"lazy_pad_closure" => {
330			let mut arguments = arguments.into_iter();
331			let operand = term(&arguments.next().expect("at least one argument"))?;
332			let pad = arguments
333				.map(|value| match value {
334					Value::Symbol(symbol) => Ok(lazy::Symbol::from(symbol)),
335					other => Err(Error::plain(format!(
336						"expected a Symbol, found {}",
337						value_sort(&other)
338					))),
339				})
340				.collect::<Result<Vec<_>, _>>()?;
341			let pad = if pad.is_empty() {
342				lazy::Pad::Last
343			} else {
344				lazy::Pad::Step(pad.into_boxed_slice())
345			};
346			engine
347				.lazy
348				.formula
349				.pad_closure(operand, pad)
350				.map(Value::Term)
351				.map_err(Error::from)
352		}
353		"lazy_project" => {
354			let [value, tapes] = take(arguments, name)?;
355			let operand = term(&value)?;
356			let tapes = as_list(&tapes)
357				.map_err(Error::plain)?
358				.iter()
359				.map(|tape| count(as_int(tape).map_err(Error::plain)?, "a tape"))
360				.collect::<Result<Vec<_>, _>>()?;
361			engine
362				.lazy
363				.formula
364				.project(operand, &tapes)
365				.map(Value::Term)
366				.map_err(Error::from)
367		}
368		"lazy_explicit" => {
369			let [value] = take(arguments, name)?;
370			let operand = term(&value)?;
371			let max_states = optional::<i64>(engine, "max-states")
372				.map(|limit| count(limit, ":max-states"))
373				.transpose()?;
374			let formula = &mut engine.lazy.formula;
375			match max_states {
376				Some(limit) => {
377					formula.explicit_with(operand, ExplicitOptions::new().max_states(limit))
378				}
379				None => formula.explicit(operand),
380			}
381			.map(Value::Term)
382			.map_err(Error::from)
383		}
384		_ => dispatch_query(handler, name, arguments, engine),
385	}
386}

388/// Serves the `lazy:` queries, each one check of the formula.
389fn dispatch_query(
390	handler: &str,
391	name: &str,
392	arguments: Vec<Value>,
393	engine: &mut Engine,
394) -> Result<Value, Error> {
395	match handler {
396		"lazy_empty" => {
397			let [value] = take(arguments, name)?;
398			let operand = term(&value)?;
399			decide(engine, operand)
400		}
401		"lazy_empty_within" => {
402			let [value] = take(arguments, name)?;
403			let operand = term(&value)?;
404			let query = query_options(engine, operand)?;
405			let outcome = engine.lazy.check(operand, &query)?;
406			Ok(Value::Bool(outcome.witness.is_none()))
407		}
408		"lazy_included" => {
409			let [lhs, rhs] = take(arguments, name)?;
410			let (lhs, rhs) = (term(&lhs)?, term(&rhs)?);
411			let difference = engine.lazy.difference(lhs, rhs)?;
412			decide(engine, difference)
413		}
414		"lazy_equivalent" => {
415			let [lhs, rhs] = take(arguments, name)?;
416			let (lhs, rhs) = (term(&lhs)?, term(&rhs)?);
417			// One check of the symmetric difference, which holds a word of either direction.
418			let missing = engine.lazy.difference(lhs, rhs)?;
419			let extra = engine.lazy.difference(rhs, lhs)?;
420			let either = engine.lazy.formula.union(missing, extra)?;
421			decide(engine, either)
422		}
423		"lazy_witness" => {
424			let [value] = take(arguments, name)?;
425			let operand = term(&value)?;
426			let query = query_options(engine, operand)?;
427			let outcome = engine.lazy.check(operand, &query)?;
428			Ok(outcome.witness.as_ref().map_or(Value::None, witness_value))
429		}
430		"lazy_materialize" => {
431			let [value] = take(arguments, name)?;
432			let operand = term(&value)?;
433			let query = query_options(engine, operand)?;
434			if !query.portfolio.is_empty() {
435				return Err(Error::plain(
436					"lazy:materialize builds the whole term, so a portfolio gains nothing".into(),
437				));
438			}
439			let explicit = engine.lazy.formula.materialize(operand, &query.options)?;
440			let nft = Nft::try_from(&explicit).map_err(|e| Error::plain(e.to_string()))?;
441			Ok(if explicit.arity() == 1 {
442				Value::Nfa(Rc::new(nft.to_nfa()))
443			} else {
444				Value::Nft(Rc::new(nft))
445			})
446		}
447		_ => Err(Error::plain(format!(
448			"operation '{name}' is not available in the Rust bindings"
449		))),
450	}
451}

453/// Checks `term` for emptiness: `true` when it is empty. The witness, a word of the term, is stored when one is
454/// wanted, as the `nfa:` decisions do.
455fn decide(engine: &mut Engine, term: Term) -> Result<Value, Error> {
456	let query = query_options(engine, term)?;
457	let outcome = engine.lazy.check(term, &query)?;
458	let wanted = engine.options.produce_witnesses.on() || engine.witness_requested.on();
459	engine.last_witness = if wanted {
460		outcome.witness.as_ref().map(witness_value)
461	} else {
462		None
463	};
464	Ok(Value::Bool(outcome.is_empty()))
465}

467/// The options of the lazy query of the optimizer that decides `decision` of `operands`. `None` when its
468/// answer could differ from the written one: a complement over the script alphabet, which a query cannot
469/// restrict itself to, of automata that read symbols outside of it.
470pub(crate) fn tree_options(
471	engine: &Engine,
472	decision: Decision,
473	operands: &[Tree<Value>],
474) -> Result<Option<Options>, Error> {
475	let over_script_alphabet =
476		decision == Decision::Universal || operands.iter().any(Tree::has_complement);
477	let script: HashSet<mata::Symbol> = engine.script_alphabet.symbols().collect();
478	for leaf in operands.iter().flat_map(Tree::leaves) {
479		let nfa = expect_nfa(leaf).map_err(Error::plain)?;
480		if over_script_alphabet
481			&& nfa
482				.used_symbols()
483				.into_iter()
484				.any(|symbol| symbol != mata::Symbol::EPSILON && !script.contains(&symbol))
485		{
486			return Ok(None);
487		}
488	}
489	let mut options = Options::new();
490	if over_script_alphabet {
491		options = options.alphabet(script.into_iter().map(lazy::Symbol::from));
492	}
493	Ok(Some(options))
494}

496/// Decides `decision` of `operands` as one lazy query with `options`, for the optimizer; the witness is
497/// stored as by the written decision.
498pub(crate) fn decide_tree(
499	engine: &mut Engine,
500	decision: Decision,
501	operands: &[Tree<Value>],
502	options: &Options,
503) -> Result<Value, Error> {
504	let mut terms = Vec::with_capacity(operands.len());
505	for operand in operands {
506		terms.push(engine.lazy.term_of(operand)?);
507	}
508	let lazy = &mut engine.lazy;
509	let term = match (decision, terms.as_slice()) {
510		(Decision::Empty, &[term]) => term,
511		(Decision::Universal, &[term]) => lazy.formula.complement(term)?,
512		(Decision::Included, &[lhs, rhs]) => lazy.difference(lhs, rhs)?,
513		(Decision::Equivalent, &[lhs, rhs]) => {
514			let missing = lazy.difference(lhs, rhs)?;
515			let extra = lazy.difference(rhs, lhs)?;
516			lazy.formula.union(missing, extra)?
517		}
518		_ => unreachable!("a decision has as many operands as its operation"),
519	};
520	let query = Query {
521		options: options.clone(),
522		portfolio: Vec::new(),
523	};
524	let outcome = engine.lazy.check(term, &query)?;
525	let wanted = engine.options.produce_witnesses.on() || engine.witness_requested.on();
526	engine.last_witness = if wanted {
527		outcome.witness.as_ref().map(witness_value)
528	} else {
529		None
530	};
531	Ok(Value::Bool(outcome.is_empty()))
532}

534/// A witness as a script value: the `Word` of a language, and a list of one `Word` per tape otherwise.
535fn witness_value(witness: &Witness) -> Value {
536	let tape = |index| {
537		let symbols = witness
538			.tape(index)
539			.expect("a witness has a word on each of its tapes");
540		Value::Word(Word::from(
541			symbols
542				.into_iter()
543				.map(mata::Symbol::from)
544				.collect::<Vec<_>>(),
545		))
546	};
547	if witness.arity() == 1 {
548		tape(0)
549	} else {
550		Value::List(Rc::new((0..witness.arity()).map(tape).collect()))
551	}
552}

554/// The options of one lazy query, and the plans of a portfolio (none for a single query).
555struct Query {
556	options: Options,
557	portfolio: Vec<Plan>,
558}

560/// The options of one lazy query, including the default annotations or an overriding policy or external plan.
561fn query_options(engine: &mut Engine, term: Term) -> Result<Query, Error> {
562	let options = options(engine)?;
563	let external = optional::<String>(engine, "plan");
564	let policy = optional::<String>(engine, "policy");
565	if external.is_some() && policy.is_some() {
566		return Err(Error::plain(":plan and :policy cannot be combined".into()));
567	}
568	let single = |plan: Plan| Query {
569		options: options.clone().plan(plan),
570		portfolio: Vec::new(),
571	};
572	if let Some(json) = external {
573		return Ok(single(engine.lazy.formula.plan_from_json(&json)?));
574	}
575	let Some(policy) = policy else {
576		return Ok(single(engine.lazy.annotations.clone()));
577	};
578	if !matches!(
579		policy.as_str(),
580		"lazy" | "eager" | "heuristic" | "portfolio"
581	) {
582		return Err(Error::plain(format!(
583			":policy must be lazy, eager, heuristic or portfolio, not {policy}"
584		)));
585	}
586	let view = engine.lazy.formula.view(term)?;
587	let heuristic = match optional::<i64>(engine, "heuristic-max-states") {
588		Some(limit) => lazy::Heuristic::new().max_states(count(limit, ":heuristic-max-states")?),
589		None => lazy::Heuristic::new(),
590	};
591	Ok(match policy.as_str() {
592		"lazy" => single(lazy::AllLazy.plan(&view)),
593		"eager" => single(lazy::AllEager.plan(&view)),
594		"heuristic" => single(heuristic.plan(&view)),
595		_ => Query {
596			options,
597			portfolio: vec![
598				lazy::AllLazy.plan(&view),
599				lazy::AllEager.plan(&view),
600				heuristic.plan(&view),
601			],
602		},
603	})
604}

606/// The options of a query: its limits and the symbols it adds to the alphabet of the checked term.
607pub(crate) fn options(engine: &mut Engine) -> Result<Options, Error> {
608	let mut options = Options::new();
609	if let Some(limit) = optional::<i64>(engine, "time-limit") {
610		let milliseconds = u64::try_from(limit)
611			.map_err(|_| Error::plain(format!(":time-limit must not be negative, not {limit}")))?;
612		options = options.time_limit(Duration::from_millis(milliseconds));
613	}
614	if let Some(limit) = optional::<i64>(engine, "max-states") {
615		options = options.max_states(count(limit, ":max-states")?);
616	}
617	if let Some(limit) = optional::<i64>(engine, "max-length") {
618		options = options.max_length(count(limit, ":max-length")?);
619	}
620	if let Some(shortest) = optional::<bool>(engine, "shortest") {
621		options = options.shortest(shortest);
622	}
623	if let Some(symbols) = alphabet(engine)? {
624		options = options.alphabet(symbols);
625	}
626	Ok(options)
627}
628
629/// The symbols of `:alphabet`, names separated by spaces, translated through the script alphabet.
630fn alphabet(engine: &mut Engine) -> Result<Option<Vec<lazy::Symbol>>, Error> {
631	let Some(names) = optional::<String>(engine, "alphabet") else {
632		return Ok(None);
633	};
634	names
635		.split_whitespace()
636		.map(|name| engine.intern_name(name).map(lazy::Symbol::from))
637		.collect::<Result<Vec<_>, _>>()
638		.map(Some)
639		.map_err(Error::plain)
640}
641
642fn term(value: &Value) -> Result<Term, Error> {
643	match value {
644		Value::Term(term) => Ok(*term),
645		other => Err(Error::plain(format!(
646			"expected a Term, found {}",
647			value_sort(other)
648		))),
649	}
650}
651
652/// `value` as a count, which must not be negative.
653fn count(value: i64, what: &str) -> Result<usize, Error> {
654	usize::try_from(value)
655		.map_err(|_| Error::plain(format!("{what} must not be negative, not {value}")))
656}
657
658impl From<lazy::Error> for Error {
659	fn from(error: lazy::Error) -> Error {
660		Error::plain(error.to_string())
661	}
662}

684	/// Regular expressions over `a` and `b` of up to `depth` nested operators, drawn by a fixed xorshift.
685	fn regexes(count: usize, depth: u32) -> Vec<String> {
686		fn draw(state: &mut u64, depth: u32) -> String {
687			*state ^= *state << 13;
688			*state ^= *state >> 7;
689			*state ^= *state << 17;
690			let choice = if depth == 0 { *state % 2 } else { *state % 5 };
691			match choice {
692				0 => "a".to_owned(),
693				1 => "b".to_owned(),
694				2 => format!("({})*", draw(state, depth - 1)),
695				3 => format!("({}|{})", draw(state, depth - 1), draw(state, depth - 1)),
696				_ => format!("{}{}", draw(state, depth - 1), draw(state, depth - 1)),
697			}
698		}
699		let mut state = 0x9e37_79b9_7f4a_7c15;
700		(0..count).map(|_| draw(&mut state, depth)).collect()
701	}
