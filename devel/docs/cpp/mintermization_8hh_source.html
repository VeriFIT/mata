83 /**
84 * The minterm partition of a sequence of source BDDs together with the incidence between the two.
85 *
86 * The minterms are numbered by their position in @c minterms, which is the order in which the refinement
87 * created them and therefore depends only on the input, not on the addresses CUDD assigned to the nodes.
88 */
89 struct MintermPartition {
90 /// Minterms of the source BDDs, in creation order.
91 std::vector<BDD> minterms{};
92 /// For each source BDD, the ascending numbers of the minterms below it.
93 std::vector<std::vector<size_t>> minterms_of_source{};
94 };

96 /**
97 * Takes a set of BDDs and build a minterm tree over it.
98 * The leaves of BDDs, which are minterms of input set, are returned
99 * @param source_bdds BDDs for which minterms are computed
100 * @return Computed minterms
101 */
102 std::unordered_set<BDD> compute_minterms(const std::unordered_set<BDD>& source_bdds) const;
103
104 /**
105 * Computes the same partition as @c compute_minterms() and additionally records, for each source BDD, which
106 * minterms lie below it.
107 *
108 * The incidence is carried along the refinement: when a region is split by a source BDD, the part inside the
109 * BDD inherits the incidence of the region plus that BDD, and the part outside inherits it unchanged. No BDD
110 * operation is therefore needed afterwards to relate a source BDD to the minterms below it.
111 * @param source_bdds BDDs for which minterms are computed
112 * @return Computed minterms and their incidence with @p source_bdds
113 */
114 MintermPartition compute_minterms_with_incidence(const std::vector<BDD>& source_bdds) const;

200 /// The manager starts with a small cache, which CUDD grows on demand: allocating and clearing the default one

203 // A copy would keep pointers to the keys of the original's maps in @c added_symbols_ and @c added_formulas_.
