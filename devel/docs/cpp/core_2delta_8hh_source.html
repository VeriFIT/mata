39/**
40 * @brief The keyed writes and lookup of a relation: one member each, at every arity.
41 *
42 * The keys are a pack of this class, not of the member, so every parameter has a declared type and
43 * `add(0, 1, 2, 3)` converts its `int`s correctly at the call. Arity 1 is specialised below.
44 *
45 * @tparam D The relation deriving from this. A friend of it, for its posts.
46 * @tparam P The relation's post chain.
47 */
48template <typename D, typename P, typename = std::make_index_sequence<arity_of<P>>> class KeyedAccess;
49
50template <typename D, typename P, size_t... KeyIdx> class KeyedAccess<D, P, std::index_sequence<KeyIdx...>> {
51 using Target = target_of<P>;
52 using State = typename TargetTraits<Target>::State;
53 using TargetArg = utils::ArgOf<Target>;
54 using TargetSet = typename PostAt<P, sizeof...(KeyIdx)>::type;
55
56 D& self() { return static_cast<D&>(*this); }
57 const D& self() const { return static_cast<const D&>(*this); }
58
59 public:
60 /**
61 * @brief Add the transition from @p source under @p keys to @p target.
62 *
63 * @param source Source state of the transition.
64 * @param keys Keys of the transition.
65 * @param target Target of the transition.
66 */
67 void add(const State source, const KeyOf<P, KeyIdx>... keys, const TargetArg target) {
68 self().resize_for_states(source, D::state_of(target));
69 // Through the post directly: mutable_state_post() would mark the count dirty.
70 if (insert_target(self().state_posts_[source], target, keys...)) { self().note_added(1); }
71 }
72
73 /**
74 * @brief Add the transition from @p source under @p keys to every one of @p targets.
75 *
76 * @param source Source state of the transition.
77 * @param keys Keys of the transition.
78 * @param targets Set of targets of the transition.
79 */
80 void add(const State source, const KeyOf<P, KeyIdx>... keys, const TargetSet& targets) {
81 const TargetSet snapshot{targets};
82 for (const Target& target : snapshot) {
83 self().resize_for_states(source, D::state_of(target));
84 if (insert_target(self().state_posts_[source], target, keys...)) { self().note_added(1); }
85 }
86 }
87
88 /**
89 * @brief Remove the transition from @p source under @p keys to @p target.
90 *
91 * @param source Source state of the transition.
92 * @param keys Keys of the transition.
93 * @param target Target of the transition.
94 * @throws Throws std::invalid_argument if @p source has no transition under @p keys. A @p source beyond the
95 * relation, or a @p target missing under existing @p keys, is ignored.
96 */
97 void remove(const State source, const KeyOf<P, KeyIdx>... keys, const TargetArg target) {
98 if (source >= self().state_posts_.size()) { return; }
99 // Looked up first: erase_target() reports whether the post emptied, not whether it erased.
100 const bool present{has_target_at(self().state_posts_[source], target, keys...)};
101 erase_target(self().state_posts_[source], target, keys...);
102 if (present) { self().note_removed(1); }
103 }
104
105 /**
106 * @brief Check whether the relation contains the transition from @p source under @p keys to @p target.
107 *
108 * @param source Source state of the transition.
109 * @param keys Keys of the transition.
110 * @param target Target of the transition.
111 * @return True if the relation contains the transition, false otherwise.
112 */
113 bool contains(const State source, const KeyOf<P, KeyIdx>... keys, const TargetArg target) const {
114 return source < self().state_posts_.size() && has_target_at(self().state_posts_[source], target, keys...);
115 }
116};

176/**
177 * @brief The generic transition relation: a vector of posts indexed by source state.
178 *
179 * A vector of posts indexed by source state, over a chain of posts ending in a set of targets.
180 * The depth is deduced from @p P as @c key_arity. Posts are named by key index, not by container count:
181 * @c Key<0> is the topmost symbol, and @c PostAt<key_arity> is the innermost post.
182 *
183 * @tparam P The post reached from one source state, itself parameterised down to the targets.
184 * @tparam TT What one transition of this relation looks like and how to build one: a traits with a
185 * @c Type and a @c make(). Defaults to the symbol-named triple at arity 1 and to the tuple-keyed
186 * transition above it. @see mata::posts::DefaultTransitionTraits.
187 * @see mata::DeltaLike.
188 */

268 /**

270 *
271 * If we try to access a state post of a @p source which is present in the automaton as an initial/final state,
272 * yet does not have allocated space in the relation, an @c empty_post is returned. Hence, the function has no side

285 * @brief Get mutable (non-constant) reference to the state post of @p source.

450 * @param[in] source Source state to look from.

601 /// @name Keys on the transitions
602 ///
603 /// Which outermost keys are in use. They ask nothing of a key beyond the ordering the posts already
604 /// rely on, so a relation keyed by an interval or a weight has them too; what those keys *stand
605 /// for* is that relation's business. Protected for the same reason: a relation re-exports the
606 /// ones it wants under the names that fit its keys, with a using-declaration or a forwarder.

659 /**
660 * @brief Collect all transitions from @p from to @p to into @p out.
661 *
662 * @param[in] from Source state of the transitions to collect.
663 * @param[in] to Target state of the transitions to collect.
664 * @param[out] out Vector to collect the transitions into.
665 */
666 void collect_transitions_to_(const State from, const State to, std::vector<TransitionType>& out) const {
667 if constexpr (key_arity == 1 && std::same_as<Target, State>) {
668 // The arity-1, bare-state case is the only one that can be done without a search.
669 for (const Entry& entry : state_post(from)) {
670 if (entry.targets.contains(to)) { out.push_back(TT::make(from, entry.key(), to)); }
671 }
672 } else {
673 state_post(from).for_each_move([&](const auto&... move) {
674 if (state_of(std::get<sizeof...(move) - 1>(std::forward_as_tuple(move...))) == to) {
675 out.push_back(TT::make(from, move...));
676 }
677 });
678 }
679 }

681 /// The largest @c State is the "no state" sentinel: states are 0, ..., max - 1.
682 static constexpr State no_state_{std::numeric_limits<State>::max()};
683 /// Throws @c std::overflow_error unless @p num_of_states states can be numbered by @c State.
684 static void check_num_of_states_(const size_t num_of_states) {
685 if constexpr (sizeof(State) < sizeof(size_t)) {
686 if (num_of_states > static_cast<size_t>(no_state_)) [[unlikely]] {
687 throw std::overflow_error("mata: the number of states does not fit below the no-state sentinel");
688 }
689 }
690 }
691 /// The number of states needed to contain @p state, throwing @c std::overflow_error for the sentinel.
692 static size_t num_of_states_for_(const State state) {
693 if (state == no_state_) [[unlikely]] {
694 throw std::overflow_error("mata: the no-state sentinel cannot become a state");
695 }
696 return static_cast<size_t>(state) + 1;
697 }

699 /// @brief Mark the cached transition count dirty, so that the next query recounts it.
700 void mark_count_dirty() { count_dirty_.store(true, std::memory_order_relaxed); }
701
702 /**
703 * @brief Record new transitions. A no-op while the count is dirty.
704 *
705 * @param added Number of transitions added to the relation.
706 */
707 void note_added(const size_t added) {
708 if (!count_dirty_.load(std::memory_order_relaxed)) {
709 cached_transition_count_.store(
710 cached_transition_count_.load(std::memory_order_relaxed) + added, std::memory_order_relaxed
711 );
712 }
713 }
714
715 /**
716 * @brief Record removed transitions. A no-op while the count is dirty.
717 *
718 * @param removed Number of transitions removed from the relation.
719 */
720 void note_removed(const size_t removed) {
721 if (!count_dirty_.load(std::memory_order_relaxed)) {
722 const size_t count{cached_transition_count_.load(std::memory_order_relaxed)};
723 MATA_ASSERT(removed <= count);
724 cached_transition_count_.store(count - removed, std::memory_order_relaxed);
725 }
726 }

728 /**
729 * @brief Set the cached transition count to @p count and mark it clean.
730 *
731 * @param count New cached transition count.
732 */
733 void set_count(const size_t count) {
734 cached_transition_count_.store(count, std::memory_order_relaxed);
735 count_dirty_.store(false, std::memory_order_release);
736 }
737
738 /**
739 * @brief Copy the cached transition count and dirty flag from @p other.
740 *
741 * @param other Relation to copy the cached transition count and dirty flag from.
742 */
743 void copy_count_from(const DeltaBase& other) {
744 // The flag first: a clean flag (acquire) publishes the count stored before it.
745 const bool dirty{other.count_dirty_.load(std::memory_order_acquire)};
746 cached_transition_count_.store(
747 other.cached_transition_count_.load(std::memory_order_relaxed), std::memory_order_relaxed
748 );
749 count_dirty_.store(dirty, std::memory_order_relaxed);
750 }

752 std::vector<PostType> state_posts_;
753 /// Exact number of transitions whenever @c count_dirty_ is false. Both are atomic because concurrent const
754 /// queries may recount; mutators run exclusively, so they use relaxed order.
755 mutable std::atomic<size_t> cached_transition_count_{0};
756 /// Set by edits whose effect on the count is not known, such as through @c mutable_state_post().
757 mutable std::atomic<bool> count_dirty_{false};
758 /// The builder installs finished posts directly; @see DeltaBuilder.
759 template <typename D> friend class DeltaBuilder;
760}; // class mata::posts::DeltaBase.

762/**
763 * @brief Builds a relation in its final form from transitions given in order: sources, then keys of a source, then
764 * targets of a key, each strictly increasing.
765 *
766 * Every post entry is created once with exact capacity, instead of being grown target by target. Arity 1 only: at a
767 * deeper key chain, the "targets" of a key are themselves keyed posts, which this builder does not shape.
768 *
769 * @tparam D The concrete relation to build. Needs no extra members on @p D: the finished posts are installed through
770 * the @c DeltaBase it derives from.
771 */
772template <typename D> class DeltaBuilder {
773 public:
774 using Base = DeltaBase<typename D::PostType, typename D::TransitionTraits>;
775 using State = typename Base::State;
776 using Key = typename Base::template Key<0>;
777 static_assert(D::key_arity == 1, "the builder shapes one level of keys only");
778
779 DeltaBuilder() = default;
780 explicit DeltaBuilder(const size_t num_of_states_hint) { state_posts_.reserve(num_of_states_hint); }
781
782 void begin_state(State source) {
783 MATA_ASSERT(!in_state_ && "begin_state() called before finish_state()");
784 MATA_ASSERT((state_posts_.empty() || source >= state_posts_.size()) && "sources must be strictly increasing");
785 // Throws for the "no state" sentinel before the builder changes, instead of wrapping `source + 1` to zero.
786 state_posts_.resize(Base::num_of_states_for_(source));
787 source_ = source;
788 in_state_ = true;
789 has_key_ = false;
790 }
791
792 void reserve_keys(const size_t num_of_keys) {
793 MATA_ASSERT(in_state_ && "reserve_keys() called outside of a state");
794 state_posts_[source_].reserve(num_of_keys);
795 }
796
797 void begin_key(const Key key) {
798 MATA_ASSERT(in_state_ && "begin_key() called outside of a state");
799 MATA_ASSERT(!in_key_ && "begin_key() called before finish_key()");
800 MATA_ASSERT((!has_key_ || key > key_) && "keys must be strictly increasing");
801 key_ = key;
802 targets_.clear();
803 in_key_ = true;
804 }
805
806 void push_sorted_target(const State target) {
807 MATA_ASSERT(in_key_ && "push_sorted_target() called outside of a key");
808 MATA_ASSERT((targets_.empty() || target > targets_.back()) && "targets must be strictly increasing");
809 targets_.push_back(target);
810 }
811
812 void finish_key() {
813 MATA_ASSERT(in_key_ && "finish_key() without begin_key()");
814 if (targets_.empty()) {
815 in_key_ = false;
816 return;
817 }
818 state_posts_[source_].emplace_back(key_, std::span<const State>{targets_});
819 // Committed only once the post entry is stored, so a throwing emplace_back() leaves the builder as it was.
820 num_of_transitions_ += targets_.size();
821 max_target_ = std::max(max_target_, targets_.back()); // Sorted, so the last target is the largest.
822 has_key_ = true;
823 in_key_ = false;
824 }
825
826 void finish_state() {
827 MATA_ASSERT(in_state_ && "finish_state() without begin_state()");
828 MATA_ASSERT(!in_key_ && "finish_state() called before finish_key()");
829 in_state_ = false;
830 }
831
832 /// Hands over the finished relation and empties the builder, which can then build another one.
833 D finish() {
834 MATA_ASSERT(!in_state_ && !in_key_ && "finish() called inside an unfinished state or key");
835 // A target state needs its own (possibly empty) post, like every other state of the relation.
836 if (!state_posts_.empty() && max_target_ >= state_posts_.size()) {
837 state_posts_.resize(Base::num_of_states_for_(max_target_));
838 }
839 D delta{};
840 Base& base{delta};
841 base.state_posts_ = std::move(state_posts_);
842 // Every stored target was counted, so the relation starts with an exact count.
843 base.set_count(num_of_transitions_);
844 // A moved-from vector is only valid, not empty, so it is cleared before the next build.
845 state_posts_.clear();
846 num_of_transitions_ = 0;
847 max_target_ = 0;
848 return delta;
849 }
850
851 private:
852 std::vector<typename Base::PostType> state_posts_{};
853 std::vector<State> targets_{};
854 Key key_{};
855 State source_{};
856 State max_target_{};
857 /// Targets of the stored post entries, which finish() hands over as the exact count.
858 size_t num_of_transitions_{0};
859 bool in_state_{false};
860 bool in_key_{false};
861 bool has_key_{false}; ///< Has the current state emitted a post entry already?
862}; // class DeltaBuilder.

868/**
869 * @brief The contract every concrete relation owes.
870 *
871 * Each check is its own @c static_assert rather than one concept, so that a failure names the check
872 * that failed. A module asks for all of them with a single line,
873 * `static_assert(posts::relation_contract_holds<Delta>());`, which may be repeated in every
874 * translation unit that includes the module's header.
875 */
876template <typename D> constexpr bool relation_contract_holds() {
877 using P = typename D::PostType;
878 static_assert(PostLike<P>, "the relation's PostType must be one post of the relation.");
879 static_assert(PostEntryLike<typename D::Entry>, "the relation's Entry must be its PostType's entry type.");
880 static_assert(TargetSetLike<typename D::TargetSet>, "the innermost post must be a set of targets.");
881 static_assert(DeltaLike<D>, "the relation must satisfy the contract mata::AutomatonBase is written against.");
882
883 using Base = DeltaBase<P, typename D::TransitionTraits>;
884 static_assert(
885 sizeof(D) == sizeof(Base) && alignof(D) == alignof(Base),
886 "a relation class must add no members to DeltaBase. It exists only to shorten a name in diagnostics."
887 );
888 return true;
889}
