38/**
39 * @brief The keyed writes and lookup of a relation: one member each, at every arity.
40 *
41 * The keys are a pack of this class, not of the member, so every parameter has a declared type and
42 * `add(0, 1, 2, 3)` converts its `int`s correctly at the call. Arity 1 is specialised below.
43 *
44 * @tparam D The relation deriving from this. A friend of it, for its posts.
45 * @tparam P The relation's post chain.
46 */
47template <typename D, typename P, typename = std::make_index_sequence<arity_of<P>>> class KeyedAccess;
48
49template <typename D, typename P, size_t... KeyIdx> class KeyedAccess<D, P, std::index_sequence<KeyIdx...>> {
50 using Target = target_of<P>;
51 using State = typename TargetTraits<Target>::State;
52 using TargetArg = utils::ArgOf<Target>;
53 using TargetSet = typename PostAt<P, sizeof...(KeyIdx)>::type;
54
55 D& self() { return static_cast<D&>(*this); }
56 const D& self() const { return static_cast<const D&>(*this); }
57
58 public:
59 /**
60 * @brief Add the transition from @p source under @p keys to @p target.
61 *
62 * @param source Source state of the transition.
63 * @param keys Keys of the transition.
64 * @param target Target of the transition.
65 */
66 void add(const State source, const KeyOf<P, KeyIdx>... keys, const TargetArg target) {
67 self().resize_for_states(source, D::state_of(target));
68 // Through the post directly: mutable_state_post() would mark the count dirty.
69 if (insert_target(self().state_posts_[source], target, keys...)) { self().note_added(1); }
70 }
71
72 /**
73 * @brief Add the transition from @p source under @p keys to every one of @p targets.
74 *
75 * @param source Source state of the transition.
76 * @param keys Keys of the transition.
77 * @param targets Set of targets of the transition.
78 */
79 void add(const State source, const KeyOf<P, KeyIdx>... keys, const TargetSet& targets) {
80 const TargetSet snapshot{targets};
81 for (const Target& target : snapshot) {
82 self().resize_for_states(source, D::state_of(target));
83 if (insert_target(self().state_posts_[source], target, keys...)) { self().note_added(1); }
84 }
85 }
86
87 /**
88 * @brief Remove the transition from @p source under @p keys to @p target.
89 *
90 * @param source Source state of the transition.
91 * @param keys Keys of the transition.
92 * @param target Target of the transition.
93 * @throws Throws std::invalid_argument if @p source has no transition under @p keys. A @p source beyond the
94 * relation, or a @p target missing under existing @p keys, is ignored.
95 */
96 void remove(const State source, const KeyOf<P, KeyIdx>... keys, const TargetArg target) {
97 if (source >= self().state_posts_.size()) { return; }
98 // Looked up first: erase_target() reports whether the post emptied, not whether it erased.
99 const bool present{has_target_at(self().state_posts_[source], target, keys...)};
100 erase_target(self().state_posts_[source], target, keys...);
101 if (present) { self().note_removed(1); }
102 }
103
104 /**
105 * @brief Check whether the relation contains the transition from @p source under @p keys to @p target.
106 *
107 * @param source Source state of the transition.
108 * @param keys Keys of the transition.
109 * @param target Target of the transition.
110 * @return True if the relation contains the transition, false otherwise.
111 */
112 bool contains(const State source, const KeyOf<P, KeyIdx>... keys, const TargetArg target) const {
113 return source < self().state_posts_.size() && has_target_at(self().state_posts_[source], target, keys...);
114 }
115};

175/**
176 * @brief The generic transition relation: a vector of posts indexed by source state.
177 *
178 * A vector of posts indexed by source state, over a chain of posts ending in a set of targets.
179 * The depth is deduced from @p P as @c key_arity. Posts are named by key index, not by container count:
180 * @c Key<0> is the topmost symbol, and @c PostAt<key_arity> is the innermost post.
181 *
182 * @tparam P The post reached from one source state, itself parameterised down to the targets.
183 * @tparam TT What one transition of this relation looks like and how to build one: a traits with a
184 * @c Type and a @c make(). Defaults to the symbol-named triple at arity 1 and to the tuple-keyed
185 * transition above it. @see mata::posts::DefaultTransitionTraits.
186 * @see mata::DeltaLike.
187 */
188template <typename P, typename TT = DefaultTransitionTraits<P>>
189class DeltaBase : public KeyedAccess<DeltaBase<P, TT>, P> {
190 public:
191 /// @name Post protocol
192 ///@{
193 using PostType = P; ///< The post reached from one source state.
194 using Target = typename PostType::Target; ///< What visiting the successors yields.
195 using TargetArg = utils::ArgOf<Target>; ///< How the keyed members take a target.
196 using State = typename TargetTraits<Target>::State; ///< What indexes this relation.
197 static constexpr size_t key_arity{PostType::key_arity};
198
199 /// The post at index @p I, counted from the outermost one.
200 template <size_t I> using PostAt = typename posts::PostAt<PostType, I>::type;
201 /// The key at index @p I, counted from the outermost one.
202 template <size_t I> using Key = typename PostAt<I>::Key;
203 /// The reserved keys at index @p I, counted from the outermost one.
204 template <size_t I> using Reserved = typename PostAt<I>::Reserved;
205 using TargetSet = PostAt<key_arity>;
206 using KeyedSuccessors = typename PostType::KeyedSuccessors; ///< The result type of the keyed @c get_successors.
207 /// The result type of the unkeyed @c get_successors: whatever the post returns, e.g. a @c StateSet for

265 /**
266 * @brief Get constant reference to the state post of @p source.
267 *
268 * If we try to access a state post of a @p source which is present in the automaton as an initial/final state,
269 * yet does not have allocated space in the relation, an @c empty_post is returned. Hence, the function has no side
270 * effects (no allocation is performed and iterators remain valid).
271 *
272 * @param source Source state of a state post to access.
273 * @return State post of @p source.
274 */
275 const PostType& state_post(const State source) const {

277 return state_posts_[source];

281 /**
282 * @brief Get mutable (non-constant) reference to the state post of @p source.
283 *
284 * The function allows modifying the state post. BEWARE, IT HAS A SIDE EFFECT.
285 * If we try to access a state post of a @p source which is present in the automaton as an initial/final state,
286 * yet does not have allocated space in the relation, a new state post for @p source will be allocated along with
287 * all state posts for all previous states. This in turn may cause that the entire post data structure is

289 * The relation cannot see edits made through the reference, so the next query of the transition count recounts.
290 * The reference may be used for edits only until that query: @c num_of_transitions(), @c empty(), or any
291 * operation that may call them. Obtain the reference again after such a query.
292 * @param source Source state of a state post to access.

444 /**
445 * @brief Checks if @p source have @p target among its successors, under any key.
446 *
447 * @param[in] source Source state to look from.
448 * @param[in] target Target state to look for.
449 * @return True if @p target is among the successors of @p source, false otherwise.
450 */
451 bool is_successor(const State source, const State target) const {
452 if constexpr (std::same_as<Target, State>) {
453 // A bare state is the target itself, so the post can binary-search for it.
454 return state_post(source).has_target(target);
455 } else {
456 // A target with a payload is found by its state, which its order need not lead with.
457 return any_target(state_post(source), [target](const Target& t) { return state_of(t) == target; });
458 }
459 }

604 /// @note Indexed *by* key, so it allocates up to the largest key used. Already unusable when the automaton has
605 /// epsilons.
606 /// @throws std::length_error if a key is too large to index a vector.
607 BoolVector get_used_keys_chv() const
608 requires std::unsigned_integral<typename P::Key>;

610 /**
611 * @brief The greatest key used, or nothing when there are no transitions.
612 *
613 * Over *every* key, epsilons included. That is what the one caller needs: NFT's simulation mints
614 * fresh symbols at `max + 1` and above, which has to clear the epsilons too or the fresh
615 * symbols collide with real transitions.
616 */
617 std::optional<typename P::Key> get_max_key() const
618 requires std::totally_ordered<typename P::Key>;

621 /**
622 * @brief Collect all transitions from @p from to @p to into @p out.
623 *
624 * @param[in] from Source state of the transitions to collect.
625 * @param[in] to Target state of the transitions to collect.
626 * @param[out] out Vector to collect the transitions into.
627 */
628 void collect_transitions_to_(const State from, const State to, std::vector<TransitionType>& out) const {
629 if constexpr (key_arity == 1 && std::same_as<Target, State>) {
630 // The arity-1, bare-state case is the only one that can be done without a search.
631 for (const Entry& entry : state_post(from)) {
632 if (entry.targets.contains(to)) { out.push_back(TT::make(from, entry.key(), to)); }
633 }
634 } else {
635 state_post(from).for_each_move([&](const auto&... move) {
636 if (state_of(std::get<sizeof...(move) - 1>(std::forward_as_tuple(move...))) == to) {
637 out.push_back(TT::make(from, move...));
638 }
639 });
640 }
641 }
642
643 /// The largest @c State is the "no state" sentinel: states are 0, ..., max - 1.
644 static constexpr State no_state_{std::numeric_limits<State>::max()};
645 /// Throws @c std::overflow_error unless @p num_of_states states can be numbered by @c State.
646 static void check_num_of_states_(const size_t num_of_states) {
647 if constexpr (sizeof(State) < sizeof(size_t)) {
648 if (num_of_states > static_cast<size_t>(no_state_)) [[unlikely]] {
649 throw std::overflow_error("mata: the number of states does not fit below the no-state sentinel");
650 }
651 }
652 }
653 /// The number of states needed to contain @p state, throwing @c std::overflow_error for the sentinel.
654 static size_t num_of_states_for_(const State state) {
655 if (state == no_state_) [[unlikely]] {
656 throw std::overflow_error("mata: the no-state sentinel cannot become a state");
657 }
658 return static_cast<size_t>(state) + 1;
659 }

661 /// @brief Mark the cached transition count dirty, so that the next query recounts it.
662 void mark_count_dirty() { count_dirty_.store(true, std::memory_order_relaxed); }
663
664 /**
665 * @brief Record new transitions. A no-op while the count is dirty.
666 *
667 * @param added Number of transitions added to the relation.
668 */
669 void note_added(const size_t added) {
670 if (!count_dirty_.load(std::memory_order_relaxed)) {
671 cached_transition_count_.store(
672 cached_transition_count_.load(std::memory_order_relaxed) + added, std::memory_order_relaxed
673 );
674 }
675 }
676
677 /**
678 * @brief Record removed transitions. A no-op while the count is dirty.
679 *
680 * @param removed Number of transitions removed from the relation.
681 */
682 void note_removed(const size_t removed) {
683 if (!count_dirty_.load(std::memory_order_relaxed)) {
684 const size_t count{cached_transition_count_.load(std::memory_order_relaxed)};
685 MATA_ASSERT(removed <= count);
686 cached_transition_count_.store(count - removed, std::memory_order_relaxed);
687 }
688 }
689
690 /**
691 * @brief Set the cached transition count to @p count and mark it clean.
692 *
693 * @param count New cached transition count.
694 */
695 void set_count(const size_t count) {
696 cached_transition_count_.store(count, std::memory_order_relaxed);
697 count_dirty_.store(false, std::memory_order_release);
698 }

700 /**
701 * @brief Copy the cached transition count and dirty flag from @p other.
702 *
703 * @param other Relation to copy the cached transition count and dirty flag from.
704 */
705 void copy_count_from(const DeltaBase& other) {
706 // The flag first: a clean flag (acquire) publishes the count stored before it.
707 const bool dirty{other.count_dirty_.load(std::memory_order_acquire)};
708 cached_transition_count_.store(
709 other.cached_transition_count_.load(std::memory_order_relaxed), std::memory_order_relaxed
710 );
711 count_dirty_.store(dirty, std::memory_order_relaxed);
712 }

718 /// Set by edits whose effect on the count is not known, such as through @c mutable_state_post().
719 mutable std::atomic<bool> count_dirty_{false};

726/**
727 * @brief The contract every concrete relation owes.
728 *
729 * Each check is its own @c static_assert rather than one concept, so that a failure names the check
730 * that failed. A module asks for all of them with a single line,
731 * `static_assert(posts::relation_contract_holds<Delta>());`, which may be repeated in every
732 * translation unit that includes the module's header.
733 */
734template <typename D> constexpr bool relation_contract_holds() {
735 using P = typename D::PostType;
736 static_assert(PostLike<P>, "the relation's PostType must be one post of the relation.");
737 static_assert(PostEntryLike<typename D::Entry>, "the relation's Entry must be its PostType's entry type.");
738 static_assert(TargetSetLike<typename D::TargetSet>, "the innermost post must be a set of targets.");
739 static_assert(DeltaLike<D>, "the relation must satisfy the contract mata::AutomatonBase is written against.");
740
741 using Base = DeltaBase<P, typename D::TransitionTraits>;
742 static_assert(
743 sizeof(D) == sizeof(Base) && alignof(D) == alignof(Base),
744 "a relation class must add no members to DeltaBase. It exists only to shorten a name in diagnostics."
745 );
746 return true;
747}
