22// Returns the default largest ordinary key for an integral key type, given the smallest epsilon key.
23template <typename K> consteval K default_max_ordinary(const K min_epsilon) {
24 static_assert(
25 std::integral<K>, "ReservedKeys<K>: only an integral key has a default reserved tail (min_epsilon = max, "
26 "max_ordinary = min_epsilon - 1). For any other key spell all three arguments: "
27 "ReservedKeys<K, MinEpsilon, MaxOrdinary>."
28 );
29 if constexpr (std::integral<K>) {
30 return min_epsilon - 1;
31 } else {
32 return K{};
33 } // K{} is never used (just to satisfy the compiler)
34}
35
36// Returns whether @p min_epsilon is the greatest key.
37// Only an integral key is known to have one: for any other, std::numeric_limits<K>::max() is just K{}.
38template <typename K> consteval bool default_epsilon_is_greatest(const K min_epsilon) {
39 if constexpr (std::integral<K>) {
40 return min_epsilon == std::numeric_limits<K>::max();
41 } else {
42 return false;
43 }
44}

47/**
48 * @brief A traits specifying which state a target denotes.
49 *
50 * A target is whatever the innermost post stores: a state, or a state with a payload.
51 * The structural algorithms need two things from it: @c state_of(target) to know the state it denotes,
52 * and @c with_state(target, state) to rebuild a target with a new state when reverting, renumbering or
53 * trimming. Both are the identity for a bare state.
54 *
55 * The default covers only a bare state, an integral type. Any other target has to specialise this,
56 * and one that does not is stopped here, rather than being taken for a state and failing deep inside
57 * an algorithm that indexes by it.
58 *
59 * The order of the targets has to follow their states: renaming the states by a strictly increasing map, as trimming
60 * does, must not reorder the targets. Any order comparing the state by @c <, wherever it comes in the comparison,
61 * does so, which includes every defaulted @c <=>. An order that is not monotone in the state, such as even states
62 * before odd ones, does not, and trimming would leave its target sets unsorted.
63 *
64 * An example of specialization for a target with MyPayload is:
65 * ```cpp
66 * template <> struct mata::TargetTraits<MyPayload> {
67 * using State = mata::State;
68 * static State state_of(const MyPayload& t) { return t.state; }
69 * static MyPayload with_state(const MyPayload& t, State s) { return {s, t.payload}; }
70 * };
71 * ```
72 */
73template <typename T> struct TargetTraits {
74 static_assert(
75 std::integral<T>,
76 "TargetTraits<T>: a target that is not a bare state has to say which state it denotes. "
77 "Specialise mata::TargetTraits<T> with State, state_of() and with_state()."
78 );
79 using State = T; ///< The state a target denotes.
80
81 // Returns the state a target denotes.
82 static constexpr State state_of(const T& target) { return target; }
83 // Returns a target with a new state name.
84 static constexpr T with_state(const T& /* target */, const State state) { return state; }
85};

87/**
88 * @brief The structure defines where the ordinary keys stop and the reserved tail begins.
89 *
90 * The key order splits into three regions:
91 * - **ordinary**, up to and including @c max_ordinary
92 * - **reserved but not epsilon**, above @c max_ordinary and below @c min_epsilon
93 * - **epsilon**, from @c min_epsilon upwards
94 *
95 * The middle region is empty by default, since @c MaxOrdinary defaults to one below @p MinEpsilon.
96 * Widening the reserved tail opens it, which is how a key can be reserved.
97 *
98 * @tparam K The key type.
99 * @tparam MinEpsilon The smallest epsilon key: every key at or above it is an epsilon. Defaults to
100 * the largest value. A non-integral key has no default and must be spelled explicitly.
101 * @tparam MaxOrdinary The largest key that is not reserved. Defaults to one below @p MinEpsilon.
102 * Can be defined explicitly to a smaller value to open the reserved-but-not-epsilon region.
103 * A non-integral key has no default and must be spelled explicitly.
104 * @tparam EpsilonIsGreatest Whether no key sorts above @p MinEpsilon. Defaults to true only for an integral key whose
105 * @p MinEpsilon is its largest value.
106 */
107template <
108 typename K,
109 K MinEpsilon = std::numeric_limits<K>::max(),
110 K MaxOrdinary = internal::default_max_ordinary<K>(MinEpsilon),
111 bool EpsilonIsGreatest = internal::default_epsilon_is_greatest<K>(MinEpsilon)>
112struct ReservedKeys {
113 using Key = K;
114 static constexpr K min_epsilon{MinEpsilon}; ///< The smallest epsilon key.
115 static constexpr K max_ordinary{MaxOrdinary}; ///< The largest key that is not reserved.
116
117 // When true, an epsilon will always be at the end of any sorted container (simplifying the search for it).
118 static constexpr bool epsilon_is_greatest{EpsilonIsGreatest};
119
120 static_assert(MaxOrdinary < MinEpsilon, "The ordinary keys must stop below the epsilons.");
121 static_assert(
122 !(std::integral<K> && EpsilonIsGreatest) || MinEpsilon == std::numeric_limits<K>::max(),
123 "ReservedKeys<K, ..., EpsilonIsGreatest>: an integral key cannot claim its epsilon is the "
124 "greatest key unless min_epsilon is that key. Claiming it wrongly makes epsilon_symbol_posts() "
125 "break the algorithms that rely on it."
126 );
127};

130/// @brief A traits specifying the key arity and target type of a post, and the aliases reading them.
131///
132/// A type is classified: a post declares both @c key_arity and @c Target, and a target set is a
133/// container (with a @c value_type) declaring neither. Anything else has no traits, so a post that has lost one of
134/// the two, e.g. to a base that failed, is reported here as an incomplete type rather than taken for a target set.
135///@{
136template <typename X> struct PostTraits;
137/// @cond
138template <typename X>
139 requires requires {
140 X::key_arity;
141 typename X::Target;
142 }
143struct PostTraits<X> {
144 static constexpr size_t key_arity{X::key_arity};
145 using Target = typename X::Target;
146};
147template <typename X>
148 requires(
149 !requires { X::key_arity; } && !requires { typename X::Target; } && requires { typename X::value_type; }
150 )
151struct PostTraits<X> {
152 static constexpr size_t key_arity{0};
153 using Target = typename X::value_type;
154};
155/// @endcond
156template <typename X> inline constexpr size_t arity_of = PostTraits<X>::key_arity;
157template <typename X> using target_of = typename PostTraits<X>::Target;
158///@}
159
160/// @brief The post at depth @p I in a chain of posts, outermost first.
161///@{
162template <typename P, size_t I> struct PostAt {
163 static_assert(I <= arity_of<P>, "no post that deep in this chain");
164 using type = typename PostAt<typename P::Nested, I - 1>::type;
165};
166template <typename P> struct PostAt<P, 0> {
167 using type = P;
168};
169///@}
