21/**
22 * @brief Where an epsilon closure is applied around a transition step.
23 *
24 * The values are bit flags, so @c BeforeAndAfter must be tested with @c closes_before() and @c closes_after()
25 * rather than compared for equality. For @c 0 -ε-> 1 -a-> 2 -ε-> 3 and the step over @c a from @c {0}:
26 *
27 * | Option | Applied | Result |
28 * |---|---|---|
29 * | @c None | post(S) | @c {} |
30 * | @c Before | post(closure(S)) | @c {2} |
31 * | @c After | closure(post(S)) | @c {} |
32 * | @c BeforeAndAfter | closure(post(closure(S))) | @c {2,3} |
33 */
34enum class EpsilonClosureOpt : std::uint8_t {
35 None = 1 << 0, ///< No epsilon closure.
36 Before = 1 << 1, ///< Epsilon closure before the transition.
37 After = 1 << 2, ///< Epsilon closure after the transition.
38 BeforeAndAfter = Before | After ///< Epsilon closure before and after the transition.
39};
40
41/// Does @p opt ask for an epsilon closure before the transition step?
42constexpr bool closes_before(const EpsilonClosureOpt opt) {
43 return (static_cast<std::uint8_t>(opt) & static_cast<std::uint8_t>(EpsilonClosureOpt::Before)) != 0;
44}
45
46/// Does @p opt ask for an epsilon closure after the transition step?
47constexpr bool closes_after(const EpsilonClosureOpt opt) {
48 return (static_cast<std::uint8_t>(opt) & static_cast<std::uint8_t>(EpsilonClosureOpt::After)) != 0;
49}

52/**
53 * @brief Reusable buffers of @c post() and @c epsilon_closure(), so that a loop of steps allocates once.
54 *
55 * A call writes all of its buffers, so one workspace serves one call at a time. Concurrent calls, even const queries
56 * of the same delta, each need their own; in particular a workspace must never be a function-local @c static.
57 */
58struct PostWorkspace {
59 std::vector<State> sources{}; ///< Epsilon closure of the source states of a step.
60 std::vector<State> gathered{}; ///< Targets of a step, or the closure being built.
61 std::vector<State> worklist{}; ///< Closure states whose epsilon posts are still unread.
62 std::vector<State> visited{}; ///< Open-addressed hash table of the states of the closure being built.
63};
