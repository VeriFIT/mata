43/**
44 * @brief The moves of a post as @c Move instances: every (symbol, target) pair, or a range of them.
45 *
46 * Available only at @c key_arity 1, since a @c Move holds a single key. Above that, use @c for_each_move().
47 *
48 * @pre No symbol post in the range is without targets: for speed, the iterator does not skip one. Mata's own writes
49 * keep a post so, as @c remove() and @c defragment() erase the symbol posts they empty. A symbol post created
50 * empty through @c mutable_state_post() has to get its targets before its moves are iterated.
51 * @c for_each_move() and the successor cursors have no such precondition.
52 * @tparam P The post visited.
53 */
54template <typename P> class PostMoves {
55 static_assert(P::key_arity == 1, "a Move holds one key: iterate a deeper post with for_each_move()");
56
57 using post_iterator = typename P::const_iterator;
58
59 public:
60 using MoveType = typename internal::MoveTypeOf<P>::type;
