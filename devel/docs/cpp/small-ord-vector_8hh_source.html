27/**
28 * @brief A set of keys kept as a sorted, duplicate-free contiguous array, like @c OrdVector, that stores up to
29 * @p InlineCapacity keys inside the object and allocates only for larger sets.
30 *
31 * The object is a pointer-sized buffer plus a 32-bit size word: the buffer holds either the inline keys or a pointer
32 * to a heap block, which keeps its capacity in a header in front of the keys. A set lives inline until it
33 * outgrows @p InlineCapacity or a larger capacity is reserved. Like @c std::vector, it then keeps its heap block
34 * when it shrinks, while copies are made in their final form: inline when they fit, otherwise with exact capacity.
35 * On 64-bit targets the object ends with four bytes of tail padding, so a @c [[no_unique_address]] member followed
36 * by a 32-bit field packs both into 16 bytes.
37 *
38 * Iterators are plain pointers; any operation that grows the set may invalidate them.
39 */
40// The class is the raw storage itself: the inline keys share a union with the heap block, and both are walked by
41// pointer.
42// NOLINTBEGIN(cppcoreguidelines-pro-type-union-access,
43// cppcoreguidelines-avoid-c-arrays, modernize-avoid-c-arrays, cppcoreguidelines-pro-bounds-array-to-pointer-decay)

153 /**
154 * @brief Insert @p key, keeping the set sorted.
155 * @return Iterator to @p key in the set and whether it was inserted.
156 */
157 std::pair<iterator, bool> insert(const Key key) {
158 MATA_ASSERT(is_sorted_unique());
159 const size_type n{size()};
160 if (n == 0 || data()[n - 1] < key) {
161 push_back(key);
162 return {end() - 1, true};
163 }
164 // Returned as the mutable 'iterator' of the pair below, which the check does not follow.
165 // NOLINTNEXTLINE(misc-const-correctness)
166 Key* pos{std::lower_bound(data(), data() + n, key)};
167 if (*pos == key) { return {pos, false}; }
168 const auto index{static_cast<size_type>(pos - data())};
169 if (n == capacity()) { reallocate(grown_capacity(n + 1)); }
170 Key* const keys{data()};
171 std::memmove(keys + index + 1, keys + index, (n - index) * sizeof(Key));
172 keys[index] = key;
173 set_size(n + 1);
174 return {keys + index, true};
175 }

177 /**
178 * @brief Insert all of the sorted, duplicate-free @p keys.
179 * @return True if any key was inserted.
180 */
181 bool insert(const std::span<const Key> keys) {
182 MATA_ASSERT(is_sorted_unique());
183 if (keys.empty() || keys.data() == data()) { return false; }
184 const size_type n{size()};
185 if (n == 0 || data()[n - 1] < keys.front()) {
186 append(keys);
187 return true;
188 }
189 // Count the union, then merge back to front in place: no temporary and at most one reallocation.
190 const Key* lhs{data()};
191 const Key* const lhs_end{lhs + n};
192 const Key* rhs{keys.data()};
193 const Key* const rhs_end{rhs + keys.size()};
194 size_type union_size{n};
195 while (rhs != rhs_end) {
196 lhs = std::lower_bound(lhs, lhs_end, *rhs);
197 if (lhs == lhs_end) {
198 union_size += static_cast<size_type>(rhs_end - rhs);
199 break;
200 }
201 if (*lhs != *rhs) { ++union_size; }
202 ++rhs;
203 }
204 if (union_size == n) { return false; }
205 reserve(union_size);
206 Key* const out{data()};
207 auto i{static_cast<std::ptrdiff_t>(n) - 1};
208 auto j{static_cast<std::ptrdiff_t>(keys.size()) - 1};
209 auto w{static_cast<std::ptrdiff_t>(union_size) - 1};
210 while (j >= 0) {
211 if (i >= 0 && out[i] > keys[static_cast<size_type>(j)]) {
212 out[w--] = out[i--];
213 } else {
214 if (i >= 0 && out[i] == keys[static_cast<size_type>(j)]) { --i; }
215 out[w--] = keys[static_cast<size_type>(j--)];
216 }
217 }
218 set_size(union_size);
219 return true;
220 }

222 /// Append @p key without keeping the order; restore it with @c sort_and_rmdupl().
223 void push_back(const Key key) {
224 const size_type n{size()};

234 /// Sort the keys and drop duplicates, restoring the invariant after unordered @c push_back().
235 void sort_and_rmdupl() {
236 Key* const first{data()};
237 Key* last{first + size()};
238 if (!std::is_sorted(first, last)) { std::sort(first, last); }
239 last = std::unique(first, last);
240 set_size(static_cast<size_type>(last - first));
241 }

243 size_type erase(const Key key) {
244 const const_iterator it{find(key)};
245 if (it == end()) { return 0; }
246 erase(it);
247 return 1;
248 }
249 /// A template so that a literal @c 0 selects @c erase(Key) rather than converting to a null iterator.
250 template <std::convertible_to<const_iterator> Iterator> iterator erase(const Iterator pos) {
251 return erase(pos, pos + 1);
252 }
253 iterator erase(const const_iterator first, const const_iterator last) {
254 Key* const keys{data()};
255 const auto from{static_cast<size_type>(first - keys)};
256 const auto to{static_cast<size_type>(last - keys)};
257 std::memmove(keys + from, keys + to, (size() - to) * sizeof(Key));
258 set_size(size() - (to - from));
259 return keys + from;
260 }
261 template <class Pred> size_type erase_if(Pred&& should_erase) {
262 Key* const first{data()};
263 Key* const last{first + size()};
264 const Key* const new_last{std::remove_if(first, last, std::forward<Pred>(should_erase))};
265 const auto erased{static_cast<size_type>(last - new_last)};
266 set_size(static_cast<size_type>(new_last - first));
267 return erased;
268 }

285 /// Keys reserved at the start of a heap block to keep its capacity.
286 static constexpr size_type header_keys{(sizeof(std::uint32_t) + sizeof(Key) - 1) / sizeof(Key)};
287
288 /// The set lives on the heap when the lowest bit of @c size_ is set; the size sits in the upper bits of the
289 /// same word, so a move is a copy of the buffer union plus this word.
290 union Buffer {
291 Key inline_[InlineCapacity];
292 Key* block_; ///< Heap block: the capacity header, then the keys.
293 };
294 Buffer buffer_{};
295 std::uint32_t size_{0};
296
297 static constexpr std::uint32_t heap_flag{1};
298 bool on_heap() const noexcept { return (size_ & heap_flag) != 0; }
299 void mark_heap() noexcept { size_ |= heap_flag; }
300
301 static Key* allocate(const size_type capacity) {
302 MATA_ASSERT(capacity <= max_size);
303 Key* const block{Allocator{}.allocate(header_keys + capacity)};
304 const auto stored_capacity{static_cast<std::uint32_t>(capacity)};
305 std::memcpy(block, &stored_capacity, sizeof(stored_capacity));
306 return block;
307 }
308 size_type heap_capacity() const noexcept {
309 std::uint32_t capacity;
310 std::memcpy(&capacity, buffer_.block_, sizeof(capacity));
311 return capacity;
312 }
313 void release() noexcept {
314 if (on_heap()) { Allocator{}.deallocate(buffer_.block_, header_keys + heap_capacity()); }
315 }
316 size_type grown_capacity(const size_type needed) const noexcept { return std::max(needed, 2 * capacity()); }
317 void reallocate(const size_type capacity) {
318 Key* const block{allocate(capacity)};
319 std::memcpy(block + header_keys, data(), size() * sizeof(Key));
320 release();
321 buffer_.block_ = block;
322 mark_heap();
323 }
324 void steal(SmallOrdVector& other) noexcept {
325 // `Buffer` is an implicit-lifetime type, so copying its object representation is well-defined regardless of
326 // which member is active, and it does not inspect bytes past the active representation.
327 // __builtin_memcpy, not std::memcpy: under _FORTIFY_SOURCE, a set whose tail padding holds an entry's
328 // [[no_unique_address]] key gets no static object size, and the copy becomes a __memcpy_chk call.
329 __builtin_memcpy(&buffer_, &other.buffer_, sizeof(buffer_));
330 size_ = other.size_;
331 other.size_ = 0;
332 // Reset the bytes too: GCC otherwise reads the stale block while destroying a moved-from set. Do not
333 // assign the union: a trivially copyable Key with a const member deletes its assignment.
334 const Buffer empty{};
335 __builtin_memcpy(&other.buffer_, &empty, sizeof(empty));
336 }
337 /// Replace the content with the sorted, duplicate-free @p keys, which must not alias this set.
338 void assign_sorted(const std::span<const Key> keys) {
339 MATA_ASSERT(std::ranges::adjacent_find(keys, std::ranges::greater_equal{}) == keys.end());
340 const size_type n{keys.size()};
341 if (n > capacity()) {
342 Key* const block{allocate(n)};
343 release();
344 buffer_.block_ = block;
345 mark_heap();
346 }
347 if (n != 0) { std::memcpy(data(), keys.data(), n * sizeof(Key)); }
348 set_size(n);
349 }
350 void append(const std::span<const Key> keys) {
351 const size_type n{size()};
352 if (n + keys.size() > capacity()) { reallocate(grown_capacity(n + keys.size())); }
353 std::memcpy(data() + n, keys.data(), keys.size() * sizeof(Key));
354 set_size(n + keys.size());
355 }
356 bool is_sorted_unique() const {
357 return std::adjacent_find(begin(), end(), [](const Key& a, const Key& b) { return !(a < b); }) == end();
358 }
359 void set_size(const size_type size) noexcept {
360 MATA_ASSERT(size <= max_size);
361 size_ = static_cast<std::uint32_t>(size << 1) | (size_ & heap_flag);
362 }
