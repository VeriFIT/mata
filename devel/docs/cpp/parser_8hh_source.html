21/**
22 * An append-only storage of names: the tokens of a parsed section and the formula nodes of an intermediate automaton
23 * keep views into it instead of strings of their own.
24 * The bytes handed out are never moved, so a view stays valid for as long as the storage lives. Copies of
25 * @c parser::ParsedSection and of @c IntermediateAut share the storage, which keeps the views of the copies valid.
26 */
27class NameStore {
28 public:
29 /**
30 * @brief Copies @p name into the storage. Threads may store concurrently; each view points to its own copy.
31 * @return A view of the copy, valid for as long as the storage lives.
32 */
33 std::string_view store(const std::string_view name) {
34 const std::scoped_lock lock{store_mutex_};
35 if (chunks_.empty() || chunks_.back().capacity - used_ < name.size()) { add_chunk(name.size()); }
36 char* const stored{chunks_.back().data.get() + used_};
37 std::memcpy(stored, name.data(), name.size());
38 used_ += name.size();
39 return {stored, name.size()};
40 }
41
42 /**
43 * Whether @p name is a view into the storage.
44 * @param hint Where to look first, updated to where @p name was found: names looked up in the order they were
45 * stored are mostly found right away. Start with 0.
46 */
47 bool contains(const std::string_view name, std::size_t& hint) const {
48 return (hint < ranges_.size() && ranges_[hint].spans(name)) || find(name, hint);
49 }
50
51 private:
52 /// One block of the storage; the bytes of the blocks handed out earlier stay where they are.
53 struct Chunk {
54 std::unique_ptr<char[]> data{};
55 std::size_t capacity{};
56 };
57 /// The addresses a chunk spans.
58 struct Range {
59 std::uintptr_t begin{};
60 std::uintptr_t end{};
61
62 bool spans(const std::string_view name) const {
63 const auto address{reinterpret_cast<std::uintptr_t>(name.data())};
64 return begin <= address && address + name.size() <= end;
65 }
66 };
67
68 /// @c contains() without the look at @p hint.
69 bool find(std::string_view name, std::size_t& hint) const;
70 /// Adds a chunk with room for at least @p size bytes.
71 void add_chunk(std::size_t size);
72
73 static constexpr std::size_t chunk_capacity_ = 16 * 1'024;
74
75 std::vector<Chunk> chunks_{};
76 /// The ranges of the chunks, by address.
77 std::vector<Range> ranges_{};
78 /// The number of bytes used in the last chunk.
79 std::size_t used_{};
80 /// Serializes the writes to @c chunks_, @c ranges_, and @c used_.
81 std::mutex store_mutex_{};
82};

95/**
96 * One token of a body line, with the flag telling whether the reader read it in quotes.
97 * @c create_node() lets a quoted token be an operand even when it starts with @c & @c | or @c !, which the
98 * unquoted reader would take for an operator of the transition formula.
99 * The token is a view: @c parse_mf() points it into the name storage of its section, other code has to keep the
100 * bytes alive for as long as the token is used.
101 */
102struct BodyToken {
103 std::string_view token{};
104 bool quoted{false};
105
106 BodyToken() = default;
107 BodyToken(const std::string_view token, const bool quoted) : token{token}, quoted{quoted} {}
108 /// An unquoted token, so that a body line reads as a list of plain names in tests and call sites.
109 BodyToken(const char* token) : token{token} {}
110 /// A literal, which the overloads for strings and views could both take.
111 BodyToken(const char* const token, const bool quoted) : token{token}, quoted{quoted} {}
112 // A view of a temporary string would dangle.
113 BodyToken(std::string&&, bool) = delete;
114 bool operator==(const BodyToken& rhs) const = default;
115};
116
117inline std::ostream& operator<<(std::ostream& os, const BodyToken& token) { return os << token.token; }
