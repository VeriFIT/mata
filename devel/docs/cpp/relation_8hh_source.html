33/**
34 * @brief A transition (source, symbol, target).
35 *
36 * Built from all three fields or none, so that a forgotten target does not become state 0.
37 */
38struct Transition {
39 State source{}; ///< Source state.
40 Symbol symbol{}; ///< Transition symbol.
41 State target{}; ///< Target state.
42
43 Transition() = default;
44 Transition(const State source, const Symbol symbol, const State target)
45 : source{source},
46 symbol{symbol},
47 target{target} {}
48
49 auto operator<=>(const Transition&) const = default;
50
51 /**
52 * @brief Get the parts of @p t as a tuple.
53 *
54 * @param t Transition to get the parts of.
55 * @return A tuple (source, symbol, target) of @p t.
56 */
57 static std::tuple<State, Symbol, State> parts(const Transition& t) noexcept {
58 return {t.source, t.symbol, t.target};
59 }
60
61 std::string to_string() const;
62};

324 /**
325 * @brief Remove the transition from @p source over @p symbol to @p target.
326 *
327 * @param source Source state of the transition.
328 * @param symbol Symbol of the transition.
329 * @param target Target of the transition.
330 * @return Whether the transition was in the relation. Unlike the generic relation's @c remove(), a missing
331 * transition does not throw.

333 bool remove(const mata::State source, const mata::Symbol symbol, const mata::State target) {
334 if (source >= state_posts_.size()) { return false; }
335 StatePost& state_post{state_posts_[source]};
336 if (state_post.empty() || state_post.back().symbol < symbol) { return false; }
337 const auto symbol_post_it{state_post.find(symbol)};
338 if (symbol_post_it == state_post.end() || !symbol_post_it->erase(target)) { return false; }
339 note_removed(1);
340 // Erase through the iterator already found instead of searching for the symbol post again.
341 if (symbol_post_it->empty()) { state_post.erase(symbol_post_it); }
342 return true;
343 }

526/**
527 * @brief Builds a @c Delta in its final form from transitions given in order: sources, then symbols of a source, then
528 * targets of a symbol, each strictly increasing.
529 *
530 * Every symbol post is created once with exact capacity, instead of being grown target by target.
531 */
532class DeltaBuilder {
533 public:
534 DeltaBuilder() = default;
535 explicit DeltaBuilder(const size_t num_of_states_hint) { state_posts_.reserve(num_of_states_hint); }
536
537 void begin_state(State source) {
538 MATA_ASSERT(!in_state_ && "begin_state() called before finish_state()");
539 MATA_ASSERT((state_posts_.empty() || source >= state_posts_.size()) && "sources must be strictly increasing");
540 // Throws for the "no state" sentinel before the builder changes, instead of wrapping `source + 1` to zero.
541 state_posts_.resize(num_of_states_for(source));
542 source_ = source;
543 in_state_ = true;
544 has_symbol_ = false;
545 }
546
547 void reserve_symbols(const size_t num_of_symbols) {
548 MATA_ASSERT(in_state_ && "reserve_symbols() called outside of a state");
549 state_posts_[source_].reserve(num_of_symbols);
550 }
551
552 void begin_symbol(const Symbol symbol) {
553 MATA_ASSERT(in_state_ && "begin_symbol() called outside of a state");
554 MATA_ASSERT(!in_symbol_ && "begin_symbol() called before finish_symbol()");
555 MATA_ASSERT((!has_symbol_ || symbol > symbol_) && "symbols must be strictly increasing");
556 symbol_ = symbol;
557 targets_.clear();
558 in_symbol_ = true;
559 }
560
561 void push_sorted_target(const State target) {
562 MATA_ASSERT(in_symbol_ && "push_sorted_target() called outside of a symbol");
563 MATA_ASSERT((targets_.empty() || target > targets_.back()) && "targets must be strictly increasing");
564 targets_.push_back(target);
565 }
566
567 void finish_symbol() {
568 MATA_ASSERT(in_symbol_ && "finish_symbol() without begin_symbol()");
569 if (targets_.empty()) {
570 in_symbol_ = false;
571 return;
572 }
573 state_posts_[source_].emplace_back(symbol_, std::span<const State>{targets_});
574 // Committed only once the symbol post is stored, so a throwing emplace_back() leaves the builder as it was.
575 num_of_transitions_ += targets_.size();
576 max_target_ = std::max(max_target_, targets_.back()); // Sorted, so the last target is the largest.
577 has_symbol_ = true;
578 in_symbol_ = false;
579 }
580
581 void finish_state() {
582 MATA_ASSERT(in_state_ && "finish_state() without begin_state()");
583 MATA_ASSERT(!in_symbol_ && "finish_state() called before finish_symbol()");
584 in_state_ = false;
585 }
586
587 /// Hands over the finished relation and empties the builder, which can then build another one.
588 Delta finish() {
589 MATA_ASSERT(!in_state_ && !in_symbol_ && "finish() called inside an unfinished state or symbol");
590 // A target state needs its own (possibly empty) post, like every other state of the relation.
591 if (!state_posts_.empty() && max_target_ >= state_posts_.size()) {
592 state_posts_.resize(num_of_states_for(max_target_));
593 }
594 Delta delta{};
595 delta.state_posts_ = std::move(state_posts_);
596 // Every stored target was counted, so the relation starts with an exact count.
597 delta.set_count(num_of_transitions_);
598 // A moved-from vector is only valid, not empty, so it is cleared before the next build.
599 state_posts_.clear();
600 num_of_transitions_ = 0;
601 max_target_ = 0;
602 return delta;
603 }
604
605 private:
606 std::vector<StatePost> state_posts_{};
607 std::vector<State> targets_{};
608 Symbol symbol_{0};
609 State source_{0};
610 State max_target_{0};
611 /// Targets of the stored symbol posts, which finish() hands over as the exact count.
612 size_t num_of_transitions_{0};
613 bool in_state_{false};
614 bool in_symbol_{false};
615 bool has_symbol_{false}; ///< Has the current state emitted a symbol post already?
616}; // class DeltaBuilder.
