33/**
34 * @brief A transition (source, symbol, target).
35 *
36 * Built from all three fields or none, so that a forgotten target does not become state 0.
37 */
38struct Transition {
39 State source{}; ///< Source state.
40 Symbol symbol{}; ///< Transition symbol.
41 State target{}; ///< Target state.
42
43 Transition() = default;
44 Transition(const State source, const Symbol symbol, const State target)
45 : source{source},
46 symbol{symbol},
47 target{target} {}
48
49 auto operator<=>(const Transition&) const = default;
50
51 /**
52 * @brief Get the parts of @p t as a tuple.
53 *
54 * @param t Transition to get the parts of.
55 * @return A tuple (source, symbol, target) of @p t.
56 */
57 static std::tuple<State, Symbol, State> parts(const Transition& t) noexcept {
58 return {t.source, t.symbol, t.target};
59 }
60
61 std::string to_string() const;
62};

64/// Defines how @c Delta builds its transitions.
65struct TransitionTraits {
66 using Type = Transition;
67 static Transition make(const State source, const Symbol symbol, const State target) {
68 return Transition{source, symbol, target};
69 }

308 /**
309 * @brief Remove the transition from @p source over @p symbol to @p target.
310 *
311 * @param source Source state of the transition.
312 * @param symbol Symbol of the transition.
313 * @param target Target of the transition.
314 * @return Whether the transition was in the relation. Unlike the generic relation's @c remove(), a missing
315 * transition does not throw.
316 */
317 bool remove(const mata::State source, const mata::Symbol symbol, const mata::State target) {
318 if (source >= state_posts_.size()) { return false; }
319 StatePost& state_post{state_posts_[source]};
320 if (state_post.empty() || state_post.back().symbol < symbol) { return false; }
321 const auto symbol_post_it{state_post.find(symbol)};
322 if (symbol_post_it == state_post.end() || !symbol_post_it->erase(target)) { return false; }
323 note_removed(1);
324 // Erase through the iterator already found instead of searching for the symbol post again.
325 if (symbol_post_it->empty()) { state_post.erase(symbol_post_it); }
326 return true;
327 }
