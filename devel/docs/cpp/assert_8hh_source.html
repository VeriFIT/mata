19namespace mata::internal {
20[[noreturn]] inline void
21 assert_fail(const char* expr, const char* file, int line, const char* func, const std::string& msg) {
22 if (msg.empty()) {
23 std::println(stderr, "{}:{}: {}: Assertion '{}' failed.", file, line, func, expr);
24 } else {
25 std::println(stderr, "{}:{}: {}: Assertion '{}' failed: {}", file, line, func, expr, msg);

31 /**
32 * @brief Assert that a condition is true, and if not, print an error message and abort the program.
33 *
34 * @param cond The condition to assert.
35 * @param ... Optional format string and arguments for the error message.
36 *
37 * Usage:
38 * ```cpp
39 * MATA_ASSERT(state < num_states, "state={}, num_states={}", state, num_states);
40 * MATA_ASSERT(state < num_states);
41 * ```
42 * In release builds (when `NDEBUG` is defined), the assertions are disabled and have no effect.
43 */
44 #define MATA_ASSERT(cond, ...)
45 {
46 ((cond) ? ((void) 0)
47 : mata::internal::assert_fail(
48 #cond, __FILE__, __LINE__, __func__, std::string(__VA_OPT__(std::format(__VA_ARGS__)))
49 ));
50 }
51#endif // NDEBUG

53/**
