27/**
28 * A node of graph representing transition formula. A node could be operator (!,&,|) or operand (symbol, state, node).
29 * Each node has its raw name as written in the input, its name (the raw name without the type marker, the initial
30 * character, under the marked naming), and information about its type.
31 */
32class FormulaNode {
33 public:
34 enum class OperandType : std::uint8_t { Symbol, State, Node, True, False, NotOperand };
35
36 enum class OperatorType : std::uint8_t { Neg, And, Or, NotOperator };
37
38 enum class Type : std::uint8_t { Operand, Operator, LeftParenthesis, RightParenthesis, Unknown };
39
40 /// Define whether a node is operand or operator
41 Type type{};
42 /**
43 * Raw name of node as it was specified in input text, i.e., including type marker.
44 * The name is not owned: it has to point into the name storage of the automaton the node belongs to
45 * (@c IntermediateAut::name_store or @c IntermediateAut::section_names) or into a literal, so that it lives for
46 * as long as the node does.
47 */
48 std::string_view raw{};
49 /// if a node is operator, it defines which one
50 OperatorType operator_type{};
51 /// if a node is operand, it defines which one
52 OperandType operand_type{};
53
54 /// The parsed name: @c raw without the type marker, if the node has one. Valid while @c raw is unchanged.
55 std::string_view name() const { return std::string_view{raw}.substr(marked_ ? 1 : 0); }
56
57 bool is_operand() const { return type == Type::Operand; }
58
59 bool is_operator() const { return type == Type::Operator; }
60
61 bool is_right_parenthesis() const { return type == Type::RightParenthesis; }
62
63 bool is_left_parenthesis() const { return type == Type::LeftParenthesis; }
64
65 bool is_state() const { return operand_type == OperandType::State; }
66
67 bool is_symbol() const { return operand_type == OperandType::Symbol; }
68
69 bool is_and() const { return type == Type::Operator && operator_type == OperatorType::And; }
70
71 bool is_neg() const { return type == Type::Operator && operator_type == OperatorType::Neg; }
72
73 // TODO: should constant be its own operand type?
74 bool is_constant() const { return is_true() || is_false(); }
75 bool is_true() const { return is_operand() && operand_type == OperandType::True; }
76 bool is_false() const { return is_operand() && operand_type == OperandType::False; }

169/// The children of a node of a @c FormulaGraph: none for an operand, one for a negation, two for a conjunction or a
170/// disjunction.
171class FormulaGraphChildren {
172 public:
173 size_t size() const { return size_; }
174 bool empty() const { return size_ == 0; }
175 const FormulaGraphView& operator[](const size_t index) const { return views_[index]; }
176 const FormulaGraphView& at(const size_t index) const {
177 if (index >= size_) { throw std::out_of_range("FormulaGraphChildren::at"); }
178 return views_[index];
179 }
180 const FormulaGraphView& front() const { return views_[0]; }
181 const FormulaGraphView& back() const { return views_[size_ - 1]; }
182 const FormulaGraphView* begin() const { return views_.data(); }
183 const FormulaGraphView* end() const { return views_.data() + size_; }
184
185 private:
186 friend class FormulaGraphView;
187 std::array<FormulaGraphView, 2> views_{};
188 size_t size_{0};
189};
190
191inline FormulaGraphChildren FormulaGraphView::children() const {
192 FormulaGraphChildren children;
193 if (!entry_->node.is_operator()) { return children; }
194 const FormulaGraphEntry* const last{entry_ - 1};
195 if (entry_->node.is_neg()) {
196 children.views_[0] = FormulaGraphView{last};
197 children.size_ = 1;
198 } else {
199 children.views_[0] = FormulaGraphView{last - last->subtree_size};
200 children.views_[1] = FormulaGraphView{last};
201 children.size_ = 2;
202 }
203 return children;
204}

206/**
207 * A transition formula as a tree whose nodes are the operators and operands of the formula.
208 * E.g., a formula q1 & s1 is a tree with & as the root node and q1 and s1 as the children of the root.
209 * The nodes are stored in one vector in postfix order, so a whole formula takes a single allocation. A negation has
210 * one child, a conjunction and a disjunction two, and any other node none.
211 */
212class FormulaGraph {
213 public:
214 /// An empty graph, viewed as a single node of the type @c FormulaNode::Type::Unknown.
215 FormulaGraph() = default;
216 /// A graph of the single node @p node, which has no children, so it must not be an operator.
217 explicit FormulaGraph(const FormulaNode& node);
218 /// A copy of the subtree @p tree.
219 explicit FormulaGraph(FormulaGraphView tree);
220 /// A graph of the negation @p op of @p operand.
221 FormulaGraph(const FormulaNode& op, FormulaGraphView operand);
222 /// A graph of the conjunction or disjunction @p op of @p lhs and @p rhs.
223 FormulaGraph(const FormulaNode& op, FormulaGraphView lhs, FormulaGraphView rhs);
224
225 /**
226 * Builds the graph of the formula in postfix notation @p postfix.
227 * @throws std::runtime_error if @p postfix is not a single formula, as when an operator lacks operands.
228 */
229 static FormulaGraph from_postfix(std::span<const FormulaNode> postfix);
230
231 bool empty() const { return entries_.empty(); }
232 FormulaGraphView root() const { return empty() ? FormulaGraphView{} : FormulaGraphView{&entries_.back()}; }
233 const FormulaNode& node() const { return root().node(); }
234 FormulaGraphChildren children() const { return root().children(); }
235 /// The nodes in postfix order, see @c FormulaGraphView::entries().
236 std::span<const FormulaGraphEntry> entries() const { return entries_; }
237
238 /// Stores the names of all nodes in @p names, so that the graph no longer points into the storage it did.
239 void store_names(NameStore& names);
240
241 std::unordered_set<std::string> collect_node_names() const { return root().collect_node_names(); }
242 void print_tree(std::ostream& os) const { root().print_tree(os); }
243
244 private:
245 /// Appends the subtree @p tree, as the next operand of the node appended after it.
246 void append(FormulaGraphView tree);
247 /// Appends @p node as the root of the subtrees its children are, which end right before it.
248 void append_root(const FormulaNode& node);
249
250 std::vector<FormulaGraphEntry> entries_{};
251};

255/**
256 * Structure for a general intermediate representation of parsed automaton.
257 * It contains information about type of automata, type of naming of nodes, symbols and states
258 * and type of alphabet. It contains also the transitions formula and formula for initial and final
259 * states. The formulas are represented as tree where nodes are either operands or operators.
260 */
261class IntermediateAut {
262 public:
263 /**
264 * Type of automaton. So far we support nondeterministic finite automata (NFA) and nondeterministic finite
265 * transducers (NFT).
266 */
267 enum class AutomatonType : std::uint8_t {
268 Nfa,
269 // TODO(afa): Re-add the alternating automata support. The AFA code is kept, but not compiled:
270 // `MATA_AFA_SUPPORT` is never defined.
271#ifdef MATA_AFA_SUPPORT
272 Afa,
273#endif
274 Nft,
275 };
276
277 /**
278 * The possible kinds of naming of items in sets of states, nodes, or symbols.
279 * It implies how the given set is defined.
280 * Naming could be automatic (all things in formula not belonging to other sets will be assigned to a
281 * set with automatic naming), marker based (everything beginning with `q` is a state, with `s` is a symbol,
282 * with `n` is a node), or enumerated (the given set is defined by enumeration).
283 * There are two special cases used for alphabet - symbols could be any character (CHARS) or anything from utf
284 * (UTF).
285 */
286 enum class Naming : std::uint8_t { Auto, Marked, Enum, Chars, Utf };
287
288 /**
289 * Type of alphabet. We can have explicit alphabet (containing explicit symbols), or bitvector, or class of symbols
290 * (e.g., alphabet is everything in utf), or intervals.
291 * So far, only explicit representation is supported.
292 */
293 enum class AlphabetType : std::uint8_t { Explicit, Bitvector, Class, Intervals };
294
295 Naming state_naming = Naming::Marked;
296 Naming symbol_naming = Naming::Marked;
297 Naming node_naming = Naming::Marked;
