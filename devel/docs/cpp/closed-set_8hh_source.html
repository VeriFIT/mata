1/** @file
2 * @brief Definition of a downward and upward closed set.
3 *
4 * This file contains definition of a closed set and function which
5 * allow us to work with them (membership, inclusion, union, intersection)
6 *
7 * Description:
8 * In this context, an upward-closed set is a set of sets of elements of type T,
9 * (set of nodes of type T, where a node is a set of elements of type T)
10 * where the elements come from the discrete range <min_val; max_val>. If the upward
11 * closed set contains a subset A of the range <min_val; max_val>, it also contains
12 * all the supersets of A (in context of the discrete range <min_val; max_val>).
13 * Thus, the upward closed set could be easily represented by its 1) type (upward closed),
14 * 2) discrete range borders (min_val, max_val), 3) its antichain.
15 * Analogously, a downward closed set contains all the subsets of the antichain elements.
16 *
17 * It is possible to:
18 *
19 * -> ==- and !=-compare two closed sets
20 * -> <=- and >=-compare two closed sets of the same type
21 * -> get a text representation of the closed set
22 * -> chceck whether the closed set contains a given node/nodes
23 * -> insert a node/more nodes to the closed set
24 * -> perform an union over two closed sets of the same type and with the same carrier
25 * -> perform an intersection over two closed sets of the same type and with the same carrier
26 * -> compute a complement of a closed set
27 *
28 * It is not possible to:
29 *
30 * -> choose a custom carrier which is not a discrete range <min_val; max_val>
31 * -> <=- and >=-compare two closed sets of the different types (nonsense)
32 * -> remove a node/more nodes from the closed set (nonsense)
33 * -> perform an union over two closed sets of different types or different carriers (nonsense)
34 * -> perform an intersection over two closed sets of different types or different carriers (nonsense)
35 *
36 * Examples:
37 *
38 * Let us have the carrier {0, 1, 2, 3} and the upward-closed set with the antichain
39 * {{0}, {1, 2}}. We can write↑{{0}, {1, 2}} = {{0}, {0, 1}, {0, 2}, {0, 3}, {0, 1, 2},
40 * {0, 1, 3}, {0, 2, 3}, {1, 2}, {1, 2, 3}, {0, 1, 2, 3}}.
41 *
42 * Let us have the carrier {0, 1, 2, 3} and the downward-closed set with the antichain
43 * {{0}, {1, 2}}. We can write ↓{{0}, {1, 2}} = {{0}, {1, 2}, {1}, {2}, {}}.
44 *
45 * @author Tomáš Kocourek
46 */

67 * @brief Closed set.
68 *
69 * Contains discrete range borders, its type and the corresponding anti-chain. It is necessary to be able to evaluate
70 * relation operators <, <=, >, >=, ==, != over instances of this datatype T.
71 * @tparam T Datatype the closed set contains.

120 // A closed set is considered to be smaller than the other one iff
121 // it is a subset of the other one
122 // It is not possible to perform <=-comparisons accros upward- and downward-closed
123 // sets, each argument has to be upward- or downward-closed set

195/** This function decides whether a set of elements is part of the closed set
196 * by subset-comparing the input with all elements of the antichain
197 * @brief decides whether the closed set contains a given element
198 * @param node a given ordered vector of elements of the type T
199 * @return true iff the given ordered vector belongs to the current closed set
200 */
201template <typename T> bool ClosedSet<T>::contains(const Node& node) const {
202 if (type_ == ClosedSetType::UpwardClosedSet) {
203 for (auto element : antichain_) {
204 if (element.is_subset_of(node)) { return true; }
205 }
206 } else if (type_ == ClosedSetType::DownwardClosedSet) {
207 for (auto element : antichain_) {
208 if (node.is_subset_of(element)) { return true; }
209 }
210 }
211 return false;
212}

214/** This function decides whether a set of sets of elements is a part of the closed set
215 * by subset-compraring the input with all elements of the antichain
216 * @brief decides whether the closed set contains a given set of sets of elements
217 * @param nodes a given ordered vector of ordered vectors of elements of the type T
218 * @return true iff the given ordered vector of ordered vectors belongs to the current closed set
219 */
220template <typename T> bool ClosedSet<T>::contains(const Nodes& nodes) const {
221 for (auto node : nodes) {
222 if (!contains(node)) { return false; }
223 }
224 return true;
225}

227/** This function decides whether a given ordered vector of elements of the datatype T
228 * belongs to the discrete interval of the current closed set
229 * @brief decides whether the given vector respects the borders
230 * @param node a given ordered vector of elements of the type T which will be inspected
231 * @return true iff the given ordered vector respects the borders
232 */
233template <typename T> bool ClosedSet<T>::in_interval(const Node& node) const {
234 for (auto value : node) {
235 if (value < min_val_ || value > max_val_) { return false; }
236 }
237 return true;

240/** Adds a new element to the closed set. If the element is already contained in the closed
241 * set, the closed set remains unchanged. Otherwise, the antichain will be recomputed.
242 * @brief inserts a new element to the closed set
243 * @param node a given node which will be added to the closed set
244 */
245template <typename T> void ClosedSet<T>::insert(const Node& node) {
246 MATA_ASSERT(in_interval(node), "Each element of the given node has to respect the carrier of the closed set.");
247 // If the closed set is empty, the antichain could be simply changed
248 // by adding the given node as the only node to the antichain
249 if (antichain_.empty()) {
250 antichain_.insert(node);
251 return;
252 }
253 // If the closed set already contains the given node, there is no
254 // need to change the closed set
255 if (contains(node)) { return; }
256 // We need to collect all the elements off the antichain which could be removed
257 // as soon as the new element is added to keep the antichain <=-uncomparable
258 OrdVec<OrdVec<T>> to_erase{};
259
260 // If the closed set is upward-closed, we have to erase all the elements of
261 // the antichain which are supersets of the inserted node
262 // Example: Let us have an upward-closed set ↑{{0, 1}, {2}} with a corresponding
263 // antichain {{0, 1}, {2}}. If we add {0} to the closed set, the element {0, 1}
264 // needs to be erased from the antichain since the set {{0}, {0, 1}, {2}} contains
265 // <=-comparable elements and the result should be upward-closed.
266 if (type_ == ClosedSetType::UpwardClosedSet) {
267 for (auto element : antichain_) {
268 if (node.is_subset_of(element)) { to_erase.insert(element); }
269 }
270 }
271
272 // If the closed set is downward-closed, we have to erase all the elements of
273 // the antichain which are subsets of the inserted node
274 // Example: Let us have an upward-closed set ↓{{0, 1}, {2}} with a corresponding
275 // antichain {{0, 1}, {2}}. If we add {1, 2} to the closed set, the element {2}
276 // needs to be erased from the antichain since the set {{0}, {1, 2}, {2}} contains
277 // <=-comparable elements and the result should be downward-closed.
278 else if (type_ == ClosedSetType::DownwardClosedSet) {
279 for (auto element : antichain_) {
280 if (element.is_subset_of(node)) { to_erase.insert(element); }
281 }
282 }
283
284 for (auto element : to_erase) { antichain_.erase(element); }
285 antichain_.insert(node);
286} // insert }}}

288/** Performs a union over two closed sets with the same type and carrier.
289 * This function simply adds all the
290 * elements of the antichain of the first closed set to the second closed set
291 * @brief performs a union over closed sets
292 * @param rhs a given closed set which will be united with the current one
293 * @return a union of the given closed sets
294 */
295template <typename T> ClosedSet<T> ClosedSet<T>::set_union(const ClosedSet& rhs) const {
296 MATA_ASSERT(
297 type_ == rhs.type_ && min_val_ == rhs.min_val_ && max_val_ == rhs.max_val_,
298 "Types and borders of given closed sets must be the same to compute their union."
299 );
300 ClosedSet result(type_, min_val_, max_val_, antichain_);
301 result.insert(rhs.antichain());
302 return result;
303}

305/** Performs an intersection over two closed sets with the same type and carrier.
306 * @brief performs an intersection over closed sets
307 * @param rhs a given closed set which will be intersectioned with the current one
308 * @return an intersection of the given closed sets
309 */
310template <typename T> ClosedSet<T> ClosedSet<T>::intersection(const ClosedSet& rhs) const {
311 MATA_ASSERT(
312 type_ == rhs.type_ && min_val_ == rhs.min_val_ && max_val_ == rhs.max_val_,
313 "Types and borders of given closed sets must be the same to compute their intersection."
314 );
315 ClosedSet result(type_, min_val_, max_val_);
316
317 // Iterates through all the tuples from Antichan1 X Antichan2
318 // and creates a union of them
319 if (type_ == ClosedSetType::UpwardClosedSet) {
320 for (const Node& element1 : antichain_) {
321 for (const Node& element2 : rhs.antichain()) {
322 Node tmp = element1;
323 tmp.insert(element2);
324 result.insert(tmp);
325 }
326 }
327 }
328
329 // Iterates through all the tuples from Antichan1 X Antichan2
330 // and creates an intersection of them
331 if (type_ == ClosedSetType::DownwardClosedSet) {
332 for (auto element1 : antichain_) {
333 for (auto element2 : rhs.antichain()) { result.insert(element1.intersection(element2)); }
334 }
335 }
336 return result;
337} // intersection }}}

339/** Performs a complementation over a closed set. The result will
340 * contain nodes which are not elements of the former closed set.
341 * The complement of an upward-closed set is always downward-closed and vice versa.
342 * @brief performs a complementation over a closed set
343 * @return a complement of a closed set
344 */
345template <typename T> ClosedSet<T> ClosedSet<T>::complement() const {
346 // The complement of an upward-closed set is
347 // always downward-closed and vice versa.
348 ClosedSetType new_type = ClosedSetType::UpwardClosedSet;
349 if (type_ == ClosedSetType::UpwardClosedSet) { new_type = ClosedSetType::DownwardClosedSet; }
350 ClosedSet result = ClosedSet(new_type, get_min(), get_max(), antichain());
351
352 if (type_ == ClosedSetType::UpwardClosedSet) {
353 // Initially, a complement contains all possible elements
354 // which will be then (possibly) removed.
355 Node initial_values{};
356 for (long unsigned i = 0; i <= max_val_; ++i) { initial_values.insert(i); }
357 result.insert(initial_values);
358
359 // For each element of the closed set {xa, xb, xc, ...}, we
360 // create a set of all sets Xa, Xb, Xc,... such that Xn contains
361 // all elements of the carrier except of xn
362 // For example, for a node {0, 2, 3} (the maximal element is 4),
363 // we create a set {{1, 2, 3, 4}, {0, 1, 3, 4}, {0, 1, 2, 4}}.
364 // This set corresponds to an antichain of a new downward-closed set.
365 // The result will be an intersection of all downward-closed sets
366 // created using this procedure.
367 for (const auto& element : antichain()) {
368 ClosedSet preparing_antichain(ClosedSetType::DownwardClosedSet, get_min(), get_max());
369 for (T i = 0; i <= max_val_; ++i) {
370 if (!element.count(i)) { continue; }
371 Node candidates{};
372 for (T j = 0; j <= max_val_; ++j) {
373 if (i != j) { candidates.insert(j); }
374 }
375 preparing_antichain.insert(candidates);
376 }
377 result = result.intersection(preparing_antichain);
378 }
379 }
380
381 else if (type_ == ClosedSetType::DownwardClosedSet) {
382 // Initially, a complement contains all possible elements
383 // which will be then (possibly) removed.
384 result.insert(Node());
385
386 // For each element of the closed set {xa, xb, xc, ...}, we
387 // create a set of all sets Xa, Xb, Xc,... such that Xn contains
388 // only xn. For example, for a node {0, 2, 3}, we create a set
389 // {{0}, {2}, {3}}.
390 // This set corresponds to an antichain of a new upward-closed set.
391 // The result will be an intersection of all upward-closed sets
392 // created using this procedure.
393 for (Node element : antichain()) {
394 ClosedSet preparing_antichain(ClosedSetType::UpwardClosedSet, min_val_, max_val_);
395 for (T i = min_val_; i <= max_val_; ++i) {
396 Node candidates{i};
397 if (!element.count(i)) { preparing_antichain.insert({i}); }
398 }
399 result = result.intersection(preparing_antichain);
400 }
401 }
402
403 return result;
404} // complement }}}
