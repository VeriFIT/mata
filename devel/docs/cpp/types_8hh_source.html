54// Minimum and maximum values of the data types used for states and symbols.
55struct Limits {
56 static constexpr State min_state = std::numeric_limits<State>::min();
57 /// The "no state" sentinel; the largest usable state is @c max_state - 1.
58 static constexpr State max_state = std::numeric_limits<State>::max();
59 /// The largest number of states an automaton can have (states @c 0 to @c max_state - 1).
60 static constexpr size_t max_num_of_states = max_state;
61 static constexpr Symbol min_symbol = std::numeric_limits<Symbol>::min();
62 static constexpr Symbol max_symbol = std::numeric_limits<Symbol>::max();
63};

65/// Throws @c std::overflow_error: a state number does not fit below @c Limits::max_state.
66[[noreturn]] void throw_state_overflow();

68/// Throws @c std::overflow_error unless @p num_of_states states can be numbered by @c State.
69inline void check_num_of_states(const size_t num_of_states) {
70 if (num_of_states > Limits::max_num_of_states) [[unlikely]] { throw_state_overflow(); }

73/// Converts @p value to a usable @c State, throwing @c std::overflow_error if it is @c max_state or larger.
74inline State to_state(const size_t value) {
75 if (value >= Limits::max_num_of_states) [[unlikely]] { throw_state_overflow(); }
76 return static_cast<State>(value);
77}

79/// The number of states needed to contain @p state, @p state + 1, computed without wrapping around.
80/// Throws @c std::overflow_error if @p state is the @c Limits::max_state sentinel.
81inline size_t num_of_states_for(const State state) {
82 if (state == Limits::max_state) [[unlikely]] { throw_state_overflow(); }

89/// @name Options for algorithms and functions in Mata.
90///@{
91/**
92 * @brief Where an epsilon closure is applied around a transition step.
93 *
94 * The values are bit flags, so @c BeforeAndAfter must be tested with @c closes_before() and @c closes_after()
95 * rather than compared for equality. For @c 0 -ε-> 1 -a-> 2 -ε-> 3 and the step over @c a from @c {0}:
96 *
97 * | Option | Applied | Result |
98 * |---|---|---|
99 * | @c None | post(S) | @c {} |
100 * | @c Before | post(closure(S)) | @c {2} |
101 * | @c After | closure(post(S)) | @c {} |
102 * | @c BeforeAndAfter | closure(post(closure(S))) | @c {2,3} |
103 */
104enum class EpsilonClosureOpt : std::uint8_t {
105 None = 1 << 0, ///< No epsilon closure.
106 Before = 1 << 1, ///< Epsilon closure before the transition.
107 After = 1 << 2, ///< Epsilon closure after the transition.
108 BeforeAndAfter = Before | After ///< Epsilon closure before and after the transition.
109};
110
111/// Does @p opt ask for an epsilon closure before the transition step?
112constexpr bool closes_before(const EpsilonClosureOpt opt) {
113 return (static_cast<std::uint8_t>(opt) & static_cast<std::uint8_t>(EpsilonClosureOpt::Before)) != 0;
114}
115
116/// Does @p opt ask for an epsilon closure after the transition step?
117constexpr bool closes_after(const EpsilonClosureOpt opt) {
118 return (static_cast<std::uint8_t>(opt) & static_cast<std::uint8_t>(EpsilonClosureOpt::After)) != 0;
119}
