55// Minimum and maximum values of the data types used for states and symbols.
56struct Limits {
57 static constexpr State min_state = std::numeric_limits<State>::min();
58 /// The "no state" sentinel; the largest usable state is @c max_state - 1.
59 static constexpr State max_state = std::numeric_limits<State>::max();
60 /// The largest number of states an automaton can have (states @c 0 to @c max_state - 1).
61 static constexpr size_t max_num_of_states = max_state;
62 static constexpr Symbol min_symbol = std::numeric_limits<Symbol>::min();
63 static constexpr Symbol max_symbol = std::numeric_limits<Symbol>::max();
64};
65
66/// Throws @c std::overflow_error: a state number does not fit below @c Limits::max_state.
67[[noreturn]] void throw_state_overflow();
68
69/// Throws @c std::overflow_error unless @p num_of_states states can be numbered by @c State.
70inline void check_num_of_states(const size_t num_of_states) {
71 if (num_of_states > Limits::max_num_of_states) [[unlikely]] { throw_state_overflow(); }
72}
73
74/// Converts @p value to a usable @c State, throwing @c std::overflow_error if it is @c max_state or larger.
75inline State to_state(const size_t value) {
76 if (value >= Limits::max_num_of_states) [[unlikely]] { throw_state_overflow(); }
77 return static_cast<State>(value);
78}
79
80/// The number of states needed to contain @p state, @p state + 1, computed without wrapping around.
81/// Throws @c std::overflow_error if @p state is the @c Limits::max_state sentinel.
82inline size_t num_of_states_for(const State state) {
83 if (state == Limits::max_state) [[unlikely]] { throw_state_overflow(); }
84 return static_cast<size_t>(state) + 1;
85}
