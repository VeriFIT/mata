4/** @file
5 * @brief Definition of one post of a relation, and one entry of it.
6 *
7 * @section Nesting Post structure
8 *
9 * A relation is @c DeltaBase indexed by source state, over a chain of posts ending in a set of
10 * targets. There is one post per key, so a relation with @c n keys is @c n posts deep:
11 * ```
12 * DeltaBase -> Post -> Post -> ... -> Targets
13 * key 0 key 1 (a state, or a state with a payload)
14 * ```
15 * A @c Post is an ordered map from one key to the post nested under it. Iterating over one yields its
16 * @c PostEntry objects, each one key paired with the post beneath it. The same two classes make up
17 * every step of the chain. @c mata::StatePost and @c mata::SymbolPost are aliases for the depth-2
18 * instantiation. @c PostChain spells a chain outermost key first.
19 *
20 * @section Sortedness Sortedness
21 *
22 * Every post is sorted by key, and the innermost one by target. Lookups binary-search on it, and
23 * @c Post::first_epsilon_it() scans backwards relying on that order alone: a sorted range
24 * splits at any threshold into "below" then "at or above". A post that does not maintain the
25 * order still compiles, and silently returns wrong iterators. The order may be broken
26 * *temporarily*: @c push_back and @c emplace_back append without restoring it, which is faster
27 * when building from unordered input. It has to be then restored by sorting before any lookup,
28 * iteration order or comparison is relied upon. A post advertises that it
29 * maintains the invariant with @c sorted_by_key / @c sorted_by_target and exposes @c is_sorted().
30 */

307/**
308 * @brief One post of the relation: an ordered map from a key to the post nested under it.
309 *
310 * An ordered vector of @p E kept sorted by key, which is what every lookup binary-searches on.
311 * Every step of a chain is one of these. A relation of @c key_arity @c n is @c n of them deep,
312 * ending in a @c mata::posts::TargetSetLike.
313 *
314 * For relation with @c key_arity 1, the top level post uses a name alias StatePost,
315 * and the next level down uses SymbolPost.
316 *
317 * @tparam E The entry type: one key plus what is nested under it.
318 * @tparam R Where this post's ordinary keys stop. @see mata::ReservedKeys.
319 * @tparam A The allocator of the entries. @c mata::StatePost uses @c mata::utils::DeltaAllocator.
320 * @see mata::posts::PostLike.
321 */
322template <typename E, typename R = ReservedKeys<typename E::Key>, typename A = std::allocator<E>>
323class Post : utils::OrdVector<E, A> {
324 using super = utils::OrdVector<E, A>;
325 using post_iterator = typename super::const_iterator;

327 public:
328 /// @name Post protocol
329 /// @see mata::posts::PostLike.
330 ///@{
331 using Entry = E; ///< What iterating this post yields.
332 using Key = typename Entry::Key; ///< What this post is keyed by (the symbol, for an NFA).
333 using Nested = typename Entry::Nested; ///< The post (or target set) under one key.
334 using Target = typename Entry::Target; ///< What visiting the successors yields, propagated up.
335 using Reserved = R; ///< Where this post's ordinary keys stop.
336 /// The innermost post: what is left once every key has been supplied.
337 using TargetSet = typename posts::PostAt<Nested, arity_of<Nested>>::type;
338 /// Number of keys from here down to a target.
339 static constexpr size_t key_arity{arity_of<Nested> + 1};
340 /// @see @ref sortedness. Ordered by the key of the contained entries.
341 static constexpr bool sorted_by_key{true};
342 /// Teturn whether the post is sorted by key. @see @ref sortedness.
343 bool is_sorted() const { return std::ranges::is_sorted(*this); }

385 /**
386 * @brief Returns an iterator to the smallest epsilon, or end() if there is no epsilon.
387 *
388 * @param first_epsilon The first epsilon key to look for. Defaults to the smallest reserved epsilon.
389 * @return An iterator to the smallest epsilon, or end() if there is no epsilon.
390 */
391 const_iterator first_epsilon_it(const Key first_epsilon = Reserved::min_epsilon) const {
392 // The post need to be ordered in order for the backward scan to work.
393 static_assert(
394 SortedWithReservedKeys<Post>,
395 "finding the epsilons by scanning back from the end needs the post ordered by key"
396 );
397 const auto end_it = cend();
398 auto it = end_it;
399 while (it != begin()) {
400 --it;
401 if (it->key() < first_epsilon) {
402 // We crossed the threshold, we need to return the previous position.
403 return it + 1;
404 }
405 }
406
407 if (it != end_it && it->key() >= first_epsilon) {
408 // Begin is the smallest epsilon.
409 return it;
410 }
411 return end_it;
412 }

414 /// Returns the set of all target states in the @c Post.
415 TargetSet get_successors() const {
416 TargetSet successors;
417 if constexpr (key_arity == 1 && requires(TargetSet& set, const TargetSet& other) { set.insert(other); }) {
418 // One sorted merge per key, where the set has one.
419 for (const Entry& entry : *this) { successors.insert(entry.targets); }
420 } else {
421 // One target at a time: above arity 1 the entries hold posts, not targets, and a set
422 // need not have a bulk insert.
423 visit_targets(*this, [&successors](const Target& target) { successors.insert(target); });
424 }
425 return successors;
426 }

428 /**
429 * @brief The result type of the keyed @c get_successors.
430 *
431 * At @c key_arity 1, the set of targets can be returned by reference.
432 * Above that, the set of targets has to be gathered.
433 */
434 using KeyedSuccessors = std::conditional_t<key_arity == 1, const TargetSet&, TargetSet>;
435
436 /**
437 * @brief The targets reachable from this post under @p key.
438 *
439 * @param symbol The key to get the successors for.
440 * @return The targets under the given key.
441 */
442 KeyedSuccessors get_successors(const Key symbol) const {
443 const auto entry_it = find(symbol);
444 if constexpr (key_arity == 1) {
445 if (entry_it == this->end()) {
446 // Returned by const reference, so it outlives the call and no caller can touch it.
447 static const TargetSet empty_set{};
448 return empty_set;
449 }
450 return entry_it->targets;
451 } else {
452 TargetSet successors{};
453 if (entry_it != this->end()) {
454 visit_targets(entry_it->nested(), [&successors](const Target& target) { successors.insert(target); });
455 }
456 return successors;
457 }

494 /**
495 * @brief Return an Iterator over alphabet (normal) symbols (not over epsilons).
496 *
497 * @param[in] last_symbol The last symbol to iterate over. Defaults to the largest ordinary key.
498 * @return An iterator over alphabet moves in @c Post represented as @c Move instances
499 * @throws std::runtime_error if @p last_symbol is itself a reserved key, since the range would
500 * then have to run past the end of the ordinary keys.
501 */
502 Moves moves_symbols(const Key last_symbol = Reserved::max_ordinary) const
503 requires(key_arity == 1)
504 {
505 if (last_symbol > Reserved::max_ordinary) {
506 throw std::runtime_error("Using a reserved key as a last symbol to iterate over.");
507 }
508 // Scanned back from the end, where only the few keys above @p last_symbol are.
509 auto symbols_end{cend()};
510 while (symbols_end != cbegin() && last_symbol < std::prev(symbols_end)->key()) { --symbols_end; }
511 return {cbegin(), symbols_end};
512 }

520 /**
521 * @brief Apply @p fn to every target reachable from this post, under any key.
522 */
523 template <typename Fn> void for_each_target(Fn&& fn) const {
524 if constexpr (key_arity == 1) {
525 // NOTE: Routing arity 1 through the generic recursion measured +18-19% on the real relation.
526 // A depth prototype had showd the recursion was free (0.99-1.01x).
527 for (const Entry& entry : *this) { entry.for_each_target(fn); }
528 } else {
529 visit_targets(*this, fn);
530 }
531 }
532
533 /**
534 * @brief Apply @p fn to every @c Move of this post, as a (key, target) pair.
535 */
536 template <typename Fn> void for_each_move(Fn&& fn) const {
537 if constexpr (key_arity == 1) {
538 // See for_each_target: the generic recursion cost +19-20% here at every size measured.
539 for (const Entry& entry : *this) {
540 entry.for_each_target([&](const Target target) { fn(entry.symbol, target); });
541 }
542 } else {
543 visit_moves(*this, fn);
544 }
545 }
546
547 /**
548 * @brief Does this post have @p target reachable under any key?
549 *
550 * @param[in] target Target state to check.
551 * @return True if @p target is reachable under any key, false otherwise.
552 */
553 bool has_target(const Target target) const {
554 // A binary search in each innermost post, rather than a walk over every target.
555 return std::ranges::any_of(*this, [&target](const Entry& entry) {
556 if constexpr (key_arity == 1) {
557 return entry.has_target(target);
558 } else {
559 return entry.nested().has_target(target);
560 }
561 });
562 }

564 private:
565 template <typename It> static It find_(It begin, It end, const utils::ArgOf<Key> key) {
566 const auto it = std::lower_bound(begin, end, key, [](const Entry& entry, const utils::ArgOf<Key> k) {
567 return entry.key() < k;
568 });
569 return (it == end || it->key() != key) ? end : it;
570 }
