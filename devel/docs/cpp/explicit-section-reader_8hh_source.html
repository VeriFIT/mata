76 /// Calls @p function with every name and its state.
77 template <class Function> void for_each(Function&& function) const {
78 for (const auto& [name, state] : others_) { function(std::string_view{name}, state); }
79 std::string name{prefix_.value_or("")};
80 const size_t prefix_size{name.size()};
81 for (size_t number{0}; number < numbered_.size(); ++number) {
82 if (numbered_[number] == NO_STATE) { continue; }
83 name.resize(prefix_size);
84 name += std::to_string(number);
85 function(std::string_view{name}, numbered_[number]);
86 }
87 }

111/**
112 * Reads the single section of a `.mata` input line by line, tokenizing every line as @c parse_mf() does.
113 *
114 * The reader covers only lines that are lists of names, and in the formula mode also the operators and parentheses of
115 * transition formulas (`&`, `|`, `!`, `(`, `)`), which it splits off as tokens of their own, as @c parse_mf() does.
116 * It gives up on anything else @c parse_mf() would turn into an operator, on a second section, on a line continuation
117 * left open at the end of the input, and on every input @c parse_mf() rejects. Giving up is reported as
118 * @c Line::Unsupported, so that the caller can hand the input to the general reader, which then either reads it or
119 * reports the error.
120 */
121class ExplicitSectionReader {
122 public:
123 enum class Line : std::uint8_t {
124 /// A `%KEY values...` line; see @c key() and @c tokens().
125 Key,
126 /// A body line; see @c tokens().
127 Body,
128 /// The end of the input.
129 End,
130 /// The input leaves the subset this reader covers.
131 Unsupported,
132 };
133
134 /// @p input has to outlive the reader and the tokens it returns. @p formulas Whether to read the operators and
135 /// parentheses of transition formulas.
136 explicit ExplicitSectionReader(std::string_view input, const bool formulas = false)
137 : input_{input},
138 formulas_{formulas} {}
139
140 /// Reads the `@TYPE` line, skipping the comment lines before it.
141 /// @return The section type, or @c std::nullopt when the input does not start with a well-formed type line.
142 std::optional<std::string_view> read_type();
143
144 /// Reads the next line, joining the lines continued by a trailing `\`.
145 Line read_line();
146
147 /// The key of the last @c Line::Key line, without the leading `%`.
148 std::string_view key() const { return key_; }
149
150 /// The values of the last @c Line::Key line, or the tokens of the last @c Line::Body line. Valid until the next
151 /// call of @c read_line().
152 const std::vector<ExplicitToken>& tokens() const { return tokens_; }
153
154 /**
155 * The text of the input that the consecutive tokens @p tokens of the last line were read from, quotes included.
156 * Equal texts read without giving up are equal tokens, as nothing before a token changes how the reader reads
157 * it but whether the reader gives up, so the text identifies the tokens.
158 * @return The text, or @c std::nullopt when a token had its escapes resolved and so has no text of the input.
159 */
160 std::optional<std::string_view> source_text(std::span<const ExplicitToken> tokens) const;
161
162 private:
163 /// Appends the tokens of one physical line to @c tokens_. @return Whether the line stays in the subset.
164 bool tokenize(std::string_view line);
165
166 std::string_view input_;
167 bool formulas_;
168 size_t position_{0};
169 std::string_view key_{};
170 std::vector<ExplicitToken> tokens_{};
171 /// Storage of the quoted tokens whose escapes had to be resolved; a deque keeps the views into it valid.
172 std::deque<std::string> unescaped_{};
173};
