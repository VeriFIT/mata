72/// Representation of a bool vector by a vector of uint8_t. Any nonzero byte is a set element, as for @c bool; every
73/// consumer (@c count(), @c get_elements(), @c SparseSet) applies this rule.
74class BoolVector : public std::vector<uint8_t> {
75 public:
76 BoolVector(const size_t size, const bool value) : std::vector<uint8_t>(size, value ? 1 : 0) {}
77 BoolVector(const BoolVector&) = default;
78 BoolVector(BoolVector&&) noexcept = default;
79 BoolVector() = default;
80 BoolVector(const std::initializer_list<uint8_t> uint8_ts) : std::vector<uint8_t>(uint8_ts) {}
81 explicit BoolVector(const std::vector<uint8_t>& uint8_ts) : std::vector<uint8_t>(uint8_ts) {}
82
83 BoolVector& operator=(const BoolVector&) = default;
84 BoolVector& operator=(BoolVector&&) noexcept = default;
85 ~BoolVector() = default;
86 /// Count the number of set elements.
87 size_t count() const {
88 size_t cnt{0};
89 for (const uint8_t value : *this) {
90 if (value != 0) { ++cnt; }
91 }
92 return cnt;
93 }
94
95 /// Collect the indices of the set bits into @p element_set, replacing its contents.
96 template <typename T> T& get_elements(T& element_set) const {
97 get_elements(&element_set, *this);
98 return element_set;
99 }
100
101 /// Collect the indices of the set bits of @p bool_vec into @p element_set, replacing its contents. Outputs without
102 /// @c reserve() (e.g. @c std::deque) are filled without reserving.
103 template <typename T> static T* get_elements(T* element_set, const BoolVector& bool_vec) {
104 element_set->clear();
105 if constexpr (requires { element_set->reserve(size_t{}); }) { element_set->reserve(bool_vec.count()); }
106 for (size_t i{0}; i < bool_vec.size(); ++i) {
107 if (bool_vec[i] != 0) { element_set->push_back(i); }
108 }
109 return element_set;
110 }
111
112}; // class BoolVector.

132 * @brief Replace all occurrences of a substring in a string
133 *
134 * @param input The input string
135 * @param needle The substring to be replaced
136 * @param replace The replacement string
137 * @return A new string with all occurrences of the substring replaced.
138 */
139inline std::string replace_all(const std::string& input, const std::string& needle, const std::string& replace) {
140 std::string result = input;
141 size_t pos = 0;
142 while ((pos = result.find(needle, pos)) != std::string::npos) {
143 result.replace(pos, needle.length(), replace);
144 pos += replace.length();
145 }
146 return result;
147}

149/** Are two sets disjoint? */
150template <class T> bool are_disjoint(const std::set<T>& lhs, const std::set<T>& rhs) { // {{{
151 auto it_lhs = lhs.begin();
152 auto it_rhs = rhs.begin();
153 while (it_lhs != lhs.end() && it_rhs != rhs.end()) {
154 if (*it_lhs == *it_rhs) { return false; }
155
156 if (*it_lhs < *it_rhs) {
157 ++it_lhs;
158 } else {
159 ++it_rhs;
160 }
161 }
162
163 return true;
164}

171/**
172 * @brief The seed every composite hash of mata starts from.
173 *
174 * Composite hashers must not start from the raw hash of their first element: @c std::hash of an integer is the
175 * identity in libstdc++ and libc++, so an unmixed first element would survive into the result and make whole
176 * families of keys collide.
177 */
178inline constexpr size_t HASH_SEED{
179 sizeof(size_t) == 8 ? static_cast<size_t>(0xcb'f2'9c'e4'84'22'23'25ULL) : static_cast<size_t>(0x81'1c'9d'c5UL)
180};

182/**
183 * @brief Spreads every input bit of @p value over all of its output bits.
184 *
185 * A multiply-xorshift finalizer: the 64-bit variant is the one Boost.ContainerHash uses since 1.81, the 32-bit
186 * variant is Murmur3's @c fmix32. Without such a finalizer, nearby keys (which is what state identifiers are)
187 * stay nearby in the hash space and share hash-table buckets.
188 */
189constexpr size_t hash_mix(size_t value) {
190 if constexpr (sizeof(size_t) == 8) {
191 uint64_t mixed{static_cast<uint64_t>(value)};
192 constexpr uint64_t multiplier{0x0e'98'46'af'9b'1a'61'5dULL};
193 mixed ^= mixed >> 32;
194 mixed *= multiplier;
195 mixed ^= mixed >> 32;
196 mixed *= multiplier;
197 mixed ^= mixed >> 28;
198 return static_cast<size_t>(mixed);
199 } else {
200 uint32_t mixed{static_cast<uint32_t>(value)};
201 mixed ^= mixed >> 16;
202 mixed *= 0x21'f0'aa'adUL;
203 mixed ^= mixed >> 15;
204 mixed *= 0x73'5a'2d'97UL;
205 mixed ^= mixed >> 15;
206 return static_cast<size_t>(mixed);
207 }
208}

210/**
211 * @brief Combines the hash of @p rhs into the running hash value @p lhs.
212 *
213 * @note The result depends on the order in which the elements are combined, which is what distinguishes the hash
214 * of @c (a, b) from the hash of @c (b, a).
215 */
216template <class T> inline size_t hash_combine(size_t lhs, const T& rhs) { // {{{
217 constexpr size_t golden_ratio{
218 sizeof(size_t) == 8 ? static_cast<size_t>(0x9e'37'79'b9'7f'4a'7c'15ULL) : static_cast<size_t>(0x9e'37'79'b9UL)
219 };
220 return hash_mix(lhs + golden_ratio + std::hash<T>{}(rhs));
221} // hash_combine }}}

227/**
228 * @brief Hashes the range @p first to @p last.
229 *
230 * Every element is combined into a constant seed, so that a one-element range does not hash to its only element.
231 */
232template <typename It> size_t hash_range(It first, It last) { // {{{
233 size_t accum{HASH_SEED};
234
235 for (; first != last; ++first) { accum = hash_combine(accum, *first); }
236
237 return accum;
238} // hash_range(It, It) }}}

240/**
241 * @brief A hasher for @c std::set, usable as the @c Hash template argument of unordered containers.
242 *
243 * @note We deliberately do NOT specialize @c std::hash<std::set<A>> for arbitrary @p A: the standard only allows
244 * program-defined specializations of standard templates when at least one template argument is a program-defined
245 * type, so specializing it for e.g. @c std::set<unsigned> would be undefined behavior (and could silently collide
246 * with another library's specialization for the same type).
247 */
248template <class A> struct SetHash {
249 size_t operator()(const std::set<A>& cont) const { return hash_range(cont.begin(), cont.end()); }
250};

261/**
262 * @brief A hasher for @c std::pair, usable as the @c Hash template argument of unordered containers.
263 *
264 * @note See @c SetHash for why we do not specialize @c std::hash<std::pair<A, B>> directly.
265 */
266template <class A, class B> struct PairHash {
267 size_t operator()(const std::pair<A, B>& pair) const {
268 // Pack a unique unsigned 32-bit pair payload once instead of mixing it twice.
269 if constexpr (sizeof(size_t) == 8 && std::unsigned_integral<A> && std::unsigned_integral<B> &&
270 sizeof(A) == sizeof(uint32_t) && sizeof(B) == sizeof(uint32_t)) {
271 return hash_mix((static_cast<uint64_t>(pair.first) << 32) | static_cast<uint64_t>(pair.second));
272 }
273 return hash_combine(hash_combine(HASH_SEED, pair.first), pair.second);
274 }
275};

295 Map<T2, T1> result;

306// This reserves space in a vector, to be used before push_back or insert.
307// Assuming the doubling extension strategy, it only makes the first reserve large, after that it leaves it to the
308// doubling. The revert speed-ups that once justified it are gone: no revert calls it any more, and #179 measured the
309// default revert faster without it.
310template <class Vector> void inline reserve_on_insert(Vector& vec, size_t needed_capacity = 0, size_t extension = 32) {
311 // return; //Try this to see the effect of calling this. It should not affect functionality.

319// This function reindexes vector, that is, the content of each index i will be moved to the index renaming[i].
320// It might be useful in revert and trim, but so far it is useless. It was hard to get right, so I am reluctant to
321// remove .... It assumes that renaming[i] <= i. It assumes that vec is not longer than renaming. The function is very
322// fragile.
323template <class Vector, typename Index> void defragment(Vector& vec, const std::vector<Index>& renaming) {
324 // MATA_ASSERT(vec.size() <= renaming.size());
325 size_t i = 0;
326 for (size_t rsize = renaming.size(), vsize = vec.size(); i < vsize && i < rsize; i++) {
327 if (renaming[i] != i) {
328 if (!(renaming[i] < vsize)) { break; }
329 MATA_ASSERT(renaming[i] < i);
330 vec[i] = std::move(vec[renaming[i]]);
331 }
332 }
333 vec.reserve(i);
334}

336// In a vector of numbers, rename the numbers according to the renaming: renaming[old_name]=new_name
337template <class Vector, typename Index> void rename(Vector& vec, const std::vector<Index>& renaming) {
338 // Every element is renamed, also one that happens to equal its index.
339 for (size_t i = 0, size = vec.size(); i < size; ++i) { vec[i] = renaming[vec[i]]; }

368/**
369 * @brief Sorts @p vec and removes its duplicate elements.
370 *
371 * @c std::sort stays the sorting algorithm: it is the fastest of the ones measured on the mixed sizes mata
372 * produces, and a specialized sort would have to be restricted to unsigned integral keys anyway, because
373 * @c SymbolPost compares by symbol only and would lose targets under a comparison sort.
374 * Already-sorted input is common (ordered construction, conversion from an ordered container), and the linear
375 * @c std::is_sorted check that detects it costs far less than the sort it avoids.
376 */
377template <class Vector> void inline sort_and_rmdupl(Vector& vec) {
378 if (!std::is_sorted(vec.begin(), vec.end())) { std::sort(vec.begin(), vec.end()); }
379 const auto it = std::unique(vec.begin(), vec.end());
380 vec.resize(static_cast<size_t>(it - vec.begin()));
381}
