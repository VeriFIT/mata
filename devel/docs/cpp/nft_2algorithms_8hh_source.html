13/**
14 * Concrete NFT implementations of algorithms, such as complement, inclusion, or universality checking.
15 *
16 * This is a separation of the implementation from the interface defined in mata::nft.
17 * Note, that in mata::nft interface, there are particular dispatch functions calling
18 * these function according to parameters provided by a user.

23 * 1. Complementation,

75/**
76 * Inclusion implemented by antichain algorithms.
77 *
78 * The operands are read as the flat automata that @c Nft::unwind_jumps() with the DONT_CARE replacements below would
79 * produce, EPSILON being an ordinary letter. When unwinding would change an operand (a jump transition, or a DONT_CARE
80 * spelled out over the replacements), the jumps and DONT_CAREs are explored lazily during the search instead: no
81 * operand is unwound and DONT_CARE is not expanded over the alphabet. The answer is that of
82 * @c is_included_antichains_unwound().
83 * @param[in] smaller Automaton which language should be included in the bigger one
84 * @param[in] bigger Automaton which language should include the smaller one
85 * @param[in] alphabet Alphabet whose symbols replace @c DONT_CARE; @c nullptr for the symbols of both automata
86 * without @c DONT_CARE.
87 * @param[out] cex A potential counterexample word which breaks inclusion, one symbol per level. When the jumps are
88 * explored lazily, the positions of @c cex->path inside a jump hold @c Limits::max_state.
89 * @param[out] jump_mode Specifies if the symbol on a jump transition (a transition with a length greater than 1)
90 * is interpreted as a sequence repeating the same symbol or as a single instance of the symbol followed by a sequence
91 * of @c DONT_CARE.
92 * @return True if smaller language is included, i.e., if the final intersection of smaller complement of bigger is
93 * empty.
94 * @throw std::invalid_argument when the level vector of an operand does not cover all its states or holds a level
95 * that is out of range.
96 */

98 const Nft& smaller,
99 const Nft& bigger,

194/**
195 * @brief Concatenate two NFTs.

197 * Connects @p lhs and @p rhs by adding a run of ε-transitions, one per level, from each final state of @p lhs to
198 * each initial state of @p rhs. The ε is set to @p epsilon. If @p use_epsilon is false, @c remove_epsilon(@p
199 * epsilon) is run on the result, so every transition over @p epsilon disappears, including the operands' own

208 * @param[out] lhs_state_renaming Map mapping lhs states to result states. Filled only when requested; it
209 * describes the shift of the full concatenation even when the result is empty.
210 * @param[out] rhs_state_renaming Map mapping rhs states to result states. Filled only when requested; it

255/**
256 * @brief The common level layout of the general composition of two NFTs.
257 *
258 * The aligned levels come in blocks, one per synchronization level and one after the last: the levels only @c lhs
259 * reads, then the levels only @c rhs reads, then the synchronization level, which the last block lacks. The result
260 * of the composition has the aligned levels in this order, without the synchronization levels when they are
261 * projected out. Example:
262 *
263 * LHS | RHS
264 * --------------------------------------------+---------------------------------------------
265 * num_of_levels = 5 | num_of_levels = 4
266 * sync_levels = { 2, 4 } | sync_levels = { 1, 2 }
267 * --------------------------------------------+---------------------------------------------
268 * positions: lhs 0, lhs 1, rhs 0, sync (2, 1), lhs 3, sync (4, 2), rhs 3
269 * lhs_new_levels_mask() = { 0, 0, 1, 0, 0, 0, 1 }, rhs_new_levels_mask() = { 1, 1, 0, 0, 1, 0, 0 }
270 */
271struct CompositionSchedule {
272 /// Which operand reads an aligned level.
273 enum class Owner : uint8_t {
274 Lhs, ///< Only @c lhs reads it; @c rhs is not inserted into it.
275 Rhs, ///< Only @c rhs reads it.
276 Sync, ///< Both read it: a synchronization level.
277 };
278 /// An aligned level.
279 struct Position {
280 Owner owner;
281 /// The level of the @c lhs state when this aligned level is read: the level @c lhs reads here, or, where only
282 /// @c rhs reads, the next level @c lhs reads (0 once it has read its last level of the round).
283 Level lhs_level;
284 /// The level of the @c rhs state when this aligned level is read, as @c lhs_level.
285 Level rhs_level;
286 bool operator==(const Position&) const = default;
287 };
288
289 /// The aligned levels, in the order of the result levels.
290 std::vector<Position> positions{};
291
292 /**
293 * @brief Lay out the composition of an NFT with @p lhs_num_of_levels levels and one with @p rhs_num_of_levels
294 * levels on the given synchronization levels, which must be valid (see @c compose()).
295 */
296 CompositionSchedule(
297 size_t lhs_num_of_levels,
298 size_t rhs_num_of_levels,
299 const utils::OrdVector<Level>& lhs_sync_levels,
300 const utils::OrdVector<Level>& rhs_sync_levels
301 );
302
303 /// The mask of the aligned levels @c insert_levels() inserts into @c lhs: those only @c rhs reads.
304 BoolVector lhs_new_levels_mask() const;
305 /// The mask of the aligned levels @c insert_levels() inserts into @c rhs: those only @c lhs reads.
306 BoolVector rhs_new_levels_mask() const;
307 /// The aligned synchronization levels.
308 utils::OrdVector<Level> sync_positions() const;
309};

311/**
312 * @brief Composes two NFTs.
313 *
314 * This function computes the composition of two NFTs, `lhs` and `rhs`, by aligning their synchronization levels.
315 * Vectors of synchronization levels must be non-empty and of the same size. The levels of the resulting NFT
316 * are formed, iteratively, from the non-synchronizing levels of `lhs`, followed by the non-synchronizing levels of
317 * `rhs`, followed by the next synchronizing level (if not projected out by setting project_out_sync_levels to true).
318 * For example, given `lhs` with 7 levels, `rhs` with 8 levels, lhs_sync_levels=<0,2,5>, rhs_sync_levels=<1,4,5>, the
319 * resulting NFT levels will correspond to (assuming that we are not projecting out the synchronized levels)
320 * - `rhs` level 0
321 * - synchronized level from level 0 of `lhs` and level 1 of `rhs`
322 * - `lhs` level 1
323 * - `rhs` levels 2 and 3
324 * - synchronized level from level 2 of `lhs` and level 4 of `rhs`
325 * - `lhs` levels 3 and 4
326 * - synchronized level from level 5 of `lhs` and level 5 of `rhs`
327 * - `lhs` level 6
328 * - `rhs` levels 6 and 7
329 *
330 * NOTE: If you have only a single synchronization level per NFT, consider @c compose_fast() for better performance; it
331 * reads jump transitions as well.
332 *
333 * @param[in] lhs First transducer to compose.
334 * @param[in] rhs Second transducer to compose.
335 * @param[in] lhs_sync_levels Ordered vector of synchronization levels of the @p lhs.
336 * @param[in] rhs_sync_levels Ordered vector of synchronization levels of the @p rhs.
337 * @param[in] project_out_sync_levels Whether we want to project out the synchronization levels.
338 * @param[in] jump_mode Specifies if the symbol on a jump transition (a transition with a length greater than 1)
339 * is interpreted as a sequence repeating the same symbol or as a single instance of the symbol followed by a sequence
340 * of @c DONT_CARE.
341 * @param[in] normalization Only @c CompositionNormalization::Full and @c Trim change the result: the general
342 * composition generates no states of its own for @c GeneratedOnly to rewrite.
343 *
344 * Operands whose transitions all go to a later level of their round or end it are composed by
345 * @c compose_general_direct() (with more than one aligned level), all others by the pipeline of
346 * @c compose_general_reference(); both accept the same relation.
347 *
348 * @return A new NFT after the composition.
349 * @throws std::invalid_argument When the request is invalid; the same checks as @c nft::compose() runs, so calling
350 * this entry point directly does not skip them.
351 */
352Nft compose_general(
353 const Nft& lhs,
354 const Nft& rhs,
355 const utils::OrdVector<Level>& lhs_sync_levels,
356 const utils::OrdVector<Level>& rhs_sync_levels,
357 bool project_out_sync_levels = true,
358 JumpMode jump_mode = JumpMode::RepeatSymbol,
359 CompositionNormalization normalization = CompositionNormalization::Auto
360);

402/**
403 * @brief Composes two NFTs with a single synchronization level, reading their jump transitions in @p jump_mode.
404 *
405 * The levels of the resulting NFT are in the following order:
406 * - levels of `lhs` before its synchronization level
407 * - levels of `rhs` before its synchronization level
408 * - the synchronizing level (possibly missing if project_out_sync_levels is true)
409 * - levels of `lhs` after its synchronization level
410 * - levels of `rhs` after its synchronization level
411 *
412 * A jump transition is read lazily, one level at a time: the jumps of a symbol post share a cursor per level they cover
413 * after their first one, which reads the symbol of the jump under @c JumpMode::RepeatSymbol and DONT_CARE under
414 * @c JumpMode::AppendDontCares, and a configuration of the composition pairs such cursors as it pairs states. A level
415 * inside a jump thus synchronizes on the symbol the jump reads there. The operands are not copied, and DONT_CARE is
416 * not spelled out. Under @c JumpMode::RepeatSymbol, an EPSILON transition between two zero-level states is a whole
417 * round of EPSILON, as under @c JumpMode::NoJump.
418 *
419 * The result has no jump transitions: a jump is unwound into one transition per level, through the result states of
420 * the configurations with its cursors, so it reads the same in every jump mode. Its relation is the one of
421 * @c compose_fast_no_jump() on the operands with their jumps unwound in @p jump_mode (DONT_CARE kept), which is the
422 * relation of @c compose_general() whenever the operands have no EPSILON or DONT_CARE, no jump of @p jump_mode
423 * @c JumpMode::AppendDontCares reads DONT_CARE on a synchronization level, and, under @c JumpMode::RepeatSymbol,
424 * neither operand with a jump is composed with a one-level operand. The two compositions place EPSILON differently,
425 * and @c compose_general() matches DONT_CARE with EPSILON (#561) and lets an operand composed with a one-level
426 * operand choose its jump again inside it.
427 *
428 * @param[in] lhs First transducer to compose.
429 * @param[in] rhs Second transducer to compose.
430 * @param[in] lhs_sync_level The synchronization level of the @p lhs.
431 * @param[in] rhs_sync_level The synchronization level of the @p rhs.
432 * @param[in] project_out_sync_levels Whether we want to project out the synchronization levels.
433 * @param[in] jump_mode How a jump transition reads the levels after its first one. With @c JumpMode::NoJump, the
434 * operands must not have jump transitions other than fast EPSILON transitions, as in @c compose_fast_no_jump().
435 * @param[in] normalization How to normalize the result after the construction; see @c CompositionNormalization.
436 *
437 * @return A new NFT after the composition.
438 * @throws std::invalid_argument When a synchronization level is out of range for its operand, when the composition
439 * would leave no levels, under a jump mode other than @c JumpMode::NoJump when a transition goes to an earlier level
440 * of its round, and under @c JumpMode::NoJump when a transition goes to another level than the next one of its
441 * round and is no fast EPSILON transition, which takes one pass over the transitions.
442 */
443Nft compose_fast(
444 const Nft& lhs,
445 const Nft& rhs,
446 Level lhs_sync_level,
447 Level rhs_sync_level,
448 bool project_out_sync_levels,
449 JumpMode jump_mode,
450 CompositionNormalization normalization = CompositionNormalization::Auto
451);
