30/**
31 * @brief A class representing the structural part of an automaton.
32 * Every operation defined here is a pass over @c delta, @c initial and @c final.
33 *
34 * @tparam D The transition relation. Every other type is read off @p D rather than passed
35 * alongside it, so the two cannot disagree. @see @c mata::DeltaLike for what @p D must provide.
36 *
37 * @note A data-owning mixin, not a polymorphic base: it owns @c delta, @c initial and @c final, and
38 * has no virtual functions. The members that need their most-derived type take an explicit object
39 * parameter (`deducing this`) instead.
40 * @note @p D needs no keys: every operation here follows successors only, so the relation may be a plain graph.
41 * @note The largest @c State is reserved as "no state", for @c initial and @c final as much as for @c delta: the
42 * algorithms count states in @c State and use the largest one as a marker.
43 */
44template <DeltaLike D> class AutomatonBase {
45 public:
46 using DeltaType = D; ///< The transition relation this automaton is built on.
47 using State = typename D::State; ///< What indexes the automaton: @c delta, @c initial, @c final.
48 using Target = typename D::Target; ///< The leaf payload visiting the successors yields. @see @c State.
49 using StateSet = utils::OrdVector<State>;
50 using StateRenaming = std::unordered_map<State, State>;
51
52 D delta; ///< Transition relation of the automaton. delta[q] contains transitions from state q.
53 utils::SparseSet<State> initial{}; ///< Set of initial states of the automaton.
54 utils::SparseSet<State> final{}; ///< Set of final states of the automaton.
55
56 public:
57 /**
58 * @brief Construct a new AutomatonBase with optional @p delta, @p initial_states and @p final_states.
59 *
60 * @param[in] delta Transition relation of the automaton.
61 * @param[in] initial_states Set of initial states of the automaton.
62 * @param[in] final_states Set of final states of the automaton.
63 */
64 explicit AutomatonBase(
65 D delta = {}, utils::SparseSet<State> initial_states = {}, utils::SparseSet<State> final_states = {}
66 )
67 : delta(std::move(delta)),
68 initial(std::move(initial_states)),
69 final(std::move(final_states)) {}
70
71 /**
72 * @brief Construct a new AutomatonBase with @p num_of_states states and optionally @p initial_states and @p
73 * final_states.
74 *
75 * @param[in] num_of_states Number of states for which to preallocate Delta.
76 * @param[in] initial_states Set of initial states of the automaton.
77 * @param[in] final_states Set of final states of the automaton.
78 */
79 explicit AutomatonBase(
80 const size_t num_of_states,
81 utils::SparseSet<State> initial_states = {},
82 utils::SparseSet<State> final_states = {}
83 )
84 : delta(num_of_states),
85 initial(std::move(initial_states)),
86 final(std::move(final_states)) {}

88 AutomatonBase(const AutomatonBase& other) = default;

189 /**
190 * @brief Is the accepted language (or relation) empty?
191 *
192 * Structural, so every automaton has it, also one without runs. @see has_no_accepting_path().
193 * @return true iff no accepting path exists.
194 */
195 bool is_lang_empty() const { return has_no_accepting_path(); }

197 /**
198 * @brief Is the accepted language (or relation) empty, with a counter-example?
199 *
200 * @note Only for an automaton that has runs: the run type comes from @p Self.
201 * @note Unifying the two searches -- having the Tarjan search emit a witness on demand -- would
202 * save a traversal on the counter-example path, which @c is_included() and @c is_universal()
203 * take whenever the caller wants one. It would also stop the witness being *shortest*, and
204 * callers do depend on which one comes back (see the exact-word check in the
205 * "mata::nfa::Nfa::get_word()" tests). Worth doing only together with a decision about that.
206 * @param[out] cex Counter-example run, filled when an accepting path exists. With @c nullptr, none is built.
207 * @return true iff no accepting path exists.
208 */
209// Sphinx cannot parse an explicit object parameter, so Doxygen sees the call-site form.
210#ifdef BUILDING_DOX
211 template <AutomatonWithRuns Self> bool is_lang_empty(typename Self::Run* cex) const;
212#else
213 template <AutomatonWithRuns Self> bool is_lang_empty(this const Self& self, typename Self::Run* cex) {
214 if (cex == nullptr) { return self.has_no_accepting_path(); }
215 if (!self.find_accepting_path_(cex->path)) { return true; }
216 cex->word = self.get_word_for_path(*cex).first.word;
217 return false;
218 }
219#endif

259 /**
260 * @brief Remove unreachable and non-terminating states in-place.
261 * Remaining states are renumbered densely in ascending order.
262 *
263 * @note A state is reachable when the state is the endpoint of a path starting from an initial state.
264 * A state is terminating when the state is the starting point of a path ending in a final state.
265 * @note Reached only through a leaf's own one-line @c trim(), on @c *this, so an lvalue @p Self suffices here.
266 * @param[out] state_renaming Mapping of trimmed states to new states.
267 * @return @c this after trimming.
268 */
