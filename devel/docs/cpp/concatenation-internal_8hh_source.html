20/**
21 * @brief Connect the operands of an in-place concatenation whose rhs delta is already appended at @p offset.
22 *
23 * Every lhs final state takes over the transitions of every shifted rhs initial state. The posts of the rhs initial
24 * states are first united into one connector post, which each lhs final state then merges in with one sorted union
25 * per symbol, instead of one @c Delta::add() per target.
26 *
27 * @param[in] lhs_final The lhs final states; may be the final states of the automaton owning @p delta.
28 * @param[in] num_of_states The number of states of the concatenation, which the returned set covers.
29 * @return The final states of the concatenation: the shifted rhs final states, and the lhs final states when an rhs
30 * initial state is final.
31 */
32inline utils::SparseSet<State> connect_concatenation(
33 Delta& delta,
34 const utils::SparseSet<State>& lhs_final,
35 const utils::SparseSet<State>& rhs_initial,
36 const utils::SparseSet<State>& rhs_final,
37 const State offset,
38 const size_t num_of_states
39) {
40 utils::SparseSet<State> result_final{};
41 result_final.reserve(num_of_states);
42 for (const State rhs_final_state : rhs_final) { result_final.insert(offset + rhs_final_state); }
43
44 // With several rhs initial states, their (symbol, target) pairs are sorted at once: uniting the posts one by one
45 // costs a merge with the growing connector for each of them.
46 StatePost connector{};
47 bool rhs_initial_is_final{false};
48 if (rhs_initial.size() == 1) {
49 const State rhs_initial_state{*rhs_initial.begin()};
50 rhs_initial_is_final = rhs_final.contains(rhs_initial_state);
51 connector = delta[offset + rhs_initial_state];
52 } else {
53 std::vector<std::pair<Symbol, State>> moves{};
54 for (const State rhs_initial_state : rhs_initial) {
55 rhs_initial_is_final = rhs_initial_is_final || rhs_final.contains(rhs_initial_state);
56 for (const SymbolPost& symbol_post : delta[offset + rhs_initial_state]) {
57 for (const State target : symbol_post.targets) { moves.emplace_back(symbol_post.symbol, target); }
58 }
59 }
60 std::ranges::sort(moves);
61 const auto [duplicates_begin, duplicates_end]{std::ranges::unique(moves)};
62 moves.erase(duplicates_begin, duplicates_end);
63 for (const auto& [symbol, target] : moves) {
64 if (connector.empty() || connector.back().symbol != symbol) { connector.push_back(SymbolPost{symbol}); }
65 // The targets of one symbol come sorted and duplicate-free.
66 connector.back().push_back(target);
67 }
68 }
69
70 for (const State lhs_final_state : lhs_final) {
71 if (rhs_initial_is_final) { result_final.insert(lhs_final_state); }
72 for (const SymbolPost& symbol_post : connector) {
73 delta.add(lhs_final_state, symbol_post.symbol, std::span<const State>{symbol_post.targets});
74 }
75 }
76 return result_final;
77}
