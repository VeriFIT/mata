27/**
28 * @brief Transition with a source state, a key, and a target.
29 *
30 * The key field is spelled @c symbol. A relation that wants other names, or more
31 * than one key, passes its own traits to @c DeltaBase instead. @see DefaultTransitionTraits.
32 *
33 * @note `Transition{source, symbol, target}` builds one, and
34 * @c parts() takes it apart again in the same order.
35 * @tparam St The state type.
36 * @tparam K The key type.
37 * @tparam T The target type.
38 */
39template <typename St, typename K, typename T> struct Transition {
40 St source{}; ///< Source state.
41 K symbol{}; ///< Transition key, spelled for an NFA's sake.
42 T target{}; ///< Target.
43
44 auto operator<=>(const Transition&) const = default;
45
46 /**
47 * @brief Returns `(source, symbol, target)` as a tuple of references or values.
48 *
49 * @note References are used for large or non-trivially-copyable types, values otherwise.
50 * The @c utils::ArgOf decides.
51 * @param t The transition to take apart.
52 */
53 static auto parts(const Transition& t) noexcept {
54 return std::tuple<utils::ArgOf<St>, utils::ArgOf<K>, utils::ArgOf<T>>{t.source, t.symbol, t.target};
55 }
56
57 // Returns `"(source, symbol, target)"`, as the @c std::formatter below prints it.
58 std::string to_string() const { return std::format("{}", *this); }
59};

61/**
62 * @brief Transition with a source state, a tuple of keys, and a target.
63 *
64 * The key field is spelled @c keys. A relation that wants other names, or only one key,
65 * passes its own traits to @c DeltaBase instead. @see DefaultTransitionTraits.
66 *
67 * @note `KeyedTransition{source, keys, target}` builds one, and
68 * @c parts() takes it apart again in the same order.
69 * @tparam St The state type.
70 * @tparam Keys The tuple of key types.
71 * @tparam T The target type.
72 */
73template <typename St, typename Keys, typename T> struct KeyedTransition {
74 St source{}; ///< Source state.
75 Keys keys{}; ///< One key per post, outermost first.
76 T target{}; ///< Target.
77
78 auto operator<=>(const KeyedTransition&) const = default;
79
80 /**
81 * @brief Returns `(source, keys..., target)` as a tuple of references or values.
82 *
83 * @note References are used for large or non-trivially-copyable types, values otherwise.
84 * The @c utils::ArgOf decides.
85 * @param t The transition to take apart.
86 */
87 static auto parts(const KeyedTransition& t) noexcept {
88 // The keys too go through ArgOf: copying an owning key here could throw from inside the noexcept hash.
89 const auto by_arg = [](const auto&... key) {
90 return std::tuple<utils::ArgOf<std::remove_cvref_t<decltype(key)>>...>{key...};
91 };
92 return std::tuple_cat(
93 std::tuple<utils::ArgOf<St>>{t.source}, std::apply(by_arg, t.keys), std::tuple<utils::ArgOf<T>>{t.target}
94 );
95 }
96 /// `"(source, keys..., target)"`, as the @c std::formatter below prints it.
97 std::string to_string() const { return std::format("{}", *this); }
98};

105/**
106 * @brief Concept for a transition that has a static @c parts() returning a tuple of references or values.
107 *
108 * @note @c parts() must not throw: the hash built on it is @c noexcept, which lets the unordered containers skip
109 * storing a hash per element.
110 */
111template <typename T>
112concept TransitionLike = requires(const T& t) {
113 { T::parts(t) } noexcept;
114 requires internal::is_tuple<std::remove_cvref_t<decltype(T::parts(t))>>::value;
115};

117/**
118 * @brief Concept for a type that can be hashed with @c std::hash.
119 */
120template <typename T>
121concept Hashable = std::is_default_constructible_v<std::hash<T>>;

123/**
124 * The hash of a transition.
125 *
126 * @note The transition's @c parts() is used, so the hash is the same for any two transitions
127 * that have the same source, keys and target, even if they are different types.
128 */
129template <TransitionLike T> size_t hash_of(const T& t) noexcept {
130 return std::apply(
131 [](const auto&... v) {
132 size_t acc{0};
133 ((acc = utils::hash_combine(acc, v)), ...);
134 return acc;
135 },
136 T::parts(t)
137 );
138}

140/**
141 * @brief Formats a transition as `(source, keys..., target)`.
142 *
143 * @param out The output iterator to write to.
144 * @param t The transition to format.
145 * @return The output iterator after the last character written.
146 */
147template <TransitionLike T, typename Out> Out format_parts(Out out, const T& t) {
148 *out++ = '(';
149 std::apply(
150 [&](const auto& first, const auto&... rest) {
151 out = std::format_to(out, "{}", first);
152 ((out = std::format_to(out, ", {}", rest)), ...);
153 },
154 T::parts(t)
155 );
156 *out++ = ')';
157 return out;
158}

217/**
218 * @brief Traits for a transition type: how to build one and how to take it apart.
219 *
220 * A traits names a @c Type and builds transition with `make(source, keys..., target)`, one key per post.
221 * The type itself owes a static @c parts(), which is what lets the hash,
222 * the formatter and the transition-shaped members of @c DeltaBase work without knowing the type.
223 *
224 * Two are shipped. @c SymbolTransitionTraits yields the `{source, symbol, target}` triple and is the
225 * default at arity 1. @c KeyedTransitionTraits yields `{source, keys, target}` with the keys as a tuple
226 * and is the default above arity 1. @c DefaultTransitionTraits picks between them.
227 *
228 * A module that wants its own field names writes a traits and passes it as the second argument of
229 * @c DeltaBase. Nothing is specialised, so two modules over the *same* chain can name their fields
230 * differently without touching each other:
231 * ```cpp
232 * struct Transition {
233 * State source{}; Symbol input{}, output{}; State target{};
234 * bool operator==(const Transition&) const = default;
235 * static auto parts(const Transition& t) noexcept {
236 * using utils::ArgOf;
237 * return std::tuple<ArgOf<State>, ArgOf<Symbol>, ArgOf<Symbol>, ArgOf<State>>{
238 * t.source, t.input, t.output, t.target};
239 * }
240 * };
241 * struct TransitionTraits {
242 * using Type = Transition;
243 * static Type make(State s, Symbol in, Symbol out, State t) { return {s, in, out, t}; }
244 * };
245 * // Right after the type, before anything hashes or formats one:
246 * template <> struct std::hash<Transition> {
247 * size_t operator()(const Transition& t) const noexcept { return mata::posts::hash_of(t); }
248 * };
249 * template <> struct std::formatter<Transition> { ... return mata::posts::format_parts(ctx.out(), t); ... };
250 *
251 * class Delta : public mata::posts::DeltaBase<StatePost, TransitionTraits> { ... };
252 * ```
253 */
254///@{
255/// The `{source, symbol, target}` triple: one key, named as an NFA names it. The default at arity 1.
256template <typename P> struct SymbolTransitionTraits {
257 static_assert(arity_of<P> == 1, "a symbol-named transition has room for exactly one key");
258 using State = typename TargetTraits<target_of<P>>::State;
259 using Type = Transition<State, KeyOf<P, 0>, target_of<P>>;
260 static Type make(utils::ArgOf<State> source, utils::ArgOf<KeyOf<P, 0>> key, utils::ArgOf<target_of<P>> target) {
261 return Type{source, key, target};
262 }
263};
264
265/// The `{source, keys, target}` transition with one key per post. The default above arity 1.
266template <typename P> struct KeyedTransitionTraits {
267 using State = typename TargetTraits<target_of<P>>::State;
268 using Type = KeyedTransition<State, KeysOf<P>, target_of<P>>;
269 // `make(source, keys..., target)`. The target comes last, as in every keyed overload, so the keys
270 // are peeled off the front of the pack.
271 template <typename... Rest>
272 requires(sizeof...(Rest) == arity_of<P> + 1)
273 static Type make(utils::ArgOf<State> source, const Rest&... rest) {
274 return make_(source, std::forward_as_tuple(rest...), std::make_index_sequence<arity_of<P>>{});
275 }
276
277 private:
278 template <typename Tup, size_t... KeyIdx>
279 static Type make_(utils::ArgOf<State> source, const Tup& rest, std::index_sequence<KeyIdx...>) {
280 return Type{source, KeysOf<P>{std::get<KeyIdx>(rest)...}, std::get<sizeof...(KeyIdx)>(rest)};
281 }
282};

302/**
303 * @brief Iterator over the transitions of a relation.
304 *
305 * A non-owning view: it must not outlive the relation, and changing the relation invalidates its
306 * iterators. Transitions come by ascending source, then in the order of the relation's @c CursorType.
307 * Each is built on the fly with `TransitionTraits::make`, so `*it` refers to a value held by the
308 * iterator and is invalidated by `++it`.
309 *
310 */
311template <typename D> class DeltaTransitions {
312 using State = typename D::State;
313 using CursorType = typename D::CursorType;
314 using TransitionType = typename D::TransitionType;
315 using TT = typename D::TransitionTraits;
316
317 public:
318 /**
319 * Iterator over transitions.
320 *
321 * @note Inline, like @c PostMoves::const_iterator and for the same reason: as a nested class of
322 * a template, each out-of-line definition would need two layers of qualification.
323 */
324 class const_iterator {
325 private:
326 const D* delta_{nullptr};
327 State source_{};
328 typename CursorType::const_iterator cursor_{};
329 bool is_end_{true};
330 TransitionType transition_{};
331
332 void load() {
333 transition_ =
334 std::apply([&](const auto&... keys) { return TT::make(source_, keys..., *cursor_); }, cursor_.keys());
335 }
336 /// Position on the first transition leaving @c source_ or a later state, or become the end.
337 void seek() {
338 for (const size_t num_of_states{delta_->num_of_states()}; source_ < num_of_states; ++source_) {
339 cursor_ = CursorType{delta_->state_post(source_)}.begin();
340 if (cursor_ != std::default_sentinel) {
341 is_end_ = false;
342 load();
343 return;
344 }
345 }
346 is_end_ = true;
347 }
348
349 public:
350 using iterator_category = std::forward_iterator_tag;
351 using iterator_concept = std::forward_iterator_tag;
352 using value_type = TransitionType;
353 using difference_type = std::ptrdiff_t;
354 using pointer = const TransitionType*;
355 /// By value: a reference into the iterator would dangle once the copy that produced it is destroyed.
356 using reference = TransitionType;
357
358 const_iterator() = default; ///< The end.
359 explicit const_iterator(const D& delta) : const_iterator{delta, 0} {}
360 /// The first transition leaving @p from or a later state.
361 const_iterator(const D& delta, const State from) : delta_{&delta}, source_{from} { seek(); }
362
363 TransitionType operator*() const { return transition_; }
364 const TransitionType* operator->() const { return &transition_; }
365
366 const_iterator& operator++() {
367 MATA_ASSERT(!is_end_);
368 if (++cursor_ != std::default_sentinel) {
369 load();
370 return *this;
371 }
372 ++source_;
373 seek();
374 return *this;
375 }
376 const_iterator operator++(int) {
377 const const_iterator tmp{*this};
378 ++(*this);
379 return tmp;
380 }
381
382 bool operator==(const const_iterator& other) const {
383 if (is_end_ || other.is_end_) { return is_end_ == other.is_end_; }
384 return source_ == other.source_ && cursor_ == other.cursor_;
385 }
386 }; // class const_iterator.
387
388 /// An empty range, with no relation behind it.
389 DeltaTransitions() = default;
390 explicit DeltaTransitions(const D* delta) : delta_{delta} {}
391
392 const_iterator begin() const { return delta_ == nullptr ? end() : const_iterator{*delta_}; }
393 static const_iterator end() { return const_iterator{}; }
394
395 private:
396 const D* delta_{nullptr};
397}; // class mata::posts::DeltaTransitions.
398} // namespace mata::posts.
