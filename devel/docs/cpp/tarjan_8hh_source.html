23/// Calls @p callback unless it is @c nullptr or an empty @c std::function. Returns true iff it asks to stop.
24template <typename Callback, typename... Args> bool call_callback(const Callback& callback, const Args&... args) {
25 if constexpr (std::is_same_v<Callback, std::nullptr_t>) {
26 return false;
27 } else {
28 if constexpr (std::is_constructible_v<bool, const Callback&>) {
29 if (!static_cast<bool>(callback)) { return false; }
30 }
31 if constexpr (std::is_void_v<std::invoke_result_t<const Callback&, const Args&...>>) {
32 callback(args...);
33 return false;
34 } else {
35 return callback(args...);
36 }
37 }
38}

48template <typename Delta> struct TarjanNodeData {
49 using State = typename Delta::State;
50 using Cursor = decltype(std::declval<const Delta&>().successor_cursor(std::declval<State>()));
51 /// The iterator type SuccessorCursorLike checks: by value, from begin() on a const cursor.
52 using SuccessorIterator = CursorIterator<Cursor>;
53 SuccessorIterator current_successor_it{};
54 unsigned long index{0};
55 unsigned long lowlink{0};
56 bool initialized{false};
57 bool on_stack{false};
58
59 TarjanNodeData() = default;
60 TarjanNodeData(const State state, const Delta& delta, const unsigned long index)
61 : current_successor_it{first_successor(delta, state)},
62 index{index},
63 lowlink{index},
64 initialized{true},
65 on_stack{true} {}
66
67 private:
68 /// Calls the begin() that SuccessorCursorLike checks. The iterator outlives the cursor, as the concept requires.
69 static SuccessorIterator first_successor(const Delta& delta, const State state) {
70 const Cursor cursor{delta.successor_cursor(state)};
71 return cursor.begin();
72 }
73};

75/**
76 * Run the non-recursive Tarjan traversal over @p delta starting at @p initial_states.
77 *
78 * The SCC callback receives its members in Tarjan-stack order (bottom to top, i.e., the reverse of the order in which
79 * they are popped) and the remaining Tarjan stack. Both spans are valid only for the duration of that callback.
80 * @c nullptr callbacks are skipped at compile time. The traversal stops when a state- or SCC-level callback returns
81 * true.
82 */
83template <typename Delta, typename Callbacks>
84void tarjan_scc_discover(
85 const Delta& delta,
86 const size_t num_of_states,
87 const utils::SparseSet<typename Delta::State>& initial_states,
88 Callbacks&& callbacks
89) {
90 using State = typename Delta::State;
91 std::vector<TarjanNodeData<Delta>> node_info(num_of_states);
92 std::vector<State> program_stack{};
93 std::vector<State> tarjan_stack{};
94 program_stack.reserve(num_of_states);
95 tarjan_stack.reserve(num_of_states);
96 unsigned long index_cnt{0};
97 for (const State state : initial_states) { program_stack.push_back(state); }
98
99 while (!program_stack.empty()) {
100 const State act_state{program_stack.back()};
101 TarjanNodeData<Delta>& act_state_data{node_info[act_state]};
102 if (act_state_data.initialized && !act_state_data.on_stack) {
103 program_stack.pop_back();
104 continue;
105 }
106 if (!act_state_data.initialized) {
107 act_state_data = TarjanNodeData<Delta>(act_state, delta, index_cnt++);
108 tarjan_stack.push_back(act_state);
109 if (call_callback(callbacks.state_discover_, act_state)) { return; }
110 } else {
111 const State act_succ{Delta::state_of(*act_state_data.current_successor_it)};
112 act_state_data.lowlink = std::min(act_state_data.lowlink, node_info[act_succ].lowlink);
113 ++act_state_data.current_successor_it;
114 }
115
116 bool rec_call{false};
117 for (; act_state_data.current_successor_it != std::default_sentinel; ++act_state_data.current_successor_it) {
118 const State next_state{Delta::state_of(*act_state_data.current_successor_it)};
119 call_callback(callbacks.succ_state_discover_, act_state, next_state);
120 if (!node_info[next_state].initialized) {
121 program_stack.push_back(next_state);
122 rec_call = true;
123 break;
124 }
125 if (node_info[next_state].on_stack) {
126 act_state_data.lowlink = std::min(act_state_data.lowlink, node_info[next_state].index);
127 }
128 }
129 if (rec_call) { continue; }
130
131 if (act_state_data.lowlink == act_state_data.index) {
132 // The SCC is the top of the stack down to act_state; search from the top to stay O(|SCC|).
133 const auto act_state_it{std::find(tarjan_stack.rbegin(), tarjan_stack.rend(), act_state)};
134 const size_t stack_pos{static_cast<size_t>(tarjan_stack.rend() - act_state_it) - 1};
135 for (size_t i{tarjan_stack.size()}; i > stack_pos; --i) {
136 const State state{tarjan_stack[i - 1]};
137 node_info[state].on_stack = false;
138 call_callback(callbacks.scc_state_discover_, state);
139 }
140 const std::span<const State> stack_view{tarjan_stack};
141 if (call_callback(callbacks.scc_discover_, stack_view.subspan(stack_pos), stack_view.first(stack_pos))) {
142 return;
143 }
144 tarjan_stack.resize(stack_pos);
145 }
146 program_stack.pop_back();
147 }
148}
