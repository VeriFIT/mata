1/** @file
2 * @brief Allocator of the heap blocks of @c Delta: the symbol posts of a state post and the spilled target sets.
3 *
4 * The backend is @c std (global @c operator @c new) or @c pool (thread-local bump chunks with size-class free lists).
5 * The environment variable @c MATA_DELTA_ALLOCATOR selects it; when it is unset or empty, the CMake option of the
6 * same name chose the default when libmata was built. The selection is read on the first allocation and fixed for
7 * the rest of the process, so every block is released by the backend that allocated it. The backend lives out of
8 * line, so header code inlined into a consumer and code inside libmata always pair the same allocation and release.
9 */

29/// Stateless allocator over @c delta_allocate(): all instances are interchangeable.
30template <class T> struct DeltaAllocator {
31 static_assert(alignof(T) <= alignof(std::max_align_t));
32 using value_type = T;
33 using is_always_equal = std::true_type;
34
35 DeltaAllocator() noexcept = default;
36 // NOLINTNEXTLINE(google-explicit-constructor,hicpp-explicit-conversions): rebinding copies implicitly.
37 template <class U> DeltaAllocator(const DeltaAllocator<U>& /*other*/) noexcept {}
38 T* allocate(const std::size_t n) { return static_cast<T*>(delta_allocate(n * sizeof(T))); }
39 void deallocate(T* const ptr, const std::size_t n) noexcept { delta_deallocate(ptr, n * sizeof(T)); }
40
41 template <class U> bool operator==(const DeltaAllocator<U>& /*other*/) const noexcept { return true; }
42};
