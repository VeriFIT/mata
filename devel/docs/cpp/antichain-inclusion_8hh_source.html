17/// A pair (q, S) of a state of the smaller automaton and a macrostate of the bigger one, as a node of the search.

35/**
36 * @brief The macrostates already processed for one state of the smaller automaton, bucketed by cardinality.
37 *
38 * All entries stored here share the same state of the smaller automaton, so the dominance relation reduces to
39 * set inclusion between the macrostates of the bigger automaton. Cardinality alone then decides which entries
40 * can possibly be related to a candidate @c succ:
41 * - @c anti ⊆ @c succ requires @c |anti| ≤ @c |succ|, so the subsumption test reads nothing above @c |succ|;
42 * - @c succ ⊆ @c d requires @c |d| ≥ @c |succ|, so pruning reads nothing below @c |succ|;
43 * - at equal cardinality both reduce to equality, which is one comparison of two sorted vectors, and an entry
44 * equal to @c succ subsumes it, so pruning never has to consider that bucket.
45 * Inside a bucket the stored signature rejects most of the remaining pairs in O(1), before the node is read and
46 * the two sorted vectors are merged; see @c algorithms::macrostate_signature().
47 *
48 * Only cardinalities that actually hold an entry get a bucket, kept in increasing order: the macrostates of one
49 * run occupy few distinct cardinalities out of the @c |bigger| possible ones, so indexing by cardinality would
50 * pay for (and scan) a bucket header per unused size.
51 *
52 * Every node on the worklist is stored here as well, so marking the pruned nodes as dominated also removes them
53 * from the worklist, without scanning it.
54 *
55 * The stored nodes are read through @c bigger_set and @c signature only, so the HKC inclusion
56 * (@c algorithms::is_included_hkc()) indexes its pairs of macrostates with the same class.
57 */
58class ProcessedAntichain {
59 public:
60 /// Is some stored macrostate a subset of @p succ_set, and therefore dominates it?
61 template <class NodeT>
62 bool subsumes(const std::vector<NodeT>& nodes, const StateSet& succ_set, const uint64_t succ_signature) const {
63 const size_t cardinality{succ_set.size()};
64 for (const Bucket& bucket : buckets_) {
65 if (bucket.cardinality > cardinality) { break; }
66 if (bucket.cardinality == cardinality) {
67 // A subset of the same cardinality is the set itself.
68 return std::ranges::any_of(bucket.entries, [&](const Entry& anti) {
69 return anti.signature == succ_signature && nodes[anti.node].bigger_set == succ_set;
70 });
71 }
72 for (const Entry& anti : bucket.entries) {
73 if ((anti.signature & ~succ_signature) != 0) { continue; }
74 if (nodes[anti.node].bigger_set.is_subset_of(succ_set)) { return true; }
75 }
76 }
77 return false;
78 }

80 /// Removes every stored macrostate that @p succ_set dominates, that is, every strict superset of it, and
81 /// marks its node as dominated.
82 void prune(std::vector<Node>& nodes, const StateSet& succ_set, const uint64_t succ_signature) {
83 prune(nodes, succ_set, succ_signature, [](Node& node) {
84 node.dominated = true;
85 node.bigger_set = StateSet{};
86 });
87 }

89 /// Removes every stored macrostate that @p succ_set dominates and passes its node to @p on_pruned.
90 template <class NodeT, class OnPruned>
91 void prune(
92 std::vector<NodeT>& nodes, const StateSet& succ_set, const uint64_t succ_signature, OnPruned&& on_pruned
93 ) {
94 auto bucket{std::ranges::upper_bound(buckets_, succ_set.size(), {}, &Bucket::cardinality)};
95 while (bucket != buckets_.end()) {
96 std::erase_if(bucket->entries, [&](const Entry& stored) {
97 if ((succ_signature & ~stored.signature) != 0) { return false; }
98 NodeT& node{nodes[stored.node]};
99 if (!succ_set.is_subset_of(node.bigger_set)) { return false; }
100 on_pruned(node);
101 return true;
102 });
103 bucket = bucket->entries.empty() ? buckets_.erase(bucket) : bucket + 1;
104 }
105 }

107 template <class NodeT> void insert(const std::vector<NodeT>& nodes, const size_t node_idx) {
108 const NodeT& node{nodes[node_idx]};
109 const size_t cardinality{node.bigger_set.size()};
110 auto bucket{std::ranges::lower_bound(buckets_, cardinality, {}, &Bucket::cardinality)};
111 if (bucket == buckets_.end() || bucket->cardinality != cardinality) {
112 bucket = buckets_.insert(bucket, Bucket{.cardinality = cardinality, .entries = {}});
113 }
114 bucket->entries.push_back(Entry{.signature = node.signature, .node = node_idx});
115 }

131/**
132 * @brief Prepend to @p cex the symbols and the smaller states of the search steps that lead from an initial node to
133 * @p node_idx.
134 *
135 * @p cex holds the part of the counterexample after @p node_idx in reverse order (the last symbol first); on return it
136 * is the whole run up to there in reading order.
137 */
138inline void trace_back(const std::vector<Node>& nodes, const std::vector<Step>& steps, size_t node_idx, Run& cex) {
139 for (; steps[node_idx].parent != node_idx; node_idx = steps[node_idx].parent) {
140 cex.word.push_back(steps[node_idx].symbol);
141 cex.path.push_back(nodes[steps[node_idx].parent].smaller_state);
142 }
143 std::ranges::reverse(cex.word);
144 std::ranges::reverse(cex.path);
145}
