38/**
39 * @brief Implementation of a set of non-negative numbers using sparse-set.
40 *
41 * This class implements the interface of a set (similar to std::set) using sparse-set date structure,
42 * that is, a pair of vectors dense and sparse (... google it).
43 * Importantly
44 * - Insertion and removal are constant time.
45 * - Iteration is linear in the number of stored elements.
46 * - Only the sparse index allocates as many indexes as the maximal stored number; the dense array grows with the
47 * number of stored elements.
48 *
49 * @tparam Number Number type: type of numbers contained in the container.
50 */
51template <typename Number> class SparseSet {
52 static_assert(std::is_unsigned_v<Number>, "SparseSet can only contain unsigned integers");
53
54 private:
55 std::vector<Number> dense_{}; // Dense set of elements.
56 std::vector<Number> sparse_{}; // Map of elements to dense set indices.
57
58 /// Number of elements which are in the set (current size).
59 size_t size_ = 0;
60
61 /// @brief Over-approximation of Number in sparse set. It represents the Numbers which were in the set throughout
62 /// the
63 /// life of the set until now.
64 ///
65 /// `truncate()` will update the domain size to the current maximal Number + 1.
66 /// Constructor can preallocate the sparse set and set domain size to the requested value.
67 /// The sparse index is preallocated to at least @c domain_size_ size; the dense array tracks @c size_ instead.
68 size_t domain_size_ = 0;
69
70 public:
71 using iterator = typename std::vector<Number>::const_iterator;
72 using const_iterator = typename std::vector<Number>::const_iterator;
73
74 iterator begin() { return dense_.begin(); }
75
76 const_iterator begin() const { return dense_.begin(); }
77
78 iterator end() { return dense_.begin() + static_cast<long>(size_); }
79
80 const_iterator end() const { return dense_.begin() + static_cast<long>(size_); }
81
82 size_t size() const { return size_; }
83 size_t domain_size() const { return domain_size_; }
84
85 bool empty() const { return size_ == 0; }
86
87 void clear() {
88 size_ = 0;
89 dense_.clear();
90 }
91
92 // TODO: maybe we could reserve space more efficiently, by something as doubling?
93 // But we should not create havoc with domain_size, which is used outside, namely for determining the states of an
94 // automaton.
95 void reserve(size_t u) {
96 if (u > domain_size_) {
97 sparse_.resize(u, 0);
98 domain_size_ = u;
99 }
100 MATA_ASSERT(consistent());
101 }
102
103 bool contains(const Number val) const {
104 return val < domain_size_ && sparse_[val] < size_ && dense_[sparse_[val]] == val;
105 }
106
107 void insert(const Number val) {
108 MATA_ASSERT(consistent());
109
110 if (!contains(val)) {
111 if (static_cast<size_t>(val) >= domain_size_) { reserve(static_cast<size_t>(val) + 1); }
112 dense_.push_back(val);
113 sparse_[val] = static_cast<Number>(size_);
114 ++size_;
115 }
116
117 MATA_ASSERT(consistent());
118 }
119
120 /**
121 * Erase @p number from the set without checking for its existence.
122 * @param number Number to be erased.
123 * @pre @p number exists in the set.
124 */
125 void erase_nocheck(const Number number) {
126 dense_[sparse_[number]] = dense_[size_ - 1];
127 sparse_[dense_[size_ - 1]] = sparse_[number];
128 --size_;
129 dense_.pop_back();
130 }
131
132 void erase(const Number val) {
133 if (contains(val)) { erase_nocheck(val); }
134 }

158 /// Construct from any range of numbers convertible to @c Number.
159 ///
160 /// A forwarding reference keeps ranges whose @c begin()/@c end() are callable
161 /// on a non-const object only (e.g. a named @c std::views::filter result) usable.
162 /// Elements are converted from the references the range yields, built-in arrays work through range access, and
163 /// the sentinel may differ from the iterator type.
164 /// @c SparseSet and @c BoolVector are excluded so that the copy/move and
165 /// bit-index constructors win over this greedy template.
166 template <class T>
167 requires std::ranges::input_range<T> && std::convertible_to<std::ranges::range_reference_t<T>, Number> &&
168 (!std::same_as<std::remove_cvref_t<T>, SparseSet>) &&
169 (!std::same_as<std::remove_cvref_t<T>, BoolVector>)
170 explicit SparseSet(T&& container) {
171 // The sentinel may have a type of its own: one combined declaration would require a common type.
172 const auto end{std::ranges::end(container)};
173 for (auto it{std::ranges::begin(container)}; it != end; ++it) {
174 const Number element = *it;
175 insert(element);
176 }
177 MATA_ASSERT(consistent());
178 }
179
180 // This is actually weird, what if the vector is not boolean values but values themselves?
181 explicit SparseSet(const BoolVector& bv) {
182 reserve(bv.size());
183 for (size_t i = 0; i < bv.size(); i++) {
184 if (bv[i]) { insert(i); }
185 }
186 MATA_ASSERT(consistent());
187 }

215 /// Compare the contents: the sparse index keeps stale slots, and the order of the dense array and the domain size
216 /// depend on the insertion history, so equal sets may differ member-wise.
217 bool operator==(const SparseSet<Number>& other) const {
218 if (size_ != other.size_) { return false; }

255 // Ranges over our own dense array are erased back to front: erasing position i moves the last element into i, so
256 // positions below i keep their elements while iterators into the popped tail would be invalidated.
257 template <Iterable T> void erase(const T& set) {
258 const auto first{std::ranges::begin(set)};
259 const auto last{std::ranges::end(set)};
260 const auto address = [](const auto& iterator) -> const void* {
261 if constexpr (std::contiguous_iterator<std::remove_cvref_t<decltype(iterator)>>) {
262 return std::to_address(iterator);
263 } else {
264 return nullptr;
265 }
266 };
267 const auto first_address{reinterpret_cast<std::uintptr_t>(address(first))};
268 const auto last_address{reinterpret_cast<std::uintptr_t>(address(last))};
269 const auto dense_begin{reinterpret_cast<std::uintptr_t>(dense_.data())};
270 const auto dense_end{reinterpret_cast<std::uintptr_t>(dense_.data() + dense_.size())};
271 if (first_address >= dense_begin && first_address <= dense_end && last_address >= dense_begin &&
272 last_address <= dense_end) {
273 const auto first_index{(first_address - dense_begin) / sizeof(Number)};
274 for (auto index{(last_address - dense_begin) / sizeof(Number)}; index > first_index;) {
275 --index;
276 erase_nocheck(dense_[index]);
277 }
278 MATA_ASSERT(consistent());
279 } else {
280 erase(first, last);
281 }
282 }

292 /*
293 * Complements the set with respect to a given number of elements = the maximum number + 1.
294 */
295 void complement(const size_t new_domain_size) {
296 const size_t old_domain_size = domain_size_;
297 for (size_t i = 0; i < new_domain_size; ++i) {
298 const auto number{static_cast<Number>(i)};
299 if (contains(number)) {
300 erase_nocheck(number);
301 } else {
302 insert(number);
303 }
304 }
305 for (size_t i = new_domain_size; i < old_domain_size; ++i) { erase(static_cast<Number>(i)); }
306
307 MATA_ASSERT(consistent());
308 }

310 template <typename F> void filter(F is_staying) {
311 for (size_t i{0}; i < size_; ++i) {
312 while (i < size_ && !is_staying(dense_[i])) { erase(dense_[i]); }
313 }

315 MATA_ASSERT(consistent());

318 void sort() {
319 std::sort(dense_.begin(), dense_.begin() + static_cast<long>(size_));

325 template <typename F> void rename(F renaming) {
326 for (size_t i = 0; i < size_; ++i) {
327 if (dense_[i] != renaming(dense_[i])) {
328 dense_[i] = renaming(dense_[i]);
329 if (dense_[i] >= domain_size_) { reserve(static_cast<size_t>(dense_[i]) + 1); }
330 sparse_[dense_[i]] = static_cast<Number>(i);
331 }
332 }
333
334 MATA_ASSERT(consistent());
335 }

337 /// @brief Maximal Number in set.
338 ///
339 /// Expensive operation as it has to compute the maximal Number in linear time. The index is a @c size_t: a full
340 /// @c SparseSet<uint8_t> holds 256 elements, which a @c Number index could never reach.
341 Number max() const {
342 Number max = 0;
343 for (size_t i = 0; i < size_; ++i) {
344 if (max < dense_[i]) { max = dense_[i]; }
345 }
346 return max;
347 }
348
349 void truncate() {
350 if (size_ == 0) {
351 domain_size_ = 0;
352 } else {
353 domain_size_ = max() + 1;
354 }
355
356 MATA_ASSERT(consistent());
357 }
