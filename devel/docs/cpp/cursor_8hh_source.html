18/**
19 * @brief A generic resumable cursor over every target reachable from one state post.
20 *
21 * Flattens the descent down the post chain into a single pointer sweep,
22 * so a position can be stored and continued later.
23 *
24 * @tparam P The whole post chain, not one post of it.
25 */
26template <typename P, size_t Arity = P::key_arity> class SuccessorCursor {
27 template <size_t I> using PostAt = typename posts::PostAt<P, I>::type;
28
29 /// The position in post @p I. One base per post keeps the iterator flat, outermost post first.
30 template <size_t I> struct Position {
31 typename PostAt<I>::const_iterator it{}, end{};
32 bool operator==(const Position&) const = default;
33 };
34 template <typename> struct Positions;
35 template <size_t... KeyIdx> struct Positions<std::index_sequence<KeyIdx...>> : Position<KeyIdx>... {
36 bool operator==(const Positions&) const = default;
37 };
38
39 // Note: Having one cursor struct per post, each delegating downwards is slower
40 // by 24.4% at arity 2 and 18.6% at arity 3 against hand-written nested loops on the same shapes.
41 public:
42 using Target = typename P::Target;
43
44 class const_iterator : Positions<std::make_index_sequence<Arity>> {
45 public:
46 const Target *target_it_{nullptr}, *target_end_{nullptr};
47
48 template <size_t I> Position<I>& at() { return *this; }
49 template <size_t I> const Position<I>& at() const { return *this; }
50
51 /**
52 * @brief Scan post @p I for the next target, descending into post `I + 1` as needed.
53 *
54 * @tparam I The post index to scan.
55 * @return true if a target was found, false if post @p I is exhausted.
56 */
57 template <size_t I> bool scan() {
58 auto& [cur, last]{at<I>()};
59 for (; cur != last; ++cur) {
60 if constexpr (I + 1 == Arity) {
61 const std::span<const Target> targets{cur->target_span()};
62 if (!targets.empty()) {
63 target_it_ = targets.data();
64 target_end_ = target_it_ + targets.size();
65 return true;
66 }
67 } else {
68 at<I + 1>().it = cur->nested().begin();
69 at<I + 1>().end = cur->nested().end();
70 if (scan<I + 1>()) { return true; }
71 }
72 }
73 return false;
74 }
75
76 /// Advance post @p I and rescan. When it is exhausted, carry into post `I - 1`.
77 template <size_t I> void carry() {
78 ++at<I>().it;
79 if (scan<I>()) { return; }
80 if constexpr (I > 0) {
81 carry<I - 1>();
82 } else {
83 exhaust();
84 }
85 }
86
87 /// Exhaust the cursor: no more targets to yield.
88 void exhaust() { target_it_ = target_end_ = nullptr; }
89
90 const Target& operator*() const { return *target_it_; }
91 const_iterator& operator++() {
92 if (++target_it_ == target_end_) { carry<Arity - 1>(); }
93 return *this;
94 }
95 bool operator==(std::default_sentinel_t) const { return at<0>().it == at<0>().end; }
96 bool operator==(const const_iterator&) const = default;
97
98 /// Returns the key path to the current target, outermost first.
99 auto keys() const { return keys(std::make_index_sequence<Arity>{}); }
100
101 private:
102 template <size_t... KeyIdx> auto keys(std::index_sequence<KeyIdx...>) const {
103 return std::tuple<typename PostAt<KeyIdx>::Key...>{at<KeyIdx>().it->key()...};
104 }
105 };
106
107 explicit SuccessorCursor(const P& post) : post_{&post} {}
108
109 const_iterator begin() const {
110 const_iterator it;
111 it.template at<0>().it = post_->begin();
112 it.template at<0>().end = post_->end();
113 if (!it.template scan<0>()) { it.exhaust(); }
114 return it;
115 }
116 std::default_sentinel_t end() const { return std::default_sentinel; }
117
118 private:
119 const P* post_;
120};

122/**
123 * @brief Specialization of @c SuccessorCursor for arity 1.
124 *
125 * @note The generic version is slower at arity 1 (on come cases by up to 24.4%).
126 */
127template <typename P> class SuccessorCursor<P, 1> {
128 public:
129 using Target = typename P::Target;
130
131 /// @cond
132 class const_iterator {
133 public:
134 typename P::const_iterator symbol_post_it_{}, symbol_post_end_{};
135 const Target *target_it_{nullptr}, *target_end_{nullptr};
136
137 /// Load the targets of the current symbol post into the target iterator.
138 void load_targets() {
139 const std::span<const Target> targets{symbol_post_it_->target_span()};
140 target_it_ = targets.data();
141 target_end_ = target_it_ + targets.size();
142 }
143
144 /// Seek the first target of a later symbol post, skipping symbol posts without targets.
145 /// @note Returning from inside the loop keeps the common case (the next symbol post has targets) free of jumps
146 /// back; a loop testing for an empty post first is up to 8% slower when iterating over large relations.
147 void seek() {
148 while (++symbol_post_it_ != symbol_post_end_) {
149 load_targets();
150 if (target_it_ != target_end_) [[likely]] { return; }
151 }
152 target_it_ = target_end_ = nullptr;
153 }
154
155 const Target& operator*() const { return *target_it_; }
156 const_iterator& operator++() {
157 if (++target_it_ == target_end_) { seek(); }
158 return *this;
159 }
160 /// Compares against @c std::default_sentinel: the end is stateless, so no end iterator is stored.
161 bool operator==(std::default_sentinel_t) const { return symbol_post_it_ == symbol_post_end_; }
162 bool operator==(const const_iterator&) const = default;
163
164 /// Returns the key path to the current target, outermost first.
165 auto keys() const { return std::tuple<typename P::Key>{symbol_post_it_->key()}; }
166 };
167 /// @endcond
168
169 explicit SuccessorCursor(const P& state_post) : state_post_{&state_post} {}
170
171 const_iterator begin() const {
172 const_iterator it;
173 it.symbol_post_it_ = state_post_->begin();
174 it.symbol_post_end_ = state_post_->end();
175 if (it.symbol_post_it_ == it.symbol_post_end_) { return it; }
176 it.load_targets();
177 if (it.target_it_ == it.target_end_) [[unlikely]] { it.seek(); }
178 return it;
179 }
180 std::default_sentinel_t end() const { return std::default_sentinel; }
181
182 private:
183 const P* state_post_;
184};
