74/**
75 * @brief Get the longest words (regarding their length) of the automaton.
76 *
77 * The language has longest words iff it is finite, i.e., iff no cycle passes through a useful state (a state reachable
78 * from an initial state that reaches a final state). Cycles through useless states are allowed. Every symbol,
79 * epsilon included, counts as one letter, as in @c get_shortest_words().
80 *
81 * Costs O(|Q| + |delta|) to compute the length of the longest words from every state, and then a depth-first search
82 * over macrostates that follows only the transitions on some longest accepting path, so every branch of the search
83 * ends in a word to report and every word is reported once, however many accepting paths spell it.
84 *
85 * @return Set of the longest words (empty if the language is empty), or @c std::nullopt if the language is infinite.
86 */
87std::optional<std::set<Word>> get_longest_words(const Nfa& nfa);

303 /**
304 * Breadth-first queue of state and depth pairs; each state is queued at most once.
305 *
306 * The first time a state is queued is the time it is processed: a FIFO queue pops the first copy of a state before
307 * any later one, so skipping the later copies at push keeps the order and the depths of the processed states.
308 */
309 struct Worklist {
310 std::deque<StateDepthTuple> queue{};
311 BoolVector queued{};
312
313 void push(const StateDepthTuple& state_depth_pair) {
314 if (queued[state_depth_pair.state]) { return; }
315 queued[state_depth_pair.state] = true;
316 queue.push_back(state_depth_pair);
317 }
318 };

353/**
354 * @brief A segment of a noodle: the part of a segment automaton between one of its initial states and one of its final
355 * states.
356 *
357 * @c automaton is the reduced part of the segment automaton that reaches the final state, shared by all segments that
358 * end in that final state; the segment is the language of @c initial in it. The initial states of @c automaton itself

360 * one automaton per final state instead of one per pair of an initial and a final state.

368 /// Whether the language of the segment is {ε}.
369 bool is_epsilon_only() const;
370
371 bool operator==(const NoodleSegment&) const = default;
372};

557/**
558 * @brief The order in which the noodlify_*_lazy() functions emit the noodles.
559 *
560 * The search is depth-first over the segments; at every node it orders the candidates for the next segment (the pairs
561 * of an initial and a final state of it). The order is greedy: a candidate ranked first leads to the next noodle, not
562 * necessarily to the best noodle overall.
563 */
564struct NoodleOrder {
