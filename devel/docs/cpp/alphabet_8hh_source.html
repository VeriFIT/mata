24/// The name under which @c EPSILON is printed in the .mata format and in DOT, and which the .mata parser maps back to
25// @c EPSILON. The name is reserved: alphabets map it straight to @c EPSILON and never register it as an ordinary
26// symbol, so it cannot collide with a translated symbol.
27constexpr std::string_view EPSILON_NAME{"<eps>"};
28/// The name under which the NFT @c DONT_CARE symbol is printed in the .mata format and in DOT, and which the .mata
29/// parser
30// maps back to @c DONT_CARE. Reserved the same way as @c EPSILON_NAME.
31constexpr std::string_view DONT_CARE_NAME{"<dntcr>"};
32
33/// Whether @p name is a reserved special-symbol name (@c EPSILON_NAME or @c DONT_CARE_NAME) that
34/// alphabets must not register as an ordinary symbol.
35constexpr bool is_reserved_symbol_name(const std::string_view name) {
36 return name == EPSILON_NAME || name == DONT_CARE_NAME;
37}

39/**
40 * @brief Returns @p name written as a single token of the .mata format.

43 * (@c "#@%()!&|\\"), writing an inner @c " as @c \", the only escape the reader knows.

459 /**
460 * @brief Try to add symbol to the alphabet map.
461 *
462 * Does not throw an exception when the adding fails. Throws on a reserved special-symbol name (@c EPSILON_NAME or
463 * @c DONT_CARE_NAME); those names belong to no alphabet.
464 *
465 * @param[in] key User-space representation of the symbol.
466 * @param[in] value Number of the symbol to be used on transitions.
467 * @return Result of the insertion as @c InsertionResult.
468 */
469 InsertionResult try_add_new_symbol(const std::string& key, Symbol value) {
470 if (is_reserved_symbol_name(key)) {
471 throw std::runtime_error{"Cannot register the reserved special-symbol name '" + key + "'."};

474 if (insertion_result.second) { index_name(insertion_result.first); }

505 /// Sorted symbols of @c symbol_map_, built on the first read after the set of symbols changes. Concurrent const
506 /// reads stay race-free: only a build takes @c mutex, and @c valid publishes the finished build to the readers.
507 struct SortedSymbols {
508 utils::OrdVector<Symbol> symbols{};
509 std::atomic<bool> valid{false};
510 std::mutex mutex{};
511
512 SortedSymbols() = default;
513 SortedSymbols(SortedSymbols&& other) noexcept
514 : symbols{std::move(other.symbols)},
515 valid{other.valid.exchange(false, std::memory_order_relaxed)} {}
516 SortedSymbols& operator=(SortedSymbols&& other) noexcept {
517 if (this != &other) {
518 symbols = std::move(other.symbols);
519 valid.store(other.valid.exchange(false, std::memory_order_relaxed), std::memory_order_relaxed);
520 }
521 return *this;
522 }
523 ~SortedSymbols() = default;

525 SortedSymbols& operator=(const SortedSymbols&) = delete;

609/**
610 * @brief Per-level alphabets for transducer-like automata.
611 *
612 * Standalone (does NOT inherit from @c Alphabet) wrapper holding a vector of @c std::shared_ptr<Alphabet>, one per
613 * level.
614 * Operates in one of two modes (see @c Mode):
615 * - @c Global — a single shared alphabet (typically stored in @c alphabets[0]) applies to every level. The level
616 * argument is ignored.
617 * - @c MultiLevel — distinct per-level alphabets. The level argument is required and indexes @c alphabets.
618 *
619 * The underlying @c Alphabet objects are held via @c std::shared_ptr and can be shared between multiple
620 * @c AlphabetLevels instances (and other automata).
621 */
622class AlphabetLevels {
623 public:
624 /// Operating mode of the @c AlphabetLevels instance.
625 enum class Mode : std::uint8_t {
626 Global, ///< @c alphabets[0] is used for every level; level argument is ignored.
627 MultiLevel, ///< Each level uses its own alphabet from @c alphabets[level]; level argument is required.
628 };
629
630 explicit AlphabetLevels(std::vector<std::shared_ptr<Alphabet>> alphabets = {}, Mode mode = Mode::MultiLevel)
631 : alphabets_{std::move(alphabets)},
632 mode_{mode} {
633 check_can_grow_to_(alphabets_.size());
634 }
635
636 /**
637 * @brief Use the same @p alphabet for every level (Global mode).
