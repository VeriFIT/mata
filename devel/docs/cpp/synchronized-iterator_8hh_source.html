16/** @page synchronized_iterator Synchronized Iterator
17 * Two classes that provide "synchronized" iterators through a vector of ordered vectors,
18 * (or of some ordered OrdContainer that have a similar iterator),
19 * needed in computation of post
20 * in subset construction, product, and non-determinization.
21 *
22 * The Type stored in OrdContainers must be comparable with `<`,`>`,`==`,`!=`,`<=`,`>=`,
23 * and it must be a total (linear) ordering.
24 * The intended usage in, for instance, determinisation is for Type to be TransSymbolStates.
25 * TransSymbolStates is ordered by the symbol.
26 *
27 * SynchronisedIterator is the base class holding the data shared by both synchronized iterators. The hierarchy is
28 * non-polymorphic on purpose: every user instantiates one of the two derived classes directly, so the base needs no
29 * vtable.
30 * It stores a vector of end-iterators for the OrdContainer v and a vector of current positions.
31 * They are filled in using the function push_back(begin,end), that adds begin and end iterators of v to positions and
32 * ends, respectively.
33 * Method advance advances all positions forward so that they are synchronized on the next smallest equiv class
34 * (next smallest symbol in the case of TransSymbolStates).
35 *
36 * There are two versions of the class.
37 * i) In product, ALL positions must point to currently the smallest equiv. class (Moves with the same symbol).
38 * Method get_current then returns the vector of all position iterators, synchronized.
39 * ii) In determinization, it is enough that there EXISTS a position that points to the smallest class.
40 * Method get_current then returns the vector of only those positions that point to the smallest equiv. class.
41 *
42 * Usage: 0) construct, 1) fill in using @c SynchronizedIterator::push_back(), iterate using
43 * @c SynchronizedIterator::advance() and @c SynchronizedIterator::get_current(), 2) @c SynchronizedIterator::reset()
44 * and go back to 1).
45 *
46 * @note The memory allocated internally for positions and ends is kept after @c SynchronizedIterator::reset(), so it is
47 * advisable to use the same iterator for many iterations, as opposed to creating a new one for each iteration.
48 */
49

93template <typename Iterator> class SynchronizedUniversalIterator : public SynchronizedIterator<Iterator> {
94 public:
95 /** "minimum" would be the smallest class bounded from below by all positions that appears in all OrdContainers.
96 * Are we sure that all positions at this class?
97 * Invariant: it can be true only if all positions are indeed synchronized.
98 */
99 bool synchronized_at_current_minimum = false;
100
101 /**
102 * Advances all positions to the NEXT minimum and returns true (though the next minimum might be the current state
103 * if synchronized_at_current_minimum is false), or returns false if positions cannot be synchronized.
104 *
105 * If positions are synchronized to start with, then synchronized_at_current_minimum decides whether to stay or
106 * advance further.
107 * The general of the algorithm is to synchronize everybody with position[0].
108 *
109 * @pre At least one input was pushed since construction or the last @c reset() (asserted in debug builds);
110 * universal synchronization over no inputs is not meaningful. In release builds this keeps returning false,
111 * and both callers (NFT intersection in @c src/nft/intersection.cc and composition in
112 * @c src/nft/composition.cc) push two inputs anyway.
113 */
114 bool advance() {
115 MATA_ASSERT(!this->positions.empty());
116 // Nothing to synchronize: there is no position to return.
117 if (this->positions.empty()) { return false; }
118
119 // If already synchronized, start moving forward by advancing position[0] (and so break synchronization).
120 if (this->synchronized_at_current_minimum) {
121 ++this->positions[0];
122 synchronized_at_current_minimum = false;
123 }
124
125 // If positions[0] has nowhere to go, then sync is not possible.
126 if (this->positions[0] == this->ends[0]) { return false; }
127
128 // Synchronise all positions with position[0].
129 for (size_t i = 1, positions_size = this->positions.size(); i < positions_size; ++i) {
130 // If some positions has nowhere to go, then sync is not possible.
131 if (this->positions[i] == this->ends[i]) { return false; }
132
133 // Advance position[i] and position[0] to the closest equal values.
134 // Remember where position[0] was: when it moves, every earlier position has to be re-synchronized.
135 // Only position[0] is compared with its own earlier value; comparing positions of different containers
136 // (the former 'positions[0] > positions[1]') depended on where the containers lay in memory.
137 const Iterator position_0_before{this->positions[0]};
138 while (*this->positions[i] != *this->positions[0]) {
139 // Advance position[i] to or beyond position[0].
140 while (*this->positions[i] < *this->positions[0]) {
141 ++this->positions[i];
142 if (this->positions[i] == this->ends[i]) { return false; }
143 }
144
145 // Advance position[0] to or beyond position[i].
146 while (*this->positions[i] > *this->positions[0]) {
147 ++this->positions[0];
148 if (this->positions[0] == this->ends[0]) { return false; }
149 }
150
151 // If position[0] moved, start from position 1 again (i is incremented by the for-loop).
152 if (this->positions[0] != position_0_before) {
153 i = 0;
154 break;
155 }
156 }
157 }
158 this->synchronized_at_current_minimum = true;
159 return true;
160 }
161
162 /// Returns the vector of current positions.
163 const std::vector<Iterator>& get_current() const { return this->positions; }
164
165 explicit SynchronizedUniversalIterator(const size_t size = 0) : SynchronizedIterator<Iterator>(size) {}
166
167 void reset(const size_t size = 0) {
168 SynchronizedIterator<Iterator>::reset(size);
169 this->synchronized_at_current_minimum = false;
170 }
171}; // class SynchronizedUniversalIterator.

173template <typename Iterator> class SynchronizedExistentialIterator : public SynchronizedIterator<Iterator> {
174 public:
175 Iterator get_current_minimum() {
176 if (currently_synchronized.empty()) {
177 throw std::runtime_error(
178 "Trying to get minimum from sync. ex. iterator which has no minimum. Don't do "
179 "that ever again or your nose falls off!"
180 );
181 }
182 return currently_synchronized[0];
183 }
184
185 std::vector<Iterator> currently_synchronized{}; // Positions that are currently synchronized.
186 Iterator next_minimum{}; // The value we should synchronise on after the first next call of advance().
187
188 bool is_synchronized() const { return !currently_synchronized.empty(); }
189
190 /**
191 * Advances all positions just above current_minimum,
192 * that is, to or above next_minimum.
193 * Those at next_minimum are added to currently_synchronized.
194 * Since next_minimum becomes the current minimum,
195 * new next_minimum must be updated too.
196 */
197 bool advance() {
198 // The next_minimum becomes the current current_minimum.
199 auto current_minimum = this->next_minimum;
200
201 // Here we collect the result.
202 currently_synchronized.clear();
203
204 for (size_t i = 0, positions_size = this->positions.size(); i < positions_size;) {
205 Iterator& position_i_it{this->positions[i]};
206 Iterator& end_i_it{this->ends[i]};
207 if (position_i_it == end_i_it) {
208 // If there is nothing left at the position, it is removed, that is,
209 // swapped with a position from the end of the vector,
210 // and the vector is shortened.
211 // The same is done with ends.
212 // Check the bound first: once i == positions_size, the slot at i was popped.
213 while (i < positions_size && position_i_it == end_i_it) {
214 position_i_it = this->positions[positions_size - 1];
215 end_i_it = this->ends[positions_size - 1];
216 // This must be here, because in the next call, position_size is set again with positions.size().
217 this->positions.pop_back();
218 this->ends.pop_back();
219 --positions_size;
220 }
221 }
222 // If the i-th position, i.e., where we are now, was erased,
223 // then we reached the end of active positions.
224 if (i == positions_size) { return !currently_synchronized.empty(); }
225
226 // If we are at the current_minimum, then save it and advance the position.
227 if (*position_i_it == *current_minimum) {
228 this->currently_synchronized.emplace_back(position_i_it);
229 ++position_i_it;
230 continue;
231 }
232
233 // If the position (now larger than current_minimum) is smaller than next_minimum,
234 // or next_minimum has not yet been updated, then update the next_minimum.
235 if (*this->next_minimum > *position_i_it || *this->next_minimum == *current_minimum) {
236 this->next_minimum = position_i_it;
237 }
238
239 ++i; // This cannot be in the for statement line, because of the continue in the if body above.
240 }
241 return !currently_synchronized.empty();
242 }
243
244 /**
245 * @brief Returns the vector of current still active positions.
246 *
247 * Beware, they will be ordered differently from how there were input into the iterator.
248 * This is due to swapping of the emptied positions with positions at the end.
249 */
250 const std::vector<Iterator>& get_current() const { return this->currently_synchronized; }
251
252 void push_back(const Iterator& begin, const Iterator& end) {
253 // Empty vector would not have any effect (unlike in the case of the universal iterator).
254 if (begin == end) { return; }
255
256 // Initialise next_minimum as the first position at the first vector.
257 if (this->positions.empty()) {
258 this->next_minimum = begin;
259 } else if (*this->next_minimum > *begin) {
260 // If the first position is of the new vector is smaller than minimum, update minimum.
261 this->next_minimum = begin;
262 }
263
264 // Let position point to the beginning the vector,
265 // save the end of the vector.
266 this->positions.emplace_back(begin);
267 this->ends.emplace_back(end);
268 }
269
270 explicit SynchronizedExistentialIterator(const size_t size = 0) : SynchronizedIterator<Iterator>(size) {
271 this->currently_synchronized.reserve(size);
272 }
273
274 void reset(const size_t size = 0) {
275 SynchronizedIterator<Iterator>::reset(size);
276 if (size > 0) { this->currently_synchronized.reserve(size); }
277 this->currently_synchronized.clear();
278 }
279};

281/**
282 * In order to make initialization of the sync. iterator nicer than inputting v.begin() and v.end()
283 * as the two parameters of the method push_back,
284 * this function wraps the method push_back,
285 * takes the iterator and v and extracts the v.begin() and v.end() from v.
286 *
287 * @note Templated on the concrete iterator type: 'SynchronizedExistentialIterator::push_back()' does more than the one
288 * in the base class, and the hierarchy is not polymorphic, so the base type must not appear in the signature.
289 */
290template <class SynchronizedIteratorType, class Container>
291void push_back(SynchronizedIteratorType& i, const Container& container) {
292 i.push_back(container.begin(), container.end());
293}
