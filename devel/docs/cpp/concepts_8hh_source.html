24/**
25 * @brief A concept for an alphabet that can be *extended* with symbols it has not seen.
26 *
27 * Only some alphabets can: a fixed @c EnumAlphabet cannot grow, an @c OnTheFlyAlphabet can.
28 * @c mata::posts::DeltaBase::add_keys_to() needs the growing kind.
29 *
30 * The symbol type is the alphabet's own @c Symbol alias, which @c mata::Alphabet supplies. An
31 * alphabet that is not mata's declares one too: it has to name its members the way this concept
32 * does anyway, so it is written for mata and has nothing to adapt from outside.
33 */
34template <typename A>
35concept ExtensibleAlphabet = requires(A& alphabet, typename A::Symbol symbol) {
36 { alphabet.update_next_symbol_value(symbol) };
37 { alphabet.try_add_new_symbol(std::string{}, symbol) };
38};

54/**
55 * @brief A concept for a reserved-key convention (see @c mata::ReservedKeys).
56 */
57template <typename R>
58concept ReservedKeysLike = requires {
59 typename R::Key;
60 requires std::totally_ordered<typename R::Key>;
61 { R::min_epsilon } -> std::convertible_to<typename R::Key>;
62 { R::max_ordinary } -> std::convertible_to<typename R::Key>;
63 { R::epsilon_is_greatest } -> std::convertible_to<bool>;
64 typename std::bool_constant<R::epsilon_is_greatest>;
65 requires R::max_ordinary < R::min_epsilon;
66};

68/**
69 * @brief A concept for a post ordered by key, whose reserved keys are described over that same key.
70 *
71 * Three requirements:
72 * - the post declares @c sorted_by_key, so a lookup may binary-search it and
73 * @c mata::posts::Post::first_epsilon_it() may scan back from its end
74 * - it names a @c Reserved satisfying @c ReservedKeysLike;
75 * - that @c Reserved is over the post's own @c Key.
76 */
77template <typename L>
78concept SortedWithReservedKeys = requires {
79 typename L::Key;
80 typename L::Reserved;
81 requires ReservedKeysLike<typename L::Reserved>;
82 requires std::same_as<typename L::Key, typename L::Reserved::Key>;
83 requires L::sorted_by_key;
84};

97/**
98 * @brief A concept for a set of targets that can be iterated, updated, and queried.
99 */
100template <typename T>
101concept TargetSetLike = IterableRange<T> && requires(const T t, const target_of<T>& target) {
102 typename target_of<T>;
103 requires arity_of<T> == 0;
104 // The successor cursors sweep the targets by pointer, which is what makes them fast.
105 requires std::ranges::contiguous_range<const T>;
106 { std::ranges::is_sorted(t) } -> std::convertible_to<bool>;
107 { std::declval<T&>().push_back(std::declval<const target_of<T>&>()) };
108 { std::declval<T&>().insert(std::declval<const target_of<T>&>()) };
109 { std::declval<T&>().erase(target) };
110 { t.contains(target) } -> std::convertible_to<bool>;

126/**
127 * @brief A concept for a post, a range of entries that can be iterated, updated, and queried.
128 *
129 * @note Moving a post must leave its entries where they are, as moving a @c std::vector does, and must not throw. A
130 * relation moves its posts whenever it grows, and the targets passed to its @c add() may be borrowed from one of
131 * them. A @c std::vector that grows moves its elements only if that cannot throw, and copies them otherwise, which
132 * would leave such borrowed targets dangling.
133 */
134template <typename L>
135concept PostLike =
136 IterableRange<L> && SortedWithReservedKeys<L> && std::is_nothrow_move_constructible_v<L> && requires(const L l) {
137 typename L::Entry;
138 requires PostEntryLike<typename L::Entry>;
139 typename L::Key;
140 typename L::Nested;
141 typename L::Target;
142 { L::key_arity } -> std::convertible_to<size_t>;
143 requires L::key_arity >= 1;
144 { l.is_sorted() } -> std::convertible_to<bool>;
145 { std::declval<L&>().push_back(std::declval<const typename L::Entry&>()) };
146 { std::declval<L&>().find(std::declval<const typename L::Key&>()) };
147 { std::declval<L&>().insert(std::declval<const typename L::Entry&>()) };
148 { l.find(std::declval<const typename L::Key&>()) };
149 { std::declval<L&>().erase(std::declval<const typename L::Entry&>()) };
150 };

170/**
171 * @brief A concept for a cursor over the targets of one state, as Tarjan's SCC traversal walks it.
172 *
173 * The traversal keeps one iterator per state, to resume it after visiting a successor. So the iterator has to be
174 * default-constructible and copyable, yield the targets, and end at @c std::default_sentinel.
175 *
176 * @note The traversal keeps only the iterator, not the cursor it came from, which is a temporary. So the iterator
177 * has to stay valid after the cursor is destroyed, for as long as the relation is not modified: it has to point into
178 * the relation, not into the cursor.
179 */
180template <typename C, typename Target>
181concept SuccessorCursorLike =
182 std::semiregular<internal::CursorIterator<C>> && requires(internal::CursorIterator<C> it) {
183 { *it } -> std::convertible_to<const Target&>;
184 { ++it };
185 { it != std::default_sentinel } -> std::convertible_to<bool>;
186 };

188/**
189 * @brief A concept for a transition relation an @c mata::AutomatonBase can be built on.
190 *
191 * Exactly what the structural algorithms use: a graph over states, with no keys and no post. A relation built from
192 * @c mata::posts::Post satisfies it, but so does a plain adjacency list.
193 *
194 * @note One contract, deliberately whole so the errors are reported immediately, rather
195 * then at the calls inside functions that need them.
196 * @note The largest @c State is reserved as "no state": a relation never holds it as a state, so it has at most
197 * that many states.
198 * @note The automaton's states can outnumber the relation's: an initial or final state may have no transitions
199 * stored. The traversal members (@c for_each_successor(), @c successor_cursor(), @c has_self_loop()) are called for
200 * every state of the automaton, and have to treat such a state as one without transitions, rather than reject it.
201 */
202template <typename D>
203concept DeltaLike = requires(
204 D d,
205 const D cd,
206 const typename D::State s,
207 const typename D::Target t,
208 const BoolVector& is_staying,
209 const std::vector<typename D::State>& renaming
210) {
211 typename D::Target;
212 typename D::State;
213 // A state indexes the relation and every per-state array of the algorithms over it.
214 requires std::unsigned_integral<typename D::State>;
215 requires std::same_as<typename D::State, typename TargetTraits<typename D::Target>::State>;
216 // Exactly the state, not a type converting to it: the algorithms index and do arithmetic with the result.
217 { D::state_of(t) } -> std::same_as<typename D::State>;
218
219 // Structure.
220 { cd.num_of_states() } -> std::convertible_to<size_t>;
221 { d.allocate(size_t{}) };
222 { d.clear() };
223
224 // Traversal.
225 {
226 cd.for_each_successor(s, [](const typename D::Target&) {})
227 };
228 { cd.successor_cursor(s) } -> SuccessorCursorLike<typename D::Target>;
229 { cd.has_self_loop(s) } -> std::convertible_to<bool>;
230
231 // Defragmentation.
232 { d.defragment(is_staying, renaming) };
233
234 // Comparison, for @c is_identical().
235 requires std::equality_comparable<D>;
236
237 // Value semantics.
238 requires std::default_initializable<D>;
239 requires std::movable<D>;
240 requires std::constructible_from<D, size_t>;
241};
