15/// Support class for the bindings-facing *_lazy_ptr() functions (e.g. nfa::plumbing::get_words_lazy_ptr()).
16/// Pull-based adaptor over a @c std::generator<T>, exposing @c done()/@c next() instead of C++23 range-for
17/// iteration. Exists for callers that cannot drive a @c std::generator directly — namely the Python bindings,
18/// generated by Cython, which has no notion of C++23 coroutines or sentinels — so they can still consume one value
19/// at a time instead of materializing the whole sequence.
20///
21/// The generator is resumed past a value only once done() or next() asks for the one after it, never right when
22/// next() takes it, so a caller that stops after k values does not pay for the search of the (k + 1)-th, which may
23/// dominate (e.g. noodlify_mult_eps_lazy() may walk exponentially many routes after its last noodle). Not
24/// thread-safe: done() and next() may run the generator.
25template <class T> class LazyGenerator {
26 public:
27 /// Runs the generator to its first value.
28 explicit LazyGenerator(std::generator<T> generator)
29 : generator_{std::move(generator)},
30 it_{generator_.begin()},
31 holds_value_{it_ != std::default_sentinel} {}
32
33 LazyGenerator(const LazyGenerator&) = delete;
34 LazyGenerator& operator=(const LazyGenerator&) = delete;
35 LazyGenerator(LazyGenerator&&) = delete;
36 LazyGenerator& operator=(LazyGenerator&&) = delete;
37
38 /// True when there are no more values to yield. Runs the generator to its next value unless one is held.
39 bool done() {
40 if (!holds_value_ && it_ != std::default_sentinel) {
41 ++it_;
42 holds_value_ = it_ != std::default_sentinel;
43 }
44 return !holds_value_;
45 }
46
47 /// Take the next value. @pre !done().
48 /// Returns the generator's own reference: a value generator yields an rvalue, so the value is moved out, and a
49 /// reference generator (e.g. @c std::generator<int&>) keeps its reference.
50 T next() {
51 done();
52 holds_value_ = false;
53 return *it_;
54 }
55
56 private:
57 std::generator<T> generator_;
58 std::ranges::iterator_t<std::generator<T>> it_;
59 /// Whether *it_ is a value not taken yet.
60 bool holds_value_;
61};
