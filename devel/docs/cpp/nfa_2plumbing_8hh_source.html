28/// @throws std::overflow_error If @p bool_vec has more entries than there can be states.
29inline void get_elements(StateSet* element_set, const BoolVector& bool_vec) {
30 check_num_of_states(bool_vec.size());
31 element_set->clear();
32 element_set->reserve(bool_vec.count());
33 for (size_t i{0}; i < bool_vec.size(); ++i) {
34 if (bool_vec[i] != 0) { element_set->push_back(static_cast<State>(i)); }
35 }
36}

120/**
121 * @brief Compute intersection of two NFAs.
122 *
123 * Both automata can contain ε-transitions. The product preserves the ε-transitions, i.e.,
124 * for each each product state `(s, t)` with`s -ε-> p`, `(s, t) -ε-> (p, t)` is created, and vice versa.
125 *
126 * Automata must share alphabets.
127 *
128 * @param[out] res The resulting intersection NFA.
129 * @param[in] lhs Input NFA.
130 * @param[in] rhs Input NFA.
131 * @param[in] first_epsilon smallest epsilon.
132 * @param[out] prod_map Mapping of pairs of the original states (lhs_state, rhs_state) to new product states (not used
133 * internally, allocated only when !=nullptr, expensive).
134 * @return NFA as a product of NFAs @p lhs and @p rhs with ε-transitions preserved.
135 */
136inline void intersection(
137 Nfa* res,
138 const Nfa& lhs,
139 const Nfa& rhs,
140 const Symbol first_epsilon = EPSILON,
141 std::unordered_map<std::pair<State, State>, State, mata::utils::PairHash<State, State>>* prod_map = nullptr
142) {
143 *res = intersection(lhs, rhs, first_epsilon, prod_map);
144}

146/**
147 * @brief Concatenate two NFAs.
148 * @param[out] lhs_result_state_renaming Map mapping lhs states to result states.
149 * @param[out] rhs_result_state_renaming Map mapping rhs states to result states.
150 */
151inline void concatenate(
152 Nfa* res,
153 const Nfa& lhs,
154 const Nfa& rhs,
155 const bool use_epsilon = false,
156 StateRenaming* lhs_result_state_renaming = nullptr,
157 StateRenaming* rhs_result_state_renaming = nullptr
158) {
159 // The result is `lhs`: consume it instead of copying; an aliased `rhs` is snapshotted by the consuming overload.
160 if (res == &lhs) {
161 *res = concatenate(std::move(*res), rhs, use_epsilon, lhs_result_state_renaming, rhs_result_state_renaming);
162 } else {
163 *res = concatenate(lhs, rhs, use_epsilon, lhs_result_state_renaming, rhs_result_state_renaming);
164 }
165}

220/**
221 * @brief Same as @c Nfa::get_words_lazy(), but as a free function returning a heap-allocated wrapper instead of a
222 * @c std::generator.
223 *
224 * Exists for callers that cannot drive a C++23 coroutine directly — namely the Python bindings, generated by
225 * Cython, which has no notion of C++23 coroutines/sentinels. Prefer @c Nfa::get_words_lazy() everywhere else.
226 *
227 * Throws @c std::runtime_error when @c MATA_HAS_GENERATOR_SUPPORT is unset (i.e. the standard library does not
228 * implement @c std::generator — some, like Apple's system libc++, do not, even under `-std=c++23`).
229 *
230 * @param aut Automaton whose language is enumerated.
231 * @param max_length Maximum length of words to be returned. Default: "no limit"; will keep yielding forever if the
232 * language is infinite.
233 * @return A heap-allocated generator; the caller takes ownership and must delete it.
234 */
235inline utils::LazyWordGenerator* get_words_lazy_ptr(
236 [[maybe_unused]] const Nfa& aut, [[maybe_unused]] const std::optional<size_t> max_length = std::nullopt
237) {
238#if MATA_HAS_GENERATOR_SUPPORT
239 return new utils::LazyWordGenerator(aut.get_words_lazy(max_length));
240#else
241 throw std::runtime_error("mata was built without std::generator support, get_words_lazy() is unavailable");
242#endif
243}
