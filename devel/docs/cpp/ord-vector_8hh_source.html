22template <class T> bool are_disjoint(const utils::OrdVector<T>& lhs, const utils::OrdVector<T>& rhs) {
23 auto it_lhs = lhs.begin();
24 auto it_rhs = rhs.begin();
25 while (it_lhs != lhs.end() && it_rhs != rhs.end()) {
26 if (*it_lhs == *it_rhs) { return false; }
27 if (*it_lhs < *it_rhs) {
28 ++it_lhs;
29 } else {
30 ++it_rhs;
31 }
32 }
33 return true;
34}

36template <class Key, class Allocator> bool is_sorted(const std::vector<Key, Allocator>& vec) {
37 // Needs only operator<, and an empty vector has no cbegin() + 1 to start from.
38 return std::adjacent_find(vec.cbegin(), vec.cend(), [](const Key& lhs, const Key& rhs) { return !(lhs < rhs); }) ==
39 vec.cend();

159 std::pair<iterator, bool> insert(iterator itr, const Key& x) {
160 if ((itr == end() || x < *itr) && (itr == begin() || *(itr - 1) < x)) { return {vec_.insert(itr, x), true}; }

194 std::pair<iterator, bool> insert(const Key& x) {
195 MATA_EXPENSIVE_ASSERT(is_sorted());
196
197 // An append fast path for keys above the maximum and a branchless lower bound were measured without an
198 // end-to-end gain: see EXP-1 in docs/experiments/.
199 auto pos = std::lower_bound(vec_.begin(), vec_.end(), x);
200 bool inserted{false};
201 if (pos == vec_.end() || x < *pos) { // lower_bound gives !(*pos < x), so `x < *pos` is the only way to differ.
202 pos = vec_.insert(pos, x);
203 inserted = true;
204 }
205
206 MATA_ASSERT(pos == vec_.begin() || *std::prev(pos) < *pos);
207 MATA_ASSERT(std::next(pos) == vec_.end() || *pos < *std::next(pos));
208 MATA_EXPENSIVE_ASSERT(is_sorted());
209 return {pos, inserted};
210 }

239 /// Insert the sorted, duplicate-free @p keys, such as another @c OrdVector or a @c SmallOrdVector.
240 bool insert(const std::span<const Key> keys) {
241 MATA_EXPENSIVE_ASSERT(is_sorted());
242 MATA_EXPENSIVE_ASSERT(std::ranges::is_sorted(keys));
243 if (keys.empty()) { return false; }
244 if (empty() || vec_.back() < keys.front()) {
245 vec_.insert(vec_.end(), keys.begin(), keys.end());
246 return true;
247 }
248
249 // The merge buffer is local. It used to be a function-local `static` shared by every thread in
250 // the process, which corrupted the result of concurrent unions over unrelated containers.
251 VectorType merged{};
252 merged.reserve(vec_.size() + keys.size());
253 std::ranges::set_union(vec_, keys, std::back_inserter(merged));
254 const bool inserted{merged.size() > vec_.size()};
255 vec_ = std::move(merged);
256 MATA_EXPENSIVE_ASSERT(is_sorted());
257 return inserted;
258 }

409 bool is_intersection_empty_with(const OrdVector& rhs) const {
410 MATA_EXPENSIVE_ASSERT(is_sorted());
411 MATA_EXPENSIVE_ASSERT(rhs.is_sorted());
412
413 const_iterator it_lhs = begin();
414 const_iterator it_rhs = rhs.begin();
415
416 while ((it_lhs != end()) &&
417 (it_rhs != rhs.end())) { // until we drop out of the array (or find a common element)
418 if (*it_lhs == *it_rhs) { // in case there exists a common element
419 return false;
420 }
421 if (*it_lhs < *it_rhs) { // in case the element in lhs is smaller
422 ++it_lhs;
423 } else { // in case the element in rhs is smaller
424 MATA_ASSERT(*it_lhs > *it_rhs);
425 ++it_rhs;
426 }
427 }
428 return true;
429 }

438 static OrdVector difference(const OrdVector& lhs, const OrdVector& rhs) {
439 MATA_EXPENSIVE_ASSERT(lhs.is_sorted());
440 MATA_EXPENSIVE_ASSERT(rhs.is_sorted());
441
442 OrdVector result{};
443 auto lhs_it{lhs.begin()};
444 auto rhs_it{rhs.begin()};
445
446 while (lhs_it != lhs.end()) {
447 if (rhs_it == rhs.end()) {
448 result.push_back(*lhs_it);
449 ++lhs_it;
450 } else if (*lhs_it == *rhs_it) {
451 ++lhs_it;
452 ++rhs_it;
453 } else if (*lhs_it < *rhs_it) {
454 result.push_back(*lhs_it);
455 ++lhs_it;
456 } else if (*lhs_it > *rhs_it) {
457 ++rhs_it;
458 }
459 }
460
461 MATA_EXPENSIVE_ASSERT(result.is_sorted());
462 return result;
463 }

465 /**
466 * @brief Write the union of @p lhs and @p rhs into @p result.
467 *
468 * @p result has to be empty: the elements are appended to it, and the return value counts them against
469 * @p lhs. Passing a cleared buffer instead of a fresh vector keeps its capacity across calls, which is
470 * what makes this overload worth having next to @c set_union(lhs, rhs).
471 * @return @c true iff @p rhs has an element that is not in @p lhs.
472 */
473 static bool set_union(const OrdVector& lhs, const OrdVector& rhs, OrdVector& result) {
474 MATA_ASSERT(result.empty());
475 MATA_EXPENSIVE_ASSERT(lhs.is_sorted());
476 MATA_EXPENSIVE_ASSERT(rhs.is_sorted());
477
478 const bool rhs_empty{rhs.empty()};
479 if (lhs.empty()) {
480 result = rhs;
481 return !rhs_empty;
482 }
483 if (rhs_empty) {
484 result = lhs;
485 return false;
486 }
487
488 result.reserve(lhs.size() + rhs.size());
489 std::ranges::set_union(lhs.vec_, rhs.vec_, std::back_inserter(result.vec_));
490
491 MATA_EXPENSIVE_ASSERT(result.is_sorted());
492 return result.size() > lhs.size();
493 }

495 static OrdVector set_union(const OrdVector& lhs, const OrdVector& rhs) {

497 set_union(lhs, rhs, result);
498 return result;
499 }

501 static OrdVector intersection(const OrdVector& lhs, const OrdVector& rhs) {
502 MATA_EXPENSIVE_ASSERT(lhs.is_sorted());
503 MATA_EXPENSIVE_ASSERT(rhs.is_sorted());
504
505 OrdVector result{};
506
507 auto lhs_it = lhs.begin();
508 auto rhs_it = rhs.vec_.begin();
509
510 while ((lhs_it != lhs.end()) && (rhs_it != rhs.end())) { // Until we get to the end of both vectors.
511 if (*lhs_it == *rhs_it) {
512 result.push_back(*lhs_it);
513 ++lhs_it;
514 ++rhs_it;
515 } else if (*lhs_it < *rhs_it) {
516 ++lhs_it;
517 } else if (*rhs_it < *lhs_it) {
518 ++rhs_it;
519 }
520 }
521
522 MATA_EXPENSIVE_ASSERT(result.is_sorted());
523 return result;
524 }
