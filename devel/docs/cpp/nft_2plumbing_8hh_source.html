24/// @throws std::overflow_error If @p bool_vec has more entries than there can be states.
25inline void get_elements(StateSet* element_set, const BoolVector& bool_vec) {
26 nfa::check_num_of_states(bool_vec.size());
27 element_set->clear();
28 element_set->reserve(bool_vec.count());
29 for (size_t i{0}; i < bool_vec.size(); ++i) {
30 if (bool_vec[i] != 0) { element_set->push_back(static_cast<State>(i)); }
31 }
32}

77inline void union_nondet(Nft* union_automaton, const Nft& lhs, const Nft& rhs) {
78 // In place when the result is `lhs`; a fully aliased call keeps the copying path, which doubles the automaton.
79 if (union_automaton == &lhs && &lhs != &rhs) {
80 union_automaton->unite_nondet_with(rhs);
81 } else {
82 *union_automaton = union_nondet(lhs, rhs);
83 }
84}

86inline void union_nondet(Nft* union_automaton, Nft&& lhs, const Nft& rhs) {
87 // `lhs` aliased with `rhs`: the free function snapshots `rhs` before consuming `lhs`.
88 if (union_automaton == &lhs && &lhs != &rhs) {
89 union_automaton->unite_nondet_with(rhs);
90 } else {
91 *union_automaton = union_nondet(std::move(lhs), rhs);

103/**
104 * @brief Compute intersection of two NFTs.
105 *
106 * Both automata can contain ε-transitions. Epsilons will be handled as alphabet symbols.
107 *
108 * Transducers must share alphabets. //TODO: this is not implemented yet.
109 * Transducers must have equal values of @c num_of_levels.
110 *
111 * @param[out] res Resulting NFT as a product of NFTs @p lhs and @p rhs.
112 * @param[in] lhs First NFT to compute intersection for.
113 * @param[in] rhs Second NFT to compute intersection for.
114 * @param[out] prod_map Mapping of pairs of the original states (lhs_state, rhs_state) to new product states (not used
115 * internally, allocated only when !=nullptr, expensive).
116 * @param[in] jump_mode Specifies if the symbol on a jump transition (a transition with a length greater than 1)
117 * is interpreted as a sequence repeating the same symbol or as a single instance of the symbol followed by a sequence
118 * of @c DONT_CARE.
119 * @param[in] lhs_first_aux_state The first auxiliary state in @p lhs. Two auxiliary states does not form a product
120 * state.
121 * @param[in] rhs_first_aux_state The first auxiliary state in @p rhs. Two auxiliary states does not form a product
122 * state.
123 */
124inline void intersection(
125 Nft* res,
126 const Nft& lhs,
127 const Nft& rhs,
128 std::unordered_map<std::pair<State, State>, State, mata::utils::PairHash<State, State>>* prod_map = nullptr,
129 const JumpMode jump_mode = JumpMode::RepeatSymbol,
130 const State lhs_first_aux_state = Limits::max_state,
131 const State rhs_first_aux_state = Limits::max_state
132) {

136/**
137 * @brief Concatenate two NFAs.
138 * @param[out] res Resulting concatenated automaton.
139 * @param[in] lhs First automaton to concatenate.
140 * @param[in] rhs Second automaton to concatenate.
141 * @param use_epsilon Whether to concatenate over an epsilon symbol.
142 * @param[out] lhs_result_state_renaming Map mapping lhs states to result states.
143 * @param[out] rhs_result_state_renaming Map mapping rhs states to result states.
144 */
145inline void concatenate(
146 Nft* res,
147 const Nft& lhs,
148 const Nft& rhs,
149 const bool use_epsilon = false,
150 StateRenaming* lhs_result_state_renaming = nullptr,
151 StateRenaming* rhs_result_state_renaming = nullptr
152) {
153 // The result is `lhs`: consume it instead of copying; an aliased `rhs` is snapshotted by the consuming overload.
154 if (res == &lhs) {
155 *res = concatenate(std::move(*res), rhs, use_epsilon, lhs_result_state_renaming, rhs_result_state_renaming);
156 } else {
157 *res = concatenate(lhs, rhs, use_epsilon, lhs_result_state_renaming, rhs_result_state_renaming);
158 }
159}

184/**
185 * @brief Same as @c Nft::get_words_lazy(), but as a free function returning a heap-allocated wrapper instead of a
186 * @c std::generator.
187 *
188 * Exists for callers that cannot drive a C++23 coroutine directly — namely the Python bindings, generated by
189 * Cython, which has no notion of C++23 coroutines/sentinels. Prefer @c Nft::get_words_lazy() everywhere else.
190 *
191 * Throws @c std::runtime_error when @c MATA_HAS_GENERATOR_SUPPORT is unset (i.e. the standard library does not
192 * implement @c std::generator — some, like Apple's system libc++, do not, even under `-std=c++23`).
193 *
194 * @param aut Automaton whose language is enumerated.
195 * @param max_length Maximum length of words to be returned. Default: "no limit"; will keep yielding forever if the
196 * language is infinite.
197 * @param jump_mode Specifies how to interpret the jump transitions.
198 * @return A heap-allocated generator; the caller takes ownership and must delete it.
199 */
200inline utils::LazyWordGenerator* get_words_lazy_ptr(
201 [[maybe_unused]] const Nft& aut,
202 [[maybe_unused]] const std::optional<size_t> max_length = std::nullopt,
203 [[maybe_unused]] const JumpMode jump_mode = JumpMode::RepeatSymbol
204) {
205#if MATA_HAS_GENERATOR_SUPPORT
206 return new utils::LazyWordGenerator(aut.get_words_lazy(max_length, jump_mode));
207#else
208 throw std::runtime_error("mata was built without std::generator support, get_words_lazy() is unavailable");
209#endif
210}
