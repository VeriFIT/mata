2 * @brief The visits over a chain of posts: reading, writing and rebuilding at any nesting depth.
3 *
4 * Free functions over a post @p P, each recursing on @c mata::arity_of until it reaches the targets.
5 * They are what @c mata::posts::Post, @c mata::posts::DeltaBase and @c mata::AutomatonBase share, so

24/**
25 * @brief The last target of the non-empty @p targets: the largest, as the set is sorted.
26 *
27 * Read through the contiguous storage @c mata::posts::TargetSetLike guarantees, so the set needs no @c back().
28 */
29template <typename Targets> const auto& last_target(const Targets& targets) {
30 MATA_ASSERT(!targets.empty());
31 return std::ranges::data(targets)[targets.size() - 1];

34/**
35 * @brief Apply @p fn to every target under @p post, at any nesting depth.
36 *
37 * @note `if constexpr` on @c key_arity turns the recursion into nested loops (same as a caller would write by hand),
38 * making this depth-generic visit free: measured at 0.99-1.01x of hand-written loops at every arity from 1 to 9.
39 * @p Fn is a template parameter and never a @c std::function: routing the same visit through one would cost 1.08x
40 to 2.18x.
41
42 * @param post The post to visit. @c mata::arity_of is used to recurse down to the innermost post.
43 * @param fn The callable to apply to each target. It must accept a single argument of type @c target_of<P>.
44 */
45template <typename P, typename Fn> void visit_targets(const P& post, Fn&& fn) {
46 if constexpr (arity_of<P> == 0) {
47 for (const target_of<P>& target : post) { fn(target); }
48 } else {
49 for (const auto& entry : post) { visit_targets(entry.nested(), fn); }
50 }
51}

53/**
54 * @brief Does any target under @p post satisfy @p pred?
55 *
56 * @note Stops at the first that does.
57 * @param post The post to visit. @c mata::arity_of is used to recurse down to the innermost post.
58 * @param pred The predicate to apply to each target. It must accept a single argument of type @c target_of<P> and
59 * return a @c bool.
60 * @return @c true if any target satisfies @p pred, @c false otherwise.
61 */
62template <typename P, typename Pred> bool any_target(const P& post, Pred&& pred) {
63 if constexpr (arity_of<P> == 0) {
64 for (const target_of<P>& target : post) {
65 if (pred(target)) { return true; }
66 }
67 } else {
68 for (const auto& entry : post) {
69 if (any_target(entry.nested(), pred)) { return true; }
70 }
71 }
72 return false;
73}

75/**
76 * @brief Returns number of targets under @p post.
77 *
78 * @note Recurses to the innermost post and takes its @c size() rather than counting targets one by one.
79 * @param post The post to visit. @c mata::arity_of is used to recurse down to the innermost post.
80 * @return The number of targets under @p post.
81 */
82template <typename P> size_t count_targets(const P& post) {
83 if constexpr (arity_of<P> == 0) {
84 return post.size();
85 } else {
86 size_t total{0};
87 for (const auto& entry : post) { total += count_targets(entry.nested()); }
88 return total;
89 }
90}

92/**
93 * @brief Apply @p fn to every move under @p post, as `fn(keys..., target)`.
94 *
95 * @note The keys accumulate down the nesting and arrive as a pack, so at @c key_arity 1 this expands to
96 * exactly `fn(key, target)`. A deeper relation simply passes more key arguments.
97 * @param post The post to visit. @c mata::arity_of is used to recurse down to the innermost post.
98 * @param fn The callable to apply to each move. It must accept a parameter pack of
99 * the keys followed by a single argument of type @c target_of<P>.
100 * @param keys The keys accumulated down the nesting, passed to @p fn along with each target.
101 */
102template <typename P, typename Fn, typename... Keys> void visit_moves(const P& post, Fn&& fn, const Keys&... keys) {
103 if constexpr (arity_of<P> == 0) {
104 for (const target_of<P>& target : post) { fn(keys..., target); }
105 } else {
106 for (const auto& entry : post) { visit_moves(entry.nested(), fn, keys..., entry.key()); }
107 }
108}

110/**
111 * @brief Insert @p target into @p post, which must be the innermost post.
112 *
113 * @param post The post to insert into. @c mata::arity_of is used to recurse down to the innermost post.
114 * @param target The target to insert.
115 * @return Whether @p target was inserted, i.e., was not there already.
116 */
117template <typename P> bool insert_target(P& post, const target_of<P>& target) {
118 static_assert(arity_of<P> == 0, "the key path ran out before the innermost post");
119 if constexpr (requires { post.insert(target).second; }) {
120 return post.insert(target).second;
121 } else {
122 // The set does not say whether it inserted, so its size does.
123 const size_t old_size{post.size()};
124 post.insert(target);
125 return post.size() != old_size;
126 }
127}

129/**
130 * @brief Insert @p target under the key path @p k0, @p rest, creating posts along the way.
131 *
132 * @param post The post to insert into. @c mata::arity_of is used to recurse down to the innermost post.
133 * @param target The target to insert.
134 * @param k0 The first key to descend through.
135 * @param rest The remaining keys to descend through, one per post.
136 * The number of keys must match the @c key_arity of @p post minus one.
137 * @return Whether @p target was inserted, i.e., was not there already.
138 */
139template <typename P, typename K0, typename... Rest>
140bool insert_target(P& post, const target_of<P>& target, const K0& k0, const Rest&... rest) {
141 static_assert(arity_of<P> == sizeof...(Rest) + 1, "the key path must be one key per post");
142 if (const auto entry_it{post.find(k0)}; entry_it != post.end()) {
143 return insert_target(entry_it->nested(), target, rest...);
144 }
145 // A new key is published only with its target: if inserting the target throws, no empty entry is left behind.
146 typename P::Nested nested{};
147 const bool inserted{insert_target(nested, target, rest...)};
148 post.insert(typename P::Entry{k0, std::move(nested)});
149 return inserted;
150}

152/**
153 * @brief Returns whether @p target is reachable in the innermost post @p post.
154 *
155 * @param post The post to search in. @c mata::arity_of is used to recurse down to the innermost post.
156 * @param target The target to search for.
157 * @return @c true if @p target is reachable, @c false otherwise.
158 */
159template <typename P> bool has_target_at(const P& post, const target_of<P>& target) {
160 static_assert(arity_of<P> == 0, "the key path ran out before the innermost post");
161 return post.contains(target);

164/**
165 * @brief Returns whether @p target is reachable from @p post down exactly the key path @p k0, @p rest.
166 *
167 * @param post The post to search in. @c mata::arity_of is used to recurse down to the innermost post.
168 * @param target The target to search for.
169 * @param k0 The first key to descend through.
170 * @param rest The remaining keys to descend through, one per post.
171 * The number of keys must match the @c key_arity of @p post minus one.
172 * @return @c true if @p target is reachable down the key path, @c false otherwise.
173 */
174template <typename P, typename K0, typename... Rest>
175bool has_target_at(const P& post, const target_of<P>& target, const K0& k0, const Rest&... rest) {
176 static_assert(arity_of<P> == sizeof...(Rest) + 1, "the key path must be one key per post");
177 const auto entry_it{post.find(k0)};
178 return entry_it != post.end() && has_target_at(entry_it->nested(), target, rest...);
179}

181/**
182 * @brief Erase @p target from the innermost post @p post.
183 *
184 * @param post The post to erase from. @c mata::arity_of is used to recurse down to the innermost post.
185 * @param target The target to erase.
186 * @return Whether @p post is empty once the erase is done.
187 */
188template <typename P> bool erase_target(P& post, const target_of<P>& target) {
189 static_assert(arity_of<P> == 0, "the key path ran out before the innermost post");
190 post.erase(target);
191 return post.empty();
192}

194/**
195 * @brief Erase @p target from under the key path @p k0, @p rest, pruning every post it empties.
196 *
197 * @param post The post to erase from. @c mata::arity_of is used to recurse down to the innermost post.
198 * @param target The target to erase.
199 * @param k0 The first key to descend through.
200 * @param rest The remaining keys to descend through, one per post.
201 * The number of keys must match the @c key_arity of @p post minus one.
202 * @return Whether @p post is empty once the erase and any pruning below it are done.
203 * @throws std::invalid_argument if any key on the path is missing.
204 */
205template <typename P, typename K0, typename... Rest>
206bool erase_target(P& post, const target_of<P>& target, const K0& k0, const Rest&... rest) {
207 static_assert(arity_of<P> == sizeof...(Rest) + 1, "the key path must be one key per post");
208 const auto entry_it{post.find(k0)};
209 if (entry_it == post.end()) { throw std::invalid_argument("The transition does not exist."); }
210 if (erase_target(entry_it->nested(), target, rest...)) { post.erase(*entry_it); }
211 return post.empty();
212}

214/**
215 * @brief Returns whether @p a and @p b structurally equal, including every target under them.
216 *
217 * @note This is a deep comparison, not a shallow one.
218 * Standard @c operator== on a post compares only the keys of its entries, so it ignores every difference in the
219 * targets. To properly compare two posts, you need to use this function.
220 *
221 * @param a The first post to compare.
222 * @param b The second post to compare.
223 * @return @c true if @p a and @p b are structurally equal, @c false otherwise.
224 */
225template <typename P> bool posts_equal(const P& a, const P& b) {
226 if constexpr (arity_of<P> == 0) {
227 return std::ranges::equal(a, b);
228 } else {
229 if (a.size() != b.size()) { return false; }
230 for (auto it_a{a.begin()}, it_b{b.begin()}; it_a != a.end(); ++it_a, ++it_b) {
231 if (!(it_a->key() == it_b->key())) { return false; }
232 if (!posts_equal(it_a->nested(), it_b->nested())) { return false; }
233 }
234 return true;
235 }
236}

238/**
239 * @brief Rebuild @p post with every target renamed by function @p rename.
240 *
241 * The targets stay sorted and duplicate-free for any @p rename.
242 * A strictly increasing one costs one comparison per target.
243 * One that merges states or reorders them falls back to an ordered insert.
244 *
245 * @param post The post to rebuild. @c mata::arity_of is used to recurse down to the innermost post.
246 * @param rename The callable to apply to each target's state. It must accept a single argument
247 * of type @c state_of<target_of<P>> and return a new state of the same type.
248 * @return A new post with the same structure as @p post, but with every target's state renamed by @p rename.
249 */
250template <typename P, typename Fn> P renumbered(const P& post, Fn&& rename) {
251 P out{};
252 if constexpr (arity_of<P> == 0) {
253 using Traits = TargetTraits<target_of<P>>;
254 for (const target_of<P>& target : post) {
255 const target_of<P> renamed{Traits::with_state(target, rename(Traits::state_of(target)))};
256 if (out.empty() || last_target(out) < renamed) {
257 out.push_back(renamed);
258 } else {
259 // A merged state repeats a target, which this skips, or puts a payload target out of order.
260 out.insert(renamed);
261 }
262 }
263 } else {
264 for (const auto& entry : post) {
265 out.push_back(typename P::Entry{entry.key(), renumbered(entry.nested(), rename)});
266 }
267 }
268 return out;
269}

274/**
275 * @brief Keep in @p post only the targets @p is_staying admits, renaming their states, in place.
276 *
277 * The renaming is monotone on the staying states, so the sets stay sorted, given a target order that follows the states
278 * (see @c mata::TargetTraits). Entries left without targets are erased.
279 *
280 * @param post The post to filter. @c mata::arity_of is used to recurse down to the innermost post.
281 * @param is_staying Indexed by old state, true for a state that stays. It must cover the state of every target in
282 * @p post.
283 * @param renaming Indexed by old state, the new state of a staying one. It must cover every staying state of a target
284 * in @p post.
285 */
286template <typename P, typename Renaming>
287inline void defragment_in_place(P& post, const BoolVector& is_staying, const Renaming& renaming) {
288 if constexpr (arity_of<P> == 0) {
289 using Traits = TargetTraits<target_of<P>>;
290 const auto is_leaving{[&is_staying](const target_of<P>& target) {
291 return !is_staying[Traits::state_of(target)];
292 }};
293 if constexpr (requires { post.erase_if(is_leaving); }) {
294 post.erase_if(is_leaving);
295 for (target_of<P>& target : post) {
296 target = Traits::with_state(target, renaming[Traits::state_of(target)]);
297 }
298 MATA_ASSERT(
299 std::ranges::is_sorted(post), "renaming reordered the targets: their order has to follow their states"
300 );
301 } else {
302 post = defragmented(post, is_staying, renaming);
303 }
304 } else {
305 for (auto entry_it{post.begin()}; entry_it != post.end();) {
306 defragment_in_place(entry_it->nested(), is_staying, renaming);
307 if (entry_it->nested().empty()) {
308 entry_it = post.erase(entry_it);
309 } else {
310 ++entry_it;
311 }
312 }
313 }
314}

316/**
317 * @brief Rebuild @p post keeping only the targets @p is_staying admits, and renaming their states.
318 *
319 * @param post The post to rebuild. @c mata::arity_of is used to recurse down to the innermost post.
320 * @param is_staying Indexed by old state, true for a state that stays. It must cover the state of every target in
321 * @p post.
322 * @param renaming Indexed by old state, the new state of a staying one. It must cover every staying state of a target
323 * in @p post.
324 * @return A new post with the same structure as @p post, but with only the targets whose states are staying,
325 * and with their states renamed according to @p renaming.
326 */
327template <typename P, typename Renaming>
328P defragmented(const P& post, const BoolVector& is_staying, const Renaming& renaming) {
329 P out{};
330 if constexpr (arity_of<P> == 0) {
331 for (const target_of<P>& target : post) {
332 using Traits = TargetTraits<target_of<P>>;
333 const auto state{Traits::state_of(target)};
334 if (is_staying[state]) { out.push_back(Traits::with_state(target, renaming[state])); }
335 }
336 MATA_ASSERT(
337 std::ranges::is_sorted(out), "renaming reordered the targets: their order has to follow their states"
338 );
339 } else {
340 for (const auto& entry : post) {
341 typename P::Nested nested{defragmented(entry.nested(), is_staying, renaming)};
342 if (!nested.empty()) { out.push_back(typename P::Entry{entry.key(), std::move(nested)}); }
343 }
344 }
345 return out;
346}
