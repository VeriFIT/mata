26/**
27 * @brief A two-dimensional map that can be used to store pairs of values and their associated value.
28 * This class can handle both small and large maps efficiently.
29 *
30 * Values are stored in a matrix of @c Cell cells (4 Bytes each, unless @p T is smaller). The matrix rows are
31 * allocated lazily: a row is allocated and zeroed when the first value is inserted into it, so the memory
32 * footprint is proportional to the number of first-dimension keys actually used, not to the size of the first
33 * dimension. A cell holds the stored value incremented by one; zero means that no value is stored.
34 *
35 * The largest matrix we are brave enough to allocate is @p MaxMatrixBytes Bytes, 200 MB by default, which is
36 * 50 million cells of 4 Bytes. For larger dimensions we do not allocate a matrix, but keep one small hash table
37 * (open addressing, linear probing) per used first-dimension key, holding the second-dimension keys paired with
38 * it. A product construction looks up many pairs that share a key in a row, which then stay in one small table
39 * instead of being scattered over a table of all pairs. The array of these tables is as long as the first
40 * dimension; when that array alone would exceed @p MaxMatrixBytes, a single hash table of all pairs is used
41 * instead (hash_storage_), keyed by the pair of keys packed into 64 bits. That table grows geometrically with the
42 * number of stored pairs, independently of the dimensions. The product of the dimensions must therefore fit into
43 * 64 bits; with keys of at most 32 bits it always does.
44 *
45 * @tparam T Type of the values stored in the map. Must be an unsigned type.
46 * @tparam TrackInverted Whether to track inverted indices for the first and second dimensions.
47 * To use this feature correctly, the mapping has to be reversible.
48 * @tparam MaxMatrixBytes Maximum size of the matrix in Bytes before switching to the hash tables.
49 */
50template <typename T, bool TrackInverted = true, size_t MaxMatrixBytes = 200'000'000> class TwoDimensionalMap {
51 static_assert(std::is_unsigned_v<T>, "TwoDimensionalMap requires an unsigned type");
52
53 public:
54 using Map = std::unordered_map<std::pair<T, T>, T, PairHash<T, T>>;
55 /// A matrix cell stores the value incremented by one; zero denotes a missing value.
56 using Cell = std::conditional_t<(sizeof(T) < sizeof(uint32_t)), T, uint32_t>;
57 /// A row is a dynamically sized array: its length is a runtime value, so 'std::array' does not apply, and
58 /// 'std::vector' would add a second size and capacity per row.
59 // NOLINTNEXTLINE(cppcoreguidelines-avoid-c-arrays,modernize-avoid-c-arrays): runtime-sized array, see above.
60 using Row = std::unique_ptr<Cell[]>;
61 using MatrixStorage = std::vector<Row>;
62 /// A slot of a sparse row: a second-dimension key, or @c no_value in an empty slot, and its value.
63 struct SparseSlot {
64 T key;
65 T value;
66 };
67 /// The hash table of the second-dimension keys paired with one first-dimension key.
68 struct SparseRow {
69 // NOLINTNEXTLINE(cppcoreguidelines-avoid-c-arrays,modernize-avoid-c-arrays): runtime-sized array, as 'Row'.
70 std::unique_ptr<SparseSlot[]> slots{};
71 /// Wide enough to count the keys of a row, which are distinct values of @c T: a row of 32-bit keys then
72 /// takes two words.
73 using Counter = std::conditional_t<(sizeof(T) <= sizeof(uint32_t)), uint32_t, uint64_t>;
74 /// Number of stored keys.
75 Counter size{0};
76 /// Bits of a 64-bit hash discarded to index the slots: 64 - log2 of the number of slots. Always 64, never
77 /// @c Counter: capacity() must stay representable even when a row has @c Counter-many keys.
78 Counter shift{0};
79
80 size_t capacity() const noexcept { return size_t{1} << (64 - shift); }
81 };
82 using SparseStorage = std::vector<SparseRow>;
83 /// A slot of the single hash table used for the largest dimensions: the packed pair of keys and its value.
84 struct Slot {
85 uint64_t key;
86 T value;
87 };
88 using HashStorage = std::vector<Slot>;
89 using InvertedStorage = std::vector<T>;
90
91 /// Returned by get() when no value is stored for the pair of keys.
92 static constexpr T no_value{std::numeric_limits<T>::max()};
93 /// The largest value a matrix cell can hold.
94 static constexpr T max_value{static_cast<T>(std::numeric_limits<Cell>::max() - 1)};
95
96 /**
97 * @brief Constructor for TwoDimensionalMap.
98 * @param first_dim_size Size of the first dimension.
99 * @param second_dim_size Size of the second dimension.
100 * @throw std::length_error if the single hash table is needed and @p first_dim_size * @p second_dim_size does not
101 * fit into 64 bits.
102 */
103 TwoDimensionalMap(const size_t first_dim_size, const size_t second_dim_size)
104 : storage_{choose_storage(first_dim_size, second_dim_size)},
105 first_dim_size_{first_dim_size},
106 second_dim_size_{second_dim_size} {
107 MATA_ASSERT(first_dim_size_ < std::numeric_limits<T>::max());
108 MATA_ASSERT(second_dim_size_ < std::numeric_limits<T>::max());
109 // Only the vector of rows is allocated here. The rows themselves are allocated on first insert().
110 switch (storage_) {
111 case Storage::Matrix:
112 matrix_storage_ = MatrixStorage(first_dim_size_);
113 break;
114 case Storage::SparseRows:
115 sparse_storage_ = SparseStorage(first_dim_size_);
116 break;
117 case Storage::Hash:
118 // Packed keys are below first_dim_size_ * second_dim_size_, so empty_key is never a packed key.
119 if (first_dim_size_ > std::numeric_limits<uint64_t>::max() / second_dim_size_) {
120 throw std::length_error{"TwoDimensionalMap: the pairs of keys do not fit into 64 bits"};
121 }
122 hash_storage_ = HashStorage(min_hash_capacity, Slot{.key = empty_key, .value = 0});
123 break;
124 }
125 }
126
127 /**
128 * @brief Get the value associated with a pair of keys.
129 * @param first First key.
130 * @param second Second key.
131 * @return The value associated with the pair, or no_value (std::numeric_limits<T>::max()) if not found.
132 */
133 T get(const T first, const T second) const {
134 MATA_ASSERT(first < first_dim_size_);
135 MATA_ASSERT(second < second_dim_size_);
136 if (storage_ == Storage::Matrix) {
137 const Row& row{matrix_storage_[first]};
138 if (row == nullptr) { return no_value; }
139 const Cell cell{row[second]};
140 return cell == 0 ? no_value : static_cast<T>(cell - 1);
141 }
142 if (storage_ == Storage::SparseRows) {
143 const SparseRow& row{sparse_storage_[first]};
144 if (row.slots == nullptr) { return no_value; }
145 const SparseSlot& slot{row.slots[find_sparse_slot(row, second)]};
146 return slot.key == second ? slot.value : no_value;
147 }
148
149 const Slot& slot{hash_storage_[find_slot(pack(first, second))]};
150 return slot.key == empty_key ? no_value : slot.value;
151 }

153 /**
154 * @brief Insert a value associated with a pair of keys.
155 * @param first First key.
156 * @param second Second key.
157 * @param value Value to associate with the pair.
158 */
159 void insert(const T first, const T second, const T value) {
160 MATA_ASSERT(first < first_dim_size_);
161 MATA_ASSERT(second < second_dim_size_);
162 if (storage_ == Storage::Matrix) {
163 MATA_ASSERT(value <= max_value, "value {} does not fit into a matrix cell", value);
164 Row& row{matrix_storage_[first]};
165 // std::make_unique<Cell[]>() value-initialises the cells, marking the whole row as empty.
166 // NOLINTNEXTLINE(cppcoreguidelines-avoid-c-arrays,modernize-avoid-c-arrays): see 'Row' above.
167 if (row == nullptr) { row = std::make_unique<Cell[]>(second_dim_size_); }
168 row[second] = static_cast<Cell>(value + 1);
169 } else if (storage_ == Storage::SparseRows) {
170 claim_sparse_slot(second, first).first.value = value;
171 } else {
172 const uint64_t key{pack(first, second)};
173 size_t index{find_slot(key)};
174 if (hash_storage_[index].key == empty_key) {
175 // Keep the load factor at most 1/2 so that the probe sequences stay short.
176 if (2 * (hash_size_ + 1) > hash_storage_.size()) {
177 grow_hash_storage();
178 index = find_slot(key);
179 }
180 hash_storage_[index].key = key;
181 ++hash_size_;
182 }
183 hash_storage_[index].value = value;
184 }
185 if constexpr (TrackInverted) {
186 // The inverted arrays only ever grow: resizing them to value + 1 unconditionally would drop the
187 // entries of all larger values inserted before.
188 if (first_dim_inverted_.size() <= value) {
189 first_dim_inverted_.resize(value + 1);
190 second_dim_inverted_.resize(value + 1);
191 }
192 first_dim_inverted_[value] = first;
193 second_dim_inverted_[value] = second;
194 }
195 }
196
197 /**
198 * @brief Insert @p value for the pair of keys, unless the pair already has a value.
199 *
200 * One probe of the hash table decides whether the pair is new, so a caller that would look the pair up and then
201 * insert it pays for a single probe, not two.
202 *
203 * @param first First key.
204 * @param second Second key.
205 * @param value Value to associate with the pair when it has none yet.
206 * @return The value stored for the pair (the new one, or the one inserted earlier) and whether @p value was
207 * inserted.
208 */
209 std::pair<T, bool> insert_if_absent(const T first, const T second, const T value) {
210 MATA_ASSERT(first < first_dim_size_);
211 MATA_ASSERT(second < second_dim_size_);
212 if (storage_ == Storage::Matrix) {
213 MATA_ASSERT(value <= max_value, "value {} does not fit into a matrix cell", value);
214 Row& row{matrix_storage_[first]};
215 // NOLINTNEXTLINE(cppcoreguidelines-avoid-c-arrays,modernize-avoid-c-arrays): see 'Row' above.
216 if (row == nullptr) { row = std::make_unique<Cell[]>(second_dim_size_); }
217 Cell& cell{row[second]};
218 if (cell != 0) { return {static_cast<T>(cell - 1), false}; }
219 cell = static_cast<Cell>(value + 1);
220 } else if (storage_ == Storage::SparseRows) {
221 const auto [slot, is_new]{claim_sparse_slot(second, first)};
222 if (!is_new) { return {slot.value, false}; }
223 slot.value = value;
224 } else {
225 const uint64_t key{pack(first, second)};
226 size_t index{find_slot(key)};
227 if (hash_storage_[index].key == key) { return {hash_storage_[index].value, false}; }
228 // Keep the load factor at most 1/2 so that the probe sequences stay short.
229 if (2 * (hash_size_ + 1) > hash_storage_.size()) {
230 grow_hash_storage();
231 index = find_slot(key);
232 }
233 hash_storage_[index].key = key;
234 hash_storage_[index].value = value;
235 ++hash_size_;
236 }
237 if constexpr (TrackInverted) {
238 // The inverted arrays only ever grow: resizing them to value + 1 unconditionally would drop the
239 // entries of all larger values inserted before.
240 if (first_dim_inverted_.size() <= value) {
241 first_dim_inverted_.resize(value + 1);
242 second_dim_inverted_.resize(value + 1);
243 }
244 first_dim_inverted_[value] = first;
245 second_dim_inverted_[value] = second;
246 }
247 return {value, true};
248 }

271 /// Where the pairs are stored; see the description of the class.
272 enum class Storage : uint8_t { Matrix, SparseRows, Hash };

274 static constexpr Storage choose_storage(const size_t first_dim_size, const size_t second_dim_size) noexcept {
275 constexpr size_t max_num_of_cells{MaxMatrixBytes / sizeof(Cell)};
276 // Division instead of multiplication: first_dim_size * second_dim_size can overflow.
277 if (second_dim_size == 0 || first_dim_size <= max_num_of_cells / second_dim_size) { return Storage::Matrix; }
278 return first_dim_size <= MaxMatrixBytes / sizeof(SparseRow) ? Storage::SparseRows : Storage::Hash;
279 }

281 /// Initial number of slots of a sparse row, a power of two.
282 static constexpr size_t min_sparse_row_capacity{4};
283
284 // NOLINTNEXTLINE(cppcoreguidelines-avoid-c-arrays,modernize-avoid-c-arrays): see 'SparseRow'.
285 static std::unique_ptr<SparseSlot[]> make_sparse_slots(const size_t capacity) {
286 // NOLINTNEXTLINE(cppcoreguidelines-avoid-c-arrays,modernize-avoid-c-arrays): see 'SparseRow'.
287 auto slots{std::make_unique_for_overwrite<SparseSlot[]>(capacity)};
288 std::fill_n(slots.get(), capacity, SparseSlot{.key = no_value, .value = 0});
289 return slots;
290 }
291
292 /// Find the slot of the allocated @p row holding @p key, or the empty slot where @p key would be inserted.
293 static size_t find_sparse_slot(const SparseRow& row, const T key) noexcept {
294 // Fibonacci hashing in 64 bits, however wide the keys are: the shift counts the bits discarded from this
295 // 64-bit hash, so growing a row cannot shift by the width of a narrower hash (undefined behavior).
296 constexpr uint64_t multiplier{0x9e'37'79'b9'7f'4a'7c'15ULL};
297 const size_t mask{row.capacity() - 1};
298 size_t index{static_cast<size_t>((static_cast<uint64_t>(key) * multiplier) >> row.shift)};
299 while (row.slots[index].key != key && row.slots[index].key != no_value) { index = (index + 1) & mask; }
300 return index;
301 }
302
303 /// The slot of the pair of keys in the sparse rows, and whether the pair was added to them by this call.
304 std::pair<SparseSlot&, bool> claim_sparse_slot(const T key, const T row_key) {
305 SparseRow& row{sparse_storage_[row_key]};
306 if (row.slots == nullptr) {
307 row.slots = make_sparse_slots(min_sparse_row_capacity);
308 row.shift = static_cast<typename SparseRow::Counter>(64 - std::countr_zero(min_sparse_row_capacity));
309 }
310 size_t index{find_sparse_slot(row, key)};
311 if (row.slots[index].key == key) { return {row.slots[index], false}; }
312 // Keep the load factor at most 1/2 so that the probe sequences stay short.
313 if (2 * (size_t{row.size} + 1) > row.capacity()) {
314 const size_t old_capacity{row.capacity()};
315 const auto old_slots{std::exchange(row.slots, make_sparse_slots(2 * old_capacity))};
316 --row.shift;
317 for (size_t old_index{0}; old_index < old_capacity; ++old_index) {
318 const SparseSlot& slot{old_slots[old_index]};
319 if (slot.key != no_value) { row.slots[find_sparse_slot(row, slot.key)] = slot; }
320 }
321 index = find_sparse_slot(row, key);
322 }
323 row.slots[index].key = key;
324 ++row.size;
325 return {row.slots[index], true};
326 }

328 /// Marks an empty slot of the hash table.
329 static constexpr uint64_t empty_key{std::numeric_limits<uint64_t>::max()};
330 /// Initial number of slots of the hash table, a power of two.
331 static constexpr size_t min_hash_capacity{16};
332
333 uint64_t pack(const T first, const T second) const noexcept {
334 return (static_cast<uint64_t>(first) * second_dim_size_) + second;
335 }
336
337 /**
338 * @brief Find the slot holding @p key, or the empty slot where @p key would be inserted.
339 *
340 * Fibonacci hashing: the top bits of the product with 2^64 / phi index the power-of-two sized table.
341 */
342 size_t find_slot(const uint64_t key) const noexcept {
343 const size_t mask{hash_storage_.size() - 1};
344 size_t index{static_cast<size_t>((key * 0x9e'37'79'b9'7f'4a'7c'15ULL) >> hash_shift_)};
345 while (hash_storage_[index].key != key && hash_storage_[index].key != empty_key) { index = (index + 1) & mask; }
346 return index;
347 }
348
349 /// Double the number of slots of the hash table and reinsert the stored pairs.
350 void grow_hash_storage() {
351 const HashStorage old_storage{
352 std::exchange(hash_storage_, HashStorage(2 * hash_storage_.size(), Slot{.key = empty_key, .value = 0}))
353 };
354 --hash_shift_;
355 for (const Slot& slot : old_storage) {
356 if (slot.key != empty_key) { hash_storage_[find_slot(slot.key)] = slot; }
357 }
358 }
